def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype
    bs, L, _, dim = input_tensor.shape
    H = config["hidden_dim"]
    eps = 1e-5
    # LN input
    x_flat = input_tensor.reshape(bs * L * L, dim).contiguous()
    norm_w = weights["norm.weight"].to(device).contiguous()
    norm_b = weights["norm.bias"].to(device).contiguous()
    x_norm_flat = triton_layernorm(x_flat, norm_w, norm_b, eps=eps)
    x_norm = x_norm_flat.view(bs, L, L, dim)

    # FP16
    x_norm_h = x_norm.to(torch.float16)

    # stacked weight for left/right proj+gate
    w_left_proj = weights["left_proj.weight"]
    w_left_gate = weights["left_gate.weight"]
    w_right_proj = weights["right_proj.weight"]
    w_right_gate = weights["right_gate.weight"]
    # stack: left_proj, left_gate, right_proj, right_gate
    stacked_w = torch.cat([w_left_proj, w_left_gate, w_right_proj, w_right_gate], dim=0)  # shape (4*H, dim)
    stacked_w = stacked_w.to(device, torch.float16).contiguous()
    # linear
    combined = F.linear(x_norm_h, stacked_w)  # shape [B, L, L, 4*H]
    # split
    cp = combined
    left_proj = cp[..., :H]
    left_gate = torch.sigmoid(cp[..., H:2*H])
    right_proj = cp[..., 2*H:3*H]
    right_gate = torch.sigmoid(cp[..., 3*H:4*H])
    # elementwise multiply
    left = left_proj * left_gate
    right = right_proj * right_gate
    if mask is not None and not config.get("nomask", False):
        mask_h = mask.to(device, torch.float16).unsqueeze(-1)
        left = left * mask_h
        right = right * mask_h

    # out_gate
    out_gate_w = weights["out_gate.weight"]
    out_gate_w = out_gate_w.to(device, torch.float16).contiguous()
    out_gate = torch.sigmoid(F.linear(x_norm_h, out_gate_w))

    # contract
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    # matmul B*H batch

"""
TriMul “outgoing” forward‑pass – Triton + fused PyTorch kernels
================================================================
The original PyTorch implementation performs a sequence of LayerNorms,
four bias‑free linear projections (left/right + gates), a masked element‑wise
product, an N³ tensor contraction (implemented as a batched GEMM) and a
final linear projection.  The heavy work is the five matrix‑multiplications
on the (B·N·N) × D input tensor.

Optimisations
-------------
1. **Fused LayerNorm** – a 2‑pass Triton kernel (mean / variance → affine)
   is used for the two normalisations (input and hidden‑dim).

2. **Projection‑gate fusion** – the four linear layers
   (left_proj, left_gate, right_proj, right_gate) are concatenated into a
   single weight matrix of shape (4·H, D).  A single ``F.linear`` call yields
   all four tensors at once, reducing the number of passes over the large
   input from *four* to *one*.  Gates are obtained by applying a sigmoid on
   the appropriate slice and fused with the projected values.

3. **Mask handling** – the mask (if present) is applied after gating; the
   order is commutative for a 0/1 mask and saves an extra broadcast.

4. **Batched GEMM** – the contraction
   ``out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]``
   is expressed as a batched matrix multiplication
   ``(B·H, N, N) @ (B·H, N, N)ᵀ`` which cuBLAS executes with Tensor‑Cores on
   the H100.

5. **Mixed precision** – all large matrix operations are performed in
   FP16 (half) and the final result is cast back to FP32, matching the
   reference implementation while maximising Tensor‑Core utilisation.

The function ``custom_kernel`` below follows these steps and returns the
output tensor of shape ``[B, N, N, D]`` (float32).  It runs without gradient
tracking and works for any configuration from the test matrix.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton row‑wise LayerNorm (two‑pass: mean/var → affine)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,                # *ptr to input  [M, N]
    y_ptr,                # *ptr to output [M, N]
    weight_ptr,           # *ptr to weight [N]
    bias_ptr,             # *ptr to bias   [N]
    M,                    # rows
    N,                    # columns (features)
    eps,                  # epsilon
    stride_xm, stride_xn,  # strides for x
    stride_ym, stride_yn,  # strides for y
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
):
    pid = tl.program_id(0)

    # -------------------------- row offsets ---------------------------
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # ----------------------- first pass: mean/var --------------------
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_N)
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        sum_val += tl.sum(x, axis=1)
        sum_sq  += tl.sum(x * x, axis=1)

    mean = sum_val / N
    var  = sum_sq / N - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # ----------------------- second pass: normalise ------------------
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        norm_x = (x - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_n, mask=mask_n, other=0.0)
        b = tl.load(bias_ptr   + offs_n, mask=mask_n, other=0.0)

        y = norm_x * w[None, :] + b[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn,
            y,
            mask=mask_m[:, None] & mask_n[None, :],
        )


def triton_layernorm(x: torch.Tensor,
                     weight: torch.Tensor,
                     bias: torch.Tensor,
                     eps: float = 1e-5) -> torch.Tensor:
    """
    Row‑wise LayerNorm implemented in Triton.
    ``x`` must be 2‑D ``[M, N]``.
    Returns a tensor of the same shape and dtype as ``x``.
    """
    assert x.dim() == 2, "triton_layernorm expects a 2‑D tensor"
    M, N = x.shape
    y = torch.empty_like(x)

    x = x.contiguous()
    weight = weight.contiguous()
    bias = bias.contiguous()

    stride_xm, stride_xn = x.stride()
    stride_ym, stride_yn = y.stride()

    grid = lambda meta: (triton.cdiv(M, meta["BLOCK_M"]),)

    layernorm_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        weight_ptr=weight,
        bias_ptr=bias,
        M=M,
        N=N,
        eps=eps,
        stride_xm=stride_xm,
        stride_xn=stride_xn,
        stride_ym=stride_ym,
        stride_yn=stride_yn,
        BLOCK_M=256,            # tuned for the large row count (≈B·N²)
        BLOCK_N=64,             # column tile – a multiple of 64 is good for fp16
    )
    return y


# ----------------------------------------------------------------------
# Entry point for the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor   shape [B, L, L, D] (float32)
        - mask         : torch.Tensor   shape [B, L, L] or ``None``
        - weights      : dict mapping weight names to torch.Tensors
        - config       : dict with keys ``dim``, ``hidden_dim`` and optional ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, L, L, D] (float32)
    """
    # --------------------------------------------------------------
    # 0️⃣ unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype  # float32

    bs, L, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    eps = 1e-5

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm (Triton) over the last dimension D
    # --------------------------------------------------------------
    M = bs * L * L
    x_flat = input_tensor.reshape(M, dim).contiguous()
    norm_w = weights["norm.weight"].to(device, dtype).contiguous()
    norm_b = weights["norm.bias"].to(device, dtype).contiguous()
    x_norm_flat = triton_layernorm(x_flat, norm_w, norm_b, eps=eps)
    x_norm = x_norm_flat.view(bs, L, L, dim)

    # --------------------------------------------------------------
    # 2️⃣ Cast to fp16 for all heavy work
    # --------------------------------------------------------------
    x_norm_h = x_norm.to(torch.float16)

    # --------------------------------------------------------------
    # 3️⃣ Fused projection + gate for left/right (single matmul)
    # --------------------------------------------------------------
    # Stack the four weight matrices (proj + gate for left & right)
    w_left_proj  = weights["left_proj.weight"]          # (H, D)
    w_left_gate  = weights["left_gate.weight"]          # (H, D)
    w_right_proj = weights["right_proj.weight"]         # (H, D)
    w_right_gate = weights["right_gate.weight"]         # (H, D)

    # shape → (4*H, D)
    stacked_w = torch.cat([w_left_proj, w_left_gate,
                           w_right_proj, w_right_gate], dim=0)
    stacked_w = stacked_w.to(device, torch.float16).contiguous()

    # One huge linear projection – result shape [B, L, L, 4*H]
    combined = F.linear(x_norm_h, stacked_w)  # bias is None

    # Split the concatenated tensor
    H = hidden_dim
    left_proj  = combined[..., :H]                       # (B, L, L, H)
    left_gate  = torch.sigmoid(combined[..., H:2*H])    # (B, L, L, H)
    right_proj = combined[..., 2*H:3*H]                 # (B, L, L, H)
    right_gate = torch.sigmoid(combined[..., 3*H:])    # (B, L, L, H)

    # Apply gating (and optional mask)
    left = left_proj * left_gate
    right = right_proj * right_gate

    if (mask is not None) and (not nomask):
        # mask shape [B, L, L] → [B, L, L, 1] in fp16
        mask_h = mask.to(device, torch.float16).unsqueeze(-1)
        left = left * mask_h
        right = right * mask_h

    # --------------------------------------------------------------
    # 4️⃣ Output‑gate (separate linear + sigmoid)
    # --------------------------------------------------------------
    out_gate_w = weights["out_gate.weight"]          # (H, D)
    out_gate_w = out_gate_w.to(device, torch.float16).contiguous()
    out_gate = torch.sigmoid(F.linear(x_norm_h, out_gate_w))  # (B, L, L, H)

    # --------------------------------------------------------------
    # 5️⃣ N³ contraction – batched GEMM
    #     out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    # --------------------------------------------------------------
    # Bring the hidden dimension to a batch axis for cuBLAS
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, L, L)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # (B, H, L, L)

    # Merge batch & head → (B*H, L, L)
    left_mat  = left_perm.view(bs * H, L, L)
    right_mat = right_perm.view(bs * H, L, L)

    # Batched matrix multiplication (FP16, Tensor‑Core)
    out_mat = torch.matmul(left_mat, right_mat.transpose(-2, -1))  # (B*H, L, L)

    # Restore original layout → (B, L, L, H)
    out_h = out_mat.view(bs, H, L, L).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 6️⃣ Hidden‑dim LayerNorm (Triton)
    # --------------------------------------------------------------
    out_fp32 = out_h.to(torch.float32)                     # LN works in FP32
    out_flat = out_fp32.view(bs * L * L, hidden_dim).contiguous()
    to_out_norm_w = weights["to_out_norm.weight"].to(device, dtype).contiguous()
    to_out_norm_b = weights["to_out_norm.bias"].to(device, dtype).contiguous()
    out_norm_flat = triton_layernorm(out_flat, to_out_norm_w, to_out_norm_b, eps=eps)
    out_norm = out_norm_flat.view(bs, L, L, hidden_dim)

    # --------------------------------------------------------------
    # 7️⃣ Apply output‑gate and final projection back to D
    # --------------------------------------------------------------
    out_norm = out_norm * out_gate.to(dtype)               # broadcast over H
    to_out_w = weights["to_out.weight"].to(device, dtype).contiguous()
    out = F.linear(out_norm, to_out_w)                     # (B, L, L, D), FP32

    return out
