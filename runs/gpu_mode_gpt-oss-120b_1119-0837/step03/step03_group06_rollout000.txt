"""
TriMul (outgoing) – highly‑fused Triton implementation.

The core operation is a batched matrix multiply over the third dimension:
    out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]

We:
  1) Apply the first LayerNorm (FP32 → FP16) and all linear projections /
     sigmoid gates in FP16 with torch.nn.functional.
  2) Pack the hidden dimension together with the batch dimension (C = B*H)
     and reshape the tensors to (C, M, K) and (C, K, N) – M = N = seq_len.
  3) Launch a Triton kernel that computes a batched GEMM for each C slice.
     The kernel uses FP16 inputs, FP32 accumulation and writes the result
     back in FP16.
  4) Unpack the result, apply the second LayerNorm, out‑gate and final
     linear projection, and return FP32 output.

The Triton kernel is deliberately simple (tiling + dot) but uses
block‑size autotuning based on the sequence length and a modest number of
warps to keep occupancy high on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    out_ptr,                # fp16*  output   (C, M, N)
    a_ptr,                  # fp16*  left     (C, M, K)
    b_ptr,                  # fp16*  right    (C, K, N)
    M, N, K,                # matrix dimensions (all = seq_len)
    stride_out_m, stride_out_n, stride_out_c,   # output strides
    stride_a_m, stride_a_k, stride_a_c,        # left  strides
    stride_b_k, stride_b_n, stride_b_c,        # right strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM: for each channel `c` (packed batch*hidden) compute
    out_c[M,N] += left_c[M,K] @ right_c[K,N].

    All inputs are fp16, accumulation is fp32.
    """
    pid_m = tl.program_id(0)   # block row
    pid_n = tl.program_id(1)   # block col
    pid_c = tl.program_id(2)   # channel (batch*hidden)

    # -----------------------------------------------------------------
    # Tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -----------------------------------------------------------------
    # Loop over K dimension
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Pointers to current tiles
        a_tile = a_ptr + pid_c * stride_a_c \
                 + offs_m[:, None] * stride_a_m \
                 + offs_k[None, :] * stride_a_k   # (M, K)

        b_tile = b_ptr + pid_c * stride_b_c \
                 + offs_k[:, None] * stride_b_k \
                 + offs_n[None, :] * stride_b_n   # (K, N)

        # Load with masks – out‑of‑bounds elements are zeroed
        a = tl.load(a_tile,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)                       # fp16
        b = tl.load(b_tile,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)                       # fp16

        # Accumulate: fp16·fp16 → fp32
        acc += tl.dot(a, b)

    # -----------------------------------------------------------------
    # Write out tile (convert back to fp16)
    out_tile = out_ptr + pid_c * stride_out_c \
               + offs_m[:, None] * stride_out_m \
               + offs_n[None, :] * stride_out_n

    tl.store(out_tile, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module (AlphaFold3 style).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor[B, N, N, C] (float32)
            - mask         : torch.Tensor[B, N, N] (bool/float) or None
            - weights      : dict of model parameters (see keys below)
            - config       : dict with entries
                * "dim"        – input channel dimension C
                * "hidden_dim" – hidden dimension H
                * "nomask"     – bool, True → ignore mask

    Returns:
        torch.Tensor[B, N, N, C] (float32) – module output
    """
    # -----------------------------------------------------------------
    # 0) Unpack
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    device = input_tensor.device

    # -----------------------------------------------------------------
    # 1) Input LayerNorm (FP32 -> FP16)
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )                                   # [B, N, N, C]  fp32
    x = x.to(torch.float16)            # cast to fp16 for the rest

    # -----------------------------------------------------------------
    # 2) Linear projections (bias=False) – fp16
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left = F.linear(x, left_proj_w)    # [B, N, N, H] fp16
    right = F.linear(x, right_proj_w)  # [B, N, N, H] fp16

    # -----------------------------------------------------------------
    # 3) Gates (sigmoid) – fp16
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))   # [B,N,N,H] fp16
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    # -----------------------------------------------------------------
    # 4) Apply gates (and optional mask)
    left = left * left_gate
    right = right * right_gate

    if not nomask and mask is not None:
        # mask is 0/1, broadcast over hidden dim
        mask_h = mask.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_h
        right = right * mask_h

    # -----------------------------------------------------------------
    # 5) Prepare tensors for the batched GEMM.
    #    left: (B, H, M, K)   – M=i, K=k
    #    right: (B, H, K, N)  – K=k, N=j (note the transposition)
    B, N, _, _ = left.shape
    left = left.permute(0, 3, 1, 2).contiguous()          # [B, H, M, K]
    right = right.permute(0, 3, 2, 1).contiguous()        # [B, H, K, N]

    C_total = B * hidden_dim                               # packed batch*hidden

    # reshape to (C, M, K) and (C, K, N)
    left_flat = left.reshape(C_total, N, N)                # (C, M, K)
    right_flat = right.reshape(C_total, N, N)              # (C, K, N)

    # Output buffer for GEMM result (fp16)
    out_flat = torch.empty_like(left_flat, dtype=torch.float16, device=device)

    # -----------------------------------------------------------------
    # 6) Choose tiling parameters based on sequence length.
    if N <= 256:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 32
    elif N <= 512:
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 64
    else:  # up to 1024
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 64

    # Strides in elements (not bytes)
    stride_out_c = out_flat.stride(0)
    stride_out_m = out_flat.stride(1)
    stride_out_n = out_flat.stride(2)

    stride_a_c = left_flat.stride(0)
    stride_a_m = left_flat.stride(1)
    stride_a_k = left_flat.stride(2)

    stride_b_c = right_flat.stride(0)
    stride_b_k = right_flat.stride(1)   # K dimension after transpose
    stride_b_n = right_flat.stride(2)   # N dimension after transpose

    # Grid: (M‑blocks, N‑blocks, C‑slices)
    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        C_total,
    )

    # Number of warps – a bit more for larger N to keep occupancy high
    num_warps = 8 if N >= 512 else 4

    # -----------------------------------------------------------------
    # 7) Launch the Triton kernel
    tri_mul_kernel[grid](
        out_flat,
        left_flat,
        right_flat,
        N, N, N,                     # M, N, K
        stride_out_m, stride_out_n, stride_out_c,
        stride_a_m, stride_a_k, stride_a_c,
        stride_b_k, stride_b_n, stride_b_c,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=num_warps,
    )

    # -----------------------------------------------------------------
    # 8) Reshape back to (B, N, N, H) – still fp16
    out = out_flat.reshape(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 9) Post‑processing: second LayerNorm, out‑gate and final linear.
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)

    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )                                   # fp16

    out = out * out_gate                # apply out‑gate (fp16)

    to_out_w = weights["to_out.weight"].to(torch.float16)
    out = F.linear(out, to_out_w)       # (B,N,N,dim) fp16

    # -----------------------------------------------------------------
    # 10) Cast back to float32 for the expected return type
    return out.to(torch.float32)
