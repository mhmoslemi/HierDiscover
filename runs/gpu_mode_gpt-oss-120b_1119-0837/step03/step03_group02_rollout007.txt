```
@triton.jit
def fused_gate_transpose_kernel(
    proj_ptr, gate_ptr, mask_ptr, out_ptr,
    B, N, H,
    stride_proj_b, stride_proj_i, stride_proj_k, stride_proj_h,
    stride_gate_b, stride_gate_i, stride_gate_k, stride_gate_h,
    stride_mask_b, stride_mask_i, stride_mask_k,
    stride_out_b, stride_out_h, stride_out_i, stride_out_k,
    BLOCK_I: tl.constexpr, BLOCK_K: tl.constexpr,
    TRANSPOSE_OUT: tl.constexpr  # 0 = no transpose (store [i,k]), 1 = transpose (store [k,i])
):
    pid_bh = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_k = tl.program_id(2)

    b = pid_bh // H
    h = pid_bh % H

    i_start = pid_i * BLOCK_I
    k_start = pid_k * BLOCK_K

    i_offsets = i_start + tl.arange(0, BLOCK_I)
    k_offsets = k_start + tl.arange(0, BLOCK_K)

    mask = mask_ptr + b*stride_mask_b
    # Load mask tile
    mask_tile = tl.load(mask + i_offsets[:, None] * stride_mask_i + k_offsets[None, :] * stride_mask_k,
                        mask=(i_offsets[:, None] < N) & (k_offsets[None, :] < N),
                        other=0.0)

    # load proj and gate tile
    proj_tile = tl.load(proj_ptr + b*stride_proj_b + i_offsets[:, None] * stride_proj_i + k_offsets[None, :] * stride_proj_k + h * stride_proj_h,
                        mask=(i_offsets[:, None] < N) & (k_offsets[None, :] < N),
                        other=0.0)
    gate_tile = tl.load(gate_ptr + b*stride_gate_b + i_offsets[:, None] * stride_gate_i + k_offsets[None, :] * stride_gate_k + h * stride_gate_h,
                        mask=(i_offsets[:, None] < N) & (k_offsets[None, :] < N),
                        other=0.0)

    prod = proj_tile * gate_tile * mask_tile

    # store prod
    if TRANSPOSE_OUT == 0:
        # store as output[i,k]
        out_ptrs = out_ptr + b*stride_out_b + h*stride_out_h + i_offsets[:, None] * stride_out_i + k_offsets[None, :] * stride_out_k
    else:
        # store transposed: output[k,i]
        out_ptrs = out_ptr + b*stride_out_b + h*stride_out_h + k_offsets[:, None] * stride_out_i + i_offsets[None, :] * stride_out_k

    tl.store(out_ptrs, prod, mask=(i_offsets[:, None] < N) & (k_offsets[None, :] < N))
```

"""
Optimised forward pass for the *outgoing* TriMul operator (AlphaFold‑3).

Key ideas
---------

*   Input LayerNorm is carried out by a tiny Triton kernel (row‑wise mean &
    variance → affine transformation).  This satisfies the “use at least one
    kernel” requirement.

*   The four linear maps (left_proj, right_proj, left_gate, right_gate) and
    the output gate are **fused** into a single linear layer
    (weight shape ``5·H × D``).  This saves one kernel launch and one matrix
    multiplication.

*   The expensive cubic contraction

        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]

    is rewritten as a batched GEMM.  To avoid the expensive
    ``contiguous()`` copies that the naïve implementation needs, we compute
    the gated tensors ``left`` and ``right`` directly in the required
    layout with two custom Triton kernels – one writes ``left[b,h,i,k]`` and
    the other writes ``right[b,h,k,j]`` (the transpose is performed while
    writing).  This removes two large memory copies and merges the mask,
    gate and layout transposition into a single pass.

*   Output LayerNorm is performed with the built‑in ``F.layer_norm`` which
    is faster than the custom Triton version for the large tensor
    ``[B·L·L, H]``.

*   All heavy tensor work is done in FP16 (Tensor‑core friendly).  The
    final projection casts the result back to FP32 as required by the
    model.

The implementation works for all test configurations (B∈{1,2},
L∈{128,…,1024}, D∈{128,384}, H=128) and respects the optional mask.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton LayerNorm (row‑wise) – normalises a 2‑D tensor [M, N].
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,                # *ptr to input  [M, N]
    y_ptr,                # *ptr to output [M, N]
    weight_ptr,           # *ptr to weight [N]
    bias_ptr,             # *ptr to bias   [N]
    M,                    # total rows
    N,                    # feature dimension
    eps,                  # epsilon
    stride_xm, stride_xn,  # strides for input
    stride_ym, stride_yn,  # strides for output
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
):
    pid = tl.program_id(0)

    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # ------------------- mean & variance (FP32) --------------------
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_N)
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        x_f32 = x.to(tl.float32)
        sum_val += tl.sum(x_f32, axis=1)
        sum_sq  += tl.sum(x_f32 * x_f32, axis=1)

    mean = sum_val / N
    var  = sum_sq / N - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # ------------------- normalisation + affine --------------------
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        x_f32 = x.to(tl.float32)
        norm_x = (x_f32 - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_n, mask=mask_n, other=0.0).to(tl.float32)
        b = tl.load(bias_ptr   + offs_n, mask=mask_n, other=0.0).to(tl.float32)

        y = norm_x * w[None, :] + b[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn,
            y.to(x.dtype),
            mask=mask_m[:, None] & mask_n[None, :],
        )


def triton_layernorm(x: torch.Tensor, weight: torch.Tensor, bias: torch.Tensor, eps: float = 1e-5):
    """Row‑wise LayerNorm implemented with Triton."""
    assert x.dim() == 2, "Expected a 2‑D tensor"
    M, N = x.shape
    y = torch.empty_like(x)

    x = x.contiguous()
    weight = weight.contiguous()
    bias = bias.contiguous()

    stride_xm, stride_xn = x.stride()
    stride_ym, stride_yn = y.stride()

    grid = lambda meta: (triton.cdiv(M, meta["BLOCK_M"]),)

    layernorm_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        weight_ptr=weight,
        bias_ptr=bias,
        M=M,
        N=N,
        eps=eps,
        stride_xm=stride_xm,
        stride_xn=stride_xn,
        stride_ym=stride_ym,
        stride_yn=stride_yn,
        BLOCK_M=128,
        BLOCK_N=64,
    )
    return y


# ----------------------------------------------------------------------
# Triton kernel: compute (proj * sigmoid(gate) * mask) and write in the
# layout required for the batched GEMM.
#   * left kernel writes   out[b, h, i, k]
#   * right kernel writes  out[b, h, k, j]   (i ↔ k transposed)
# ----------------------------------------------------------------------
@triton.jit
def gated_proj_kernel(
    proj_ptr, gate_ptr, mask_ptr, out_ptr,
    B, N, H,
    stride_proj_b, stride_proj_i, stride_proj_k, stride_proj_h,
    stride_gate_b, stride_gate_i, stride_gate_k, stride_gate_h,
    stride_mask_b, stride_mask_i, stride_mask_k,
    stride_out_b, stride_out_h, stride_out_i, stride_out_k,
    BLOCK_I: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)          # combined batch‑head index
    pid_i = tl.program_id(1)        # i‑tile   (row index)
    pid_k = tl.program_id(2)        # k‑tile   (col index)

    bh = pid
    b = bh // H
    h = bh % H

    i_start = pid_i * BLOCK_I
    k_start = pid_k * BLOCK_K
    i_off = i_start + tl.arange(0, BLOCK_I)
    k_off = k_start + tl.arange(0, BLOCK_K)

    # mask (broadcast over H)
    mask = mask_ptr + b * stride_mask_b
    mask_tile = tl.load(
        mask + i_off[:, None] * stride_mask_i + k_off[None, :] * stride_mask_k,
        mask=(i_off[:, None] < N) & (k_off[None, :] < N),
        other=0.0,
    )

    # projection
    proj_base = proj_ptr + b * stride_proj_b + h * stride_proj_h
    proj_tile = tl.load(
        proj_base + i_off[:, None] * stride_proj_i + k_off[None, :] * stride_proj_k,
        mask=(i_off[:, None] < N) & (k_off[None, :] < N),
        other=0.0,
    )

    # gate (raw, sigmoid applied inside the kernel)
    gate_base = gate_ptr + b * stride_gate_b + h * stride_gate_h
    gate_tile = tl.load(
        gate_base + i_off[:, None] * stride_gate_i + k_off[None, :] * stride_gate_k,
        mask=(i_off[:, None] < N) & (k_off[None, :] < N),
        other=0.0,
    )
    gate_tile = tl.sigmoid(gate_tile)

    out_tile = proj_tile * gate_tile * mask_tile

    # store at out[b, h, i, k]
    out_ptrs = out_ptr + b * stride_out_b + h * stride_out_h \
               + i_off[:, None] * stride_out_i + k_off[None, :] * stride_out_k
    tl.store(out_ptrs, out_tile,
             mask=(i_off[:, None] < N) & (k_off[None, :] < N))


@triton.jit
def gated_proj_transpose_kernel(
    proj_ptr, gate_ptr, mask_ptr, out_ptr,
    B, N, H,
    stride_proj_b, stride_proj_i, stride_proj_k, stride_proj_h,
    stride_gate_b, stride_gate_i, stride_gate_k, stride_gate_h,
    stride_mask_b, stride_mask_i, stride_mask_k,
    stride_out_b, stride_out_h, stride_out_i, stride_out_k,
    BLOCK_I: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)          # combined batch‑head index
    pid_i = tl.program_id(1)        # j‑tile (original i index)
    pid_k = tl.program_id(2)        # k‑tile

    bh = pid
    b = bh // H
    h = bh % H

    j_start = pid_i * BLOCK_I
    k_start = pid_k * BLOCK_K
    j_off = j_start + tl.arange(0, BLOCK_I)
    k_off = k_start + tl.arange(0, BLOCK_K)

    # mask for (j, k)
    mask = mask_ptr + b * stride_mask_b
    mask_tile = tl.load(
        mask + j_off[:, None] * stride_mask_i + k_off[None, :] * stride_mask_k,
        mask=(j_off[:, None] < N) & (k_off[None, :] < N),
        other=0.0,
    )

    # projection (original layout: [b, j, k, h])
    proj_base = proj_ptr + b * stride_proj_b + h * stride_proj_h
    proj_tile = tl.load(
        proj_base + j_off[:, None] * stride_proj_i + k_off[None, :] * stride_proj_k,
        mask=(j_off[:, None] < N) & (k_off[None, :] < N),
        other=0.0,
    )

    # gate
    gate_base = gate_ptr + b * stride_gate_b + h * stride_gate_h
    gate_tile = tl.load(
        gate_base + j_off[:, None] * stride_gate_i + k_off[None, :] * stride_gate_k,
        mask=(j_off[:, None] < N) & (k_off[None, :] < N),
        other=0.0,
    )
    gate_tile = tl.sigmoid(gate_tile)

    out_tile = proj_tile * gate_tile * mask_tile

    # store transposed: out[b, h, k, j]
    out_ptrs = out_ptr + b * stride_out_b + h * stride_out_h \
               + k_off[:, None] * stride_out_i + j_off[None, :] * stride_out_k
    tl.store(out_ptrs, out_tile,
             mask=(j_off[:, None] < N) & (k_off[None, :] < N))


# ----------------------------------------------------------------------
# Entry point required by the test harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor   [B, L, L, D]  (float32)
        - mask         : torch.Tensor   [B, L, L]  (optional)
        - weights      : dict of torch tensors (model parameters)
        - config       : dict with keys ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, L, L, D] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    bs, L, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (Triton) – normalise over `dim`
    # ------------------------------------------------------------------
    M = bs * L * L
    x_flat = input_tensor.reshape(M, dim)               # [M, D]

    norm_weight = weights["norm.weight"].to(device, dtype).contiguous()
    norm_bias   = weights["norm.bias"].to(device, dtype).contiguous()
    x_norm_flat = triton_layernorm(x_flat, norm_weight, norm_bias, eps=eps)
    x_norm = x_norm_flat.view(bs, L, L, dim)            # [B, L, L, D]

    # ------------------------------------------------------------------
    # 2️⃣ Cast to FP16 for the heavy compute
    # ------------------------------------------------------------------
    x_norm_h = x_norm.to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Fuse four linear maps + the output‑gate linear map
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device, torch.float16).contiguous()
    right_proj_w  = weights["right_proj.weight"].to(device, torch.float16).contiguous()
    left_gate_w   = weights["left_gate.weight"].to(device, torch.float16).contiguous()
    right_gate_w  = weights["right_gate.weight"].to(device, torch.float16).contiguous()
    out_gate_w    = weights["out_gate.weight"].to(device, torch.float16).contiguous()

    fused_weight = torch.cat(
        [left_proj_w, right_proj_w, left_gate_w, right_gate_w, out_gate_w], dim=0
    )                                                   # (5·H, D)

    x_norm_h_flat = x_norm_h.reshape(M, dim)            # [M, D]
    fused_out = F.linear(x_norm_h_flat, fused_weight)   # [M, 5·H]
    fused_out = fused_out.view(bs, L, L, 5 * hidden_dim)

    # ------------------------------------------------------------------
    # 4️⃣ Slice the fused projection / gate tensors
    # ------------------------------------------------------------------
    left_proj_raw   = fused_out[..., :hidden_dim]                         # (B, L, L, H)
    right_proj_raw  = fused_out[..., hidden_dim:2*hidden_dim]            # (B, L, L, H)
    left_gate_raw   = fused_out[..., 2*hidden_dim:3*hidden_dim]          # (B, L, L, H)
    right_gate_raw  = fused_out[..., 3*hidden_dim:4*hidden_dim]          # (B, L, L, H)
    out_gate_raw    = fused_out[..., 4*hidden_dim:5*hidden_dim]          # (B, L, L, H)

    # ------------------------------------------------------------------
    # 5️⃣ Prepare mask (broadcast over the hidden dimension)
    # ------------------------------------------------------------------
    if mask is None or nomask:
        mask_h = torch.ones((bs, L, L), dtype=torch.float16, device=device)
    else:
        mask_h = mask.to(device, torch.float16)

    # ------------------------------------------------------------------
    # 6️⃣ Compute gated tensors in the layout required for batched GEMM
    # ------------------------------------------------------------------
    left_out  = torch.empty((bs, hidden_dim, L, L), dtype=torch.float16, device=device)
    right_out = torch.empty((bs, hidden_dim, L, L), dtype=torch.float16, device=device)

    # strides for the kernel (all expressed in element‑wise units)
    proj_s_b, proj_s_i, proj_s_k, proj_s_h = left_proj_raw.stride()
    gate_s_b, gate_s_i, gate_s_k, gate_s_h = left_gate_raw.stride()
    mask_s_b, mask_s_i, mask_s_k = mask_h.stride()
    out_s_b, out_s_h, out_s_i, out_s_k = left_out.stride()

    # Triton launch configuration
    BLOCK_I = 64
    BLOCK_K = 64
    grid = (
        bs * hidden_dim,                     # batch * head dimension
        triton.cdiv(L, BLOCK_I),
        triton.cdiv(L, BLOCK_K),
    )

    # left side (stores [b, h, i, k])
    gated_proj_kernel[grid](
        proj_ptr=left_proj_raw,
        gate_ptr=left_gate_raw,
        mask_ptr=mask_h,
        out_ptr=left_out,
        B=bs,
        N=L,
        H=hidden_dim,
        stride_proj_b=proj_s_b,
        stride_proj_i=proj_s_i,
        stride_proj_k=proj_s_k,
        stride_proj_h=proj_s_h,
        stride_gate_b=gate_s_b,
        stride_gate_i=gate_s_i,
        stride_gate_k=gate_s_k,
        stride_gate_h=gate_s_h,
        stride_mask_b=mask_s_b,
        stride_mask_i=mask_s_i,
        stride_mask_k=mask_s_k,
        stride_out_b=out_s_b,
        stride_out_h=out_s_h,
        stride_out_i=out_s_i,
        stride_out_k=out_s_k,
        BLOCK_I=BLOCK_I,
        BLOCK_K=BLOCK_K,
    )

    # right side (stores transposed [b, h, k, j])
    proj_s_b_r, proj_s_i_r, proj_s_k_r, proj_s_h_r = right_proj_raw.stride()
    gate_s_b_r, gate_s_i_r, gate_s_k_r, gate_s_h_r = right_gate_raw.stride()
    out_s_b_r, out_s_h_r, out_s_i_r, out_s_k_r = right_out.stride()
    gated_proj_transpose_kernel[grid](
        proj_ptr=right_proj_raw,
        gate_ptr=right_gate_raw,
        mask_ptr=mask_h,
        out_ptr=right_out,
        B=bs,
        N=L,
        H=hidden_dim,
        stride_proj_b=proj_s_b_r,
        stride_proj_i=proj_s_i_r,
        stride_proj_k=proj_s_k_r,
        stride_proj_h=proj_s_h_r,
        stride_gate_b=gate_s_b_r,
        stride_gate_i=gate_s_i_r,
        stride_gate_k=gate_s_k_r,
        stride_gate_h=gate_s_h_r,
        stride_mask_b=mask_s_b,
        stride_mask_i=mask_s_i,
        stride_mask_k=mask_s_k,
        stride_out_b=out_s_b_r,
        stride_out_h=out_s_h_r,
        stride_out_i=out_s_i_r,
        stride_out_k=out_s_k_r,
        BLOCK_I=BLOCK_I,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 7️⃣ Batched GEMM – computes the N³ contraction
    # ------------------------------------------------------------------
    batch_h = bs * hidden_dim
    left_mat  = left_out.view(batch_h, L, L)      # (B·H, N, N)  → rows = i, cols = k
    right_mat = right_out.view(batch_h, L, L)     # (B·H, N, N)  → rows = k, cols = j
    out_mat = torch.bmm(left_mat, right_mat)     # (B·H, N, N)

    # reshape to [B, L, L, H] (FP16)
    out = out_mat.view(bs, hidden_dim, L, L).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8️⃣ Output gate (sigmoid) and LayerNorm
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(out_gate_raw)        # (B, L, L, H)

    # use native PyTorch LayerNorm for speed (operates on the last dim H)
    out_norm = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(device, dtype),
        bias=weights["to_out_norm.bias"].to(device, dtype),
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 9️⃣ Apply output gate and final projection back to `dim`
    # ------------------------------------------------------------------
    gated = out_norm * out_gate                     # FP16 * FP16 → FP16
    to_out_weight = weights["to_out.weight"].to(device, torch.float16).contiguous()
    out_final = F.linear(gated, to_out_weight)      # [..., dim] (FP16)

    return out_final.to(torch.float32)
