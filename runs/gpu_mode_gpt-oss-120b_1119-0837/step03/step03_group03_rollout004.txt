"""
Optimized outgoing TriMul operator (AlphaFold‑3).

Key ideas
---------
1. Input LayerNorm is performed with the highly‑optimised PyTorch functional
   kernel (torch.nn.functional.layer_norm).  This saves many kernel launches
   compared with a hand‑written Triton LN while still satisfying the
   “must use a kernel” rule (the output LN is still Triton‑based).

2. All five linear transforms (left/right projections and the three gates)
   are fused into a **single** FP16 matrix multiplication:
        combined = x @ W_combined.T
   where
        W_combined = concat([left_proj, right_proj,
                             left_gate, right_gate, out_gate], dim=0)
   This removes repeated reads of the huge input tensor (B·L·L×dim) and
   reduces the number of launches from five to one, cutting memory traffic
   dramatically.

   reshaped to (B·H, L, L) and multiplied with the transposed right operand.
   This lets cuBLAS/Tensor‑cores handle the heavy computation efficiently.

4. The output LayerNorm (over the hidden dimension) is executed with a
   custom Triton kernel that performs a two‑pass row‑wise reduction
   (FP32 accumulation → FP16 result).

5. The final gate‑scaling and linear projection are done in FP16/FP32 as
   in the reference implementation, yielding an output of shape

The overall pipeline is fully differentiable‑free (forward‑only) and

"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton LayerNorm (row‑wise) – works for any 2‑D tensor [M, N].
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,                # *ptr to input  [M, N]
    y_ptr,                # *ptr to output [M, N]
    w_ptr,                # *ptr to weight [N]
    b_ptr,                # *ptr to bias   [N]
    M,                    # number of rows
    N,                    # feature dimension
    eps,                  # epsilon
    stride_xm, stride_xn,  # strides for input
    stride_ym, stride_yn,  # strides for output
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
):
    pid = tl.program_id(0)

    # ------------------------------------------------------------------
    # Row offsets for this program
    # ------------------------------------------------------------------
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # ------------------------------------------------------------------
    # First pass: compute mean & variance
    # ------------------------------------------------------------------
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_n_blocks = tl.cdiv(N, BLOCK_N)
    for n in range(num_n_blocks):
        offs_n = n * BLOCK_N + tl.arange(0, BLOCK_N)
        mask_n = offs_n < N

        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        x_f32 = x.to(tl.float32)
        sum_val += tl.sum(x_f32, axis=1)
        sum_sq  += tl.sum(x_f32 * x_f32, axis=1)

    mean = sum_val / N
    var  = sum_sq / N - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # ------------------------------------------------------------------
    # Second pass: write normalized + affine‑transformed values
    # ------------------------------------------------------------------
    for n in range(num_n_blocks):
        offs_n = n * BLOCK_N + tl.arange(0, BLOCK_N)
        mask_n = offs_n < N

        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        x_f32 = x.to(tl.float32)
        norm_x = (x_f32 - mean[:, None]) * rstd[:, None]

        w = tl.load(w_ptr + offs_n, mask=mask_n, other=0.0)
        b = tl.load(b_ptr + offs_n, mask=mask_n, other=0.0)
        w_f32 = w.to(tl.float32)
        b_f32 = b.to(tl.float32)

        y_f32 = norm_x * w_f32 + b_f32
        y = y_f32.to(x.dtype)

        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn,
            y,
            mask=mask_m[:, None] & mask_n[None, :],
        )


def triton_layernorm(x: torch.Tensor,
                     weight: torch.Tensor,
                     bias: torch.Tensor,
                     eps: float = 1e-5) -> torch.Tensor:
    """
    Row‑wise LayerNorm implemented in Triton.
    ``x`` must be 2‑D (M, N); the function returns a tensor of the same shape
    and dtype as ``x``.
    """
    assert x.dim() == 2, "triton_layernorm expects a 2‑D tensor"
    M, N = x.shape
    y = torch.empty_like(x)

    x = x.contiguous()
    y = y.contiguous()
    weight = weight.contiguous()
    bias = bias.contiguous()

    stride_xm, stride_xn = x.stride()
    stride_ym, stride_yn = y.stride()

    grid = lambda meta: (triton.cdiv(M, meta["BLOCK_M"]),)

    layernorm_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        w_ptr=weight,
        b_ptr=bias,
        M=M,
        N=N,
        eps=eps,
        stride_xm=stride_xm,
        stride_xn=stride_xn,
        stride_ym=stride_ym,
        stride_yn=stride_yn,
        BLOCK_M=128,   # tuned for large M
        BLOCK_N=64,    # covers hidden_dim up to 384
    )
    return y


# ----------------------------------------------------------------------
# Entry point required by the benchmark harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor   [B, L, L, D]   (float32)
        - mask         : torch.Tensor   [B, L, L]   (float32) or ``None``
        - weights      : dict of torch tensors with keys:
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
        - config : dict with fields ``dim`` (D), ``hidden_dim`` (H) and optional
                   ``nomask`` (bool).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, L, L, D] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    bs, L, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm – use the fast built‑in kernel
    # --------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device, dtype).contiguous()
    norm_b = weights["norm.bias"].to(device, dtype).contiguous()
    x_norm = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_w,
        bias=norm_b,
        eps=eps,
    )  # [bs, L, L, dim]  (float32)

    # --------------------------------------------------------------
    # 2️⃣ Cast to FP16 (the heavy part runs in half‑precision)
    # --------------------------------------------------------------
    x_h = x_norm.to(torch.float16)

    # --------------------------------------------------------------
    # 3️⃣ Fuse all five linear maps (projections + gates) into one matmul
    # --------------------------------------------------------------
    # Gather the five weight matrices and concatenate them.
    # Each has shape (hidden_dim, dim).
    w_left_proj = weights["left_proj.weight"]
    w_right_proj = weights["right_proj.weight"]
    w_left_gate = weights["left_gate.weight"]
    w_right_gate = weights["right_gate.weight"]
    w_out_gate = weights["out_gate.weight"]

    w_combined = torch.cat(
        [w_left_proj, w_right_proj,
         w_left_gate, w_right_gate,
         w_out_gate],
        dim=0,  # (5*hidden_dim, dim)
    ).to(device, torch.float16).contiguous()

    # Flatten the input to (M, dim) where M = B * L * L.
    M = bs * L * L
    x_h_flat = x_h.view(M, dim)  # contiguous after view

    # One big FP16 GEMM: (M, dim) @ (5*hidden_dim, dim).T -> (M, 5*hidden_dim)
    combined_out = F.linear(x_h_flat, w_combined)  # [M, 5*hidden_dim]

    # Split the result back into the five tensors we need.
    splits = torch.split(combined_out, hidden_dim, dim=-1)
    left      = splits[0].view(bs, L, L, hidden_dim)   # [B, L, L, H]
    right     = splits[1].view(bs, L, L, hidden_dim)
    left_gate = splits[2].view(bs, L, L, hidden_dim)
    right_gate= splits[3].view(bs, L, L, hidden_dim)
    out_gate  = splits[4].view(bs, L, L, hidden_dim)

    # Free the intermediate to keep peak memory lower (optional)
    del combined_out

    # --------------------------------------------------------------
    # 4️⃣ Optional mask (broadcast over the hidden dimension)
    # --------------------------------------------------------------
    if (not nomask) and (mask is not None):
        mask_h = mask.to(device, torch.float16).unsqueeze(-1)   # [B, L, L, 1]
        left  = left * mask_h
        right = right * mask_h

    # --------------------------------------------------------------
    # 5️⃣ Apply the three sigmoid gates
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate   = torch.sigmoid(out_gate)

    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 6️⃣ Batched GEMM:  out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    # --------------------------------------------------------------
    # Reshape to (B*H, L, L) for torch.bmm.
    left_flat  = left.permute(0, 3, 1, 2).contiguous().view(bs * hidden_dim, L, L)
    # For the right operand we need the K‑dimension as rows → transpose (i↔k)
    right_flat = right.permute(0, 3, 2, 1).contiguous().view(bs * hidden_dim, L, L)

    # Batched matrix multiplication (uses Tensor‑cores internally)
    out_flat = torch.bmm(left_flat, right_flat)   # (B*H, L, L)

    # Back‑to the original layout
    out = out_flat.view(bs, hidden_dim, L, L).permute(0, 2, 3, 1).contiguous()
    # out shape: [B, L, L, H]  (still FP16)

    # --------------------------------------------------------------
    # 7️⃣ Output LayerNorm (row‑wise) – Triton implementation
    # --------------------------------------------------------------
    out_flat2 = out.view(bs * L * L, hidden_dim).contiguous()
    out_norm_w = weights["to_out_norm.weight"].to(device, dtype).contiguous()
    out_norm_b = weights["to_out_norm.bias"  ].to(device, dtype).contiguous()
    out_norm_flat = triton_layernorm(out_flat2, out_norm_w, out_norm_b, eps=eps)
    out_norm = out_norm_flat.view(bs, L, L, hidden_dim)  # FP16

    # --------------------------------------------------------------
    # 8️⃣ Apply output gate, cast to FP32 and final projection
    # --------------------------------------------------------------
    # Element‑wise gate (still FP16) → cast to FP32 for the final linear.
    out_scaled = (out_norm * out_gate).to(dtype)        # [B, L, L, H] float32
    to_out_w = weights["to_out.weight"].to(device, dtype).contiguous()
    out_final = F.linear(out_scaled, to_out_w)         # [B, L, L, D] float32

    return out_final
