"""
Optimized Triton‐based implementation of the “outgoing” TriMul operator
(AlphaFold‑3).  The forward pass consists of four stages:

1️⃣  Row‑wise LayerNorm on the input using a custom Triton kernel.
2️⃣  A *fused* FP16 linear projection that produces left/right tensors and,
    in a second fused pass, the three gate tensors (left, right, out).  This
    reduces the number of separate linear layers from five to two and
    eliminates most extra memory traffic over the [B, L, L, D] tensor.
3️⃣  The expensive N³ contraction
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    is carried out as a batched GEMM on the flattened tensors
    (B·H, L, L) using `torch.bmm`, which on H100 maps to a single
    Tensor‑Core kernel.
4️⃣  Row‑wise LayerNorm on the hidden dimension, out‑gate scaling, and a final
    FP32 linear projection back to the original channel size.

All kernels run on the same device as the inputs; the final output is float32.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton LayerNorm (row‑wise) – works for any 2‑D tensor [M, N].
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,                # *ptr to input  [M, N]
    y_ptr,                # *ptr to output [M, N]
    w_ptr,                # *ptr to weight [N]
    b_ptr,                # *ptr to bias   [N]
    M,                    # number of rows
    N,                    # feature dimension
    eps,                  # epsilon
    stride_xm, stride_xn,  # strides for input
    stride_ym, stride_yn,  # strides for output
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
):
    pid = tl.program_id(0)

    # ------------------------------------------------------------------
    # Row offsets for this program
    # ------------------------------------------------------------------
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # ------------------------------------------------------------------
    # First pass: compute mean & variance per row
    # ------------------------------------------------------------------
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_n_blocks = tl.cdiv(N, BLOCK_N)
    for n in range(num_n_blocks):
        offs_n = n * BLOCK_N + tl.arange(0, BLOCK_N)
        mask_n = offs_n < N

        # Load tile X (BLOCK_M x BLOCK_N)
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        x_f32 = x.to(tl.float32)
        sum_val += tl.sum(x_f32, axis=1)
        sum_sq  += tl.sum(x_f32 * x_f32, axis=1)

    mean = sum_val / N
    var  = sum_sq / N - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # ------------------------------------------------------------------
    # Second pass: write normalized + affine‑transformed values
    # ------------------------------------------------------------------
    for n in range(num_n_blocks):
        offs_n = n * BLOCK_N + tl.arange(0, BLOCK_N)
        mask_n = offs_n < N

        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        x_f32 = x.to(tl.float32)

        norm_x = (x_f32 - mean[:, None]) * rstd[:, None]

        w = tl.load(w_ptr + offs_n, mask=mask_n, other=0.0)
        b = tl.load(b_ptr + offs_n, mask=mask_n, other=0.0)
        w_f32 = w.to(tl.float32)
        b_f32 = b.to(tl.float32)

        y_f32 = norm_x * w_f32 + b_f32
        y = y_f32.to(x.dtype)

        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn,
            y,
            mask=mask_m[:, None] & mask_n[None, :],
        )


def triton_layernorm(x: torch.Tensor,
                     weight: torch.Tensor,
                     bias: torch.Tensor,
                     eps: float = 1e-5) -> torch.Tensor:
    """
    Row‑wise LayerNorm implemented in Triton.
    ``x`` must be 2‑D (M, N); the function returns a tensor of the same shape
    and dtype as ``x``.
    """
    assert x.dim() == 2, "triton_layernorm expects a 2‑D tensor"
    M, N = x.shape
    y = torch.empty_like(x)

    # Ensure contiguity (required for correct stride extraction)
    x = x.contiguous()
    y = y.contiguous()
    weight = weight.contiguous()
    bias = bias.contiguous()

    stride_xm, stride_xn = x.stride()
    stride_ym, stride_yn = y.stride()

    # Grid: one program per BLOCK_M rows
    grid = lambda meta: (triton.cdiv(M, meta["BLOCK_M"]),)

    layernorm_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        w_ptr=weight,
        b_ptr=bias,
        M=M,
        N=N,
        eps=eps,
        stride_xm=stride_xm,
        stride_xn=stride_xn,
        stride_ym=stride_ym,
        stride_yn=stride_yn,
        BLOCK_M=128,   # rows per program (tuned)
        BLOCK_N=64,    # columns per tile (covers hidden_dim up to 384)
    )
    return y


# ----------------------------------------------------------------------
# Entry point required by the benchmark harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor   [B, L, L, D]   (float32)
        - mask         : torch.Tensor   [B, L, L]     (float32) or ``None``
        - weights      : dict of torch tensors with keys:
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
        - config : dict with fields ``dim`` (D), ``hidden_dim`` (H) and optional
                   ``nomask`` (bool).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, L, L, D] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype  # expected torch.float32

    bs, L, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (Triton) – normalize over the last dimension
    # ------------------------------------------------------------------
    # reshape to (bs*L*L, dim) for row‑wise LN
    x_flat = input_tensor.reshape(bs * L * L, dim).contiguous()
    norm_w = weights["norm.weight"].to(device, dtype).contiguous()
    norm_b = weights["norm.bias"].to(device, dtype).contiguous()
    x_norm_flat = triton_layernorm(x_flat, norm_w, norm_b, eps=eps)
    x_norm = x_norm_flat.view(bs, L, L, dim)

    # ------------------------------------------------------------------
    # 2️⃣ Cast to FP16 for the heavy computation
    # ------------------------------------------------------------------
    x_h = x_norm.to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Fused linear projections and gates (all FP16, bias‑free)
    #    * left/right projection in a single matmul
    #    * three gates (left, right, out) in a second matmul
    # ------------------------------------------------------------------
    # concatenate projection weights: [left; right] -> (2*H, D)
    left_proj_w = weights["left_proj.weight"].to(device, torch.float16).contiguous()
    right_proj_w = weights["right_proj.weight"].to(device, torch.float16).contiguous()
    proj_weight = torch.cat([left_proj_w, right_proj_w], dim=0)      # (2*H, D)

    # concatenate gate weights: [left_gate; right_gate; out_gate] -> (3*H, D)
    left_gate_w = weights["left_gate.weight"].to(device, torch.float16).contiguous()
    right_gate_w = weights["right_gate.weight"].to(device, torch.float16).contiguous()
    out_gate_w = weights["out_gate.weight"].to(device, torch.float16).contiguous()
    gate_weight = torch.cat([left_gate_w, right_gate_w, out_gate_w], dim=0)  # (3*H, D)

    # ---- projection ----
    # (B, L, L, 2*H) in FP16
    proj = F.linear(x_h, proj_weight)   # bias is None by default
    proj = proj.view(bs, L, L, 2 * hidden_dim)
    left  = proj[..., :hidden_dim]      # (B, L, L, H)
    right = proj[..., hidden_dim:]      # (B, L, L, H)

    # ---- gates ----
    gates = F.linear(x_h, gate_weight)  # (B, L, L, 3*H)
    gates = gates.view(bs, L, L, 3 * hidden_dim)
    left_gate_raw  = gates[..., :hidden_dim]
    right_gate_raw = gates[..., hidden_dim:2 * hidden_dim]
    out_gate_raw   = gates[..., 2 * hidden_dim:]

    left_gate  = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)
    out_gate   = torch.sigmoid(out_gate_raw)   # remains FP16 for now

    # ------------------------------------------------------------------
    # 4️⃣ Optional mask (broadcast over hidden dimension)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_h = mask.to(device, torch.float16).unsqueeze(-1)   # (B, L, L, 1)
        left = left * mask_h
        right = right * mask_h

    # ------------------------------------------------------------------
    # 5️⃣ Apply gates to the projected tensors
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6️⃣ Batched GEMM (FP16) – computes the N³ contraction
    #    out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    # reshape to (B*H, L, L) where each batch element is an independent matmul
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(bs * hidden_dim, L, L)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(bs * hidden_dim, L, L)

    # GEMM: left_flat @ right_flatᵀ   -> (B*H, L, L)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # reshape back to (B, L, L, H)
    out = out_flat.view(bs, hidden_dim, L, L).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7️⃣ Output LayerNorm (Triton) over the hidden dimension
    # ------------------------------------------------------------------
    out_flat2 = out.view(bs * L * L, hidden_dim).contiguous()
    to_out_norm_w = weights["to_out_norm.weight"].to(device, dtype).contiguous()
    to_out_norm_b = weights["to_out_norm.bias"].to(device, dtype).contiguous()
    out_norm_flat = triton_layernorm(out_flat2, to_out_norm_w, to_out_norm_b, eps=eps)
    out_norm = out_norm_flat.view(bs, L, L, hidden_dim)

    # ------------------------------------------------------------------
    # 8️⃣ Apply the output gate and final projection back to dim (FP32)
    # ------------------------------------------------------------------
    # out_gate is FP16; promote to FP32 for the multiplication with the
    # float32 tensor produced by the layernorm.
    out = out_norm * out_gate.to(dtype)   # broadcast over hidden_dim

    to_out_w = weights["to_out.weight"].to(device, dtype).contiguous()
    out = F.linear(out, to_out_w)        # (B, L, L, D), float32

    return out
```
