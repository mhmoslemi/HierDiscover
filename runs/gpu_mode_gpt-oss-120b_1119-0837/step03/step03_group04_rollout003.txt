# TriMul outgoing Triton implementation
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr,          # FP16 [B,N,N,H] in/out
    right_ptr,         # FP16 [B,N,N,H] in/out
    left_gate_ptr,     # FP16 [B,N,N,H]
    right_gate_ptr,    # FP16 [B,N,N,H]
    mask_ptr,          # FP16 [B,N,N] flatten
    total_elements,    # int64 total #elements in left/right
    hidden_dim,        # int64
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    # mask for valid index
    valid = offs < total_elements

    # each hidden_dim entries correspond to one mask element
    mask_idx = offs // hidden_dim
    mask_val = tl.load(mask_ptr + mask_idx, mask=valid, other=tl.float16(0.0))

    # load tensors
    left = tl.load(left_ptr + offs, mask=valid, other=tl.float16(0.0))
    right = tl.load(right_ptr + offs, mask=valid, other=tl.float16(0.0))
    left_g = tl.load(left_gate_ptr + offs, mask=valid, other=tl.float16(0.0))
    right_g = tl.load(right_gate_ptr + offs, mask=valid, other=tl.float16(0.0))

    # inline fused multiplication
    left_out = left * left_g * mask_val
    right_out = right * right_g * mask_val

    # store back
    tl.store(left_ptr + offs, left_out, mask=valid)
    tl.store(right_ptr + offs, right_out, mask=valid)

def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul block.

    The implementation follows the PyTorch reference but
    (i)   computes the cheap element‑wise mask + gate fusion with a
          Triton kernel,
    (ii)  performs the O(N³) contraction with a batched GEMM (cuBLAS)
          over half‑precision tensors,
    (iii) keeps the heavy layer‑norm and final projection in
          full‑precision to preserve numerical quality.
    """
    # unpack
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    H = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", True)

    # 1. Input layer‑norm (FP32)
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )   # shape [B,N,N,dim] (float32)

    # 2. Cast to FP16 for the bulk of the computation
    x_h = x.half()

    # 3. Load FP16 weights (move to device if needed)
    def w16(key):
        return weights[key].to(device=device, dtype=torch.float16)

    left_proj_w   = w16("left_proj.weight")
    right_proj_w  = w16("right_proj.weight")
    left_gate_w   = w16("left_gate.weight")
    right_gate_w  = w16("right_gate.weight")
    out_gate_w    = w16("out_gate.weight")
    to_out_norm_w = w16("to_out_norm.weight")
    to_out_norm_b = w16("to_out_norm.bias")
    to_out_w      = w16("to_out.weight")

    # 4. Linear projections (FP16)
    left_proj = F.linear(x_h, left_proj_w)   # [B,N,N,H] FP16
    right_proj = F.linear(x_h, right_proj_w)  # [B,N,N,H] FP16

    # 5. Gate values (sigmoid) – keep them in FP16 for projection gating
    left_gate = torch.sigmoid(F.linear(x_h, left_gate_w)).half()    # [B,N,N,H]
    right_gate = torch.sigmoid(F.linear(x_h, right_gate_w)).half()  # [B,N,N,H]

    # 6. Output‑gate (used after the contraction) – keep in FP32
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))   # [B,N,N,H] float32

    # 7. Fuse mask & per‑channel gates onto the projected tensors
    if not nomask and mask is not None:
        # ensure contiguity for flattening
        left_proj = left_proj.contiguous()
        right_proj = right_proj.contiguous()
        left_gate = left_gate.contiguous()
        right_gate = right_gate.contiguous()

        mask_h = mask.to(dtype=torch.float16, device=device).contiguous()
        mask_flat = mask_h.view(-1)  # shape [B*N*N]

        total_elems = left_proj.numel()                 # B*N*N*H
        BLOCK_SIZE = 1024
        grid = (total_elems + BLOCK_SIZE - 1) // BLOCK_SIZE

        _mask_gate_fuse_kernel[grid](
            left_proj,
            right_proj,
            left_gate,
            right_gate,
            mask_flat,
            total_elems,
            H,
            BLOCK_SIZE=BLOCK_SIZE,
            num_warps=8,
        )
        left = left_proj
        right = right_proj
    else:
        # no mask – simple element‑wise multiplication
        left = left_proj * left_gate
        right = right_proj * right_gate

    # 8. Heavy O(N³) contraction with a batched GEMM (FP16 → FP16)
    #   out[b,i,j] = sum_k left[b,i,k] * right[b,j,k]   per hidden channel
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B,H,N,N)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B,H,N,N)

    batch = B * H
    left_mat = left_perm.view(batch, N, N)             # (B*H, N, N)
    right_mat = right_perm.view(batch, N, N)           # (B*H, N, N)

    # batched matrix multiply: (B*H, N, N) @ (B*H, N, N)^T
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*H, N, N)

    # reshape back to [B,N,N,H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H) FP16

    # 9. Hidden‑dim layer‑norm (FP32) + out‑gate + final projection
    out_fp32 = out.float()  # cast for a numerically stable layer‑norm

    out_norm = F.layer_norm(
        out_fp32,
        normalized_shape=(H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=eps,
    )  # (B,N,N,H) float32

    out_gated = out_norm * out_gate   # element‑wise gate (float32)

    out_final = F.linear(out_gated, to_out_w)   # (B,N,N,dim) float32

    return out_final

"""
TriMul (outgoing) – fast Triton‑augmented implementation.

Algorithm
---------
1️⃣ Layer‑norm the input in FP32 (numerical stability).
2️⃣ Cast to FP16 and apply the two linear projections (left/right) together
   with their per‑channel sigmoid gates.
3️⃣ If a mask is present, fuse the mask and the two gates onto the projected
   tensors with a small Triton kernel (single memory pass).
4️⃣ Perform the O(N³) contraction with a batched GEMM:
      out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
   This is realised as a `torch.bmm` over (B·H) matrices of size N×N.
5️⃣ Cast the result back to FP32, apply a hidden‑dim LayerNorm,
   multiply by the output gate (computed in FP32) and finish with a final
   linear projection back to the original feature dimension.
6️⃣ Return the FP32 tensor.

The heavy compute (steps 3‑4) run in half‑precision to exploit Tensor‑cores,
while the light‑weight normalisation / final projection stay in FP32 for
accuracy.  The Triton kernel is the required custom kernel.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------------------- #
# Triton kernel: fuse mask (FP16) and per‑channel gates onto the projected
# left/right tensors.  All tensors are flattened 1‑D views of shape
# (B·N·N·H).  The mask is flattened (B·N·N) and broadcast over the hidden dim.
# --------------------------------------------------------------------------- #
@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr,          # FP16 [B,N,N,H] – in‑place output
    right_ptr,         # FP16 [B,N,N,H] – in‑place output
    left_gate_ptr,     # FP16 [B,N,N,H]
    right_gate_ptr,    # FP16 [B,N,N,H]
    mask_ptr,          # FP16 [B,N,N] – flattened
    total_elements,    # i64  = B·N·N·H
    hidden_dim,        # i64  = H
    BLOCK_SIZE: tl.constexpr,
):
    pid   = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs  = start + tl.arange(0, BLOCK_SIZE)

    # Guard against out‑of‑bounds
    valid = offs < total_elements

    # One mask element per hidden_dim consecutive entries
    mask_idx = offs // hidden_dim
    mask_val = tl.load(mask_ptr + mask_idx,
                       mask=valid,
                       other=tl.float16(0.0))

    # Load tensors
    left      = tl.load(left_ptr + offs,      mask=valid, other=tl.float16(0.0))
    right     = tl.load(right_ptr + offs,     mask=valid, other=tl.float16(0.0))
    left_g    = tl.load(left_gate_ptr + offs, mask=valid, other=tl.float16(0.0))
    right_g   = tl.load(right_gate_ptr + offs,mask=valid, other=tl.float16(0.0))

    # Fuse: left = left * left_gate * mask,  right = right * right_gate * mask
    left_out  = left  * left_g  * mask_val
    right_out = right * right_g * mask_val

    # Store back in‑place
    tl.store(left_ptr  + offs, left_out,  mask=valid)
    tl.store(right_ptr + offs, right_out, mask=valid)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul block (no gradients).

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, D] (float32)
            - mask_tensor : torch.Tensor [B, N, N] (bool/float) – optional
            - weights_dict : mapping of all module parameters (float32)
            - config_dict  : {"dim": D, "hidden_dim": H, "nomask": bool, ...}

    Returns:
        torch.Tensor [B, N, N, D] (float32)
    """
    # --------------------------------------------------------------------- #
    # Unpack arguments
    # --------------------------------------------------------------------- #
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    H = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------------- #
    # Helper for weight conversion
    # --------------------------------------------------------------------- #
    def w_fp32(key):
        return weights[key].to(device=device, dtype=torch.float32)

    def w_fp16(key):
        return weights[key].to(device=device, dtype=torch.float16)

    # --------------------------------------------------------------------- #
    # 1) Input LayerNorm (FP32)
    # --------------------------------------------------------------------- #
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=w_fp32("norm.weight"),
        bias=w_fp32("norm.bias"),
        eps=eps,
    )   # [B, N, N, dim]  (float32)

    # --------------------------------------------------------------------- #
    # 2) Cast to FP16 for the bulk of the work
    # --------------------------------------------------------------------- #
    x_h = x.half()                                   # [B, N, N, dim] FP16

    # --------------------------------------------------------------------- #
    # 3) Load FP16 weights (projections & gates)
    # --------------------------------------------------------------------- #
    left_proj_w   = w_fp16("left_proj.weight")
    right_proj_w  = w_fp16("right_proj.weight")
    left_gate_w   = w_fp16("left_gate.weight")
    right_gate_w  = w_fp16("right_gate.weight")

    # FP32 weights used later (output gate, post‑norm, out‑proj)
    out_gate_w    = w_fp32("out_gate.weight")
    to_out_norm_w = w_fp32("to_out_norm.weight")
    to_out_norm_b = w_fp32("to_out_norm.bias")
    to_out_w      = w_fp32("to_out.weight")

    # --------------------------------------------------------------------- #
    # 4) Linear projections (FP16, bias‑free)
    # --------------------------------------------------------------------- #
    left_proj  = F.linear(x_h, left_proj_w)    # [B, N, N, H] FP16
    right_proj = F.linear(x_h, right_proj_w)   # [B, N, N, H] FP16

    # --------------------------------------------------------------------- #
    # 5) Gate values (sigmoid) – keep them in FP16 for projection gating
    # --------------------------------------------------------------------- #
    left_gate  = torch.sigmoid(F.linear(x_h, left_gate_w)).half()   # [B,N,N,H]
    right_gate = torch.sigmoid(F.linear(x_h, right_gate_w)).half()  # [B,N,N,H]

    # --------------------------------------------------------------------- #
    # 6) Output‑gate (used after the contraction) – keep in FP32
    # --------------------------------------------------------------------- #
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))   # [B,N,N,H] FP32

    # --------------------------------------------------------------------- #
    # 7) Fuse mask + per‑channel gates onto the projected tensors
    # --------------------------------------------------------------------- #
    if not nomask and mask is not None:
        # Ensure all tensors are contiguous for flat indexing
        left_proj  = left_proj.contiguous()
        right_proj = right_proj.contiguous()
        left_gate  = left_gate.contiguous()
        right_gate = right_gate.contiguous()

        mask_h    = mask.to(dtype=torch.float16, device=device).contiguous()
        mask_flat = mask_h.view(-1)          # [B*N*N] FP16

        total_elems = left_proj.numel()                # B*N*N*H
        BLOCK_SIZE = 1024
        grid = (total_elems + BLOCK_SIZE - 1) // BLOCK_SIZE

        _mask_gate_fuse_kernel[grid](
            left_proj,
            right_proj,
            left_gate,
            right_gate,
            mask_flat,
            total_elems,
            H,
            BLOCK_SIZE=BLOCK_SIZE,
            num_warps=8,
        )
        left, right = left_proj, right_proj
    else:
        # No mask: simple element‑wise multiplication
        left  = left_proj * left_gate
        right = right_proj * right_gate

    # --------------------------------------------------------------------- #
    # 8) Heavy O(N³) contraction via a batched GEMM (FP16 → FP16)
    #    out = Σₖ left[...,i,k,:] * right[...,j,k,:]
    # --------------------------------------------------------------------- #
    # (B,N,N,H) -> (B,H,N,N) for both tensors
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N) (k dimension transposed)

    batch = B * H
    left_mat  = left_perm.view(batch, N, N)          # (B*H, N, N)
    right_mat = right_perm.view(batch, N, N)         # (B*H, N, N)

    # Batched matrix multiplication (uses cuBLAS Tensor‑Cores)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B*H, N, N)

    # Reshape back to (B,N,N,H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # FP16

    # --------------------------------------------------------------------- #
    # 9) Hidden‑dim LayerNorm (FP32), out‑gate, and final linear projection
    # --------------------------------------------------------------------- #
    out_fp32 = out.float()   # cast for a stable layer‑norm

    out_norm = F.layer_norm(
        out_fp32,
        normalized_shape=(H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=eps,
    )   # (B,N,N,H) FP32

    out_gated = out_norm * out_gate                     # element‑wise gate (FP32)
    out_final = F.linear(out_gated, to_out_w)           # (B,N,N,dim) FP32

    return out_final
