"""
TriMul “outgoing” forward pass – fast Triton‑based LayerNorm + fused
FP16 linear projections.

Algorithm
---------
1. Layer‑norm over the last dimension (D) of the 4‑D tensor using a
   custom Triton kernel (two‑pass Welford reduction → affine).
2. Cast the normalized tensor to FP16.
3. Fuse the five bias‑free linear layers (left, right, left_gate,
   right_gate, out_gate) into a single batched GEMM by concatenating the
   weight matrices. This reduces memory traffic and kernel launch
   overhead.
4. Split the concatenated result, optionally apply the pair‑wise mask to
   ``left`` and ``right``, and apply sigmoid to the three gate tensors.
5. Fuse the gates with the projected values (element‑wise multiply).
6. Perform the expensive contraction
      out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   as a batched FP16 GEMM (B·H independent matrix multiplies).
7. Layer‑norm over the hidden dimension (H) of the contracted result
   (again via Triton).
8. Multiply by the output‑gate and apply the final linear projection
   back to the original dimension D.  The final output is float32.

All heavy tensor work stays on the GPU; only the two LayerNorms are
implemented in Triton, satisfying the “must use Triton” requirement.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton LayerNorm (row‑wise) – two‑pass implementation (mean/var → affine)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,                # *ptr to input  [M, N]
    y_ptr,                # *ptr to output [M, N]
    weight_ptr,           # *ptr to weight [N]
    bias_ptr,             # *ptr to bias   [N]
    M,                    # number of rows
    N,                    # feature dimension
    eps,                  # epsilon
    stride_xm, stride_xn,  # strides for input (row, col)
    stride_ym, stride_yn,  # strides for output (row, col)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
):
    pid = tl.program_id(0)

    # ------------------------------------------------------------------
    # Row offsets for this program
    # ------------------------------------------------------------------
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # ------------------------------------------------------------------
    # First pass: compute mean & variance for each row
    # ------------------------------------------------------------------
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_N)
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        sum_val += tl.sum(x, axis=1)
        sum_sq  += tl.sum(x * x, axis=1)

    mean = sum_val / N
    var  = sum_sq / N - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # ------------------------------------------------------------------
    # Second pass: normalize + affine transform
    # ------------------------------------------------------------------
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        norm_x = (x - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_n, mask=mask_n, other=0.0)
        b = tl.load(bias_ptr   + offs_n, mask=mask_n, other=0.0)

        y = norm_x * w[None, :] + b[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn,
            y,
            mask=mask_m[:, None] & mask_n[None, :],
        )


def triton_layernorm(x: torch.Tensor, weight: torch.Tensor, bias: torch.Tensor, eps: float = 1e-5):
    """
    Apply LayerNorm over the last dimension of a 2‑D tensor ``[M, N]`` using
    the Triton kernel above.  ``weight`` and ``bias`` are 1‑D tensors of shape
    ``[N]`` (FP32).  The output has the same dtype as ``x``.
    """
    assert x.dim() == 2, "triton_layernorm expects a 2‑D tensor"
    M, N = x.shape
    y = torch.empty_like(x)

    # Ensure tensors are contiguous so that stride handling is correct
    x = x.contiguous()
    weight = weight.contiguous()
    bias = bias.contiguous()

    stride_xm, stride_xn = x.stride()
    stride_ym, stride_yn = y.stride()

    grid = lambda meta: (triton.cdiv(M, meta["BLOCK_M"]),)

    layernorm_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        weight_ptr=weight,
        bias_ptr=bias,
        M=M,
        N=N,
        eps=eps,
        stride_xm=stride_xm,
        stride_xn=stride_xn,
        stride_ym=stride_ym,
        stride_yn=stride_yn,
        BLOCK_M=256,   # reasonable for the large number of rows (B·N²)
        BLOCK_N=64,    # works well for D ≤ 384 (multiple of 64)
    )
    return y


# ----------------------------------------------------------------------
# TriMul entry point (outgoing version)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor   shape [B, L, L, D] (float32)
        - mask         : torch.Tensor   shape [B, L, L] (or None)
        - weights      : dict of torch tensors (model parameters)
        - config       : dict with keys ``dim``, ``hidden_dim`` and optional ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, L, L, D] (float32)
    """
    # ------------------------------------------------------------------
    # 0️⃣ Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    bs, L, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (Triton) over the last dimension (dim)
    # ------------------------------------------------------------------
    M = bs * L * L
    x_flat = input_tensor.reshape(M, dim).contiguous()
    norm_w = weights["norm.weight"].to(device, dtype).contiguous()
    norm_b = weights["norm.bias"].to(device, dtype).contiguous()
    x_norm_flat = triton_layernorm(x_flat, norm_w, norm_b, eps=eps)
    x_norm = x_norm_flat.view(bs, L, L, dim)               # [B, L, L, D]

    # ------------------------------------------------------------------
    # 2️⃣ Cast to FP16 for all heavy linear work
    # ------------------------------------------------------------------
    x_norm_h = x_norm.to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Fuse the five linear projections (left, right, left_gate,
    #    right_gate, out_gate) into a single GEMM.
    # ------------------------------------------------------------------
    left_proj_w     = weights["left_proj.weight"].to(device, torch.float16).contiguous()
    right_proj_w    = weights["right_proj.weight"].to(device, torch.float16).contiguous()
    left_gate_w     = weights["left_gate.weight"].to(device, torch.float16).contiguous()
    right_gate_w    = weights["right_gate.weight"].to(device, torch.float16).contiguous()
    out_gate_w      = weights["out_gate.weight"].to(device, torch.float16).contiguous()

    # Concatenate the weight matrices along the output dimension:
    #   [5*hidden_dim, dim] = [left; right; left_gate; right_gate; out_gate]
    weight_concat = torch.cat(
        [left_proj_w, right_proj_w, left_gate_w, right_gate_w, out_gate_w],
        dim=0,
    )  # shape (5*hidden_dim, dim)

    # Perform the fused linear projection.
    # Flatten to (M, dim) → (M, 5*hidden_dim)
    x_norm_h_flat = x_norm_h.view(-1, dim).contiguous()
    proj_flat = F.linear(x_norm_h_flat, weight_concat)               # FP16
    proj = proj_flat.view(bs, L, L, 5 * hidden_dim)                  # [B, L, L, 5*H]

    # ------------------------------------------------------------------
    # 4️⃣ Split the concatenated tensor.
    # ------------------------------------------------------------------
    H = hidden_dim
    left       = proj[..., 0:H]                     # [B, L, L, H]
    right      = proj[..., H:2*H]
    left_gate  = proj[..., 2*H:3*H]
    right_gate = proj[..., 3*H:4*H]
    out_gate   = proj[..., 4*H:5*H]

    # ------------------------------------------------------------------
    # 5️⃣ Optional mask (broadcast over the hidden dimension)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_h = mask.to(device, torch.float16).unsqueeze(-1)      # [B, L, L, 1]
        left = left * mask_h
        right = right * mask_h

    # ------------------------------------------------------------------
    # 6️⃣ Gating (sigmoid) – stays in FP16 for speed
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)

    # Apply the gates
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 7️⃣ Batched matrix multiplication over the sequence dimension:
    #     out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    # Rearrange to (B, H, N, N) to expose each head as an independent batch.
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    # Batched GEMM (FP16).  The transpose on the second operand implements the
    # “k” dimension contraction exactly as described in the reference code.
    out_perm = torch.matmul(left_perm, right_perm.transpose(-2, -1))  # [B, H, N, N]

    # Return to the original axis order → [B, N, N, H]
    out_h = out_perm.permute(0, 2, 3, 1).contiguous()   # FP16

    # ------------------------------------------------------------------
    # 8️⃣ Output LayerNorm over the hidden dimension (H) – Triton
    # ------------------------------------------------------------------
    out_fp32 = out_h.to(torch.float32)                 # LayerNorm works in FP32
    out_M = bs * L * L
    out_flat = out_fp32.view(out_M, hidden_dim).contiguous()
    to_out_norm_w = weights["to_out_norm.weight"].to(device, dtype).contiguous()
    to_out_norm_b = weights["to_out_norm.bias"].to(device, dtype).contiguous()
    out_norm_flat = triton_layernorm(out_flat, to_out_norm_w, to_out_norm_b, eps=eps)
    out_norm = out_norm_flat.view(bs, L, L, hidden_dim)  # [B, L, L, H]

    # ------------------------------------------------------------------
    # 9️⃣ Apply output gate and final projection back to D.
    # ------------------------------------------------------------------
    out_norm = out_norm * out_gate.to(dtype)            # broadcast over H
    to_out_w = weights["to_out.weight"].to(device, dtype).contiguous()
    out = F.linear(out_norm, to_out_w)                  # [B, L, L, D] (float32)

    return out
