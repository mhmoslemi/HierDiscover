"""
Outgoing TriMul (AlphaFold‑3) – fast forward pass.

Algorithm
---------
1️⃣  Input LayerNorm (FP32) – 2‑pass Triton kernel.
2️⃣  One fused linear projects the normalized tensor to the five
    matrices needed for the TriMul (left, right, left‑gate, right‑gate,
    out‑gate).  No bias, FP16 weights.
3️⃣  Split the concatenated projection, optionally apply the binary mask,
    and compute the three sigmoid gates.
4️⃣  Multiply `left`/`right` by their respective gates (FP16).
5️⃣  Batched matrix multiplication over the summed axis *k*:
    for every head `h` compute   out[b,i,j,h] = Σₖ left[b,i,k,h]·right[b,j,k,h].
    This is implemented with `torch.bmm`, which on H100 uses tensor‑core
    GEMM and is faster than a custom Triton kernel for many small matrices.
6️⃣  Output LayerNorm (FP32) – same Triton kernel as step 1.
7️⃣  Apply the output gate and a final linear projection back to the model
    dimension (FP16 weight, FP32 output).

All heavy compute is performed in FP16 with FP32 accumulation where needed,
while the two LayerNorms run in FP32 for numerical stability.  The kernel
required by the problem statement is the LayerNorm implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 2‑pass LayerNorm (FP32 accumulation, FP32 output)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,                # *pointer to input*   [M, N]
    y_ptr,                # *pointer to output*  [M, N]
    weight_ptr, bias_ptr, # [N]
    M, N, eps,            # dimensions
    stride_xm, stride_xn,  # input strides
    stride_ym, stride_yn,  # output strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
):
    pid = tl.program_id(0)

    # Row range for this program instance
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # ------------------------------------
    # First pass: compute mean & variance
    # ------------------------------------
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_iter_n = tl.cdiv(N, BLOCK_N)
    for it in range(num_iter_n):
        offs_n = tl.arange(0, BLOCK_N) + it * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        x_f = x.to(tl.float32)
        sum_val += tl.sum(x_f, axis=1)
        sum_sq  += tl.sum(x_f * x_f, axis=1)

    mean = sum_val / N
    var  = sum_sq / N - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # ------------------------------------
    # Second pass: normalise + affine
    # ------------------------------------
    for it in range(num_iter_n):
        offs_n = tl.arange(0, BLOCK_N) + it * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        x_f = x.to(tl.float32)

        norm_x = (x_f - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_n, mask=mask_n, other=0.0).to(tl.float32)
        b = tl.load(bias_ptr   + offs_n, mask=mask_n, other=0.0).to(tl.float32)

        y = norm_x * w[None, :] + b[None, :]

        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn,
            y,
            mask=mask_m[:, None] & mask_n[None, :],
        )


def triton_layernorm(
    x: torch.Tensor,
    weight: torch.Tensor,
    bias: torch.Tensor,
    eps: float = 1e-5,
    BLOCK_M: int = 128,
    BLOCK_N: int = 64,
) -> torch.Tensor:
    """LayerNorm over the last dimension of a 2‑D tensor (M×N) using Triton."""
    assert x.dim() == 2, "Expected a 2‑D tensor"
    M, N = x.shape
    y = torch.empty_like(x)

    # make sure tensors are contiguous
    x = x.contiguous()
    y = y.contiguous()
    weight = weight.contiguous()
    bias = bias.contiguous()

    stride_xm, stride_xn = x.stride()
    stride_ym, stride_yn = y.stride()

    grid = (triton.cdiv(M, BLOCK_M),)

    layernorm_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        weight_ptr=weight,
        bias_ptr=bias,
        M=M,
        N=N,
        eps=eps,
        stride_xm=stride_xm,
        stride_xn=stride_xn,
        stride_ym=stride_ym,
        stride_yn=stride_yn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        num_warps=4,
        num_stages=3,
    )
    return y


# ----------------------------------------------------------------------
# Triton‑accelerated forward pass of TriMul (outgoing)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, C] (float32)
        - mask         : torch.Tensor of shape [B, N, N] (float32) or None
        - weights      : dict of torch.Tensors containing model weights
        - config       : dict with keys ``dim``, ``hidden_dim`` and optional ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, C] (float32)
    """
    # --------------------------------------------------------------
    # 0️⃣ Unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype            # expected torch.float32
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32) – Triton
    # --------------------------------------------------------------
    # reshape to (B * N * N, dim) for 2‑D LayerNorm kernel
    M = bs * seq_len * seq_len
    x_flat = input_tensor.reshape(M, dim)                # (M, dim) FP32
    norm_w = weights["norm.weight"].to(device, dtype=torch.float32)
    norm_b = weights["norm.bias"].to(device, dtype=torch.float32)
    x_norm_flat = triton_layernorm(
        x_flat, norm_w, norm_b, eps=1e-5, BLOCK_M=128, BLOCK_N=64
    )
    x_norm = x_norm_flat.view(bs, seq_len, seq_len, dim)  # (B,N,N,dim) FP32

    # --------------------------------------------------------------
    # 2️⃣ Combined projection (half‑precision)
    # --------------------------------------------------------------
    # Convert to FP16 for the heavy linear work
    x_norm_h = x_norm.half()                               # (B,N,N,dim) FP16
    x_norm_h_flat = x_norm_h.view(M, dim)                   # (M, dim) FP16

    # Concatenate the five weight matrices: (5*H, dim)
    combined_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(device, dtype=torch.float16)

    # Fused linear (no bias)
    proj_out = F.linear(x_norm_h_flat, combined_weight)    # (M, 5*H) FP16

    # --------------------------------------------------------------
    # 3️⃣ Split into left/right and their gates
    # --------------------------------------------------------------
    left_raw, right_raw, left_gate_raw, right_gate_raw, out_gate_raw = torch.split(
        proj_out, hidden_dim, dim=1
    )
    # reshape back to (B,N,N,H)
    left_raw      = left_raw.view   (bs, seq_len, seq_len, hidden_dim)
    right_raw     = right_raw.view  (bs, seq_len, seq_len, hidden_dim)
    left_gate_raw = left_gate_raw.view(bs, seq_len, seq_len, hidden_dim)
    right_gate_raw= right_gate_raw.view(bs, seq_len, seq_len, hidden_dim)
    out_gate_raw  = out_gate_raw.view(bs, seq_len, seq_len, hidden_dim)

    # --------------------------------------------------------------
    # 4️⃣ Optional binary mask (broadcast over hidden dimension)
    # --------------------------------------------------------------
    if (not nomask) and (mask is not None):
        mask_h = mask.to(device, dtype=torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left_raw  = left_raw  * mask_h
        right_raw = right_raw * mask_h

    # --------------------------------------------------------------
    # 5️⃣ Gates (sigmoid) and application
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(left_gate_raw)          # (B,N,N,H) FP16
    right_gate = torch.sigmoid(right_gate_raw)         # (B,N,N,H) FP16
    out_gate   = torch.sigmoid(out_gate_raw)           # (B,N,N,H) FP16

    left  = left_raw  * left_gate                     # (B,N,N,H) FP16
    right = right_raw * right_gate                    # (B,N,N,H) FP16

    # --------------------------------------------------------------
    # 6️⃣ Batched matrix multiplication across the summed dimension (k)
    #    out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    #    Implemented with torch.bmm (highly‑optimized GEMM on H100)
    # --------------------------------------------------------------
    # Rearrange to (B, H, N, N) where the last two axes correspond to
    # the matrix dimensions for each head.
    left_perm  = left .permute(0, 3, 1, 2).contiguous()   # (B, H, i, k)
    right_perm = right.permute(0, 3, 2, 1).contiguous()   # (B, H, k, j)

    # Collapse batch and head dimensions for bmm
    left_flat  = left_perm .reshape(bs * hidden_dim, seq_len, seq_len)  # (B*H, N, K)
    right_flat = right_perm.reshape(bs * hidden_dim, seq_len, seq_len)  # (B*H, K, N)

    # Batched GEMM: (B*H, N, N)
    out_flat = torch.bmm(left_flat, right_flat)  # FP16 accumulation in FP32 internally

    # Restore (B, H, N, N) then back to (B, N, N, H)
    out_perm = out_flat.view(bs, hidden_dim, seq_len, seq_len)          # (B, H, N, N)
    out = out_perm.permute(0, 2, 3, 1).contiguous()                     # (B, N, N, H) FP16

    # --------------------------------------------------------------
    # 7️⃣ Output LayerNorm (FP32) – Triton
    # --------------------------------------------------------------
    out_fp32 = out.float()                               # (B,N,N,H) FP32
    out_flat_fp32 = out_fp32.view(bs * seq_len * seq_len, hidden_dim)

    to_out_norm_w = weights["to_out_norm.weight"].to(device, dtype=torch.float32)
    to_out_norm_b = weights["to_out_norm.bias"  ].to(device, dtype=torch.float32)

    out_norm_flat = triton_layernorm(
        out_flat_fp32,
        to_out_norm_w,
        to_out_norm_b,
        eps=1e-5,
        BLOCK_M=128,
        BLOCK_N=64,
    )
    out_norm = out_norm_flat.view(bs, seq_len, seq_len, hidden_dim)   # (B,N,N,H) FP32

    # --------------------------------------------------------------
    # 8️⃣ Apply output gate and final projection back to model dim
    # --------------------------------------------------------------
    # Output gate (already FP16, cast to FP32 for the multiplication)
    out_gate_fp32 = out_gate.float()
    gated = out_norm * out_gate_fp32                                 # (B,N,N,H) FP32

    # Final linear projection (half‑precision weight, FP16 compute)
    to_out_w = weights["to_out.weight"].to(device, dtype=torch.float16)  # (dim, H)
    gated_h = gated.half()
    final_fp16 = F.linear(gated_h, to_out_w)                         # (B,N,N,dim) FP16
    final_fp32 = final_fp16.float()                                   # (B,N,N,dim) FP32

    return final_fp32
