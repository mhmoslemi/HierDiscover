"""
Outgoing TriMul (AlphaFold3) – fused Triton GEMM.

The heavy part of the TriMul module is the “triangle” multiplication:
    out[b, i, j, h] = Σₖ left[b, i, k, h] * right[b, j, k, h]

For each batch `b` and hidden channel `h` this is a regular matrix
multiplication of two N×N matrices (left and right^T).  We pack the
batch and hidden dimensions into a single GEMM‑batch dimension
`C = B * H` and launch a batched GEMM kernel with Triton.  The kernel
uses fp16 inputs, fp32 accumulation (tensor‑core friendly) and writes
the result in fp16.  All surrounding linear layers, gates and layer‑norms
are performed with torch​.nn.functional in half‑precision; the final
output is cast back to fp32 as required by the reference implementation.

The implementation removes the mask‑related branching when `nomask`
is true (most test cases) and selects tile sizes / warp counts that
fit the H100 compute resources.  `num_stages=3` enables double‑buffered
shared‑memory loading for higher throughput.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    out_ptr,            # fp16*   (C, M, N) – output
    a_ptr,              # fp16*   (C, M, K) – left matrix
    b_ptr,              # fp16*   (C, K, N) – right matrix (logically transposed)
    M, N, K,            # matrix sizes (all == seq_len)
    stride_out_c, stride_out_m, stride_out_n,   # output strides
    stride_a_c, stride_a_m, stride_a_k,        # left strides
    stride_b_c, stride_b_k, stride_b_n,        # right strides (after transposition)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    GROUP_M: tl.constexpr,      # not used, kept for signature compatibility
):
    """Batched GEMM: for each channel `c` (batch*hidden) compute
    out_c[M,N] = left_c[M,K] @ right_c[K,N].
    fp16 inputs → fp32 accumulation → fp16 store.
    """
    pid_m = tl.program_id(0)   # block row index (i)
    pid_n = tl.program_id(1)   # block col index (j)
    pid_c = tl.program_id(2)   # batch*hidden index

    # -----------------------------------------------------------------
    # Tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -----------------------------------------------------------------
    # Loop over K dimension in tiles
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # pointers to the current tiles of A and B
        a_tile = a_ptr + pid_c * stride_a_c \
                 + offs_m[:, None] * stride_a_m \
                 + offs_k[None, :] * stride_a_k          # (M, K)

        b_tile = b_ptr + pid_c * stride_b_c \
                 + offs_k[:, None] * stride_b_k \
                 + offs_n[None, :] * stride_b_n          # (K, N)

        # Load tiles (mask out‑of‑bounds entries)
        a = tl.load(a_tile,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)                # fp16
        b = tl.load(b_tile,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)                # fp16

        # GEMM tile: fp16 * fp16 → fp32 accumulation
        acc += tl.dot(a, b)

    # -----------------------------------------------------------------
    # Write result (convert back to fp16)
    out_tile = out_ptr + pid_c * stride_out_c \
               + offs_m[:, None] * stride_out_m \
               + offs_n[None, :] * stride_out_n
    tl.store(out_tile,
             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module (AlphaFold3 style).

    Args:
        data (tuple): (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor[B, N, N, C] (float32)
            - mask         : torch.Tensor[B, N, N] (bool/float) or None
            - weights      : dict of model parameters
            - config       : dict with keys
                * "dim"         – input channel dimension C
                * "hidden_dim" – hidden dimension H
                * "nomask"      – bool, True → ignore mask

    Returns:
        torch.Tensor[B, N, N, C] (float32)
    """
    # -----------------------------------------------------------------
    # Unpack arguments
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -----------------------------------------------------------------
    # 1) Input LayerNorm (fp32 for numerical stability, then cast to fp16)
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 2) Linear projections (no bias) – fp16
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left = F.linear(x, left_proj_w)   # (B, N, N, H)
    right = F.linear(x, right_proj_w) # (B, N, N, H)

    # -----------------------------------------------------------------
    # 3) Gates (sigmoid) – fp16
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))   # (B, N, N, H)

    # -----------------------------------------------------------------
    # 4) Apply gates (and optional mask)
    left = left * left_gate
    right = right * right_gate

    if not nomask and mask is not None:
        # mask is expected to be 0/1, broadcast over hidden dim
        mask_h = mask.to(torch.float16).unsqueeze(-1)  # (B, N, N, 1)
        left = left * mask_h
        right = right * mask_h

    # -----------------------------------------------------------------
    # 5) Prepare tensors for the batched GEMM kernel.
    #    Pack (B, H) into a single GEMM batch dimension C = B*H.
    B, N, _, _ = left.shape
    # bring hidden dim before the two sequence dims
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    C_total = B * hidden_dim
    left_flat = left.reshape(C_total, N, N)   # (C, M, K)  – K == N
    right_flat = right.reshape(C_total, N, N) # (C, K, N)

    # Output buffer (fp16)
    out_flat = torch.empty_like(left_flat, dtype=torch.float16, device=device)

    # -----------------------------------------------------------------
    # 6) Determine tile sizes.  All test seq‑len values are multiples of 64.
    if N <= 256:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 32
    else:
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 64 if N > 512 else 32

    # Strides in element units (not bytes)
    stride_out_c, stride_out_m, stride_out_n = (
        out_flat.stride(0),
        out_flat.stride(1),
        out_flat.stride(2),
    )
    stride_a_c, stride_a_m, stride_a_k = (
        left_flat.stride(0),
        left_flat.stride(1),
        left_flat.stride(2),
    )
    stride_b_c, stride_b_k, stride_b_n = (
        right_flat.stride(0),
        right_flat.stride(1),   # K dimension
        right_flat.stride(2),   # N dimension
    )

    # Grid: (M‑blocks, N‑blocks, C‑batch)
    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # pid_m
        (N + BLOCK_N - 1) // BLOCK_N,   # pid_n
        C_total,                        # pid_c
    )

    # Choose enough warps to keep the H100 SM busy.
    num_warps = 16 if N >= 768 else 8

    # Launch the kernel with double‑buffered loads (num_stages=3)
    tri_mul_kernel[grid](
        out_flat,
        left_flat,
        right_flat,
        N, N, N,
        stride_out_c, stride_out_m, stride_out_n,
        stride_a_c, stride_a_m, stride_a_k,
        stride_b_c, stride_b_k, stride_b_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        GROUP_M=0,                     # placeholder – not used
        num_warps=num_warps,
        num_stages=3,
    )

    # -----------------------------------------------------------------
    # 7) Reshape the GEMM result back to (B, N, N, H) – still fp16.
    out = out_flat.reshape(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 8) Post‑processing: 2nd LayerNorm → out‑gate → final linear projection.
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)

    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # apply the out‑gate (already fp16)
    out = out * out_gate

    # final projection back to original dim
    to_out_w = weights["to_out.weight"].to(torch.float16)
    out = F.linear(out, to_out_w)   # (B, N, N, C) fp16

    # -----------------------------------------------------------------
    # 9) Return float‑32 tensor as expected by the reference code.
    return out.to(torch.float32)
