"""
TriMul “outgoing’’ forward pass – Triton‑accelerated version.

Algorithm
---------
1️⃣  Input LayerNorm (Triton).  
2️⃣  Cast to FP16 and fuse the five Linear projections (left, right, and the three
    gating streams) into a single GEMM.  
3️⃣  Apply sigmoid to the gates and (optionally) the pair mask – all done with
    a single element‑wise kernel.  
4️⃣  Compute the O(N³·H) contraction

        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]

    with a custom Triton kernel that works directly on the original
    (B,N,N,H) layout, thus avoiding the costly ``permute(...).contiguous()``
    copies required for a batched torch‑bmm.  The kernel uses Tensor‑Core
    ``tl.dot`` and accumulates in FP32, storing the result in FP16.
5️⃣  Output LayerNorm (Triton).  
6️⃣  Apply the output gate, a final Linear projection and cast back to FP32.

Only the two LayerNorms and the N³ contraction are written in Triton – the
remaining dense operations use PyTorch’s highly‑optimised FP16 GEMM kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# LayerNorm kernel (operates on the last dimension)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr, y_ptr, weight_ptr, bias_ptr,
    M, N, eps,
    stride_xm, stride_xn,
    stride_ym, stride_yn,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr
):
    pid = tl.program_id(0)

    # --------------------------------------------------------------
    # Row range for this program
    # --------------------------------------------------------------
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # --------------------------------------------------------------
    # First pass: compute mean & variance (FP32 accumulation)
    # --------------------------------------------------------------
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_N)
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        x_f32 = tl.cast(x, tl.float32)
        sum_val += tl.sum(x_f32, axis=1)
        sum_sq  += tl.sum(x_f32 * x_f32, axis=1)

    mean = sum_val / N
    var  = sum_sq / N - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # --------------------------------------------------------------
    # Second pass: write normalised + affine transformed values
    # --------------------------------------------------------------
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        x_f32 = tl.cast(x, tl.float32)

        norm_x = (x_f32 - mean[:, None]) * rstd[:, None]

        w = tl.cast(tl.load(weight_ptr + offs_n, mask=mask_n, other=0.0), tl.float32)
        b = tl.cast(tl.load(bias_ptr + offs_n, mask=mask_n, other=0.0), tl.float32)

        y = norm_x * w[None, :] + b[None, :]    # affine
        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn,
            tl.cast(y, x.dtype),
            mask=mask_m[:, None] & mask_n[None, :],
        )

def triton_layernorm(x: torch.Tensor,
                     weight: torch.Tensor,
                     bias: torch.Tensor,
                     eps: float = 1e-5,
                     block_m: int = 128,
                     block_n: int = 64) -> torch.Tensor:
    """LayerNorm over the last dimension with the Triton kernel."""
    assert x.dim() == 2
    M, N = x.shape
    y = torch.empty_like(x)

    # make everything contiguous and fetch strides
    x = x.contiguous()
    y = y.contiguous()
    weight = weight.contiguous()
    bias = bias.contiguous()
    stride_xm, stride_xn = x.stride()
    stride_ym, stride_yn = y.stride()

    grid = lambda meta: (triton.cdiv(M, meta["BLOCK_M"]),)

    layernorm_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        weight_ptr=weight,
        bias_ptr=bias,
        M=M,
        N=N,
        eps=eps,
        stride_xm=stride_xm,
        stride_xn=stride_xn,
        stride_ym=stride_ym,
        stride_yn=stride_yn,
        BLOCK_M=block_m,
        BLOCK_N=block_n,
    )
    return y


# ----------------------------------------------------------------------
# Batched N³ contraction kernel (outgoing version)
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_outgoing_kernel(
    a_ptr, b_ptr, c_ptr,
    B, N, H,
    stride_batch, stride_i, stride_k, stride_head,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Compute out[b,i,j,h] = Σ_k a[b,i,k,h] * b[b,j,k,h]"""
    pid0 = tl.program_id(0)          # combined batch * head index
    pid1 = tl.program_id(1)          # i‑tile
    pid2 = tl.program_id(2)          # j‑tile

    b_idx = pid0 // H
    h_idx = pid0 % H

    offs_i = pid1 * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid2 * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # pointer to the output tile
    c_tile_ptr = (
        c_ptr
        + b_idx * stride_batch
        + h_idx * stride_head
        + offs_i[:, None] * stride_i
        + offs_j[None, :] * stride_i   # i‑ and j‑stride are identical for (B,N,N,H)
    )

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_K)
    for k_iter in range(num_k):
        offs_k = k_iter * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # A block : (BLOCK_M, BLOCK_K)   – rows i, cols k
        a_tile_ptr = (
            a_ptr
            + b_idx * stride_batch
            + h_idx * stride_head
            + offs_i[:, None] * stride_i
            + offs_k[None, :] * stride_k
        )
        a = tl.load(a_tile_ptr, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # B block : (BLOCK_K, BLOCK_N)   – rows k, cols j
        b_tile_ptr = (
            b_ptr
            + b_idx * stride_batch
            + h_idx * stride_head
            + offs_k[:, None] * stride_k
            + offs_j[None, :] * stride_i
        )
        b = tl.load(b_tile_ptr, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        acc += tl.dot(a, b)   # FP16 × FP16 → FP32 accumulation

    tl.store(c_tile_ptr, tl.cast(acc, tl.float16), mask=mask_i[:, None] & mask_j[None, :])

# ----------------------------------------------------------------------
# Triton‑accelerated TriMul “outgoing’’ entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing’’ operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, D] (float32)
        - mask         : torch.Tensor [B, N, N] or None
        - weights      : dict of torch.Tensors
        - config       : dict with keys ``dim``, ``hidden_dim``, ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor [B, N, N, D] (float32)
    """
    # --------------------------------------------------------------
    # Unpack
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype           # float32

    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    eps = 1e-5

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm (Triton)
    # --------------------------------------------------------------
    rows = bs * seq_len * seq_len
    x_flat = input_tensor.reshape(rows, dim)    # (M, D)

    norm_w = weights["norm.weight"].to(device, dtype)
    norm_b = weights["norm.bias"].to(device, dtype)

    # Choose a BLOCK_N that nicely divides dim (64 for ≤128, 128 otherwise)
    block_n_in = 64 if dim <= 128 else 128
    x_norm_flat = triton_layernorm(
        x_flat, norm_w, norm_b, eps=eps, block_m=256, block_n=block_n_in
    )
    x_norm = x_norm_flat.view(bs, seq_len, seq_len, dim)   # (B,N,N,D)

    # --------------------------------------------------------------
    # 2️⃣ Cast to FP16 (all heavy work in half)
    # --------------------------------------------------------------
    x_norm_f16 = x_norm.to(torch.float16)

    # --------------------------------------------------------------
    # 3️⃣ Fuse the five Linear layers (proj + gates) into one GEMM
    # --------------------------------------------------------------
    #   [left, left_gate, right, right_gate, out_gate]  →  (5*H, D)
    left_proj_w   = weights["left_proj.weight"]
    left_gate_w   = weights["left_gate.weight"]
    right_proj_w  = weights["right_proj.weight"]
    right_gate_w  = weights["right_gate.weight"]
    out_gate_w    = weights["out_gate.weight"]

    combined_weight = torch.cat(
        [left_proj_w, left_gate_w, right_proj_w, right_gate_w, out_gate_w],
        dim=0,
    ).to(device, torch.float16)                     # (5*H, D)

    # single fused projection
    proj_gate_flat = F.linear(x_norm_f16.reshape(rows, dim), combined_weight)   # (M, 5*H)

    # --------------------------------------------------------------
    # 4️⃣ Slice, apply sigmoid, and (optionally) mask
    # --------------------------------------------------------------
    proj_gate = proj_gate_flat.view(bs, seq_len, seq_len, 5 * hidden_dim)   # (B,N,N,5H)

    H = hidden_dim
    left_proj   = proj_gate[..., 0:H]                 # (B,N,N,H)
    left_gate_r = proj_gate[..., H:2*H]
    right_proj  = proj_gate[..., 2*H:3*H]
    right_gate_r= proj_gate[..., 3*H:4*H]
    out_gate_r  = proj_gate[..., 4*H:5*H]

    left_gate = torch.sigmoid(left_gate_r)
    right_gate = torch.sigmoid(right_gate_r)
    out_gate  = torch.sigmoid(out_gate_r)

    if not nomask and mask is not None:
        mask_f16 = mask.to(device, torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left  = left_proj * left_gate * mask_f16
        right = right_proj * right_gate * mask_f16
    else:
        left  = left_proj * left_gate
        right = right_proj * right_gate

    # ensure contiguous layout for the upcoming Triton kernel
    left = left.contiguous()
    right = right.contiguous()

    # --------------------------------------------------------------
    # 5️⃣ Batched N³ contraction (custom Triton kernel)
    # --------------------------------------------------------------
    # Output tensor (B,N,N,H) – initially uninitialised
    out = torch.empty_like(left)

    # strides for the (B,N,N,H) layout (contiguous, FP16)
    stride_batch, stride_i, stride_k, stride_head = left.stride()

    # Triton grid: (batch*head, ceil(N/BLOCK_M), ceil(N/BLOCK_N))
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    grid = (
        bs * hidden_dim,
        triton.cdiv(seq_len, BLOCK_M),
        triton.cdiv(seq_len, BLOCK_N),
    )

    tri_mul_outgoing_kernel[grid](
        a_ptr=left,
        b_ptr=right,
        c_ptr=out,
        B=bs,
        N=seq_len,
        H=hidden_dim,
        stride_batch=stride_batch,
        stride_i=stride_i,
        stride_k=stride_k,
        stride_head=stride_head,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # --------------------------------------------------------------
    # 6️⃣ Output LayerNorm (Triton) – stays in FP16
    # --------------------------------------------------------------
    out_rows = rows
    out_flat = out.view(out_rows, hidden_dim)          # (M_out, H)

    to_out_norm_w = weights["to_out_norm.weight"].to(device, dtype)   # FP32
    to_out_norm_b = weights["to_out_norm.bias"].to(device, dtype)     # FP32

    block_n_out = 64 if hidden_dim <= 128 else 128
    out_norm_flat = triton_layernorm(
        out_flat, to_out_norm_w, to_out_norm_b, eps=eps,
        block_m=256, block_n=block_n_out
    )
    out_norm = out_norm_flat.view(bs, seq_len, seq_len, hidden_dim)   # (B,N,N,H)

    # --------------------------------------------------------------
    # 7️⃣ Apply output gate and final linear projection
    # --------------------------------------------------------------
    gated = out_norm * out_gate                                 # (B,N,N,H)

    to_out_w = weights["to_out.weight"].to(device, torch.float16)   # (D, H)
    out_fp16 = F.linear(gated, to_out_w)                           # (B,N,N,D) FP16
    out_fp32 = out_fp16.to(torch.float32)                          # API contract

    return out_fp32
