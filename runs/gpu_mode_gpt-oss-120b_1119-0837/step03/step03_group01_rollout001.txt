"""
TriMul “outgoing” forward pass – Triton‑accelerated version.

Algorithm
---------
1️⃣  Layer‑Norm the input tensor (B,N,N,D) with a custom Triton kernel.
2️⃣  Cast to FP16 and fuse the five linear layers (left‑proj/right‑proj &
    three gates) into one GEMM to reduce memory traffic.
3️⃣  Apply sigmoid to the gating tensors and (optionally) the pairwise mask.
    The gating and masking are done **in‑place** on the projected tensors so
    no extra buffers are allocated.
4️⃣  Contract the two 4‑D tensors with the O(N³·H) pattern

        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]

    by reshaping to (B·H,N,N) and using a batched half‑precision `torch.bmm`
    (cuBLAS/Tensor‑Core).
5️⃣  Layer‑Norm the result with a second Triton kernel (output stays in FP16).
6️⃣  Apply the output gate **in‑place** and a final linear projection back to
    the model dimension.  The final tensor is returned as FP32.

Only the two LayerNorms are implemented in Triton; all dense linear
operations use highly‑optimized FP16 GEMM kernels from PyTorch.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel for LayerNorm over the last dimension (2‑D view)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,      # *ptr to input   [M, N]
    y_ptr,      # *ptr to output  [M, N]
    w_ptr,      # *ptr to weight  [N]
    b_ptr,      # *ptr to bias    [N]
    M,          # number of rows  (B*N*N)
    N,          # feature dimension (dim or hidden_dim)
    eps,        # epsilon for numerical stability
    stride_xm, stride_xn,   # strides of input
    stride_ym, stride_yn,   # strides of output
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
):
    pid = tl.program_id(0)

    # -----------------------------
    # Row offsets for this program
    # -----------------------------
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # -------------------------------------------------
    # First pass: compute per‑row mean & variance FP32
    # -------------------------------------------------
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_N)
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        x_f32 = tl.cast(x, tl.float32)
        sum_val += tl.sum(x_f32, axis=1)
        sum_sq  += tl.sum(x_f32 * x_f32, axis=1)

    mean = sum_val / N
    var  = sum_sq / N - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # -------------------------------------------------
    # Second pass: write normalised + affine transformed values
    # -------------------------------------------------
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        x_f32 = tl.cast(x, tl.float32)

        # normalisation (FP32)
        norm_x = (x_f32 - mean[:, None]) * rstd[:, None]

        # affine transform – weight/bias are FP32 → cast to float32 for safety
        w = tl.cast(tl.load(w_ptr + offs_n, mask=mask_n, other=0.0), tl.float32)
        b = tl.cast(tl.load(b_ptr + offs_n, mask=mask_n, other=0.0), tl.float32)

        y = norm_x * w[None, :] + b[None, :]

        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn,
            tl.cast(y, x.dtype),
            mask=mask_m[:, None] & mask_n[None, :],
        )


def triton_layernorm(x: torch.Tensor,
                     weight: torch.Tensor,
                     bias: torch.Tensor,
                     eps: float = 1e-5) -> torch.Tensor:
    """
    Apply LayerNorm over the last dimension of a 2‑D tensor using Triton.
    Arguments:
        x      – (M, N) tensor (any floating dtype)
        weight – (N,) affine weight (same dtype as x or FP32)
        bias   – (N,) affine bias  (same dtype as x or FP32)
    Returns:
        Normalised tensor of same shape and dtype as ``x``.
    """
    assert x.dim() == 2, "triton_layernorm expects a 2‑D tensor"
    M, N = x.shape
    y = torch.empty_like(x)

    # Ensure contiguous layout for stride handling
    x = x.contiguous()
    y = y.contiguous()
    weight = weight.contiguous()
    bias = bias.contiguous()

    stride_xm, stride_xn = x.stride()
    stride_ym, stride_yn = y.stride()

    grid = lambda meta: (triton.cdiv(M, meta["BLOCK_M"]),)

    # Tune block sizes: larger BLOCK_N reduces the loop count over N.
    layernorm_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        w_ptr=weight,
        b_ptr=bias,
        M=M,
        N=N,
        eps=eps,
        stride_xm=stride_xm,
        stride_xn=stride_xn,
        stride_ym=stride_ym,
        stride_yn=stride_yn,
        BLOCK_M=256,   # rows per program – larger for better occupancy
        BLOCK_N=128,   # columns per tile – fits typical hidden dims
    )
    return y


# ----------------------------------------------------------------------
# Triton‑accelerated TriMul “outgoing’’ entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing’’ operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, D] (float32)
        - mask         : torch.Tensor [B, N, N] (float32/float16) or None
        - weights      : dict of torch.Tensors (model parameters)
        - config       : dict with keys ``dim``, ``hidden_dim``, ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor [B, N, N, D] (float32)
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype                     # expected torch.float32

    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    eps = 1e-5

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm (Triton)
    # --------------------------------------------------------------
    M = bs * seq_len * seq_len                     # rows for layernorm
    x_flat = input_tensor.reshape(M, dim)          # (M, D) FP32

    norm_w = weights["norm.weight"].to(device, dtype)
    norm_b = weights["norm.bias"].to(device, dtype)

    x_norm_flat = triton_layernorm(x_flat, norm_w, norm_b, eps=eps)
    x_norm = x_norm_flat.view(bs, seq_len, seq_len, dim)   # (B,N,N,D) FP32

    # --------------------------------------------------------------
    # 2️⃣ Cast to FP16 for heavy linear work
    # --------------------------------------------------------------
    x_norm_f16 = x_norm.to(torch.float16)
    x_norm_f16_flat = x_norm_f16.reshape(M, dim)   # (M, D) FP16

    # --------------------------------------------------------------
    # 3️⃣ Fuse all projection & gate linear layers into a single GEMM
    # --------------------------------------------------------------
    # Collect weight matrices (no bias for any of these Linear layers)
    w_left_proj   = weights["left_proj.weight"]
    w_left_gate   = weights["left_gate.weight"]
    w_right_proj  = weights["right_proj.weight"]
    w_right_gate  = weights["right_gate.weight"]
    w_out_gate    = weights["out_gate.weight"]

    # Shape (5*H, D). Cast to fp16 once.
    combined_weight = torch.cat(
        [w_left_proj, w_left_gate,
         w_right_proj, w_right_gate,
         w_out_gate],
        dim=0,
    ).to(device, torch.float16)      # (5*H, D)

    # Single fused projection
    proj_gate_flat = F.linear(x_norm_f16_flat, combined_weight)   # (M,5H) FP16
    proj_gate = proj_gate_flat.view(bs, seq_len, seq_len, 5 * hidden_dim)

    H = hidden_dim
    # Slice the fused tensor (views – no copy)
    left_proj   = proj_gate[..., 0:H]               # (B,N,N,H)
    left_gate_r = proj_gate[..., H:2*H]
    right_proj  = proj_gate[..., 2*H:3*H]
    right_gate_r= proj_gate[..., 3*H:4*H]
    out_gate_r  = proj_gate[..., 4*H:5*H]

    # --------------------------------------------------------------
    # 4️⃣ Apply sigmoid gates (still FP16)
    # --------------------------------------------------------------
    left_gate   = torch.sigmoid(left_gate_r)   # (B,N,N,H)
    right_gate  = torch.sigmoid(right_gate_r)
    out_gate    = torch.sigmoid(out_gate_r)    # (B,N,N,H)

    # In‑place gating – no extra buffers
    left_proj.mul_(left_gate)                # left = left_proj * left_gate
    right_proj.mul_(right_gate)              # right = right_proj * right_gate

    if not nomask and mask is not None:
        # Mask is broadcastable over the hidden dimension
        mask_f16 = mask.to(device, torch.float16).unsqueeze(-1)    # (B,N,N,1)
        left_proj.mul_(mask_f16)
        right_proj.mul_(mask_f16)

    # --------------------------------------------------------------
    # 5️⃣ Batched contraction Σₖ left[b,i,k,h] * right[b,j,k,h]
    # --------------------------------------------------------------
    # Reshape to (B*H, N, N) for a batched matmul
    left_perm  = left_proj.permute(0, 3, 1, 2).contiguous()   # (B,H,N,N)
    right_perm = right_proj.permute(0, 3, 1, 2).contiguous()  # (B,H,N,N)

    BH = bs * hidden_dim
    left_mat  = left_perm.view(BH, seq_len, seq_len)          # (B*H,N,N)
    right_mat = right_perm.view(BH, seq_len, seq_len)         # (B*H,N,N)

    # (B*H,N,N) @ (B*H,N,N)^T  →  (B*H,N,N), FP16 via Tensor‑Core
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Back to (B,N,N,H)
    out = out_mat.view(bs, hidden_dim, seq_len, seq_len) \
                .permute(0, 2, 3, 1).contiguous()          # (B,N,N,H) FP16

    # --------------------------------------------------------------
    # 6️⃣ Output LayerNorm (Triton) – keep FP16
    # --------------------------------------------------------------
    out_flat = out.view(M, hidden_dim)                     # (B*N*N, H)
    to_out_norm_w = weights["to_out_norm.weight"].to(device, dtype)  # FP32
    to_out_norm_b = weights["to_out_norm.bias"].to(device, dtype)    # FP32

    out_norm_flat = triton_layernorm(out_flat, to_out_norm_w, to_out_norm_b, eps=eps)
    out_norm = out_norm_flat.view(bs, seq_len, seq_len, hidden_dim)  # (B,N,N,H) FP16

    # --------------------------------------------------------------
    # 7️⃣ Apply output gate (in‑place) and final projection
    # --------------------------------------------------------------
    out_norm.mul_(out_gate)                                 # gated, FP16

    to_out_w = weights["to_out.weight"].to(device, torch.float16)   # (D, H) FP16
    output_fp16 = F.linear(out_norm, to_out_w)                      # (B,N,N,D) FP16
    output = output_fp16.to(torch.float32)                         # cast to FP32

    return output
