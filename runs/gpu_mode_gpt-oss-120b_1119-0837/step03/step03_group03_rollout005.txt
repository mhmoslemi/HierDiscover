"""
Optimised Triton+PyTorch implementation of the *outgoing* TriMul operator
(AlphaFold‑3).  The key speed‑ups over the baseline are:

* **Fused linear projections** – left/right projections and all three gates
  (left_gate, right_gate, out_gate) are obtained with only **two** matrix‐
  multiplications instead of five independent calls.  This halves the number of
  kernel launches and reduces the total memory traffic of the large flattened
  input tensor.
* **Mixed‑precision pipeline** – the heavy tensor‑contraction is performed in
  FP16 (Tensor‑Core friendly).  The small LayerNorms are kept in FP32 for
  numerical stability but return FP16, so the rest of the pipeline stays in the
  fast half‑precision format.
* **Batched GEMM for the N³ contraction** – the `torch.bmm` on a tensor of shape
  `(B*H, L, L)` is still the fastest way to compute the contraction
  ``out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]``.  The previous code used a
  separate `permute`+`contiguous` for each tensor; we keep this only once
  after the fused projections.
* **Triton LayerNorm** – a two‑pass row‑wise LayerNorm kernel that works for
  both FP16 and FP32 inputs.

The overall algorithm is:

1. Input LayerNorm (Triton) on the flattened `[B*L*L, D]` tensor.
2. Cast to FP16.
3. **Fused projection**:
   * Linear → (`left_proj`, `right_proj`)   with a weight of shape `(2*H, D)`.
   * Linear → (`left_gate`, `right_gate`, `out_gate`) with a weight of
     shape `(3*H, D)`.
4. Optional mask multiplication (broadcast over the hidden dimension).
5. Apply sigmoid gates and element‑wise gating.
6. Reshape to `(B*H, L, L)` and perform the batched GEMM (`torch.bmm`).
7. Output LayerNorm (Triton) over the hidden dimension.
8. Multiply by the previously computed `out_gate`.
9. Final linear projection back to the model dimension `D`.

The implementation satisfies the required signature `custom_kernel(data)`.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton row‑wise LayerNorm (operates on a 2‑D tensor [M, N]).
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr, y_ptr,              # *ptr to input / output  [M, N]
    w_ptr, b_ptr,              # *ptr to weight / bias   [N]
    M, N, eps,                 # dimensions and epsilon
    stride_xm, stride_xn,      # strides for input
    stride_ym, stride_yn,      # strides for output
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
):
    pid = tl.program_id(0)

    # ---------------------------------------------------------------
    # Row offsets for this program
    # ---------------------------------------------------------------
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # ---------------------------------------------------------------
    # First pass – compute mean and variance (FP32 accumulation)
    # ---------------------------------------------------------------
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_n_blocks = tl.cdiv(N, BLOCK_N)
    for n in range(num_n_blocks):
        offs_n = n * BLOCK_N + tl.arange(0, BLOCK_N)
        mask_n = offs_n < N

        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        x_f32 = x.to(tl.float32)
        sum_val += tl.sum(x_f32, axis=1)
        sum_sq  += tl.sum(x_f32 * x_f32, axis=1)

    mean = sum_val / N
    var  = sum_sq / N - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # ---------------------------------------------------------------
    # Second pass – write normalized + affine transformed values
    # ---------------------------------------------------------------
    for n in range(num_n_blocks):
        offs_n = n * BLOCK_N + tl.arange(0, BLOCK_N)
        mask_n = offs_n < N

        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        x_f32 = x.to(tl.float32)
        norm_x = (x_f32 - mean[:, None]) * rstd[:, None]

        w = tl.load(w_ptr + offs_n, mask=mask_n, other=0.0)
        b = tl.load(b_ptr + offs_n, mask=mask_n, other=0.0)
        w_f32 = w.to(tl.float32)
        b_f32 = b.to(tl.float32)

        y_f32 = norm_x * w_f32 + b_f32
        y = y_f32.to(x.dtype)          # cast back to the input dtype

        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn,
            y,
            mask=mask_m[:, None] & mask_n[None, :],
        )


def triton_layernorm(x: torch.Tensor,
                     weight: torch.Tensor,
                     bias: torch.Tensor,
                     eps: float = 1e-5) -> torch.Tensor:
    """
    Row‑wise LayerNorm implemented in Triton.
    ``x`` must be a 2‑D tensor; the returned tensor has the same shape and dtype.
    """
    assert x.dim() == 2, "triton_layernorm expects a 2‑D tensor"
    M, N = x.shape
    y = torch.empty_like(x)

    # make sure strides are correct (contiguous tensors guarantee sane strides)
    x = x.contiguous()
    weight = weight.contiguous()
    bias = bias.contiguous()
    y = y.contiguous()

    stride_xm, stride_xn = x.stride()
    stride_ym, stride_yn = y.stride()

    # Tune BLOCK sizes – these work well for dim up to 384
    BLOCK_M = 256
    BLOCK_N = 128

    grid = lambda meta: (triton.cdiv(M, meta["BLOCK_M"]),)

    layernorm_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        w_ptr=weight,
        b_ptr=bias,
        M=M,
        N=N,
        eps=eps,
        stride_xm=stride_xm,
        stride_xn=stride_xn,
        stride_ym=stride_ym,
        stride_yn=stride_yn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
    )
    return y


# ----------------------------------------------------------------------
# Entry point required by the benchmark harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul *outgoing* operator (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor   [B, L, L, D] (float32)
        - mask         : torch.Tensor   [B, L, L] (float32) or ``None``
        - weights      : dict of torch tensors with keys:
              "norm.weight", "norm.bias",
              "left_proj.weight", "right_proj.weight",
              "left_gate.weight", "right_gate.weight", "out_gate.weight",
              "to_out_norm.weight", "to_out_norm.bias",
              "to_out.weight"
        - config : dict with fields ``dim`` (D), ``hidden_dim`` (H) and optional
                   ``nomask`` (bool).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, L, L, D] (float32)
    """
    # ---------------------------------------------------------------
    # Unpack arguments
    # ---------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    bs, L, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    eps = 1e-5

    # ---------------------------------------------------------------
    # 1️⃣ Input LayerNorm (Triton) – operates on [bs*L*L, dim]
    # ---------------------------------------------------------------
    x_flat = input_tensor.reshape(bs * L * L, dim).contiguous()
    norm_w = weights["norm.weight"].to(device, dtype).contiguous()
    norm_b = weights["norm.bias"].to(device, dtype).contiguous()
    x_norm_flat = triton_layernorm(x_flat, norm_w, norm_b, eps=eps)
    x_norm = x_norm_flat.view(bs, L, L, dim)          # shape [B, L, L, D]

    # ---------------------------------------------------------------
    # 2️⃣ Cast to FP16 for the heavy compute
    # ---------------------------------------------------------------
    x_h = x_norm.to(torch.float16)

    # ---------------------------------------------------------------
    # 3️⃣ Fused linear projections:
    #     - left_proj + right_proj  → weight (2*H, D)
    #     - left_gate, right_gate, out_gate → weight (3*H, D)
    # ---------------------------------------------------------------
    # Prepare FP16 weights
    left_proj_w = weights["left_proj.weight"].to(device, torch.float16).contiguous()
    right_proj_w = weights["right_proj.weight"].to(device, torch.float16).contiguous()
    left_gate_w = weights["left_gate.weight"].to(device, torch.float16).contiguous()
    right_gate_w = weights["right_gate.weight"].to(device, torch.float16).contiguous()
    out_gate_w = weights["out_gate.weight"].to(device, torch.float16).contiguous()

    # Combine projection weights (2*H) and gate weights (3*H)
    proj_weights = torch.cat([left_proj_w, right_proj_w], dim=0)          # (2*H, D)
    gate_weights = torch.cat([left_gate_w, right_gate_w, out_gate_w], dim=0)  # (3*H, D)

    # Flattened input for linear layers
    x_h_flat = x_h.reshape(bs * L * L, dim)

    # One GEMM for the two projections
    proj_out = F.linear(x_h_flat, proj_weights)   # (M, 2*H)
    # One GEMM for the three gates
    gate_out = F.linear(x_h_flat, gate_weights)   # (M, 3*H)

    # ---------------------------------------------------------------
    # 4️⃣ Reshape & split the fused results
    # ---------------------------------------------------------------
    # Projections
    left = proj_out[:, :hidden_dim].view(bs, L, L, hidden_dim)
    right = proj_out[:, hidden_dim:2 * hidden_dim].view(bs, L, L, hidden_dim)

    # Gates (apply sigmoid)
    left_gate = torch.sigmoid(gate_out[:, :hidden_dim].view(bs, L, L, hidden_dim))
    right_gate = torch.sigmoid(gate_out[:, hidden_dim:2 * hidden_dim].view(bs, L, L, hidden_dim))
    out_gate = torch.sigmoid(gate_out[:, 2 * hidden_dim:3 * hidden_dim].view(bs, L, L, hidden_dim))

    # ---------------------------------------------------------------
    # 5️⃣ Optional mask (broadcast over hidden dimension)
    # ---------------------------------------------------------------
    if not nomask and mask is not None:
        mask_h = mask.to(device, torch.float16).unsqueeze(-1)   # [B, L, L, 1]
        left = left * mask_h
        right = right * mask_h

    # ---------------------------------------------------------------
    # 6️⃣ Apply the multiplicative gates
    # ---------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ---------------------------------------------------------------
    # 7️⃣ Batched GEMM (the N³ contraction)
    # ---------------------------------------------------------------
    # Bring hidden dimension into the batch dimension: [B, H, L, L]
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, H, L, L)
    right = right.permute(0, 3, 1, 2).contiguous() # (B, H, L, L)

    left_flat = left.view(bs * hidden_dim, L, L)   # (B*H, L, L)
    right_flat = right.view(bs * hidden_dim, L, L)

    # out_flat[b*h] = left_flat[b*h] @ right_flat[b*h].T
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # Reshape back to [B, L, L, H]
    out = out_flat.view(bs, hidden_dim, L, L).permute(0, 2, 3, 1).contiguous()

    # ---------------------------------------------------------------
    # 8️⃣ Output LayerNorm (Triton) over the hidden dimension
    # ---------------------------------------------------------------
    out_flat2 = out.reshape(bs * L * L, hidden_dim).contiguous()
    out_norm_w = weights["to_out_norm.weight"].to(device, dtype).contiguous()
    out_norm_b = weights["to_out_norm.bias"].to(device, dtype).contiguous()
    out_norm_flat = triton_layernorm(out_flat2, out_norm_w, out_norm_b, eps=eps)
    out_norm = out_norm_flat.view(bs, L, L, hidden_dim)

    # ---------------------------------------------------------------
    # 9️⃣ Apply the final gate and the output projection (back to dim)
    # ---------------------------------------------------------------
    # Convert to float32 for the final matmul (as in the reference code)
    out_norm_fp32 = out_norm.to(torch.float32)
    out_gate_fp32 = out_gate.to(torch.float32)

    out = out_norm_fp32 * out_gate_fp32    # [B, L, L, H]

    to_out_w = weights["to_out.weight"].to(device, torch.float32).contiguous()
    out = F.linear(out, to_out_w)          # [B, L, L, D] (float32)

    return out
```
