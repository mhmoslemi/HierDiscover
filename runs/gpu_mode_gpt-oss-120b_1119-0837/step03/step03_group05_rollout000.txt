"""
Highly‑optimized Triton‑augmented forward pass of the “outgoing” TriMul
operator (AlphaFold‑3).  The implementation fuses the two LayerNorms into
custom Triton kernels while keeping the bulk of the linear algebra in
half‑precision Torch operations (cuBLAS Tensor‑cores).  No backward pass
or gradient storage is required.

Algorithm
---------
1.  Row‑wise LayerNorm over the last dimension *dim* of the 4‑D input
    tensor (shape [B, N, N, dim]) – implemented in Triton.
2.  Cast the normalized tensor to FP16 and apply the four bias‑free linear
    projections (left, right, left_gate, right_gate, out_gate).  The three
    gate projections are passed through a sigmoid.
3.  If a mask is supplied, broadcast‑multiply it with the left/right
    projections.
4.  Fuse the per‑head gating (element‑wise) with the projected tensors
    and perform a batched matrix multiplication:
        out[b, i, j, h] = Σₖ left[b, i, k, h] * right[b, j, k, h]
    (this is a GEMM per hidden dimension, executed via Torch’s
    ``torch.matmul`` which maps to cuBLAS Tensor‑cores).
5.  Row‑wise LayerNorm over the hidden dimension *hidden_dim* of the
    result – another Triton kernel.
6.  Multiply the normalized tensor by the output‑gate and apply the final
    linear projection back to the original dimension *dim*.

All heavy reductions are performed in FP32 for numerical stability,
while the dense matrix work stays in FP16.  The final output is returned
as ``torch.float32``.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton LayerNorm – two‑pass (mean/var → affine) implementation
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,                # *ptr to input  [M, N]
    y_ptr,                # *ptr to output [M, N]
    weight_ptr,           # *ptr to weight [N]
    bias_ptr,             # *ptr to bias   [N]
    M,                    # # rows   (M = batch * seq_len * seq_len)
    N,                    # # columns (feature dimension)
    eps,                  # epsilon for numerical stability
    stride_xm, stride_xn,  # strides for input  (row, col)
    stride_ym, stride_yn,  # strides for output (row, col)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
):
    pid = tl.program_id(0)

    # ------------------------------------------------------------------
    # Row offset for this block
    # ------------------------------------------------------------------
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # ------------------------------------------------------------------
    # First pass: compute mean & variance of each row
    # ------------------------------------------------------------------
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    # Number of column tiles needed to cover N
    num_tiles = tl.cdiv(N, BLOCK_N)
    for tile in range(num_tiles):
        offs_n = tl.arange(0, BLOCK_N) + tile * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        sum_val += tl.sum(x, axis=1)
        sum_sq  += tl.sum(x * x, axis=1)

    mean = sum_val / N
    var  = sum_sq / N - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # ------------------------------------------------------------------
    # Second pass: normalize, apply affine transform and store
    # ------------------------------------------------------------------
    for tile in range(num_tiles):
        offs_n = tl.arange(0, BLOCK_N) + tile * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        # Normalization (promoted to fp32 internally)
        norm_x = (x - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_n, mask=mask_n, other=0.0)
        b = tl.load(bias_ptr   + offs_n, mask=mask_n, other=0.0)

        y = norm_x * w[None, :] + b[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn,
            y,
            mask=mask_m[:, None] & mask_n[None, :],
        )

def triton_layernorm(x: torch.Tensor,
                     weight: torch.Tensor,
                     bias: torch.Tensor,
                     eps: float = 1e-5) -> torch.Tensor:
    """
    Triton‑based LayerNorm over the last dimension.
    ``x`` must be a 2‑D tensor of shape [M, N].  The result has the same dtype
    as ``weight``/``bias`` (typically fp32).  ``x`` may be fp16 – the kernel
    accumulates in fp32.
    """
    assert x.dim() == 2, "triton_layernorm expects a 2‑D tensor"
    M, N = x.shape
    y = torch.empty_like(x)

    # Ensure tensors are contiguous for reliable stride handling
    x = x.contiguous()
    weight = weight.contiguous()
    bias = bias.contiguous()

    stride_xm, stride_xn = x.stride()
    stride_ym, stride_yn = y.stride()

    # Grid: one program per row block
    grid = lambda meta: (triton.cdiv(M, meta["BLOCK_M"]),)

    layernorm_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        weight_ptr=weight,
        bias_ptr=bias,
        M=M,
        N=N,
        eps=eps,
        stride_xm=stride_xm,
        stride_xn=stride_xn,
        stride_ym=stride_ym,
        stride_yn=stride_yn,
        BLOCK_M=256,   # tuned for large M (≈ B·N²)
        BLOCK_N=64,    # columns per tile (good for fp16 & fp32)
    )
    return y

# ----------------------------------------------------------------------
# Entry‑point used by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor   shape [B, L, L, D] (float32)
        - mask         : torch.Tensor   shape [B, L, L] (float32/bool) or ``None``
        - weights      : dict of torch tensors (model parameters)
        - config       : dict with keys ``dim``, ``hidden_dim`` and optional ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, L, L, D] (float32)
    """
    # ------------------------------------------------------------------
    # 0️⃣ Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype      # expected fp32

    bs, L, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (Triton) over the channel dimension ``dim``
    # ------------------------------------------------------------------
    # Collapse the first three axes so that the kernel sees a (M, dim) matrix
    M = bs * L * L
    x_flat = input_tensor.reshape(M, dim).contiguous()
    norm_w = weights["norm.weight"].to(device, dtype).contiguous()
    norm_b = weights["norm.bias"].to(device, dtype).contiguous()
    x_norm_flat = triton_layernorm(x_flat, norm_w, norm_b, eps=eps)
    x_norm = x_norm_flat.view(bs, L, L, dim)               # [B, L, L, D] (fp32)

    # ------------------------------------------------------------------
    # 2️⃣ Cast to FP16 for the heavy linear work
    # ------------------------------------------------------------------
    x_norm_h = x_norm.to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Linear projections (bias‑free) : D → hidden_dim
    # ------------------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"].to(device, torch.float16).contiguous()
    right_proj_w = weights["right_proj.weight"].to(device, torch.float16).contiguous()

    left  = F.linear(x_norm_h, left_proj_w)   # [B, L, L, H] (fp16)
    right = F.linear(x_norm_h, right_proj_w)

    # ------------------------------------------------------------------
    # 4️⃣ Optional mask (broadcast over the hidden dimension)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # mask assumed to be 0/1 (float or bool)
        mask_h = mask.to(device, torch.float16).unsqueeze(-1)   # [B, L, L, 1]
        left  = left  * mask_h
        right = right * mask_h

    # ------------------------------------------------------------------
    # 5️⃣ Gates (sigmoid of bias‑free linear projections)
    # ------------------------------------------------------------------
    left_gate_w  = weights["left_gate.weight"].to(device, torch.float16).contiguous()
    right_gate_w = weights["right_gate.weight"].to(device, torch.float16).contiguous()
    out_gate_w   = weights["out_gate.weight"].to(device, torch.float16).contiguous()

    left_gate  = torch.sigmoid(F.linear(x_norm_h, left_gate_w))   # [B, L, L, H]
    right_gate = torch.sigmoid(F.linear(x_norm_h, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x_norm_h, out_gate_w))

    left  = left  * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6️⃣ Batched matrix multiplication per hidden dimension
    #     out[b, i, j, h] = Σₖ left[b, i, k, h] * right[b, j, k, h]
    # ------------------------------------------------------------------
    # Rearrange to (B, H, N, N) so that each head becomes an independent batch.
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N] (fp16)
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    # Tensor‑core GEMM via cuBLAS
    out_perm = torch.matmul(left_perm, right_perm.transpose(-2, -1))  # [B, H, N, N] (fp16)

    # Restore original order → [B, N, N, H]
    out_h = out_perm.permute(0, 2, 3, 1).contiguous()   # (fp16)

    # ------------------------------------------------------------------
    # 7️⃣ Output LayerNorm over hidden dimension (H) – Triton
    # ------------------------------------------------------------------
    out_M = bs * L * L
    out_flat = out_h.view(out_M, hidden_dim).contiguous()   # [M, H] (fp16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device, dtype).contiguous()
    to_out_norm_b = weights["to_out_norm.bias"].to(device, dtype).contiguous()
    out_norm_flat = triton_layernorm(out_flat, to_out_norm_w, to_out_norm_b, eps=eps)
    out_norm = out_norm_flat.view(bs, L, L, hidden_dim)   # [B, L, L, H] (fp32)

    # ------------------------------------------------------------------
    # 8️⃣ Apply output gate and final linear projection H → D
    # ------------------------------------------------------------------
    out_norm = out_norm * out_gate.to(dtype)      # element‑wise gate
    to_out_w = weights["to_out.weight"].to(device, dtype).contiguous()
    out = F.linear(out_norm, to_out_w)           # [B, L, L, D] (fp32)

    return out
