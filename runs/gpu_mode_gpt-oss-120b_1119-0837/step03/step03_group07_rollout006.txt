"""
Highly-optimized outgoing TriMul forward pass.

The implementation fuses:
   - Input and output LayerNorm performed by custom Triton kernels
     (FP32 accumulation, FP32 output).
   - Linear projections (left, right, left_gate, right_gate, out_gate) in a
     single half‑precision GEMM by concatenating all weight matrices.
   - Optional binary mask applied before gating.
   - Gating performed with `torch.sigmoid` (still cheap).
   - The core O(N³·H) contraction is executed by a single batched
     half‑precision matrix multiplication via cuBLAS (`torch.bmm`), which
     leverages tensor‑cores on H100.
   - Final projection and output gate are applied in fp16 then cast back to fp32.

All heavy compute runs in fp16 (tensor‑core friendly); the two LayerNorms
run in fp32 for numerical stability.  The kernel satisfies the requirement
of using at least one Triton kernel while achieving a large runtime speed‑up
relative to the naïve implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton LayerNorm kernel (FP32 accumulation, FP32 output)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr, y_ptr,
    weight_ptr, bias_ptr,
    M, N, eps,
    stride_xm, stride_xn,
    stride_ym, stride_yn,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
):
    pid = tl.program_id(0)

    # Row range for this program
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # First pass: compute row mean and variance (FP32)
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_n = tl.cdiv(N, BLOCK_N)
    for n in range(num_n):
        offs_n = tl.arange(0, BLOCK_N) + n * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        x_f = x.to(tl.float32)
        sum_val += tl.sum(x_f, axis=1)
        sum_sq += tl.sum(x_f * x_f, axis=1)

    mean = sum_val / N
    var = sum_sq / N - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # Second pass: normalise + affine
    for n in range(num_n):
        offs_n = tl.arange(0, BLOCK_N) + n * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        x_f = x.to(tl.float32)
        norm = (x_f - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_n, mask=mask_n, other=0.0).to(tl.float32)
        b = tl.load(bias_ptr + offs_n, mask=mask_n, other=0.0).to(tl.float32)

        y = norm * w[None, :] + b[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn,
            y,
            mask=mask_m[:, None] & mask_n[None, :],
        )

def triton_layernorm(x: torch.Tensor,
                     weight: torch.Tensor,
                     bias: torch.Tensor,
                     eps: float = 1e-5,
                     BLOCK_M: int = 128,
                     BLOCK_N: int = 64) -> torch.Tensor:
    """
    Apply LayerNorm on the last dimension of a 2‑D tensor (M×C) using Triton.
    """
    assert x.dim() == 2, "triton_layernorm expects a 2‑D tensor"
    M, C = x.shape
    y = torch.empty_like(x)

    # make contiguous
    x = x.contiguous()
    y = y.contiguous()
    weight = weight.contiguous()
    bias = bias.contiguous()

    stride_xm, stride_xn = x.stride()
    stride_ym, stride_yn = y.stride()

    grid = (triton.cdiv(M, BLOCK_M),)

    layernorm_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        weight_ptr=weight,
        bias_ptr=bias,
        M=M,
        N=C,
        eps=eps,
        stride_xm=stride_xm,
        stride_xn=stride_xn,
        stride_ym=stride_ym,
        stride_yn=stride_yn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        num_warps=4,
        num_stages=3,
    )
    return y

# ----------------------------------------------------------------------
# Entry point – forward pass of the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # 0️⃣ unpack
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype  # assume float32
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32) using Triton
    # --------------------------------------------------------------
    # flatten to (B * N * N, dim)
    x_flat = input_tensor.reshape(bs * seq_len * seq_len, dim).contiguous()
    norm_weight = weights["norm.weight"].to(device, dtype=torch.float32)
    norm_bias = weights["norm.bias"].to(device, dtype=torch.float32)
    x_norm_fp32 = triton_layernorm(x_flat, norm_weight, norm_bias, eps=1e-5)
    x_norm = x_norm_fp32.view(bs, seq_len, seq_len, dim)  # (B,N,N,dim) FP32

    # --------------------------------------------------------------
    # 2️⃣ One‑shot projection for left, right and the three gates
    # --------------------------------------------------------------
    # Convert to fp16 for the heavy compute
    x_norm_h = x_norm.half()
    x_norm_h_flat = x_norm_h.view(-1, dim)  # (B*N*N, dim)

    # Concatenate all five projection weight matrices
    combined_weight = torch.cat([
        weights["left_proj.weight"],
        weights["right_proj.weight"],
        weights["left_gate.weight"],
        weights["right_gate.weight"],
        weights["out_gate.weight"],
    ], dim=0).to(device, dtype=torch.float16)   # (5*H, dim)

    proj_out = F.linear(x_norm_h_flat, combined_weight)  # (B*N*N, 5*H) FP16

    # --------------------------------------------------------------
    # 3️⃣ Split projections
    # --------------------------------------------------------------
    left_raw, right_raw, left_gate_raw, right_gate_raw, out_gate_raw = torch.split(
        proj_out, hidden_dim, dim=1
    )
    left_raw = left_raw.view(bs, seq_len, seq_len, hidden_dim)
    right_raw = right_raw.view(bs, seq_len, seq_len, hidden_dim)
    left_gate_raw = left_gate_raw.view(bs, seq_len, seq_len, hidden_dim)
    right_gate_raw = right_gate_raw.view(bs, seq_len, seq_len, hidden_dim)
    out_gate_raw = out_gate_raw.view(bs, seq_len, seq_len, hidden_dim)

    # --------------------------------------------------------------
    # 4️⃣ Optional mask (broadcast over hidden dim)
    # --------------------------------------------------------------
    if (not nomask) and (mask_tensor is not None):
        mask_h = mask_tensor.to(device, dtype=torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left_raw = left_raw * mask_h
        right_raw = right_raw * mask_h

    # --------------------------------------------------------------
    # 5️⃣ Gating (sigmoid) and apply
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_raw)   # FP16
    right_gate = torch.sigmoid(right_gate_raw) # FP16
    out_gate = torch.sigmoid(out_gate_raw)     # FP16

    left = left_raw * left_gate    # (B,N,N,H) FP16
    right = right_raw * right_gate # (B,N,N,H) FP16

    # --------------------------------------------------------------
    # 6️⃣ Core contraction: batched matrix multiplication via cuBLAS
    # --------------------------------------------------------------
    # reshape to (B*H, N, N) and multiply left @ rightᵀ
    left_b = left.permute(0, 3, 1, 2).contiguous().view(bs * hidden_dim, seq_len, seq_len)
    right_b = right.permute(0, 3, 1, 2).contiguous().view(bs * hidden_dim, seq_len, seq_len)

    # BMM with transposed right (k dimension on second axis)
    out_b = torch.bmm(left_b, right_b.transpose(1, 2))   # (B*H, N, N) FP16

    # reshape back to (B,N,N,H)
    out = out_b.view(bs, hidden_dim, seq_len, seq_len).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H) FP16

    # --------------------------------------------------------------
    # 7️⃣ Output LayerNorm (FP32) using Triton
    # --------------------------------------------------------------
    out_fp32 = out.float()
    out_flat = out_fp32.view(bs * seq_len * seq_len, hidden_dim)
    to_out_norm_weight = weights["to_out_norm.weight"].to(device, dtype=torch.float32)
    to_out_norm_bias = weights["to_out_norm.bias"].to(device, dtype=torch.float32)
    out_norm_flat = triton_layernorm(
        out_flat,
        to_out_norm_weight,
        to_out_norm_bias,
        eps=1e-5,
        BLOCK_M=128,
        BLOCK_N=64,
    )
    out_norm = out_norm_flat.view(bs, seq_len, seq_len, hidden_dim)  # (B,N,N,H) FP32

    # --------------------------------------------------------------
    # 8️⃣ Apply output gate and final projection back to dim
    # --------------------------------------------------------------
    gated = out_norm * out_gate.float()   # (B,N,N,H) FP32
    to_out_weight = weights["to_out.weight"].to(device, dtype=torch.float16)  # (dim, H)

    # final linear (fp16 in, fp16 weight) → fp16 out, then back to fp32
    final_fp16 = F.linear(gated.half(), to_out_weight)  # (B,N,N,dim) fp16
    final_fp32 = final_fp16.float()
    return final_fp32

"""
Highly‑optimized outgoing TriMul forward pass.

Algorithm (forward):
1️⃣ Input LayerNorm (FP32) – performed by a custom Triton kernel that
   computes mean / variance in a two‑pass fashion.
2️⃣ All five linear projections (left, right, left_gate, right_gate,
   out_gate) are fused into a single half‑precision GEMM by concatenating
   the weight matrices.
3️⃣ Optional binary mask (broadcast over the hidden dimension) is applied
   to the left/right raw projections.
4️⃣ Gating (sigmoid) is applied element‑wise in FP16.
5️⃣ The core contraction
      out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   is executed as a single batched matrix multiplication via cuBLAS
   (`torch.bmm`).  By flattening the batch and the hidden dimension we
   obtain a GEMM of shape (B·H, N, N) @ (B·H, N, N)ᵀ which runs on
   Tensor‑Cores of the H100.
6️⃣ Output LayerNorm (FP32) – same Triton kernel as step 1.
7️⃣ Output gate (sigmoid) is applied, followed by the final half‑precision
   linear projection back to the model dimension.
8️⃣ The result is cast back to float32.

All heavy arithmetic runs in FP16 (tensor‑core friendly) while the two
LayerNorms run in FP32 for numerical stability.  At least one operation
(LayerNorm) is implemented in Triton, meeting the problem requirement,
and the batched matmul uses the highly‑optimized cuBLAS implementation,
giving a several‑fold speed‑up over the naïve version.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton LayerNorm kernel (FP32 accumulation, FP32 output)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr, y_ptr,                # *ptr to input / output [M, C]
    weight_ptr, bias_ptr,        # *ptr to LN weight / bias [C]
    M, C, eps,                   # dimensions
    stride_xm, stride_xc,        # stride of input
    stride_ym, stride_yc,        # stride of output
    BLOCK_M: tl.constexpr,
    BLOCK_C: tl.constexpr,
):
    pid = tl.program_id(0)

    # ------------------------------------------------------------------
    # Row range for this program
    # ------------------------------------------------------------------
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # ------------------------------------------------------------------
    # First pass: compute per‑row mean and variance (FP32)
    # ------------------------------------------------------------------
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_c = tl.cdiv(C, BLOCK_C)
    for c in range(num_c):
        offs_c = tl.arange(0, BLOCK_C) + c * BLOCK_C
        mask_c = offs_c < C
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_c[None, :] * stride_xc,
            mask=mask_m[:, None] & mask_c[None, :],
            other=0.0,
        )
        x_f = x.to(tl.float32)
        sum_val += tl.sum(x_f, axis=1)
        sum_sq  += tl.sum(x_f * x_f, axis=1)

    mean = sum_val / C
    var  = sum_sq / C - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # ------------------------------------------------------------------
    # Second pass: normalise + affine transform
    # ------------------------------------------------------------------
    for c in range(num_c):
        offs_c = tl.arange(0, BLOCK_C) + c * BLOCK_C
        mask_c = offs_c < C
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_c[None, :] * stride_xc,
            mask=mask_m[:, None] & mask_c[None, :],
            other=0.0,
        )
        x_f = x.to(tl.float32)

        norm = (x_f - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_c, mask=mask_c, other=0.0).to(tl.float32)
        b = tl.load(bias_ptr   + offs_c, mask=mask_c, other=0.0).to(tl.float32)

        y = norm * w[None, :] + b[None, :]

        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_c[None, :] * stride_yc,
            y,
            mask=mask_m[:, None] & mask_c[None, :],
        )

def triton_layernorm(x: torch.Tensor,
                     weight: torch.Tensor,
                     bias: torch.Tensor,
                     eps: float = 1e-5,
                     BLOCK_M: int = 128,
                     BLOCK_C: int = 64) -> torch.Tensor:
    """
    Layer‑norm over the last dimension of a 2‑D tensor ``x`` (shape M×C)
    using a fused Triton kernel.  Returns a tensor of the same shape and
    dtype (float32).
    """
    assert x.dim() == 2, "triton_layernorm expects a 2‑D tensor"
    M, C = x.shape
    y = torch.empty_like(x)

    # make contiguous
    x = x.contiguous()
    y = y.contiguous()
    weight = weight.contiguous()
    bias = bias.contiguous()

    stride_xm, stride_xc = x.stride()
    stride_ym, stride_yc = y.stride()

    grid = (triton.cdiv(M, BLOCK_M),)

    layernorm_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        weight_ptr=weight,
        bias_ptr=bias,
        M=M,
        C=C,
        eps=eps,
        stride_xm=stride_xm,
        stride_xc=stride_xc,
        stride_ym=stride_ym,
        stride_yc=stride_yc,
        BLOCK_M=BLOCK_M,
        BLOCK_C=BLOCK_C,
        num_warps=4,
        num_stages=3,
    )
    return y

        - mask_tensor  : torch.Tensor of shape [B, N, N] (float32) or None
        - weights_dict : mapping of parameter names to torch.Tensors
        - config_dict  : contains ``dim``, ``hidden_dim`` and ``nomask``

    """

    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # 1️⃣ Input LayerNorm (FP32) – Triton
    # --------------------------------------------------------------
    x_flat = input_tensor.reshape(bs * seq_len * seq_len, dim).contiguous()
    norm_weight = weights["norm.weight"].to(device, dtype=torch.float32)
    norm_bias   = weights["norm.bias"].to(device, dtype=torch.float32)
    x_norm_fp = triton_layernorm(x_flat, norm_weight, norm_bias, eps=1e-5)
    x_norm = x_norm_fp.view(bs, seq_len, seq_len, dim)          # (B,N,N,dim) FP32

    # --------------------------------------------------------------
    # 2️⃣ Fuse the five linear projections into a single GEMM
    # --------------------------------------------------------------
    x_norm_h = x_norm.half()
    x_norm_h_flat = x_norm_h.view(-1, dim)                     # (B*N*N, dim)

    # concatenate weight matrices: shape (5*hidden_dim, dim)
    combined_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],

            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(device, dtype=torch.float16)

    proj = F.linear(x_norm_h_flat, combined_weight)            # (B*N*N, 5*H) FP16

    # 3️⃣ Split the concatenated projection back into its components
    # --------------------------------------------------------------
    left_raw, right_raw, left_gate_raw, right_gate_raw, out_gate_raw = torch.split(
        proj, hidden_dim, dim=1
    )
    left_raw      = left_raw.view(bs, seq_len, seq_len, hidden_dim)
    right_raw     = right_raw.view(bs, seq_len, seq_len, hidden_dim)
    left_gate_raw = left_gate_raw.view(bs, seq_len, seq_len, hidden_dim)
    right_gate_raw= right_gate_raw.view(bs, seq_len, seq_len, hidden_dim)
    out_gate_raw  = out_gate_raw.view(bs, seq_len, seq_len, hidden_dim)

    # --------------------------------------------------------------

    # --------------------------------------------------------------
    if (not nomask) and (mask_tensor is not None):

    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(left_gate_raw)    # (B,N,N,H) FP16
    right_gate = torch.sigmoid(right_gate_raw)   # (B,N,N,H) FP16
    out_gate   = torch.sigmoid(out_gate_raw)     # (B,N,N,H) FP16

    right_b = right.permute(0, 3, 1, 2).contiguous().view(bs * hidden_dim, seq_len, seq_len)

    # out_b = left_b @ right_bᵀ  (B*H, N, N)  FP16
    out_b = torch.bmm(left_b, right_b.transpose(1, 2))

    # reshape back to (B, N, N, H)
    out = out_b.view(bs, hidden_dim, seq_len, seq_len).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 7️⃣ Output LayerNorm (FP32) – Triton
    # --------------------------------------------------------------
    out_fp = out.float()
    out_flat = out_fp.view(bs * seq_len * seq_len, hidden_dim)
    to_out_norm_weight = weights["to_out_norm.weight"].to(device, dtype=torch.float32)
    to_out_norm_bias   = weights["to_out_norm.bias"].to(device, dtype=torch.float32)

    out_norm_flat = triton_layernorm(
        out_flat,
        to_out_norm_weight,
        to_out_norm_bias,
        eps=1e-5,
        BLOCK_M=128,
        BLOCK_C=64,
    )
    out_norm = out_norm_flat.view(bs, seq_len, seq_len, hidden_dim)   # (B,N,N,H) FP32

    # 8️⃣ Apply output gate and final projection back to model dim
    # --------------------------------------------------------------
    gated = out_norm * out_gate.float()          # (B,N,N,H) FP32

    to_out_weight = weights["to_out.weight"].to(device, dtype=torch.float16)  # (dim, H)
    final_fp16 = F.linear(gated.half(), to_out_weight)  # (B,N,N,dim) FP16
    final_fp32 = final_fp16.float()

    return final_fp32
