@triton.jit
def bmmt_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    M, N, K,
    stride_left_batch, stride_left_m, stride_left_k,
    stride_right_batch, stride_right_k, stride_right_n,
    stride_out_batch, stride_out_m, stride_out_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # base addresses
    left_base = left_ptr + batch_id * stride_left_batch
    right_base = right_ptr + batch_id * stride_right_batch
    out_base = out_ptr + batch_id * stride_out_batch

    # Loop over k
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)  # accumulate FP32

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)

        # mask for dimensions
        mask_m = offs_m < M
        mask_n = offs_n < N
        mask_k = offs_k < K

        # Load left tile (M x K)
        a = tl.load(left_base + offs_m[:, None] * stride_left_m + offs_k[None, :] * stride_left_k,
                    mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # Load right tile (N x K) for transposed version: want B_T into shape (K,N) or (N,K)? We need right_T block of K x N.

        # We want B_T block of shape (K, N). B_T[k, n] = right[n, k] (original). So we can load as:
        b = tl.load(right_base + offs_k[:, None] * stride_right_k + offs_n[None, :] * stride_right_n,
                    mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # Now a shape (M, K), b shape (K, N)
        acc += tl.dot(a, b)  # result shape (M,N) accumulate

    # Store output
    # Cast back to same dtype as left/right, assume fp16 input and fp16 output
    # In many kernels, they use to_float32 accumulation then convert to fp16
    c = acc.to(tl.float16)

    tl.store(out_base + offs_m[:, None] * stride_out_m + offs_n[None, :] * stride_out_n,
             c,
             mask=mask_m[:, None] & mask_n[None, :])

"""
TriMul – “outgoing’’ kernel (AlphaFold‑3 style)

Algorithm
---------
1️⃣  Layer‑Norm the 4‑D input (B,N,N,D) with a small Triton kernel.
2️⃣  Cast to FP16 and fuse the five linear projections (left/right & three gates)
    into a single GEMM → one contiguous tensor of shape (B,N,N,5·H).
3️⃣  Slice, apply the three sigmoid gates and (optional) pair‑wise mask.
4️⃣  Compute the heavy O(N³·H) contraction
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    with a custom Triton batched‑matmul kernel that works directly on the
    (B,N,N,H) layout – no costly permute/contiguous copies.
5️⃣  Layer‑Norm the result (still FP16) with the same Triton kernel.
6️⃣  Apply the output gate, a final linear projection back to D and return
    a float‑32 tensor.

Only the two LayerNorms and the contraction are implemented in Triton;
all dense linear ops use highly‑optimized PyTorch FP16 GEMM kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 2‑pass LayerNorm (operates on the last dimension)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,               # *ptr to input  [M, N]
    y_ptr,               # *ptr to output [M, N]
    w_ptr,               # *ptr to weight [N]
    b_ptr,               # *ptr to bias   [N]
    M, N, eps,
    stride_xm, stride_xn,
    stride_ym, stride_yn,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
):
    pid = tl.program_id(0)

    # ------------------------------------------------------------------
    # row range for this program
    # ------------------------------------------------------------------
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # ------------------------------------------------------------------
    # first pass: compute per‑row mean & var (FP32)
    # ------------------------------------------------------------------
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_N)
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        xv = tl.cast(x, tl.float32)
        sum_val += tl.sum(xv, axis=1)
        sum_sq  += tl.sum(xv * xv, axis=1)

    mean = sum_val / N
    var  = sum_sq / N - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # ------------------------------------------------------------------
    # second pass: write normalized + affine transformed values
    # ------------------------------------------------------------------
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        xv = tl.cast(x, tl.float32)
        norm = (xv - mean[:, None]) * rstd[:, None]

        w = tl.cast(tl.load(w_ptr + offs_n, mask=mask_n, other=0.0), tl.float32)
        b = tl.cast(tl.load(b_ptr + offs_n, mask=mask_n, other=0.0), tl.float32)

        y = norm * w[None, :] + b[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn,
            tl.cast(y, x.dtype),
            mask=mask_m[:, None] & mask_n[None, :],
        )


def triton_layernorm(x: torch.Tensor,
                     weight: torch.Tensor,
                     bias: torch.Tensor,
                     eps: float = 1e-5) -> torch.Tensor:
    """LayerNorm over the last dimension using the Triton kernel."""
    assert x.dim() == 2, "triton_layernorm expects a 2‑D tensor"
    M, N = x.shape
    y = torch.empty_like(x)

    # force contiguous layout for correct stride handling
    x = x.contiguous()
    y = y.contiguous()
    weight = weight.contiguous()
    bias = bias.contiguous()

    stride_xm, stride_xn = x.stride()
    stride_ym, stride_yn = y.stride()

    grid = lambda meta: (triton.cdiv(M, meta["BLOCK_M"]),)

    layernorm_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        w_ptr=weight,
        b_ptr=bias,
        M=M,
        N=N,
        eps=eps,
        stride_xm=stride_xm,
        stride_xn=stride_xn,
        stride_ym=stride_ym,
        stride_yn=stride_yn,
        BLOCK_M=128,
        BLOCK_N=64,
    )
    return y


# ----------------------------------------------------------------------
# Batched contraction kernel (outgoing TriMul)
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    # Left strides
    left_batch_stride, left_i_stride, left_k_stride, left_h_stride,
    # Right strides
    right_batch_stride, right_i_stride, right_k_stride, right_h_stride,
    # Output strides
    out_batch_stride, out_i_stride, out_j_stride, out_h_stride,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Computes, for each (b,h) pair:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

    The three program IDs index:
        0 – combined batch*head dimension (B*H)
        1 – row block (i)
        2 – column block (j)
    """
    bh_id = tl.program_id(0)          # 0 … B*H-1
    pid_i = tl.program_id(1)          # row block
    pid_j = tl.program_id(2)          # column block

    # decode batch and head from bh_id
    b = bh_id // H
    h = bh_id % H

    # base pointers for this (b,h)
    left_base  = left_ptr  + b * left_batch_stride  + h * left_h_stride
    right_base = right_ptr + b * right_batch_stride + h * right_h_stride
    out_base   = out_ptr   + b * out_batch_stride   + h * out_h_stride

    # offsets for the current tile
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32 for numeric stability
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the reduction dimension k
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # load A = left[b,i,k,h]; shape (BLOCK_M, BLOCK_K)
        a = tl.load(
            left_base + offs_i[:, None] * left_i_stride
                     + offs_k[None, :] * left_k_stride,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )

        # load Bᵀ = right[b,j,k,h] transposed, i.e. Bᵀ[k,j] = right[b,j,k,h]
        # shape (BLOCK_K, BLOCK_N)
        b = tl.load(
            right_base + offs_k[:, None] * right_k_stride
                      + offs_j[None, :] * right_i_stride,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )

        # accumulate (A @ Bᵀ)
        acc += tl.dot(a, b)

    # write back the result (cast back to the input dtype – fp16 here)
    out_tile = acc.to(tl.float16)
    tl.store(
        out_base + offs_i[:, None] * out_i_stride + offs_j[None, :] * out_j_stride,
        out_tile,
        mask=mask_i[:, None] & mask_j[None, :],
    )

    # ------------------------------------------------------------------

    nomask = config.get("nomask", False)
    eps = 1e-5

    # ------------------------------------------------------------------

    norm_b = weights["norm.bias"].to(device, dtype_fp32)

    x_norm = x_norm_flat.view(bs, seq_len, seq_len, dim)   # (B,N,N,D) FP32

    # ------------------------------------------------------------------
    x_norm_f16 = x_norm.to(torch.float16)
    x_flat_f16 = x_norm_f16.reshape(M, dim)   # (M, D) FP16

    # ------------------------------------------------------------------
    # 3️⃣ Fuse all projection + gate linear layers into a single GEMM
    # ------------------------------------------------------------------
    #   left_proj   : (H, D)
    #   left_gate   : (H, D)
    #   right_proj  : (H, D)
    #   right_gate  : (H, D)
    #   out_gate    : (H, D)
    left_proj_w   = weights["left_proj.weight"]
    left_gate_w   = weights["left_gate.weight"]
    right_proj_w  = weights["right_proj.weight"]
    right_gate_w  = weights["right_gate.weight"]
    out_gate_w    = weights["out_gate.weight"]

        [left_proj_w, left_gate_w,
         right_proj_w, right_gate_w,
         out_gate_w],
        dim=0,
    ).to(device, torch.float16)                 # (5*H, D) FP16

    proj_gate_flat = F.linear(x_flat_f16, fused_w)   # (M, 5*H) FP16
    proj_gate = proj_gate_flat.view(bs, seq_len, seq_len, 5 * hidden_dim)   # (B,N,N,5H)

    H = hidden_dim
    left_proj   = proj_gate[..., 0:H]                # (B,N,N,H)
    left_gate_r = proj_gate[..., H:2*H]
    right_proj  = proj_gate[..., 2*H:3*H]

    out_gate_r  = proj_gate[..., 4*H:5*H]

    left_gate = torch.sigmoid(left_gate_r)           # (B,N,N,H) FP16
    right_gate = torch.sigmoid(right_gate_r)
    out_gate  = torch.sigmoid(out_gate_r)

    # ------------------------------------------------------------------

    out_s_batch, out_s_i, out_s_j, out_s_h = out.stride()

    # Kernel launch grid: (B*H, ceil(N/BM), ceil(N/BN))
    grid = (
        bs * hidden_dim,
        triton.cdiv(seq_len, 64),   # BLOCK_M
        triton.cdiv(seq_len, 64),   # BLOCK_N
    )
    tri_mul_outgoing_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=bs,
        N=seq_len,
        H=hidden_dim,
        left_batch_stride=left_s_batch,
        left_i_stride=left_s_i,
        left_k_stride=left_s_k,
        left_h_stride=left_s_h,
        right_batch_stride=right_s_batch,
        right_i_stride=right_s_i,
        right_k_stride=right_s_k,
        right_h_stride=right_s_h,
        out_batch_stride=out_s_batch,
        out_i_stride=out_s_i,
        out_j_stride=out_s_j,
        out_h_stride=out_s_h,
        BLOCK_M=64,
        BLOCK_N=64,
        BLOCK_K=32,
    )

    # ------------------------------------------------------------------
    # 6️⃣ Output LayerNorm (Triton) – keep FP16 precision for speed
    # ------------------------------------------------------------------
    out_flat = out.reshape(M, hidden_dim)          # (B*N*N, H) FP16
    out_norm_w = weights["to_out_norm.weight"].to(device, dtype_fp32)
    out_norm_b = weights["to_out_norm.bias"].to(device, dtype_fp32)

    out_norm_flat = triton_layernorm(out_flat, out_norm_w, out_norm_b, eps=eps)
    out_norm = out_norm_flat.view(bs, seq_len, seq_len, hidden_dim)   # (B,N,N,H) FP16

    # ------------------------------------------------------------------
    # 7️⃣ Apply output gate and final projection back to model dim
    # ------------------------------------------------------------------
    gated = out_norm * out_gate                      # (B,N,N,H) FP16
    to_out_w = weights["to_out.weight"].to(device, torch.float16)   # (D, H) FP16
    out_fp16 = F.linear(gated, to_out_w)            # (B,N,N,D) FP16
    out_fp32 = out_fp16.to(torch.float32)           # return in FP32 as required

    return out_fp32
```
