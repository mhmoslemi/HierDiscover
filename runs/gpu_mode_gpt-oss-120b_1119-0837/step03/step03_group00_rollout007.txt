import torch.nn.functional as F
import triton
import triton.language as tl

# Custom layernorm kernels as provided (unchanged)

# Batched GEMM kernel
@triton.jit
def batched_gemm_fp16_kernel(
    a_ptr, b_ptr, c_ptr,
    batch_heads, M, N, K, H,
    stride_ab, stride_ai, stride_ak, stride_ah,
    stride_br, stride_bj, stride_bk, stride_bh,
    stride_cb, stride_ci, stride_cj, stride_ch,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Derive batch and head indices
    bh = pid_batch
    b = bh // H
    h = bh - b * H   # same as bh % H

    # Offsets for output tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Offsets for k dimension (reduction)
    offs_k = tl.arange(0, BLOCK_K)

    # Base pointers for A and B
    a_base = a_ptr + b * stride_ab + h * stride_ah
    b_base = b_ptr + b * stride_br + h * stride_bh

    a_ptrs = a_base + offs_m[:, None] * stride_ai + offs_k[None, :] * stride_ak
    b_ptrs = b_base + offs_k[:, None] * stride_bj + offs_n[None, :] * stride_bk

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k = tl.cdiv(K, BLOCK_K)
    for k_iter in range(num_k):
        cur_k = k_iter * BLOCK_K
        cur_offs_k = offs_k + cur_k
        mask_k = cur_offs_k < K

        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        acc += tl.dot(a, b)

        # advance pointers to next K-block
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bj

    # Write result
    c_base = c_ptr + b * stride_cb + h * stride_ch
    c_ptrs = c_base + offs_m[:, None] * stride_ci + offs_n[None, :] * stride_cj

    c = acc.to(tl.float16)
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])

"""
Highly‑optimized Triton implementation of the “outgoing” TriMul operator.

Algorithm (forward only)
------------------------
1️⃣ Layer‑Norm on the model dimension (float‑32) – custom 2‑pass Triton kernel.
2️⃣ Cast to FP16 for the heavy linear work.
3️⃣ Fuse the four linear projections (left/right + their gates) into a *single*
   GEMM: W_comb ∈ℝ^{4·H×D}.  Slice the result and apply sigmoid gates.
4️⃣ Apply the optional mask (broadcasted over the hidden dimension).
5️⃣ Compute the output gate (D → H) with a second GEMM + sigmoid.
6️⃣ Core contraction  out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h] .
   This is a batched half‑precision GEMM (B·H separate heads) implemented
   in Triton, avoiding the extra copy performed by torch.bmm.
7️⃣ Layer‑Norm on the hidden dimension (float‑16) – custom Triton kernel.
8️⃣ Apply the output gate and a final linear projection back to D.
9️⃣ Cast the result back to float‑32.

All heavy linear work stays in FP16 to exploit Tensor‑Core throughput.
Two custom Triton kernels are used (LayerNorm FP32, LayerNorm FP16) and a
custom batched GEMM kernel replaces the torch.bmm call, reducing memory
traffic and launch overhead.  The implementation works for any batch size
(1 or 2), sequence length up to 1024, model dimension D ∈ {128,384},
hidden dimension H ∈ {128,384,768} and respects optional masking.
"""

import triton

# ----------------------------------------------------------------------
# 2‑pass LayerNorm kernels (float32 → float32, and float16 → float16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp32_kernel(
    x_ptr, y_ptr, weight_ptr, bias_ptr,
    M, N, eps,
    stride_xm, stride_xn,
    stride_ym, stride_yn,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr
):
    pid = tl.program_id(0)

    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # --- first pass: mean / variance ------------------------------------
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)
    num_k = tl.cdiv(N, BLOCK_N)
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        sum_val += tl.sum(x, axis=1)
        sum_sq  += tl.sum(x * x, axis=1)

    mean = sum_val / N
    var  = sum_sq / N - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # --- second pass: write normalized values ---------------------------
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        norm_x = (x - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_n, mask=mask_n, other=0.0)
        b = tl.load(bias_ptr   + offs_n, mask=mask_n, other=0.0)

        y = norm_x * w[None, :] + b[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn,
            y,
            mask=mask_m[:, None] & mask_n[None, :],
        )

def triton_layernorm_fp32(x: torch.Tensor,
                          weight: torch.Tensor,
                          bias: torch.Tensor,
                          eps: float = 1e-5) -> torch.Tensor:
    """LayerNorm (float32) using a Triton kernel."""
    assert x.dim() == 2
    M, N = x.shape
    y = torch.empty_like(x)

    x = x.contiguous()
    y = y.contiguous()
    weight = weight.contiguous()
    bias = bias.contiguous()

    stride_xm, stride_xn = x.stride()
    stride_ym, stride_yn = y.stride()

    grid = lambda meta: (triton.cdiv(M, meta["BLOCK_M"]),)

    layernorm_fp32_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        weight_ptr=weight,
        bias_ptr=bias,
        M=M,
        N=N,
        eps=eps,
        stride_xm=stride_xm,
        stride_xn=stride_xn,
        stride_ym=stride_ym,
        stride_yn=stride_yn,
        BLOCK_M=128,
        BLOCK_N=64,
    )
    return y

# Batched GEMM kernel for the N³ contraction (FP16 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_fp16_kernel(
    a_ptr, b_ptr, c_ptr,
    batch_heads, M, N, K, H,
    stride_ab, stride_ai, stride_ak, stride_ah,
    stride_br, stride_bj, stride_bk, stride_bh,
    stride_cb, stride_ci, stride_cj, stride_ch,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """C[bh,i,j] = Σ_k A[bh,i,k] * B[bh,j,k]   (B = rightᵀ)."""
    pid_batch = tl.program_id(0)   # bh = batch * hidden_dim
    pid_m = tl.program_id(1)      # output row block (i)
    pid_n = tl.program_id(2)      # output col block (j)

    # recover batch and head indices (H is compile‑time constant)
    bh = pid_batch
    b = bh // H
    h = bh - b * H   # equivalent to bh % H

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    offs_k = tl.arange(0, BLOCK_K)

    # base pointers for this (b, h) slice
    a_base = a_ptr + b * stride_ab + h * stride_ah
    b_base = b_ptr + b * stride_br + h * stride_bh

    # pointers for the first k‑block
    a_ptrs = a_base + offs_m[:, None] * stride_ai + offs_k[None, :] * stride_ak
    b_ptrs = b_base + offs_k[:, None] * stride_bj + offs_n[None, :] * stride_bk

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k = tl.cdiv(K, BLOCK_K)
    for _ in range(num_k):
        cur_k = offs_k + _ * BLOCK_K
        mask_k = cur_k < K

        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)
        acc += tl.dot(a, b)

        # advance to next block in the reduction dimension
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bj

    # write out tile
    c_base = c_ptr + b * stride_cb + h * stride_ch
    c_ptrs = c_base + offs_m[:, None] * stride_ci + offs_n[None, :] * stride_cj
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

# Entry point – Triton‑accelerated “outgoing” TriMul
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.
    Arguments:
        data = (input_tensor, mask, weights, config)
            input_tensor : torch.Tensor[B, N, N, D] (float32)
            mask         : torch.Tensor[B, N, N] or None
            weights      : dict of torch.Tensor model parameters
            config       : dict with keys
                           - dim (model dimension D)
                           - hidden_dim (H)
                           - nomask (bool, optional)
    Returns:
        torch.Tensor[B, N, N, D] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype  # expected torch.float32

    bs, N, _, dim = input_tensor.shape
    H = config["hidden_dim"]
    nomask = config.get("nomask", False)
    eps = 1e-5

    # ---------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32) – flatten to [B·N·N, D]
    # ---------------------------------------------------------------
    M = bs * N * N
    x_flat = input_tensor.reshape(M, dim)
    norm_w = weights["norm.weight"].to(device, dtype)
    norm_b = weights["norm.bias"].to(device, dtype)
    x_norm_flat = triton_layernorm_fp32(x_flat, norm_w, norm_b, eps=eps)
    x_norm = x_norm_flat.view(bs, N, N, dim)          # (B,N,N,D) float32

    # ---------------------------------------------------------------
    # 2️⃣ Cast to FP16 for all subsequent heavy work
    # ---------------------------------------------------------------
    x_fp16 = x_norm.to(torch.float16)

    # ---------------------------------------------------------------
    # 3️⃣ Fuse all four linear projections (left/right + gates)
    # ---------------------------------------------------------------
    w_comb = torch.cat(
        [
            weights["left_proj.weight"],
            weights["left_gate.weight"],
            weights["right_proj.weight"],
            weights["right_gate.weight"],
        ],
        dim=0,
    ).to(device, torch.float16)   # (4·H, D)

    # combined output: (B, N, N, 4·H)
    combined = F.linear(x_fp16, w_comb)

    # split
    left_proj = combined[..., :H]               # (B,N,N,H)
    left_gate_raw = combined[..., H:2 * H]      # (B,N,N,H)
    right_proj = combined[..., 2 * H:3 * H]     # (B,N,N,H)
    right_gate_raw = combined[..., 3 * H:]      # (B,N,N,H)

    # gates
    left_gate = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)

    # ---------------------------------------------------------------
    # 4️⃣ Optional mask (broadcast over hidden dim)
    # ---------------------------------------------------------------
    if not nomask and mask is not None:
        mask_fp16 = mask.to(device, torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left_proj = left_proj * mask_fp16
        right_proj = right_proj * mask_fp16

    # ---------------------------------------------------------------
    # 5️⃣ Apply element‑wise gates
    # ---------------------------------------------------------------
    left = left_proj * left_gate    # (B,N,N,H)
    right = right_proj * right_gate

    # ---------------------------------------------------------------
    # 6️⃣ Output gate (independent of mask)
    # ---------------------------------------------------------------
    out_gate_w = weights["out_gate.weight"].to(device, torch.float16)  # (H, D)
    out_gate = torch.sigmoid(F.linear(x_fp16, out_gate_w))            # (B,N,N,H)

    # ---------------------------------------------------------------
    # 7️⃣ Core N³ contraction – custom batched GEMM (FP16)
    # ---------------------------------------------------------------
    # Ensure contiguous layout for the kernel
    left = left.contiguous()
    right = right.contiguous()
    out = torch.empty_like(left)  # (B,N,N,H) fp16

    # strides for left  [B, N, N, H]
    s_ab, s_ai, s_ak, s_ah = left.stride()
    # strides for right (used transposed inside the kernel)
    s_br, s_bj, s_bk, s_bh = right.stride()
    # strides for output
    s_cb, s_ci, s_cj, s_ch = out.stride()

    # launch the batched GEMM kernel
    BH = bs * H
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    grid = (BH, triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N))

    batched_gemm_fp16_kernel[grid](
        a_ptr=left,
        b_ptr=right,
        c_ptr=out,
        batch_heads=BH,
        M=N,               # rows   (i dimension)
        N=N,               # cols   (j dimension)
        K=N,               # reduction length (k dimension)
        H=H,
        stride_ab=s_ab,
        stride_ai=s_ai,
        stride_ak=s_ak,
        stride_ah=s_ah,
        stride_br=s_br,
        stride_bj=s_bj,
        stride_bk=s_bk,
        stride_bh=s_bh,
        stride_cb=s_cb,
        stride_ci=s_ci,
        stride_cj=s_cj,
        stride_ch=s_ch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # ---------------------------------------------------------------
    # 8️⃣ Second LayerNorm (hidden dimension, FP16)
    # ---------------------------------------------------------------
    out_flat = out.view(M, H)   # (B·N·N, H)
    to_out_norm_w = weights["to_out_norm.weight"].to(device, torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device, torch.float16)
    out_norm_flat = triton_layernorm_fp16(out_flat, to_out_norm_w, to_out_norm_b, eps=eps)  # FP16

    # ---------------------------------------------------------------
    # 9️⃣ Apply output gate and final projection (both FP16)
    # ---------------------------------------------------------------
    out_gate_flat = out_gate.view(M, H)               # (M, H) fp16
    gated = out_norm_flat * out_gate_flat             # (M, H) fp16

    to_out_w = weights["to_out.weight"].to(device, torch.float16)  # (D, H)
    output_flat = F.linear(gated, to_out_w)          # (M, D) fp16

    output = output_flat.view(bs, N, N, dim).to(torch.float32)  # final FP32 tensor
    return output
