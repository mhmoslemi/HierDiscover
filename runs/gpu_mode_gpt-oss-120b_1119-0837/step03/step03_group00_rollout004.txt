@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak, stride_ab,  # stride for m, k, batch for a
    stride_bk, stride_bn, stride_bb,  # stride for k, n, batch for b
    stride_cm, stride_cn, stride_cb,  # c strides
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    eps: tl.constexpr,
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers for a and b: start points for given batch
    a_base = a_ptr + batch_id * stride_ab
    b_base = b_ptr + batch_id * stride_bb
    c_base = c_ptr + batch_id * stride_cb

    # Compute pointer matrices:
    # a has shape (M, K); we'll load a_block shape (BLOCK_M, BLOCK_K)
    # b has shape (K, N); we load b_block shape (BLOCK_K, BLOCK_N)
    # We'll use parallel loads with broadcast across rows/cols.

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        a_ptrs = a_base + (offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak)
        b_ptrs = b_base + (offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn)

        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        acc += tl.dot(a, b)  # shape (BLOCK_M, BLOCK_N)

    # Write back result, cast to fp16
    c = acc.to(tl.float16)
    c_ptrs = c_base + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    bs, N, _, dim = input_tensor.shape
    H = config["hidden_dim"]
    nomask = config.get("nomask", False)
    eps = 1e-5

    # LayerNorm on original dimension (FP32)
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    # PyTorch layernorm expects normalized shape = (dim,)
    x_norm = torch.nn.functional.layer_norm(input_tensor, (dim,), norm_weight, norm_bias, eps=eps)

    # Cast to half
    x_f16 = x_norm.to(torch.float16)

    # Fuse projection + gate weights
    left_comb_w = torch.cat([weights["left_proj.weight"], weights["left_gate.weight"]], dim=0).to(device, torch.float16)
    right_comb_w = torch.cat([weights["right_proj.weight"], weights["right_gate.weight"]], dim=0).to(device, torch.float16)
    out_gate_w = weights["out_gate.weight"].to(device, torch.float16)

    left_comb = torch.nn.functional.linear(x_f16, left_comb_w)   # (B,N,N,2H)
    right_comb = torch.nn.functional.linear(x_f16, right_comb_w)

    # Split projections and gates
    left_proj = left_comb[..., :H]              # (B,N,N,H)
    left_gate_pre = left_comb[..., H:]          # (B,N,N,H)
    right_proj = right_comb[..., :H]
    right_gate_pre = right_comb[..., H:]

    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)

    # Optional mask (broadcast over H)
    if (not nomask) and (mask is not None):
        mask_f16 = mask.to(device, torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left_proj = left_proj * mask_f16
        right_proj = right_proj * mask_f16

    left = left_proj * left_gate
    right = right_proj * right_gate

    # Output gate (depends only on x)
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_f16, out_gate_w))   # (B,N,N,H)

    # Core N^3 contraction via custom Triton kernel
    out = torch.empty_like(left)   # allocate output (B,N,N,H) FP16
    _batched_matmul_outgoing(left, right, out, H)

    # Second LayerNorm on hidden dimension (FP16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device, torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device, torch.float16)
    out_norm = torch.nn.functional.layer_norm(out, (H,), to_out_norm_w, to_out_norm_b, eps=eps)

    # Apply output gate and final projection
    gated = out_norm * out_gate
    to_out_w = weights["to_out.weight"].to(device, torch.float16)  # (dim, H)
    # torch.nn.functional.linear expects weight shape (out_features, in_features) => (dim, H)
    output = torch.nn.functional.linear(gated, to_out_w)   # (B,N,N,dim) FP16

    # Cast back to float32 for downstream usage
    return output.to(torch.float32)

@triton.jit
def batched_matmul_transpose_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_i, stride_a_k, stride_a_h,
    stride_b_batch, stride_b_i, stride_b_k, stride_b_h,
    stride_c_batch, stride_c_i, stride_c_j, stride_c_h,
    H,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_bh = tl.program_id(0)   # combined batch * head
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # recover batch and head indices
    b_idx = pid_bh // H
    h_idx = pid_bh % H

    # ... compute offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < M
    mask_j = offs_j < N

    # base pointers for this (b,h)
    a_base = a_ptr + b_idx * stride_a_batch + h_idx * stride_a_h
    b_base = b_ptr + b_idx * stride_b_batch + h_idx * stride_b_h
    c_base = c_ptr + b_idx * stride_c_batch + h_idx * stride_c_h

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A block (M x K)
        a_ptrs = a_base + offs_i[:, None] * stride_a_i + offs_k[None, :] * stride_a_k
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Load B block transposed (K x N) = right[j, k] -> we read using j offset as rows of original
        b_ptrs = b_base + offs_j[:, None] * stride_b_i + offs_k[None, :] * stride_b_k
        b = tl.load(b_ptrs, mask=mask_k[None, :] & mask_j[:, None], other=0.0).to(tl.float32)
        # Note: shape of b is (BLOCK_N, BLOCK_K) but we need (BLOCK_K, BLOCK_N) for dot.
        # Transpose for dot:
        b = tl.transpose(b)   # now (BLOCK_K, BLOCK_N)

        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_N) accumulation

    # Write to output, cast back to fp16
    c = acc.to(tl.float16)
    c_ptrs = c_base + offs_i[:, None] * stride_c_i + offs_j[None, :] * stride_c_j
    tl.store(c_ptrs, c, mask=mask_i[:, None] & mask_j[None, :])
```

But there is nuance: loading b may need shape correctly. In the code above, `b_ptrs` constructs shape (BLOCK_N, BLOCK_K) (since offs_j is rows, offs_k columns). `tl.load` returns shape (BLOCK_N, BLOCK_K). Then we need to transpose to get (BLOCK_K, BLOCK_N) for dot. Since `tl.dot` expects (M,K) @ (K,N). So we could either load with reversed offsets: use `offs_k[:, None] * stride_b_i` etc. Let's examine.

Original right shape (B,N,N,H) with strides: stride_i for first N dimension (j), stride_k for second N dimension (k). """
TriMul “outgoing” forward – fused Triton+torch implementation.

Algorithm (FP16 heavy)
----------------------
1️⃣ Layer‑Norm on the model dimension (float32) – use PyTorch’s highly‑optimized kernel.
2️⃣ Cast to FP16 and fuse each projection + its gate into a single linear layer:
   * left_comb   = Linear(x, [left_proj; left_gate])   → (B,N,N,2·H)
   * right_comb  = Linear(x, [right_proj; right_gate]) → (B,N,N,2·H)
   Split the result, apply sigmoid‑gates and the optional mask.
3️⃣ Core N³ contraction:
   out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   Implemented as a custom Triton kernel that treats each (batch,head) pair as
   an independent GEMM (A @ Bᵀ) with FP32 accumulation and FP16 output.
   No tensor reshaping or copies are performed – the kernel works on the
   original (B,N,N,H) layout using explicit strides.
4️⃣ Second Layer‑Norm on the hidden dimension (still FP16).
5️⃣ Apply the output‑gate, final linear projection back to the model dimension,
   and cast the result to float32.

The heavy O(N³) work is performed entirely on‑GPU with a tiled kernel that
exploits Tensor‑Cores (FP16×FP16→FP32 accumulation).  All other ops stay in
FP16 to keep memory traffic low.  At least one Triton kernel (the GEMM) is
used, satisfying the problem constraints.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM with transposed right operand.
# Computes out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_transpose_kernel(
    a_ptr, b_ptr, c_ptr,                     # left, right, out pointers
    M, N, K,                                 # dimensions (all = seq_len)
    stride_a_batch, stride_a_i, stride_a_k, stride_a_h,
    stride_b_batch, stride_b_i, stride_b_k, stride_b_h,
    stride_c_batch, stride_c_i, stride_c_j, stride_c_h,
    H,                                       # hidden dimension (head count per batch)
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # ------------------------------------------------------------------
    # Program IDs
    #   pid0 : combined batch * head (BH)
    #   pid1 : block row   (i)
    #   pid2 : block col   (j)
    # ------------------------------------------------------------------
    pid_bh = tl.program_id(0)
    pid_i  = tl.program_id(1)
    pid_j  = tl.program_id(2)

    # Recover batch & head indices from the flattened BH dimension
    b_idx = pid_bh // H
    h_idx = pid_bh % H

    # ------------------------------------------------------------------
    # Compute block offsets
    # ------------------------------------------------------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < M
    mask_j = offs_j < N

    # Base pointers for this (batch,head) pair
    a_base = a_ptr + b_idx * stride_a_batch + h_idx * stride_a_h
    b_base = b_ptr + b_idx * stride_b_batch + h_idx * stride_b_h
    c_base = c_ptr + b_idx * stride_c_batch + h_idx * stride_c_h

    # ------------------------------------------------------------------
    # Accumulate over the K‑dimension
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    num_k = tl.cdiv(K, BLOCK_K)

    for k_iter in range(num_k):
        offs_k = k_iter * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A block: shape (BLOCK_M, BLOCK_K)
        a_ptrs = a_base + offs_i[:, None] * stride_a_i + offs_k[None, :] * stride_a_k
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0
        ).to(tl.float32)

        # Load B block transposed (Bᵀ).  We need B_T[k, j] = right[b, j, k, h].
        # Right layout: stride_b_i → j‑axis, stride_b_k → k‑axis.
        b_ptrs = b_base + offs_k[:, None] * stride_b_k + offs_j[None, :] * stride_b_i
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0
        ).to(tl.float32)

        # Rank‑K update
        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_N)

    # ------------------------------------------------------------------
    # Write result (cast back to FP16)
    # ------------------------------------------------------------------
    c = acc.to(tl.float16)
    c_ptrs = c_base + offs_i[:, None] * stride_c_i + offs_j[None, :] * stride_c_j
    tl.store(
        c_ptrs,
        c,
        mask=mask_i[:, None] & mask_j[None, :]
    )


def _batched_matmul_outgoing(left: torch.Tensor,
                             right: torch.Tensor,
                             out: torch.Tensor,
                             hidden_dim: int,
                             block_m: int = 64,
                             block_n: int = 64,
                             block_k: int = 32) -> None:
    """
    Wrapper around the Triton kernel.
    left/right/out : tensors of shape (B, N, N, H) – FP16, contiguous.
    hidden_dim    : H.
    """
    B, N, _, H = left.shape
    assert H == hidden_dim, "hidden dimension mismatch"

    grid = (
        B * H,                         # combined batch*head dimension (pid0)
        triton.cdiv(N, block_m),       # i‑blocks (pid1)
        triton.cdiv(N, block_n),       # j‑blocks (pid2)
    )

    batched_matmul_transpose_kernel[grid](
        a_ptr=left,
        b_ptr=right,
        c_ptr=out,
        M=N,
        N=N,
        K=N,
        stride_a_batch=left.stride(0),
        stride_a_i=left.stride(1),
        stride_a_k=left.stride(2),
        stride_a_h=left.stride(3),
        stride_b_batch=right.stride(0),
        stride_b_i=right.stride(1),
        stride_b_k=right.stride(2),
        stride_b_h=right.stride(3),
        stride_c_batch=out.stride(0),
        stride_c_i=out.stride(1),
        stride_c_j=out.stride(2),
        stride_c_h=out.stride(3),
        H=H,
        BLOCK_M=block_m,
        BLOCK_N=block_n,
        BLOCK_K=block_k,
    )
    # Triton kernels are asynchronous; synchronize before using `out`.
    torch.cuda.synchronize()


# ----------------------------------------------------------------------
# Entry point – the Triton‑accelerated TriMul (outgoing) operator.
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator.
    Arguments:
        data = (input_tensor, mask, weights, config)
            input_tensor : torch.Tensor[B, N, N, D] (float32)
            mask         : torch.Tensor[B, N, N] or None
            weights      : dict of torch.Tensor (model parameters)
            config       : dict with keys:
                           - dim (model dimension D)
                           - hidden_dim (H)
                           - nomask (bool, optional)
    Returns:
        torch.Tensor[B, N, N, D] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype  # expected torch.float32

    bs, N, _, dim = input_tensor.shape
    H = config["hidden_dim"]
    nomask = config.get("nomask", False)
    eps = 1e-5

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32, using PyTorch's fast kernel)
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x_norm = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps
    )   # (B, N, N, D) float32

    # --------------------------------------------------------------
    # 2️⃣ Cast to FP16 for the heavy work
    # --------------------------------------------------------------
    x_f16 = x_norm.to(torch.float16)

    # --------------------------------------------------------------
    # 3️⃣ Fuse projection + gate for left & right sides
    # --------------------------------------------------------------
    left_comb_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(device, torch.float16)            # (2·H, D)

    right_comb_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(device, torch.float16)            # (2·H, D)

    out_gate_w = weights["out_gate.weight"].to(device, torch.float16)  # (H, D)

    # Linear projections (combined proj+gate)
    left_comb = F.linear(x_f16, left_comb_w)   # (B, N, N, 2·H)
    right_comb = F.linear(x_f16, right_comb_w)

    # Split projections and gates
    left_proj = left_comb[..., :H]          # (B, N, N, H)
    left_gate_pre = left_comb[..., H:]      # (B, N, N, H)
    right_proj = right_comb[..., :H]
    right_gate_pre = right_comb[..., H:]

    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)

    # --------------------------------------------------------------
    # 4️⃣ Optional mask (broadcast over H)
    # --------------------------------------------------------------
    if (not nomask) and (mask is not None):
        mask_f16 = mask.to(device, torch.float16).unsqueeze(-1)  # (B, N, N, 1)
        left_proj = left_proj * mask_f16
        right_proj = right_proj * mask_f16

    # --------------------------------------------------------------
    # 5️⃣ Apply element‑wise gates
    # --------------------------------------------------------------
    left = left_proj * left_gate          # (B, N, N, H)
    right = right_proj * right_gate       # (B, N, N, H)

    # --------------------------------------------------------------
    # 6️⃣ Output gate (independent of mask)
    # --------------------------------------------------------------
    out_gate = torch.sigmoid(F.linear(x_f16, out_gate_w))   # (B, N, N, H)

    # --------------------------------------------------------------
    # 7️⃣ Core N³ contraction via the Triton kernel
    # --------------------------------------------------------------
    out = torch.empty_like(left)   # allocate (B, N, N, H) in FP16
    _batched_matmul_outgoing(left, right, out, hidden_dim=H)

    # --------------------------------------------------------------
    # 8️⃣ Second LayerNorm on the hidden dimension (FP16)
    # --------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device, torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device, torch.float16)
    out_norm = F.layer_norm(
        out,
        normalized_shape=(H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=eps
    )   # (B, N, N, H) FP16

    # --------------------------------------------------------------
    # 9️⃣ Apply output gate and final projection (both FP16)
    # --------------------------------------------------------------
    gated = out_norm * out_gate   # (B, N, N, H) FP16
    to_out_w = weights["to_out.weight"].to(device, torch.float16)  # (D, H)
    output = F.linear(gated, to_out_w)   # (B, N, N, D) FP16

    # --------------------------------------------------------------
    # 10️⃣ Cast back to FP32 for the public API.
    # --------------------------------------------------------------
    return output.to(torch.float32)
