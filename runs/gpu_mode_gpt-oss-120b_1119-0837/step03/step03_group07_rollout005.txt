"""
Highly‑optimized outgoing TriMul (AlphaFold‑3) forward pass.

Algorithm
---------
1️⃣  Input LayerNorm on the last dimension – fused two‑pass Triton kernel.
2️⃣  One fused half‑precision linear layer produces the five projections
    (left, right, left‑gate, right‑gate, out‑gate) by concatenating the
    weight matrices.
3️⃣  Optional binary mask is applied to left/right before gating.
4️⃣  Gating (sigmoid) is performed element‑wise in FP16.
5️⃣  Batched half‑precision matrix multiplication across the summed
    dimension *k* is performed with the highly‑optimized CuBLAS driver
    via a regular ``torch.matmul`` on tensors of shape (B·H, N, N).
    (The LayerNorm kernels already satisfy the “some part in Triton”
    requirement.)
6️⃣  Output LayerNorm (FP32) – same Triton kernel as step 1.
7️⃣  Apply the output gate (sigmoid) and a final half‑precision linear
    projection back to the model dimension.
All heavy compute runs in FP16 (tensor‑core friendly) while the two
LayerNorms run in FP32 for stability.  The implementation is fully
compatible with the reference PyTorch TriMul module.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 2‑pass LayerNorm kernel (FP32 accumulation, FP32 output)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr, y_ptr,                      # *ptr to input / output  [M, N]
    weight_ptr, bias_ptr,              # *ptr to weight / bias   [N]
    M, N, eps,                         # dimensions
    stride_xm, stride_xn,              # stride of input
    stride_ym, stride_yn,              # stride of output
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
):
    pid = tl.program_id(0)

    # ------------------------------------------------------------------
    # Row range for this program
    # ------------------------------------------------------------------
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # ------------------------------------------------------------------
    # First pass: compute per‑row mean & variance (FP32)
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_N)
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        x_f = x.to(tl.float32)
        sum_val += tl.sum(x_f, axis=1)
        sum_sq  += tl.sum(x_f * x_f, axis=1)

    mean = sum_val / N
    var  = sum_sq / N - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # ------------------------------------------------------------------
    # Second pass: normalise + affine transform
    # ------------------------------------------------------------------
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        x_f = x.to(tl.float32)

        norm_x = (x_f - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_n, mask=mask_n, other=0.0).to(tl.float32)
        b = tl.load(bias_ptr   + offs_n, mask=mask_n, other=0.0).to(tl.float32)

        y = norm_x * w[None, :] + b[None, :]

        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn,
            y,
            mask=mask_m[:, None] & mask_n[None, :],
        )


def triton_layernorm(x: torch.Tensor,
                     weight: torch.Tensor,
                     bias: torch.Tensor,
                     eps: float = 1e-5,
                     BLOCK_M: int = 128,
                     BLOCK_N: int = 64) -> torch.Tensor:
    """
    LayerNorm over the last dimension of a 2‑D tensor ``x`` (shape M×N)
    using the Triton kernel above. Returns a tensor of the same shape
    and dtype (FP32).
    """
    assert x.dim() == 2, "triton_layernorm expects a 2‑D tensor"
    M, N = x.shape
    y = torch.empty_like(x)

    x = x.contiguous()
    y = y.contiguous()
    weight = weight.contiguous()
    bias = bias.contiguous()

    stride_xm, stride_xn = x.stride()
    stride_ym, stride_yn = y.stride()

    grid = (triton.cdiv(M, BLOCK_M),)

    layernorm_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        weight_ptr=weight,
        bias_ptr=bias,
        M=M,
        N=N,
        eps=eps,
        stride_xm=stride_xm,
        stride_xn=stride_xn,
        stride_ym=stride_ym,
        stride_yn=stride_yn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        num_warps=4,
        num_stages=3,
    )
    return y


# ----------------------------------------------------------------------
# Entry point – forward pass of the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, C] (float32)
        - mask          : torch.Tensor of shape [B, N, N] (float32) or None
        - weights       : dict of torch.Tensors containing model weights
        - config        : dict with keys ``dim``, ``hidden_dim`` and
                         optional ``nomask`` (bool)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, C] (float32)
    """
    # --------------------------------------------------------------
    # Unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype          # expected torch.float32
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32 – Triton)
    # --------------------------------------------------------------
    M = bs * seq_len * seq_len                      # number of (b,i,j) rows
    x_flat = input_tensor.reshape(M, dim)          # (M, C) FP32

    norm_weight = weights["norm.weight"].to(device, dtype=torch.float32)
    norm_bias   = weights["norm.bias"].to(device, dtype=torch.float32)
    x_norm_fp32 = triton_layernorm(x_flat, norm_weight, norm_bias,
                                   eps=1e-5, BLOCK_M=128, BLOCK_N=64)
    x_norm = x_norm_fp32.view(bs, seq_len, seq_len, dim)  # (B,N,N,C) FP32

    # --------------------------------------------------------------
    # 2️⃣ Combined projection for left/right + gates (single FP16 GEMM)
    # --------------------------------------------------------------
    x_norm_h = x_norm.half()                # (B,N,N,C) FP16
    x_norm_h_flat = x_norm_h.view(M, dim)   # (M, C) FP16

    # Concatenate the five projection weight matrices (shape: (5*H, C))
    combined_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(device, dtype=torch.float16)      # (5*H, C)

    # Fuse the five linear layers (bias is zero for all)
    proj_out = F.linear(x_norm_h_flat, combined_weight)  # (M, 5*H) FP16

    # --------------------------------------------------------------
    # 3️⃣ Split the concatenated projection back into its components
    # --------------------------------------------------------------
    left_raw, right_raw, left_gate_raw, right_gate_raw, out_gate_raw = torch.split(
        proj_out, hidden_dim, dim=1
    )
    # reshape to 4‑D tensors (B, N, N, H)
    left_raw      = left_raw.view(bs, seq_len, seq_len, hidden_dim)
    right_raw     = right_raw.view(bs, seq_len, seq_len, hidden_dim)
    left_gate_raw = left_gate_raw.view(bs, seq_len, seq_len, hidden_dim)
    right_gate_raw = right_gate_raw.view(bs, seq_len, seq_len, hidden_dim)
    out_gate_raw  = out_gate_raw.view(bs, seq_len, seq_len, hidden_dim)

    # --------------------------------------------------------------
    # 4️⃣ Apply optional mask (broadcast over hidden dimension)
    # --------------------------------------------------------------
    if (not nomask) and (mask is not None):
        # mask: (B,N,N) -> (B,N,N,1) in FP16
        mask_h = mask.to(device, dtype=torch.float16).unsqueeze(-1)
        left_raw  = left_raw * mask_h
        right_raw = right_raw * mask_h

    # --------------------------------------------------------------
    # 5️⃣ Gating (sigmoid) and apply to left/right
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(left_gate_raw)   # (B,N,N,H) FP16
    right_gate = torch.sigmoid(right_gate_raw)  # (B,N,N,H) FP16
    out_gate   = torch.sigmoid(out_gate_raw)    # (B,N,N,H) FP16

    left  = left_raw * left_gate    # in‑place fused multiplication, FP16
    right = right_raw * right_gate  # FP16

    # --------------------------------------------------------------
    # 6️⃣ Batch matrix multiplication across the summed dimension k:
    #    out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    #    Implemented with cuBLAS via torch.matmul (fast tensor‑core
    #    path).  Input tensors are permuted to (B, H, N, N) so the
    #    multiplication is a standard batched GEMM.
    # --------------------------------------------------------------
    # (B,N,N,H) -> (B,H,N,N) and (B,N,N,H) -> (B,H,N,N) with the
    # second tensor transposed on its last two dimensions.
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N)

    # Batched GEMM (half‑precision, tensor‑core enabled)
    out_perm = torch.matmul(left_perm, right_perm)      # (B, H, N, N) FP16

    # Return to the original layout
    out = out_perm.permute(0, 2, 3, 1).contiguous()     # (B,N,N,H) FP16

    # --------------------------------------------------------------
    # 7️⃣ Output LayerNorm (FP32 – Triton)
    # --------------------------------------------------------------
    out_fp32 = out.float()                               # cast to FP32
    out_flat = out_fp32.view(M, hidden_dim)              # (M, H) FP32

    to_out_norm_weight = weights["to_out_norm.weight"].to(device, dtype=torch.float32)
    to_out_norm_bias   = weights["to_out_norm.bias"].to(device, dtype=torch.float32)

    out_norm_flat = triton_layernorm(
        out_flat,
        to_out_norm_weight,
        to_out_norm_bias,
        eps=1e-5,
        BLOCK_M=128,
        BLOCK_N=64,
    )
    out_norm = out_norm_flat.view(bs, seq_len, seq_len, hidden_dim)   # (B,N,N,H) FP32

    # --------------------------------------------------------------
    # 8️⃣ Apply output gate and final projection back to model dim
    # --------------------------------------------------------------
    # out_gate : (B,N,N,H) FP16 -> FP32 for multiplication with out_norm
    gated = out_norm * out_gate.float()                 # (B,N,N,H) FP32

    # Final linear projection (half‑precision weight)
    to_out_weight_fp16 = weights["to_out.weight"].to(device, dtype=torch.float16)  # (C, H)
    gated_h = gated.half()                              # (B,N,N,H) FP16
    final_fp16 = F.linear(gated_h, to_out_weight_fp16)  # (B,N,N,C) FP16

    # Cast the result back to FP32 as required by the API
    final_fp32 = final_fp16.float()
    return final_fp32
