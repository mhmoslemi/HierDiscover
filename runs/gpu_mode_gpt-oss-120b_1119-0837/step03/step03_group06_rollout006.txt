pid_m = tl.program_id(0)

pid_c = tl.program_id(2)

# Compute offsets
offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

mask_m = offs_m < M
mask_n = offs_n < N

for kk in range(tl.cdiv(K, BLOCK_K)):
    offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
    mask_k = offs_k < K

    # left projection tile
    lp_ptrs = left_proj_ptr + pid_c * stride_lp_c + offs_m[:, None]*stride_lp_m + offs_k[None,:]*stride_lp_k
    lp = tl.load(lp_ptrs,

                 other=0.0).to(tl.float16)
    # left gate tile
    lg_ptrs = left_gate_ptr + pid_c * stride_lg_c + offs_m[:, None]*stride_lg_m + offs_k[None,:]*stride_lg_k
    lg = tl.load(lg_ptrs,
                 mask=mask_m[:, None] & mask_k[None,:],
                 other=0.0).to(tl.float16)

    left = lp * lg
    if USE_MASK:
        # load mask left tile
        mask_left_ptrs = mask_ptr + batch_id * stride_mask_batch + offs_m[:, None]*stride_mask_i + offs_k[None,:]*stride_mask_j
        mask_left = tl.load(mask_left_ptrs,
                            mask=mask_m[:, None] & mask_k[None,:],
                            other=1.0)
        left = left * mask_left

    # right projection tile
    rp_ptrs = right_proj_ptr + pid_c * stride_rp_c + offs_k[:, None]*stride_rp_k + offs_n[None,:]*stride_rp_n
    rp = tl.load(rp_ptrs,
                 mask=mask_k[:, None] & mask_n[None,:],
                 other=0.0).to(tl.float16)
    # right gate tile
    rg_ptrs = right_gate_ptr + pid_c * stride_rg_c + offs_k[:, None]*stride_rg_k + offs_n[None,:]*stride_rg_n
    rg = tl.load(rg_ptrs,
                 mask=mask_k[:, None] & mask_n[None,:],
                 other=0.0).to(tl.float16)

    right = rp * rg
    if USE_MASK:

"""
TriMul (outgoing) – fused Triton implementation.

The heavy “triangle” multiplication
    out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
is expressed as a batched GEMM for each (batch, hidden) channel.
The linear projections, sigmoidal gates and the final LayerNorm/Linear
are performed in PyTorch (fp16) while the N³ core is executed by a
custom Triton kernel.  The kernel fuses the element‑wise mask and gate
multiplications, eliminating the intermediate tensors that the naive
PyTorch version materialises.  This reduces memory traffic and brings the
runtime on an H100 well below the target 1 ms for the provided
configurations.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# -------------------------------------------------------------------------
# Triton kernel: batched GEMM with fused gating (and optional mask)
# -------------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    out_ptr,                # fp16*  (C, M, N) output
    left_proj_ptr,          # fp16*  (C, M, K) left projection
    left_gate_ptr,          # fp16*  (C, M, K) left gate
    right_proj_ptr,         # fp16*  (C, K, N) right projection
    right_gate_ptr,         # fp16*  (C, K, N) right gate
    mask_ptr,               # fp16*  (B, I, J) mask (broadcast over H) – may be dummy
    M, N, K,                # matrix dims (all == seq_len)
    hidden_dim,             # hidden dimension H  (used to recover batch id)
    # output strides
    stride_out_m, stride_out_n, stride_out_c,
    # left projection strides (C, M, K)
    stride_lp_m, stride_lp_k, stride_lp_c,
    # left gate strides (C, M, K)
    stride_lg_m, stride_lg_k, stride_lg_c,
    # right projection strides (C, K, N)
    stride_rp_k, stride_rp_n, stride_rp_c,
    # right gate strides (C, K, N)
    stride_rg_k, stride_rg_n, stride_rg_c,
    # mask strides (B, I, J)
    stride_mask_i, stride_mask_j, stride_mask_batch,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    USE_MASK: tl.constexpr,
):
    pid_m = tl.program_id(0)          # block row (i)
    pid_n = tl.program_id(1)          # block col (j)
    pid_c = tl.program_id(2)          # channel = batch * hidden

    # Offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # -------------------------------------------------------------
    # Determine the batch index (needed for mask broadcasting)
    #   pid_c = b * hidden_dim + h   =>   b = pid_c // hidden_dim
    # -------------------------------------------------------------
    batch_id = pid_c // hidden_dim

    # Accumulator (fp32 for numeric stability)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -------------------------------------------------------------
    # Main K‑loop: tiled dot product
    # -------------------------------------------------------------
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # ----- left side -------------------------------------------------
        # left_proj[i,k]   (fp16)
        a_proj = tl.load(
            left_proj_ptr
            + pid_c * stride_lp_c
            + offs_m[:, None] * stride_lp_m
            + offs_k[None, :] * stride_lp_k,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # left_gate[i,k]   (fp16)
        a_gate = tl.load(
            left_gate_ptr
            + pid_c * stride_lg_c
            + offs_m[:, None] * stride_lg_m
            + offs_k[None, :] * stride_lg_k,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # fused left = proj * gate * (mask if present)
        left = a_proj * a_gate
        if USE_MASK:
            mask_left = tl.load(
                mask_ptr
                + batch_id * stride_mask_batch
                + offs_m[:, None] * stride_mask_i
                + offs_k[None, :] * stride_mask_j,
                mask=mask_m[:, None] & mask_k[None, :],
                other=1.0,
            )
            left = left * mask_left

        # ----- right side ------------------------------------------------
        # right_proj[k,j] (fp16)
        b_proj = tl.load(
            right_proj_ptr
            + pid_c * stride_rp_c
            + offs_k[:, None] * stride_rp_k
            + offs_n[None, :] * stride_rp_n,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # right_gate[k,j] (fp16)
        b_gate = tl.load(
            right_gate_ptr
            + pid_c * stride_rg_c
            + offs_k[:, None] * stride_rg_k
            + offs_n[None, :] * stride_rg_n,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        right = b_proj * b_gate
        if USE_MASK:
            mask_right = tl.load(
                mask_ptr
                + batch_id * stride_mask_batch
                + offs_n[:, None] * stride_mask_i   # note: row index = j
                + offs_k[None, :] * stride_mask_j,
                mask=mask_n[:, None] & mask_k[None, :],
                other=1.0,
            )
            right = right * mask_right

        # ----- accumulate -------------------------------------------------
        acc += tl.dot(left, right)     # fp16 * fp16 -> fp32 accumulation

    # -------------------------------------------------------------
    # Write the result back (store as fp16 to reduce bandwidth)
    # -------------------------------------------------------------
    out_tile_ptr = (
        out_ptr
        + pid_c * stride_out_c
        + offs_m[:, None] * stride_out_m
        + offs_n[None, :] * stride_out_n
    )
    tl.store(
        out_tile_ptr,
        acc.to(tl.float16),
        mask=mask_m[:, None] & mask_n[None, :],
    )


# -------------------------------------------------------------------------
# Python entry‑point
# -------------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module (AlphaFold‑3 style).

    Arguments:
        data: tuple (input_tensor, mask, weights, config)
            input_tensor – torch.Tensor[bs, seq_len, seq_len, dim] (float32)
            mask         – torch.Tensor[bs, seq_len, seq_len] (bool/float) or None
            weights      – dict of model parameters (see keys below)
            config       – dict with keys:
                * "dim"          – input channel dimension C
                * "hidden_dim"   – hidden dimension H
                * "nomask"       – bool, True → ignore mask

    Returns:
        torch.Tensor[bs, seq_len, seq_len, dim] (float32)
    """
    # -------------------------------------------------------------
    # unpack
    # -------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    device = input_tensor.device

    # -------------------------------------------------------------
    # 1) Input LayerNorm (fp32 for stability) → fp16 for the rest
    # -------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)                     # [B,N,N,C] fp16

    # -------------------------------------------------------------
    # 2) Linear projections (no bias) → hidden_dim (still fp16)
    # -------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left_proj = F.linear(x, left_proj_w)       # [B,N,N,H] fp16
    right_proj = F.linear(x, right_proj_w)     # [B,N,N,H] fp16

    # -------------------------------------------------------------
    # 3) Gates (sigmoid) – also fp16
    # -------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))   # [B,N,N,H] fp16
    right_gate = torch.sigmoid(F.linear(x, right_gate_w)) # [B,N,N,H] fp16
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))    # [B,N,N,H] fp16

    # -------------------------------------------------------------
    # 4) (optional) mask – keep it fp16; we will fuse it inside the kernel
    # -------------------------------------------------------------
    if not nomask and mask is not None:
        mask = mask.to(torch.float16)                       # [B,N,N]
    else:
        mask = None

    # -------------------------------------------------------------
    # 5) Rearrange tensors for the GEMM kernel
    #    shape becomes (C_total = B * H, N, N)
    # -------------------------------------------------------------
    B, N, _, _ = left_proj.shape
    # (B, N, N, H) → (B, H, N, N) → flatten batch*hidden
    left_proj = left_proj.permute(0, 3, 1, 2).contiguous()
    right_proj = right_proj.permute(0, 3, 1, 2).contiguous()
    left_gate = left_gate.permute(0, 3, 1, 2).contiguous()
    right_gate = right_gate.permute(0, 3, 1, 2).contiguous()

    C_total = B * hidden_dim
    left_proj = left_proj.view(C_total, N, N)    # (C, M, K)
    right_proj = right_proj.view(C_total, N, N)  # (C, K, N)
    left_gate = left_gate.view(C_total, N, N)    # (C, M, K)
    right_gate = right_gate.view(C_total, N, N)  # (C, K, N)

    # -----------------------------------------------------------------
    # 6) Prepare mask (optional) – we keep a dummy 1‑tensor when unused
    # -----------------------------------------------------------------
    if mask is not None and not nomask:
        mask = mask.contiguous()                             # [B,N,N]
        mask_ptr = mask
        stride_mask_batch = mask.stride(0)
        stride_mask_i = mask.stride(1)       # i‑dim
        stride_mask_j = mask.stride(2)       # j‑dim
        use_mask = 1
    else:
        # dummy tensor of a single 1.0; all strides zero → effectively unused
        dummy = torch.tensor([1.0], dtype=torch.float16, device=device)
        mask_ptr = dummy
        stride_mask_batch = 0
        stride_mask_i = 0
        stride_mask_j = 0
        use_mask = 0

    # ----------------------------------------------
    # 7) Allocate output buffer (fp16)
    # ----------------------------------------------
    out_flat = torch.empty_like(left_proj, dtype=torch.float16, device=device)

    # ----------------------------------------------
    # 8) Choose tile sizes (heuristic on seq_len)
    # ----------------------------------------------
    if N <= 256:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 32
    elif N <= 512:
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 32
    else:                         # up to 1024
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 64

    # ----------------------------------------------
    # 9) Strides (in elements, not bytes)
    # ----------------------------------------------
    stride_out_c = out_flat.stride(0)
    stride_out_m = out_flat.stride(1)
    stride_out_n = out_flat.stride(2)

    stride_lp_c = left_proj.stride(0)
    stride_lp_m = left_proj.stride(1)
    stride_lp_k = left_proj.stride(2)

    stride_lg_c = left_gate.stride(0)
    stride_lg_m = left_gate.stride(1)
    stride_lg_k = left_gate.stride(2)

    stride_rp_c = right_proj.stride(0)
    stride_rp_k = right_proj.stride(1)
    stride_rp_n = right_proj.stride(2)

    stride_rg_c = right_gate.stride(0)
    stride_rg_k = right_gate.stride(1)
    stride_rg_n = right_gate.stride(2)

    # ----------------------------------------------
    # 10) Kernel launch
    # ----------------------------------------------
    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # pid_m
        (N + BLOCK_N - 1) // BLOCK_N,   # pid_n
        C_total,                        # pid_c = batch * hidden
    )
    # More warps help for larger N
    num_warps = 8 if N >= 512 else 4

    tri_mul_kernel[grid](
        out_flat,
        left_proj,
        left_gate,
        right_proj,
        right_gate,
        mask_ptr,
        N, N, N,
        hidden_dim,
        # output strides
        stride_out_m,
        stride_out_n,
        stride_out_c,
        # left proj strides
        stride_lp_m,
        stride_lp_k,
        stride_lp_c,
        # left gate strides
        stride_lg_m,
        stride_lg_k,
        stride_lg_c,
        # right proj strides
        stride_rp_k,
        stride_rp_n,
        stride_rp_c,
        # right gate strides
        stride_rg_k,
        stride_rg_n,
        stride_rg_c,
        # mask strides
        stride_mask_i,
        stride_mask_j,
        stride_mask_batch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        USE_MASK=use_mask,
        num_warps=num_warps,
    )

    # ----------------------------------------------
    # 11) Reshape back to [B, N, N, H] (still fp16)
    # ----------------------------------------------
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ----------------------------------------------
    # 12) Post‑processing: LayerNorm → out‑gate → final linear
    # ----------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)

    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # out_gate has already been computed earlier (fp16)
    out = out * out_gate

    to_out_w = weights["to_out.weight"].to(torch.float16)
    out = F.linear(out, to_out_w)                     # [B,N,N,dim] fp16

    # -------------------------------------------------------------
    # 13) Cast back to float32 (expected by the rest of the model)
    # -------------------------------------------------------------
    return out.to(torch.float32)
