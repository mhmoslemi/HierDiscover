def layernorm_kernel(x_ptr, y_ptr, weight_ptr, bias_ptr, M, N, eps,
                     stride_xm, stride_xn, stride_ym, stride_yn,
                     BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr):
    pid = tl.program_id(0)

    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # First pass: compute mean and variance for each row
    sum = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_N)
    for k in range(num_k):
        offs_n = k * BLOCK_N + tl.arange(0, BLOCK_N)
        mask_n = offs_n < N
        x = tl.load(x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
                    mask=mask_m[:, None] & mask_n[None, :],
                    other=0.0)  # dtype matches input
        x_fp = tl.cast(x, tl.float32)
        sum += tl.sum(x_fp, axis=1)
        sum_sq += tl.sum(x_fp * x_fp, axis=1)

    mean = sum / N
    var = sum_sq / N - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # Second pass: normalize and apply affine
    for k in range(num_k):
        offs_n = k * BLOCK_N + tl.arange(0, BLOCK_N)
        mask_n = offs_n < N
        x = tl.load(x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,

                    other=0.0)

"""
TriMul “outgoing” forward pass – heavily‑optimized Triton + PyTorch.

Algorithm
---------
1️⃣  Row‑wise LayerNorm over the last dimension (dim) using a Triton kernel.
2️⃣  Cast the normalized tensor to FP16.
3️⃣  Fuse the five bias‑free linear projections (left, left_gate,
    right, right_gate, out_gate) into a **single** matrix‑multiply
    (M×dim) @ (dim×5·hidden) via ``torch.nn.functional.linear``.
    This cuts the memory traffic of the input tensor from 5× to 1×.
4️⃣  Slice the concatenated result, apply the optional binary mask to
    the two value‑projections, and apply sigmoid to the three gate
    tensors.  Multiply the projected values by their corresponding
    gates (mask is applied before gating, as in the reference).
5️⃣  Reshape back to [B, N, N, H] and compute the contraction

       out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]

    as a batched GEMM:  (B, H, N, N) @ (B, H, N, N)ᵀ.
6️⃣  Second LayerNorm over the hidden dimension (H) – same Triton kernel
    (the output of the GEMM is cast back to FP32 for numerical stability).
7️⃣  Multiply by the output‑gate and finish with a final FP32 linear
    projection (hidden → dim).

All tensors stay on the device; the final result is ``torch.float32``.
The implementation works for the full test matrix of shapes
(B∈{1,2}, N∈{128,…,1024}, dim∈{128,384}, hidden=128) and for masked
or unmasked inputs.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton LayerNorm (row‑wise) – two‑pass implementation (mean/var → affine)
# Supports both FP32 and FP16 inputs (computes statistics in FP32).
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,                # *ptr to input  [M, N]   (FP16 or FP32)
    y_ptr,                # *ptr to output [M, N]   (same dtype as x)
    weight_ptr,           # *ptr to weight [N]    (FP32)
    bias_ptr,             # *ptr to bias   [N]    (FP32)
    M,                    # number of rows
    N,                    # feature dimension
    eps,                  # epsilon
    stride_xm, stride_xn,  # strides for input (row, col)
    stride_ym, stride_yn,  # strides for output (row, col)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
):
    pid = tl.program_id(0)

    # ------------------------------------------------------------------
    # Row offsets for this program
    # ------------------------------------------------------------------
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # ------------------------------------------------------------------
    # First pass: compute mean & variance for each row
    # ------------------------------------------------------------------
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_N)
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        x_fp = tl.cast(x, tl.float32)            # promote to FP32 for reduction
        sum_val += tl.sum(x_fp, axis=1)
        sum_sq  += tl.sum(x_fp * x_fp, axis=1)

    mean = sum_val / N
    var  = sum_sq / N - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # ------------------------------------------------------------------
    # Second pass: normalize + affine transform
    # ------------------------------------------------------------------
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        x_fp = tl.cast(x, tl.float32)
        norm_x = (x_fp - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_n, mask=mask_n, other=0.0)   # FP32 weight
        b = tl.load(bias_ptr   + offs_n, mask=mask_n, other=0.0)   # FP32 bias

        y_fp = norm_x * w[None, :] + b[None, :]
        # Cast back to the dtype of the output tensor (FP16 or FP32)
        y = tl.cast(y_fp, tl.typeof(x))
        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn,
            y,
            mask=mask_m[:, None] & mask_n[None, :],
        )


def triton_layernorm(x: torch.Tensor,
                     weight: torch.Tensor,
                     bias: torch.Tensor,
                     eps: float = 1e-5) -> torch.Tensor:
    """
    Triton LayerNorm over the last dimension.
    ``x`` may be FP32 or FP16; output has the same dtype as ``x``.
    """
    assert x.dim() == 2, "triton_layernorm expects a 2‑D tensor"
    M, N = x.shape
    y = torch.empty_like(x)

    # Ensure contiguous tensors for correct stride handling
    x = x.contiguous()
    weight = weight.contiguous()
    bias = bias.contiguous()

    stride_xm, stride_xn = x.stride()
    stride_ym, stride_yn = y.stride()

    grid = lambda meta: (triton.cdiv(M, meta["BLOCK_M"]),)

    layernorm_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        weight_ptr=weight,
        bias_ptr=bias,
        M=M,
        N=N,
        eps=eps,
        stride_xm=stride_xm,
        stride_xn=stride_xn,
        stride_ym=stride_ym,
        stride_yn=stride_yn,
        BLOCK_M=256,   # tuned for large M (≈B·N²)
        BLOCK_N=64,    # columns per tile (multiple of 64 is good for fp16)
    )
    return y

# Entry point used by the evaluator

    hidden_dim = config["hidden_dim"]

                               left_gate_w,
                               right_proj_w,
                               right_gate_w,
                               out_gate_w], dim=0)        # shape (5*H, D)

    # Flatten the input tensor to (M, D) and run a single linear layer
    x_flat_h = x_norm_h.reshape(M, dim).contiguous()
    raw = F.linear(x_flat_h, weight_concat)               # shape (M, 5*H), FP16

    # ------------------------------------------------------------------
    # 4️⃣ Slice the concatenated result and apply the optional mask
    # ------------------------------------------------------------------
    H = hidden_dim
    left_raw       = raw[:, 0:H]                # (M, H)
    left_gate_raw  = raw[:, H:2*H]
    right_raw      = raw[:, 2*H:3*H]
    right_gate_raw = raw[:, 3*H:4*H]
    out_gate_raw   = raw[:, 4*H:5*H]

    if (not nomask) and (mask is not None):
        mask_f = mask.reshape(M, 1).to(torch.float16)   # broadcast over H
        left_raw  = left_raw * mask_f
        right_raw = right_raw * mask_f

    # ------------------------------------------------------------------
    # 5️⃣ Apply sigmoid gates and fuse gate multiplication
    # ------------------------------------------------------------------
    left  = left_raw * torch.sigmoid(left_gate_raw)    # (M, H)  FP16
    right = right_raw * torch.sigmoid(right_gate_raw) # (M, H)  FP16
    out_gate = torch.sigmoid(out_gate_raw)            # (M, H)  FP16

    # ------------------------------------------------------------------
    # 6️⃣ Reshape back to [B, L, L, H] for the contraction
    # ------------------------------------------------------------------
    left = left.view(bs, L, L, H)
    right = right.view(bs, L, L, H)
    out_gate = out_gate.view(bs, L, L, H)

    # ------------------------------------------------------------------
    # 7️⃣ Batched matrix multiplication:
    #     out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    # Bring the head dimension into the batch dimension.
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, L, L)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # (B, H, L, L)

    # (B·H, L, L) @ (B·H, L, L)ᵀ  – all in FP16 on Tensor‑Cores
    out_perm = torch.matmul(left_perm, right_perm.transpose(-2, -1))  # (B, H, L, L)

    # Restore original axis order → (B, L, L, H)
    out = out_perm.permute(0, 2, 3, 1).contiguous()     # (B, L, L, H) FP16

    # ------------------------------------------------------------------
    # 8️⃣ Output LayerNorm over hidden dimension (H) – Triton (FP32)
    # ------------------------------------------------------------------
    # Cast to FP32 for numerical stability in the normalization step
    out_fp32 = out.to(torch.float32)
    out_flat = out_fp32.view(bs * L * L, H).contiguous()
    to_out_norm_w = weights["to_out_norm.weight"].to(device, dtype).contiguous()
    to_out_norm_b = weights["to_out_norm.bias"].to(device, dtype).contiguous()
    out_norm_flat = triton_layernorm(out_flat, to_out_norm_w, to_out_norm_b, eps=eps)
    out_norm = out_norm_flat.view(bs, L, L, H)          # (B, L, L, H) FP32

    # ------------------------------------------------------------------
    # 9️⃣ Apply the output‑gate and final linear projection (H → D)
    # ------------------------------------------------------------------
    out_norm = out_norm * out_gate.to(dtype)            # broadcast over H
    to_out_w = weights["to_out.weight"].to(device, dtype).contiguous()
    out = F.linear(out_norm, to_out_w)                  # (B, L, L, D) FP32

    return out
```
