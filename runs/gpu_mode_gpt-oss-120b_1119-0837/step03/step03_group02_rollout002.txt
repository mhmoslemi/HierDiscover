def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    B, M, N, K,
    stride_az, stride_ai, stride_ak,
    stride_bz, stride_bk, stride_bj,
    stride_cz, stride_ci, stride_cj,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    # Compute which batch and block_i, block_j to process
    num_pid_m = tl.cdiv(M, BLOCK_M)
    num_pid_n = tl.cdiv(N, BLOCK_N)
    # total tasks per batch: num_pid_m * num_pid_n
    # total P: B (batch-head)
    batch = pid // (num_pid_m * num_pid_n)
    pid_rem = pid % (num_pid_m * num_pid_n)
    pid_m = pid_rem // num_pid_n
    pid_n = pid_rem % num_pid_n

    # Compute row and column offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Create k offsets for iteration
    a_ptrs = a_ptr + batch * stride_az + offs_m[:, None] * stride_ai + tl.arange(0, BLOCK_K)[None, :] * stride_ak
    b_ptrs = b_ptr + batch * stride_bz + tl.arange(0, BLOCK_K)[:, None] * stride_bk + offs_n[None, :] * stride_bj

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask=(offs_m[:,None] < M) & (tl.arange(0, BLOCK_K)[None,:] + k*BLOCK_K < K), other=0.0)
        b = tl.load(b_ptrs, mask=(tl.arange(0, BLOCK_K)[:,None] + k*BLOCK_K < K) & (offs_n[None,:] < N), other=0.0)
        acc += tl.dot(a, b) # but a and b shapes? Should be (BLOCK_M, BLOCK_K) x (BLOCK_K, BLOCK_N)
        # Update pointers
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk
    # Store result

@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    batch: tl.constexpr, M: tl.constexpr, N: tl.constexpr, K: tl.constexpr,
    stride_az, stride_ai, stride_ak,
    stride_bz, stride_bk, stride_bj,
    stride_cz, stride_ci, stride_cj,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    # total number of blocks per batch:
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    # compute batch index and block indices
    batch_idx = pid // (num_blocks_m * num_blocks_n)
    rem = pid % (num_blocks_m * num_blocks_n)
    block_i = rem // num_blocks_n
    block_j = rem % num_blocks_n

    # thread block coordinates for output
    offs_i = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = block_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < M
    mask_j = offs_j < N

    # Initialize accumulation
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Iterate over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < K

        # Load A block (batch, i, k)
        a_ptrs = a_ptr + batch_idx * stride_az + offs_i[:, None] * stride_ai + k_offset[None, :] * stride_ak
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # Load B block (batch, k, j)
        b_ptrs = b_ptr + batch_idx * stride_bz + k_offset[:, None] * stride_bk + offs_j[None, :] * stride_bj
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # Multiply-accumulate
        acc += tl.dot(a, b)

    # Write output
    c_ptrs = c_ptr + batch_idx * stride_cz + offs_i[:, None] * stride_ci + offs_j[None, :] * stride_cj
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])

def batched_gemm(a, b, block_m=64, block_n=64, block_k=64):
    # a: [P, M, K], b: [P, K, N]; both half
    assert a.is_contiguous()
    assert b.is_contiguous()
    P, M, K = a.shape
    _, K2, N = b.shape
    assert K == K2
    c = torch.empty((P, M, N), dtype=a.dtype, device=a.device)
    # strides in elements
    stride_az, stride_ai, stride_ak = a.stride()
    stride_bz, stride_bk, stride_bj = b.stride()
    stride_cz, stride_ci, stride_cj = c.stride()
    # compute grid

    num_tiles_n = (N + block_n - 1) // block_n
    total_blocks = P * num_tiles_m * num_tiles_n

        stride_az, stride_ai, stride_ak,
        stride_bz, stride_bk, stride_bj,
        stride_cz, stride_ci, stride_cj,
        BLOCK_M=block_m,
        BLOCK_N=block_n,
        BLOCK_K=block_k,
    )
    return c

2. **Fused linear projections + gates** (all FP16)  
   * left_proj   + left_gate   → weight matrix (2·H, D)  
   * right_proj  + right_gate  → weight matrix (2·H, D)  
   A single `torch.nn.functional.linear` call for each pair produces the
   projected values.  The gate tensors are passed through a sigmoid.

3. **Mask (optional)** – applied after gating (mask·value = value when mask=1,
   zero otherwise).  The mask is also FP16.

4. **Batched GEMM (`batched_gemm_kernel`)**  
   The core contraction   
   `out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]`   
   is a batched matrix multiplication over the hidden dimension.
   The kernel processes all (batch·head) matrices in a single launch,
   loading tiles of A and B, performing a FP16×FP16 → FP32 `tl.dot`,
   and writing the result back in FP16.  This reduces kernel‑launch
   overhead compared with `torch.bmm` for many small GEMMs.

5. **Output LayerNorm** – another Triton LN over the hidden dim (FP16).

6. **Output gate** – a linear layer (`out_gate`) on the *normalised*
   input followed by a sigmoid (FP16).

7. **Final projection** – a linear layer (`to_out`) back to the original
   dimension (FP16).  The result is cast to FP32 for the model output.

All heavy arithmetic is performed in FP16 (Tensor‑core friendly) with
FP32 accumulation inside the kernels.  The implementation works for the
full range of test configurations (B∈{1,2}, L∈{128,…,1024},
D∈{128,384}, H=128) and respects an optional binary mask.

The only requirement is that at least one operation runs in a Triton
kernel – here we have two (LayerNorm and the batched GEMM).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton LayerNorm (row‑wise) – works for FP16 or FP32 input.
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,                # *ptr to input  [M, N]
    y_ptr,                # *ptr to output [M, N]
    weight_ptr,           # *ptr to weight [N]
    bias_ptr,             # *ptr to bias   [N]
    M,                    # total rows
    N,                    # feature dimension
    eps,                  # epsilon
    stride_xm, stride_xn,  # strides for input
    stride_ym, stride_yn,  # strides for output
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
):
    pid = tl.program_id(0)

    # ---------------------------------------------------------------
    # Row offsets for this program
    # ---------------------------------------------------------------
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # ---------------------------------------------------------------
    # First pass: compute mean & variance (FP32 accumulation)
    # ---------------------------------------------------------------
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_N)
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        x_f32 = x.to(tl.float32)
        sum_val += tl.sum(x_f32, axis=1)
        sum_sq  += tl.sum(x_f32 * x_f32, axis=1)

    mean = sum_val / N
    var  = sum_sq / N - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # ---------------------------------------------------------------
    # Second pass: write normalised + affine‑transformed values
    # ---------------------------------------------------------------
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        x_f32 = x.to(tl.float32)
        norm_x = (x_f32 - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_n, mask=mask_n, other=0.0).to(tl.float32)
        b = tl.load(bias_ptr   + offs_n, mask=mask_n, other=0.0).to(tl.float32)

        y = norm_x * w[None, :] + b[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn,
            y.to(x.dtype),
            mask=mask_m[:, None] & mask_n[None, :],
        )


def triton_layernorm(x: torch.Tensor,
                     weight: torch.Tensor,
                     bias: torch.Tensor,
                     eps: float = 1e-5) -> torch.Tensor:
    """
    Row‑wise LayerNorm implemented with Triton.
    ``x`` must be a 2‑D tensor of shape (M, N).
    Returns a tensor with the same shape and dtype as ``x``.
    """
    assert x.dim() == 2, "triton_layernorm expects a 2‑D tensor"
    M, N = x.shape
    y = torch.empty_like(x)

    # Make strides explicit (in elements, not bytes)
    stride_xm, stride_xn = x.stride()
    stride_ym, stride_yn = y.stride()

    grid = (triton.cdiv(M, 128),)      # BLOCK_M = 128 by default

    layernorm_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        weight_ptr=weight,
        bias_ptr=bias,
        M=M,
        N=N,
        eps=eps,
        stride_xm=stride_xm,
        stride_xn=stride_xn,
        stride_ym=stride_ym,
        stride_yn=stride_yn,
        BLOCK_M=128,   # good for H100
        BLOCK_N=64,
    )
    return y


# ----------------------------------------------------------------------
# Batched GEMM kernel (FP16×FP16 → FP16, FP32 accumulation)
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    batch, M, N, K,
    stride_az, stride_ai, stride_ak,
    stride_bz, stride_bk, stride_bj,
    stride_cz, stride_ci, stride_cj,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute many independent GEMMs:
        C[p, i, j] = Σₖ A[p, i, k] * B[p, k, j]
    A has shape (batch, M, K)   – row‑major
    B has shape (batch, K, N)   – row‑major
    C has shape (batch, M, N)   – row‑major
    All tensors are FP16; accumulation is FP32.
    """
    pid = tl.program_id(0)

    # ---------------------------------------------------------------
    # Decode the 3‑D program index (batch, block_i, block_j)
    # ---------------------------------------------------------------
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)

    batch_idx = pid // (num_blocks_m * num_blocks_n)
    rem = pid % (num_blocks_m * num_blocks_n)
    block_i = rem // num_blocks_n
    block_j = rem % num_blocks_n

    # ---------------------------------------------------------------
    # Offsets inside the current tile
    # ---------------------------------------------------------------
    offs_i = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = block_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < M
    mask_j = offs_j < N

    # ---------------------------------------------------------------
    # Accumulator (FP32)
    # ---------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---------------------------------------------------------------
    # Loop over K dimension in tiles
    # ---------------------------------------------------------------
    num_tiles_k = tl.cdiv(K, BLOCK_K)
    for k_tile in range(num_tiles_k):
        k_offset = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < K

        # A[p, i, k]
        a_ptrs = (
            a_ptr
            + batch_idx * stride_az
            + offs_i[:, None] * stride_ai
            + k_offset[None, :] * stride_ak
        )
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)

        # B[p, k, j]
        b_ptrs = (
            b_ptr
            + batch_idx * stride_bz
            + k_offset[:, None] * stride_bk
            + offs_j[None, :] * stride_bj
        )
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0)

        # FP16×FP16 → FP32 dot product
        acc += tl.dot(a, b)

    # ---------------------------------------------------------------
    # Write back the result (cast to FP16)
    # ---------------------------------------------------------------
    c_ptrs = (
        c_ptr
        + batch_idx * stride_cz
        + offs_i[:, None] * stride_ci
        + offs_j[None, :] * stride_cj
    )
    tl.store(c_ptrs,
             acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


def batched_gemm_fp16(
    A: torch.Tensor,
    B: torch.Tensor,
    block_m: int = 64,
    block_n: int = 64,
    block_k: int = 64,
) -> torch.Tensor:
    """
    Wrapper for ``batched_gemm_kernel``.
    A : [batch, M, K]   (fp16, contiguous)
    B : [batch, K, N]   (fp16, contiguous)
    Returns C : [batch, M, N] (fp16)
    """
    assert A.is_contiguous() and B.is_contiguous()
    batch, M, K = A.shape
    _, K2, N = B.shape
    assert K == K2, "Inner dimensions must match."

    C = torch.empty((batch, M, N), dtype=A.dtype, device=A.device)

    # Strides in *elements* (not bytes)
    stride_az, stride_ai, stride_ak = A.stride()
    stride_bz, stride_bk, stride_bj = B.stride()
    stride_cz, stride_ci, stride_cj = C.stride()

    # Grid size = batch * #tiles_m * #tiles_n
    tiles_m = (M + block_m - 1) // block_m
    tiles_n = (N + block_n - 1) // block_n
    total_blocks = batch * tiles_m * tiles_n

    grid = (total_blocks,)

    batched_gemm_kernel[grid](
        A,
        B,
        C,
        batch,
        M,
        N,
        K,
        stride_az,
        stride_ai,
        stride_ak,
        stride_bz,
        stride_bk,
        stride_bj,
        stride_cz,
        stride_ci,
        stride_cj,
        BLOCK_M=block_m,
        BLOCK_N=block_n,
        BLOCK_K=block_k,
    )
    return C


# ----------------------------------------------------------------------
# Main entry‑point for the test harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor   [B, L, L, D]   (float32)
        - mask         : torch.Tensor   [B, L, L] or ``None``
        - weights      : dict of torch tensors (model parameters)
        - config       : dict with keys ``dim``, ``hidden_dim``
                         optional ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, L, L, D] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    bs, L, _, dim = input_tensor.shape
    hidden = config["hidden_dim"]
    nomask = config.get("nomask", False)
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (Triton) – operates in FP16
    # ------------------------------------------------------------------
    # Cast input to half‑precision for the whole pipeline
    inp_fp16 = input_tensor.to(torch.float16)

    # Flatten to [B·L·L, D]
    M = bs * L * L
    inp_flat = inp_fp16.reshape(M, dim)

    # LayerNorm parameters in FP16
    norm_w = weights["norm.weight"].to(device, torch.float16).contiguous()
    norm_b = weights["norm.bias"].to(device, torch.float16).contiguous()

    norm_flat = triton_layernorm(inp_flat, norm_w, norm_b, eps=eps)
    norm = norm_flat.view(bs, L, L, dim)            # [B, L, L, D]

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections + gates (four maps fused into two matmuls)
    # ------------------------------------------------------------------
    # Gather weight matrices (all bias‑free)
    left_proj_w  = weights["left_proj.weight"].to(device, torch.float16).contiguous()
    left_gate_w  = weights["left_gate.weight"].to(device, torch.float16).contiguous()
    right_proj_w = weights["right_proj.weight"].to(device, torch.float16).contiguous()
    right_gate_w = weights["right_gate.weight"].to(device, torch.float16).contiguous()

    # Fuse (proj+gate) for left side → (2·H, D)
    left_fused_w = torch.cat([left_proj_w, left_gate_w], dim=0)   # (2H, D)
    # Fuse for right side → (2·H, D)
    right_fused_w = torch.cat([right_proj_w, right_gate_w], dim=0)  # (2H, D)

    # Flatten for matmul
    norm_flat_h = norm_flat.reshape(M, dim)      # [M, D] (already FP16)

    # Left side
    left_out = F.linear(norm_flat_h, left_fused_w)        # [M, 2H]
    left_proj = left_out[..., :hidden]                   # (M, H)
    left_gate = torch.sigmoid(left_out[..., hidden:])    # (M, H)

    # Right side
    right_out = F.linear(norm_flat_h, right_fused_w)      # [M, 2H]
    right_proj = right_out[..., :hidden]                 # (M, H)
    right_gate = torch.sigmoid(right_out[..., hidden:])  # (M, H)

    # Reshape back to 4‑D tensors (FP16)
    left_proj = left_proj.view(bs, L, L, hidden)
    left_gate = left_gate.view(bs, L, L, hidden)
    right_proj = right_proj.view(bs, L, L, hidden)
    right_gate = right_gate.view(bs, L, L, hidden)

    # ------------------------------------------------------------------
    # 3️⃣ Apply mask (if present) and gates
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_h = mask.to(device, torch.float16).unsqueeze(-1)   # [B, L, L, 1]
        left = (left_proj * left_gate) * mask_h
        right = (right_proj * right_gate) * mask_h
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # Free memory we no longer need
    del left_proj, left_gate, right_proj, right_gate, left_out, right_out, left_fused_w, right_fused_w

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM (the N³ contraction)
    #    out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    # Permute to bring the hidden dimension next to batch
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, L, L] → (i,k)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # [B, H, L, L] → (i,k)

    # Transpose the last two axes of right so that we have (K, N)
    right_t = right_perm.transpose(2, 3).contiguous()   # [B, H, L, L] (k,j)

    # Collapse batch and head into a single dimension
    batch_heads = bs * hidden
    left_mat = left_perm.reshape(batch_heads, L, L)   # (P, M, K)   M=N=L
    right_mat = right_t.reshape(batch_heads, L, L)   # (P, K, N)

    # Choose block sizes – small enough for all test seq‑len values.
    # For L ≤ 256 we use 64, otherwise 128.
    if L <= 256:
        bm, bn, bk = 64, 64, 64
    else:
        bm, bn, bk = 128, 128, 64

    out_mat = batched_gemm_fp16(left_mat, right_mat, block_m=bm, block_n=bn, block_k=bk)  # (P, L, L)

    # Reshape back to [B, L, L, H] (still half‑precision)
    out = (
        out_mat.view(bs, hidden, L, L)
        .permute(0, 2, 3, 1)
        .contiguous()
    )
    del left, right, left_perm, right_perm, right_t, out_mat, left_mat, right_mat

    # ------------------------------------------------------------------
    # 5️⃣ Output LayerNorm (Triton) over the hidden dimension
    # ------------------------------------------------------------------
    out_M = bs * L * L
    out_flat = out.view(out_M, hidden)                       # [M_out, H]

    out_norm_w = weights["to_out_norm.weight"].to(device, torch.float16).contiguous()
    out_norm_b = weights["to_out_norm.bias"].to(device, torch.float16).contiguous()
    out_norm_flat = triton_layernorm(out_flat, out_norm_w, out_norm_b, eps=eps)
    out_norm = out_norm_flat.view(bs, L, L, hidden)          # [B, L, L, H]

    # ------------------------------------------------------------------
    # 6️⃣ Output gate (linear + sigmoid) – still in FP16
    # ------------------------------------------------------------------
    out_gate_w = weights["out_gate.weight"].to(device, torch.float16).contiguous()
    out_gate = torch.sigmoid(F.linear(norm_flat_h, out_gate_w))   # [M, H]
    out_gate = out_gate.view(bs, L, L, hidden)                    # [B, L, L, H]

    # ------------------------------------------------------------------
    # 7️⃣ Apply gate and final projection back to `dim`
    # ------------------------------------------------------------------
    gated = out_norm * out_gate           # FP16 × FP16 → FP16
    gated_flat = gated.reshape(out_M, hidden)

    to_out_w = weights["to_out.weight"].to(device, torch.float16).contiguous()
    final_fp16 = F.linear(gated_flat, to_out_w)   # [M, D] (half)

    out_final = final_fp16.view(bs, L, L, dim).to(torch.float32)  # cast to FP32 as required

    return out_final
