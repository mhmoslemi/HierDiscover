import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# Triton kernel for gating & mask
@triton.jit
def gate_mask_kernel(
    proj_ptr, gate_ptr, mask_ptr, out_ptr,
    total_size, M, H,
    HAS_MASK: tl.constexpr, BLOCK_SIZE: tl.constexpr
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_size

    # Load projection and gate values (fp16)
    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)

    # Compute sigmoid(gate) in fp32 for stability
    proj_f = proj.to(tl.float32)
    gate_f = gate.to(tl.float32)
    gate_sig = 1.0 / (1.0 + tl.exp(-gate_f))

    if HAS_MASK:
        # row index = floor(offset / H)
        row = offs // H
        # Load mask per row (broadcasted)
        mask_val = tl.load(mask_ptr + row, mask=row < M, other=1.0).to(tl.float32)
        out_f = proj_f * gate_sig * mask_val
    else:
        out_f = proj_f * gate_sig

    # Cast back to fp16 and store
    out = out_f.to(proj.dtype)
    tl.store(out_ptr + offs, out, mask=mask)
```

```
def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype
    bs, L, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    eps = 1e-5

    # 1. Input LayerNorm using torch.nn.functional.layer_norm (fast)
    norm_weight = weights["norm.weight"].to(device, dtype)
    norm_bias = weights["norm.bias"].to(device, dtype)
    x_norm = torch.nn.functional.layer_norm(input_tensor,
                                            normalized_shape=(dim,),
                                            weight=norm_weight,
                                            bias=norm_bias,
                                            eps=eps)

    # 2. Cast to FP16 for heavy math
    x_norm_h = x_norm.to(torch.float16)

    # 3. Fuse four linear maps
    left_proj_w = weights["left_proj.weight"].to(device, torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device, torch.float16)
    left_gate_w = weights["left_gate.weight"].to(device, torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device, torch.float16)
    fused_weight = torch.cat([left_proj_w,
                              right_proj_w,
                              left_gate_w,
                              right_gate_w], dim=0)   # shape (4*H, D)

    M = bs * L * L
    x_flat = x_norm_h.view(M, dim)   # [M, D]
    fused_out = F.linear(x_flat, fused_weight)  # [M, 4*H]
    fused_out = fused_out.view(bs, L, L, 4 * hidden_dim)

    # 4. Split and apply gate + mask with a single Triton kernel
    # slices (contiguous)
    left_proj_raw  = fused_out[..., :hidden_dim].contiguous()
    right_proj_raw = fused_out[..., hidden_dim:2*hidden_dim].contiguous()
    left_gate_raw  = fused_out[..., 2*hidden_dim:3*hidden_dim].contiguous()
    right_gate_raw = fused_out[..., 3*hidden_dim:4*hidden_dim].contiguous()

    M = bs * L * L
    total = M * hidden_dim

    # Prepare mask if needed
    if not nomask and mask is not None:
        mask_h = mask.to(torch.float16).contiguous().view(-1)  # shape [M]
    else:
        # dummy mask (filled with 1.0) – kernel will ignore it when HAS_MASK=0
        mask_h = torch.empty(1, dtype=torch.float16, device=device)

    # Output tensors
    left = torch.empty_like(left_proj_raw)
    right = torch.empty_like(right_proj_raw)

    # Determine block size
    BLOCK_SIZE = 1024  # max threads per program
    grid = lambda meta: (triton.cdiv(total, meta["BLOCK_SIZE"]),)

    # Left side
    gate_mask_kernel[grid](
        proj_ptr=left_proj_raw.view(-1),
        gate_ptr=left_gate_raw.view(-1),
        mask_ptr=mask_h,
        out_ptr=left.view(-1),
        total_size=total,
        M=M,
        H=hidden_dim,
        HAS_MASK=int(not nomask and mask is not None),
        BLOCK_SIZE=BLOCK_SIZE,
    )
    # Right side
    gate_mask_kernel[grid](
        proj_ptr=right_proj_raw.view(-1),
        gate_ptr=right_gate_raw.view(-1),
        mask_ptr=mask_h,
        out_ptr=right.view(-1),
        total_size=total,
        M=M,
        H=hidden_dim,
        HAS_MASK=int(not nomask and mask is not None),
        BLOCK_SIZE=BLOCK_SIZE,
    )

    # 5. Batched GEMM for the contraction
    # Bring hidden dimension to second axis
    # Right needs to be transposed on inner k dimension
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, L, L]
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # [B, H, L, L]
    right_t   = right_perm.transpose(2, 3)               # [B, H, L, L] (K,N)

    batch_h = bs * hidden_dim
    left_mat  = left_perm.reshape(batch_h, L, L)
    right_mat = right_t.reshape(batch_h, L, L)

    out_mat = torch.bmm(left_mat, right_mat)  # (B*H, L, L)
    out = out_mat.view(bs, hidden_dim, L, L).permute(0, 2, 3, 1).contiguous()  # [B, L, L, H]

    # 6. Output LayerNorm (torch.nn.functional.layer_norm)
    out_norm_weight = weights["to_out_norm.weight"].to(device, dtype).contiguous()
    out_norm_bias   = weights["to_out_norm.bias"].to(device, dtype).contiguous()
    out_norm = torch.nn.functional.layer_norm(out,
                                              normalized_shape=(hidden_dim,),
                                              weight=out_norm_weight,
                                              bias=out_norm_bias,
                                              eps=eps)

    # 7. Output gate (linear + sigmoid)
    out_gate_w = weights["out_gate.weight"].to(device, torch.float16).contiguous()
    out_gate = torch.sigmoid(F.linear(x_norm_h.view(M, dim), out_gate_w))   # [M, H]
    out_gate = out_gate.view(bs, L, L, hidden_dim)

    # 8. Apply output gate and final projection
    gated = out_norm * out_gate   # fp16 * fp16 -> fp16
    to_out_w = weights["to_out.weight"].to(device, torch.float16).contiguous()
    out_final_h = F.linear(gated.view(M, hidden_dim), to_out_w)   # [M, D] fp16
    out_final = out_final_h.view(bs, L, L, dim).to(torch.float32)

    return out_final
```

"""
Optimised Triton‑accelerated outgoing TriMul (AlphaFold‑3) forward pass.

Key ideas
---------
*   Input / output LayerNorm are performed with the highly‑optimised
    torch.nn.functional.layer_norm (single‑pass, fused mean/var) – this is
    significantly faster than the two‑pass Triton version used previously.
*   The four linear maps (left/right projection + their gates) are fused into a
    single half‑precision GEMM (weight shape 4·H × D).  After the GEMM the
    projection, gate and optional mask are fused into a *single* Triton kernel:
    ``proj * sigmoid(gate) * mask``.  This eliminates an extra sigmoid pass
    and two extra element‑wise passes, cutting memory traffic.
*   The expensive N³ contraction ``out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]``
    is executed as a batched GEMM (torch.bmm) on FP16 tensors – the best
    performer on H100 Tensor‑cores.
*   The output gate and final projection remain in FP16 and the result is cast
    back to float32 as required by the model.

The implementation respects the optional ``mask`` (broadcast over the hidden
dimension) and works for all configurations listed in the problem statement.
At least one custom Triton kernel (the fused gate‑mask kernel) is used.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: fused projection * sigmoid(gate) * optional mask
# ----------------------------------------------------------------------
@triton.jit
def gate_mask_kernel(
    proj_ptr, gate_ptr, mask_ptr, out_ptr,  # pointers
    M, H,                                   # total rows (M), hidden dim (H)
    HAS_MASK: tl.constexpr,                 # compile‑time bool
    BLOCK_SIZE: tl.constexpr,               # threads per program
):
    """Compute out = proj * sigmoid(gate) * mask  (mask broadcast over H)."""
    total = M * H
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    valid = offs < total

    # Load FP16 projection & gate values
    proj = tl.load(proj_ptr + offs, mask=valid, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=valid, other=0.0)

    # FP32 sigmoid for numerical stability
    proj_f = proj.to(tl.float32)
    gate_f = gate.to(tl.float32)
    gate_sig = 1.0 / (1.0 + tl.exp(-gate_f))

    if HAS_MASK:
        # Row index for the mask (floor division)
        row = offs // H
        # Broadcast mask value (0/1) to every hidden channel
        mask_val = tl.load(mask_ptr + row, mask=row < M, other=1.0).to(tl.float32)
        out_f = proj_f * gate_sig * mask_val
    else:
        out_f = proj_f * gate_sig

    # Store result back in FP16
    out = out_f.to(proj.dtype)
    tl.store(out_ptr + offs, out, mask=valid)


# ----------------------------------------------------------------------
# Entry point required by the test harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the *outgoing* TriMul operator (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor   [B, L, L, D]   (float32)
        - mask         : torch.Tensor   [B, L, L] or ``None``
        - weights      : dict of torch tensors (model parameters)
        - config       : dict with keys ``dim``, ``hidden_dim``,
                         optional ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, L, L, D] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    bs, L, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (torch.nn.functional) – normalises over ``dim``
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(device, dtype).contiguous()
    norm_bias   = weights["norm.bias"].to(device, dtype).contiguous()
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B, L, L, D]  (float32)

    # ------------------------------------------------------------------
    # 2️⃣ Cast to FP16 for the heavy compute
    # ------------------------------------------------------------------
    x_norm_h = x_norm.to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Fuse the four linear maps (left_proj, right_proj,
    #    left_gate, right_gate) into a single matmul.
    # ------------------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"].to(device, torch.float16).contiguous()
    right_proj_w = weights["right_proj.weight"].to(device, torch.float16).contiguous()
    left_gate_w  = weights["left_gate.weight"].to(device, torch.float16).contiguous()
    right_gate_w = weights["right_gate.weight"].to(device, torch.float16).contiguous()

    fused_weight = torch.cat(
        [left_proj_w, right_proj_w, left_gate_w, right_gate_w], dim=0
    )  # (4·H, D)

    M = bs * L * L
    x_flat = x_norm_h.view(M, dim)                     # [M, D]
    fused_out = F.linear(x_flat, fused_weight)         # [M, 4·H]
    fused_out = fused_out.view(bs, L, L, 4 * hidden_dim)

    # ------------------------------------------------------------------
    # 4️⃣ Split and apply gate + optional mask with a single Triton kernel
    # ------------------------------------------------------------------
    # slices are made contiguous to get a flat FP16 buffer for the kernel
    left_proj_raw  = fused_out[..., :hidden_dim].contiguous()
    right_proj_raw = fused_out[..., hidden_dim:2 * hidden_dim].contiguous()
    left_gate_raw  = fused_out[..., 2 * hidden_dim:3 * hidden_dim].contiguous()
    right_gate_raw = fused_out[..., 3 * hidden_dim:4 * hidden_dim].contiguous()

    # flatten for the kernel (row‑major)
    left_proj_raw_f = left_proj_raw.view(-1)
    right_proj_raw_f = right_proj_raw.view(-1)
    left_gate_raw_f = left_gate_raw.view(-1)
    right_gate_raw_f = right_gate_raw.view(-1)

    # Prepare mask (broadcasted over the hidden dimension) if required
    if not nomask and mask is not None:
        mask_f = mask.to(torch.float16).contiguous().view(-1)     # [M]
        has_mask = 1
    else:
        # dummy tensor – never read when HAS_MASK==0
        mask_f = torch.tensor([0.0], dtype=torch.float16, device=device)
        has_mask = 0

    # Output containers
    left  = torch.empty_like(left_proj_raw)   # [B, L, L, H] FP16
    right = torch.empty_like(right_proj_raw)

    total_elements = M * hidden_dim
    BLOCK_SIZE = 1024  # one program = 1024 threads (max supported)

    grid = lambda meta: (triton.cdiv(total_elements, meta["BLOCK_SIZE"]),)

    # Left side
    gate_mask_kernel[grid](
        proj_ptr=left_proj_raw_f,
        gate_ptr=left_gate_raw_f,
        mask_ptr=mask_f,
        out_ptr=left.view(-1),
        M=M,
        H=hidden_dim,
        HAS_MASK=has_mask,
        BLOCK_SIZE=BLOCK_SIZE,
    )
    # Right side
    gate_mask_kernel[grid](
        proj_ptr=right_proj_raw_f,
        gate_ptr=right_gate_raw_f,
        mask_ptr=mask_f,
        out_ptr=right.view(-1),
        M=M,
        H=hidden_dim,
        HAS_MASK=has_mask,
        BLOCK_SIZE=BLOCK_SIZE,
    )

    # ------------------------------------------------------------------
    # 5️⃣ Batched GEMM – compute out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    # (B, L, L, H) → (B, H, L, L) so that batch × head can be flattened
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, L, L]
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # [B, H, L, L]

    # Transpose the last two axes of `right` → (K,N) layout for GEMM
    right_t = right_perm.transpose(2, 3).contiguous()   # [B, H, L, L]

    batch_h = bs * hidden_dim
    left_mat  = left_perm.reshape(batch_h, L, L)   # (B·H, N, K)
    right_mat = right_t.reshape(batch_h, L, L)    # (B·H, K, N)

    out_mat = torch.bmm(left_mat, right_mat)      # (B·H, N, N)
    out = out_mat.view(bs, hidden_dim, L, L).permute(0, 2, 3, 1).contiguous()  # [B, L, L, H]

    # ------------------------------------------------------------------
    # 6️⃣ Output LayerNorm (torch.nn.functional) over the hidden dimension
    # ------------------------------------------------------------------
    out_norm_w = weights["to_out_norm.weight"].to(device, dtype).contiguous()
    out_norm_b = weights["to_out_norm.bias"].to(device, dtype).contiguous()
    out_norm = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=out_norm_w,
        bias=out_norm_b,
        eps=eps,
    )  # [B, L, L, H] FP16

    # ------------------------------------------------------------------
    # 7️⃣ Output gate (linear + sigmoid)
    # ------------------------------------------------------------------
    out_gate_w = weights["out_gate.weight"].to(device, torch.float16).contiguous()
    out_gate = torch.sigmoid(F.linear(x_norm_h.view(M, dim), out_gate_w))  # [M, H]
    out_gate = out_gate.view(bs, L, L, hidden_dim)                         # [B, L, L, H]

    # ------------------------------------------------------------------
    # 8️⃣ Apply output gate and final linear projection back to `dim`
    # ------------------------------------------------------------------
    gated = out_norm * out_gate                         # FP16 * FP16 → FP16
    to_out_w = weights["to_out.weight"].to(device, torch.float16).contiguous()
    out_final_h = F.linear(gated.view(M, hidden_dim), to_out_w)   # [M, D] (FP16)
    out_final = out_final_h.view(bs, L, L, dim).to(torch.float32)

    return out_final
