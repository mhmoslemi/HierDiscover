def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_lb, stride_li, stride_lk, stride_lh,
    stride_rb, stride_ri, stride_rk, stride_rh,
    stride_ob, stride_oi, stride_oj, stride_oh,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    batch_head = tl.program_id(0) # 0..B*H-1
    # compute batch and head IDs
    bh = batch_head
    b = bh // H
    h = bh % H

    # compute start indices for tile in output matrix
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)
    off_m = pid_m * BLOCK_M
    off_n = pid_n * BLOCK_N

    # compute pointers to base
    left_base = left_ptr + b * stride_lb + h * stride_lh
    right_base = right_ptr + b * stride_rb + h * stride_rh
    out_base = out_ptr + b * stride_ob + h * stride_oh

    # compute masks for output tile boundaries
    mask_m = off_m + tl.arange(0, BLOCK_M) < N
    mask_n = off_n + tl.arange(0, BLOCK_N) < N

    # allocate accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = kk * BLOCK_K
        # offsets for k dimension
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

*   The four linear maps (left_proj, right_proj, left_gate, right_gate)
    are fused into a single FP16 matrix‑multiply.  After the matmul the
    result is sliced, a sigmoid is applied to the two gate parts and
    the mask (if present) is incorporated.

*   The costly contraction

        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]

    is performed by a custom Triton GEMM kernel.  The kernel works on
    the original layout ``[B, N, N, H]`` (no costly ``permute →
    contiguous`` copy) and computes a batched matrix multiplication for
    each ``(b, h)`` pair.  Memory accesses are tiled (M×N blocks,
    K‑loop of size 64) and the accumulation is done in FP32, the final
    result is stored as FP16.

*   The remaining lightweight operations (output gate, final linear
    projection) stay in PyTorch/F‑16 for maximum Tensor‑core utilisation.
    The final tensor is cast back to FP32 as required by the model.

The implementation works for all configurations listed in the task
statement (B∈{1,2}, N∈{128,…,1024}, dim∈{128,384},
hidden_dim∈{128}).  Only the forward pass is provided – gradients are
not needed.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton row‑wise LayerNorm (operates on a 2‑D tensor [M, N])
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr, y_ptr,      # *ptr to input / output  [M, N]
    weight_ptr, bias_ptr,  # *ptr to [N]
    M, N, eps,
    stride_xm, stride_xn,
    stride_ym, stride_yn,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr
):
    pid = tl.program_id(0)

    # ------------------------------------------------------------------
    # Row offsets for this program
    # ------------------------------------------------------------------
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # ------------------------------------------------------------------
    # First pass – mean / variance (reduction in FP32)
    # ------------------------------------------------------------------
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_n_blocks = tl.cdiv(N, BLOCK_N)
    for nb in range(num_n_blocks):
        offs_n = nb * BLOCK_N + tl.arange(0, BLOCK_N)
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        xf = x.to(tl.float32)
        sum_val += tl.sum(xf, axis=1)
        sum_sq  += tl.sum(xf * xf, axis=1)

    mean = sum_val / N
    var  = sum_sq / N - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # ------------------------------------------------------------------
    # Second pass – normalise + affine transform
    # ------------------------------------------------------------------
    for nb in range(num_n_blocks):
        offs_n = nb * BLOCK_N + tl.arange(0, BLOCK_N)
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        xf = x.to(tl.float32)
        norm = (xf - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_n, mask=mask_n, other=0.0).to(tl.float32)
        b = tl.load(bias_ptr   + offs_n, mask=mask_n, other=0.0).to(tl.float32)

        y = norm * w[None, :] + b[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn,
            y.to(x.dtype),
            mask=mask_m[:, None] & mask_n[None, :],
        )


def triton_layernorm(x: torch.Tensor,
                     weight: torch.Tensor,
                     bias: torch.Tensor,
                     eps: float = 1e-5) -> torch.Tensor:
    """Row‑wise LayerNorm implemented in Triton (input 2‑D)."""
    assert x.dim() == 2, "triton_layernorm expects a 2‑D tensor"
    M, N = x.shape
    y = torch.empty_like(x)

    x = x.contiguous()
    weight = weight.contiguous()
    bias = bias.contiguous()

    stride_xm, stride_xn = x.stride()
    stride_ym, stride_yn = y.stride()

    grid = lambda meta: (triton.cdiv(M, meta["BLOCK_M"]),)

    layernorm_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        weight_ptr=weight,
        bias_ptr=bias,
        M=M,
        N=N,
        eps=eps,
        stride_xm=stride_xm,
        stride_xn=stride_xn,
        stride_ym=stride_ym,
        stride_yn=stride_yn,
        BLOCK_M=128,   # tuned for the H100
        BLOCK_N=64,
    )
    return y


# ----------------------------------------------------------------------
# Triton batched GEMM kernel that directly computes
#   out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
# without any extra permute / contiguous copies.
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_lb, stride_li, stride_lk, stride_lh,
    stride_rb, stride_ri, stride_rk, stride_rh,
    stride_ob, stride_oi, stride_oj, stride_oh,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # ------------------------------------------------------------------
    # 0) Identify which (batch, head) we are processing
    # ------------------------------------------------------------------
    bh = tl.program_id(0)                     # 0 … B*H‑1
    b = bh // H
    h = bh % H

    # ------------------------------------------------------------------
    # 1) Tile coordinates for the output matrix (i‑axis, j‑axis)
    # ------------------------------------------------------------------
    pid_m = tl.program_id(1)                  # row block
    pid_n = tl.program_id(2)                  # col block
    off_m = pid_m * BLOCK_M
    off_n = pid_n * BLOCK_N

    # ------------------------------------------------------------------
    # 2) Base pointers for the selected (b, h) slice
    # ------------------------------------------------------------------
    left_base  = left_ptr  + b * stride_lb + h * stride_lh
    right_base = right_ptr + b * stride_rb + h * stride_rh
    out_base   = out_ptr   + b * stride_ob + h * stride_oh

    # ------------------------------------------------------------------
    # 3) Accumulator (FP32 for numerical stability)
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # 4) Bounds masks for rows / cols
    # ------------------------------------------------------------------
    mask_m = off_m + tl.arange(0, BLOCK_M) < N
    mask_n = off_n + tl.arange(0, BLOCK_N) < N

    # ------------------------------------------------------------------
    # 5) Loop over the reduction dimension k
    # ------------------------------------------------------------------
    num_k = tl.cdiv(N, BLOCK_K)
    for k in range(num_k):
        cur_k = k * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ---- left block: shape (BLOCK_M, BLOCK_K) ----
        offs_i = off_m + tl.arange(0, BLOCK_M)
        left_ptrs = left_base + offs_i[:, None] * stride_li + offs_k[None, :] * stride_lk
        left = tl.load(left_ptrs,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)

        # ---- right block (transposed): shape (BLOCK_K, BLOCK_N) ----
        offs_j = off_n + tl.arange(0, BLOCK_N)
        # we need right[j, k] → address: k*stride_rk + j*stride_ri
        right_ptrs = right_base + offs_k[:, None] * stride_rk + offs_j[None, :] * stride_ri
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_n[None, :],
                        other=0.0).to(tl.float32)

        # ---- MAC ----
        acc += tl.dot(left, right)          # (BLOCK_M, BLOCK_N)

    # ------------------------------------------------------------------
    # 6) Write the result (store as FP16)
    # ------------------------------------------------------------------
    out_ptrs = out_base + off_m[:, None] * stride_oi + off_n[None, :] * stride_oj
    tl.store(out_ptrs,
             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry point required by the test harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor   [B, L, L, D]   (float32)
        - mask         : torch.Tensor   [B, L, L] or None
        - weights      : dict of torch tensors (model parameters)
        - config       : dict with keys ``dim``, ``hidden_dim``,
                         optional ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, L, L, D] (float32)
    """
    # ------------------------------------------------------------------
    # 0️⃣ unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype  = input_tensor.dtype

    bs, L, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (Triton) – normalises over *dim*
    # ------------------------------------------------------------------
    M = bs * L * L                       # flatten batch × seq²
    x_flat = input_tensor.reshape(M, dim)               # [M, D]

    norm_w = weights["norm.weight"].to(device, dtype).contiguous()
    norm_b = weights["norm.bias"  ].to(device, dtype).contiguous()
    x_norm_flat = triton_layernorm(x_flat, norm_w, norm_b, eps=eps)
    x_norm = x_norm_flat.view(bs, L, L, dim)               # [B, L, L, D]

    # ------------------------------------------------------------------
    # 2️⃣ Cast to FP16 for the heavy compute
    # ------------------------------------------------------------------
    x_norm_h = x_norm.to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Fuse the four linear maps (proj + gates) into one matmul
    # ------------------------------------------------------------------
    # weights are bias‑free → simple concatenation
    left_proj_w  = weights["left_proj.weight"].to(device, torch.float16).contiguous()
    right_proj_w = weights["right_proj.weight"].to(device, torch.float16).contiguous()
    left_gate_w  = weights["left_gate.weight"].to(device, torch.float16).contiguous()
    right_gate_w = weights["right_gate.weight"].to(device, torch.float16).contiguous()

    fused_w = torch.cat([left_proj_w,
                         right_proj_w,
                         left_gate_w,
                         right_gate_w], dim=0)                 # (4·H, D)

    # single linear projection
    fused_out = F.linear(x_norm_h.reshape(M, dim), fused_w)    # (M, 4·H)

    # reshape/split
    fused_out = fused_out.view(bs, L, L, 4 * hidden_dim)      # (B, L, L, 4·H)
    left_proj  = fused_out[..., :hidden_dim]                      # (B, L, L, H)
    right_proj = fused_out[..., hidden_dim:2*hidden_dim]
    left_gate  = torch.sigmoid(fused_out[..., 2*hidden_dim:3*hidden_dim])
    right_gate = torch.sigmoid(fused_out[..., 3*hidden_dim:4*hidden_dim])

    # ------------------------------------------------------------------
    # 4️⃣ Apply mask (if any) and the gates
    # ------------------------------------------------------------------
    if (not nomask) and (mask is not None):
        mask_h = mask.to(device, torch.float16).unsqueeze(-1)    # (B, L, L, 1)
        left_proj  = left_proj  * mask_h
        right_proj = right_proj * mask_h

    left  = left_proj  * left_gate        # (B, L, L, H)
    right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Triton batched GEMM – computes the N³ contraction
    # ------------------------------------------------------------------
    out = torch.empty_like(left)         # will hold FP16 result
    # strides of the 4‑D tensors
    stride_lb, stride_li, stride_lk, stride_lh = left.stride()
    stride_rb, stride_ri, stride_rk, stride_rh = right.stride()
    stride_ob, stride_oi, stride_oj, stride_oh = out.stride()

    # tile sizes (tuned for H100 – can be tweaked per N)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    num_blocks_m = triton.cdiv(L, BLOCK_M)
    num_blocks_n = triton.cdiv(L, BLOCK_N)

    # launch grid: (batch*head, row‑tiles, col‑tiles)
    grid = (bs * hidden_dim, num_blocks_m, num_blocks_n)

    tri_mul_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=bs,
        N=L,
        H=hidden_dim,
        stride_lb=stride_lb,
        stride_li=stride_li,
        stride_lk=stride_lk,
        stride_lh=stride_lh,
        stride_rb=stride_rb,
        stride_ri=stride_ri,
        stride_rk=stride_rk,
        stride_rh=stride_rh,
        stride_ob=stride_ob,
        stride_oi=stride_oi,
        stride_oj=stride_oj,
        stride_oh=stride_oh,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 6️⃣ Output LayerNorm (Triton) over the hidden dimension
    # ------------------------------------------------------------------
    out_fp32 = out.to(torch.float32)                # LayerNorm expects FP32
    out_M = bs * L * L
    out_flat = out_fp32.reshape(out_M, hidden_dim)

    to_out_norm_w = weights["to_out_norm.weight"].to(device, dtype).contiguous()
    to_out_norm_b = weights["to_out_norm.bias"  ].to(device, dtype).contiguous()
    out_norm_flat = triton_layernorm(out_flat, to_out_norm_w, to_out_norm_b, eps=eps)
    out_norm = out_norm_flat.view(bs, L, L, hidden_dim)   # (B, L, L, H)

    # ------------------------------------------------------------------
    # 7️⃣ Output gate (linear + sigmoid) – same FP16 path as in the reference
    # ------------------------------------------------------------------
    out_gate_w = weights["out_gate.weight"].to(device, torch.float16).contiguous()
    out_gate = torch.sigmoid(F.linear(x_norm_h.reshape(M, dim), out_gate_w))
    out_gate = out_gate.view(bs, L, L, hidden_dim)      # (B, L, L, H)

    # ------------------------------------------------------------------
    # 8️⃣ Apply output gate and final linear projection back to `dim`
    # ------------------------------------------------------------------
    gated = out_norm * out_gate                 # (FP32 * FP32) → FP32
    gated_h = gated.to(torch.float16)           # FP16 for the final matmul

    to_out_w = weights["to_out.weight"].to(device, torch.float16).contiguous()
    out_final_h = F.linear(gated_h, to_out_w)   # (B, L, L, D)  FP16
    out_final = out_final_h.to(torch.float32)   # required output dtype

    return out_final
