"""
TriMul outgoing forward – Triton‑accelerated implementation.

Algorithm
---------
1️⃣ Input LayerNorm (FP32) – a 2‑pass Triton kernel that computes per‑row
    mean/variance and applies the affine transform.
2️⃣ A single fused linear projection (FP16) produces the five tensors
    needed for the update (left_proj, right_proj, left_gate, right_gate,
    out_gate) by concatenating the corresponding weight matrices.
3️⃣ Mask (if present) and sigmoid gates are applied in FP16.
4️⃣ The core contraction
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   is performed by a custom batched GEMM Triton kernel.  The kernel
   processes the (batch × hidden) dimension as the third program‑id and
   computes each (i,j) tile with a Tensor‑Core‑friendly BLOCK_M×BLOCK_N
   layout, accumulating in FP32 for numerical safety.
5️⃣ Output LayerNorm (FP32) – the same Triton kernel as step 1.
6️⃣ Output gate (sigmoid) and final projection back to the model dimension
   are performed with a half‑precision linear layer (the small cost is
   negligible compared to the N³ contraction).

All heavy compute runs in FP16 (Tensor‑Core friendly) while the two
LayerNorms run in FP32 for stability.  At least one operation (the
contraction) is fully implemented in Triton as required.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# 2‑pass LayerNorm kernel (FP32 accumulation, FP32 output)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr, y_ptr,                # *ptr to input / output [M, N] (N = dim)
    weight_ptr, bias_ptr,        # *ptr to weight / bias   [N]
    M, N, eps,                   # tensor dimensions
    stride_xm, stride_xn,        # strides for input
    stride_ym, stride_yn,        # strides for output
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
):
    pid = tl.program_id(0)

    # ------------------------------------------------------------------
    # Row range for this program
    # ------------------------------------------------------------------
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # ------------------------------------------------------------------
    # First pass: compute per‑row mean & variance (FP32)
    # ------------------------------------------------------------------
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_N)
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        x_f = x.to(tl.float32)
        sum_val += tl.sum(x_f, axis=1)
        sum_sq  += tl.sum(x_f * x_f, axis=1)

    mean = sum_val / N
    var  = sum_sq / N - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # ------------------------------------------------------------------
    # Second pass: normalise and apply affine parameters
    # ------------------------------------------------------------------
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        x_f = x.to(tl.float32)

        # Normalise
        norm_x = (x_f - mean[:, None]) * rstd[:, None]

        # Load scale / bias (still in FP32)
        w = tl.load(weight_ptr + offs_n, mask=mask_n, other=0.0).to(tl.float32)
        b = tl.load(bias_ptr   + offs_n, mask=mask_n, other=0.0).to(tl.float32)

        y = norm_x * w[None, :] + b[None, :]

        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn,
            y,
            mask=mask_m[:, None] & mask_n[None, :],
        )


def triton_layernorm(x: torch.Tensor,
                     weight: torch.Tensor,
                     bias: torch.Tensor,
                     eps: float = 1e-5,
                     BLOCK_M: int = 128,
                     BLOCK_N: int = 64) -> torch.Tensor:
    """LayerNorm over the last dimension of a 2‑D tensor ``x`` (M×N)."""
    assert x.dim() == 2, "triton_layernorm expects a 2‑D tensor"
    M, N = x.shape
    y = torch.empty_like(x)

    # Ensure contiguous strides for the kernel
    x = x.contiguous()
    y = y.contiguous()
    weight = weight.contiguous()
    bias = bias.contiguous()

    stride_xm, stride_xn = x.stride()
    stride_ym, stride_yn = y.stride()

    grid = (triton.cdiv(M, BLOCK_M),)

    layernorm_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        weight_ptr=weight,
        bias_ptr=bias,
        M=M,
        N=N,
        eps=eps,
        stride_xm=stride_xm,
        stride_xn=stride_xn,
        stride_ym=stride_ym,
        stride_yn=stride_yn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
    )
    return y


# ----------------------------------------------------------------------
# Batched half‑precision matrix multiplication kernel (outgoing)
#   out[b,h,i,j] = Σₖ left[b,h,i,k] * right[b,h,k,j]
# ----------------------------------------------------------------------
@triton.jit
def triton_matmul_outgoing_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,                         # all equal to seq_len
    stride_ah, stride_am, stride_ak, # strides for A (head, row, col)
    stride_bh, stride_bk, stride_bn, # strides for B (head, row, col)
    stride_ch, stride_cm, stride_cn, # strides for C (head, row, col)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_head = tl.program_id(2)   # index = batch * hidden
    pid_m = tl.program_id(0)      # tile row
    pid_n = tl.program_id(1)      # tile col

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k = tl.cdiv(K, BLOCK_K)
    for k in range(num_k):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load a tile of A and B (both FP16)
        a = tl.load(
            A_ptr
            + pid_head * stride_ah
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        b = tl.load(
            B_ptr
            + pid_head * stride_bh
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # Tensor‑core dot, accumulate in FP32
        acc += tl.dot(a, b)

    # Write back (cast to FP16 for downstream ops)
    c = acc.to(tl.float16)
    tl.store(
        C_ptr
        + pid_head * stride_ch
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn,
        c,
        mask=mask_m[:, None] & mask_n[:, None],
    )


def triton_batched_matmul(A: torch.Tensor, B: torch.Tensor,
                         BLOCK_M: int = 64,
                         BLOCK_N: int = 64,
                         BLOCK_K: int = 128) -> torch.Tensor:
    """
    Batched matmul for tensors of shape (B, H, N, N) (FP16).
    Returns C of the same shape and dtype.
    """
    B_dim, H_dim, N, _ = A.shape
    BH = B_dim * H_dim

    # Collapse batch‑head into a single dimension for the kernel
    A_flat = A.view(BH, N, N)
    B_flat = B.view(BH, N, N)
    C_flat = torch.empty_like(A_flat)

    stride_ah, stride_am, stride_ak = A_flat.stride()
    stride_bh, stride_bk, stride_bn = B_flat.stride()
    stride_ch, stride_cm, stride_cn = C_flat.stride()

    grid = (
        triton.cdiv(N, BLOCK_M),   # rows (i)
        triton.cdiv(N, BLOCK_N),   # cols (j)
        BH,                        # batch * hidden
    )

    triton_matmul_outgoing_kernel[grid](
        A_ptr=A_flat,
        B_ptr=B_flat,
        C_ptr=C_flat,
        M=N,
        N=N,
        K=N,
        stride_ah=stride_ah,
        stride_am=stride_am,
        stride_ak=stride_ak,
        stride_bh=stride_bh,
        stride_bk=stride_bk,
        stride_bn=stride_bn,
        stride_ch=stride_ch,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,      # more warps → better occupancy on H100
        num_stages=4,     # double‑buffering
    )

    # Reshape back to (B, H, N, N)
    return C_flat.view(B_dim, H_dim, N, N)

    # ------------------------------------------------------------------

    right_gate_raw = right_gate_raw.view(bs, seq_len, seq_len, hidden_dim)

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------

    right = right_raw * right_gate

    # ------------------------------------------------------------------
    # 6️⃣ Batched matrix multiplication (outgoing contraction)
    #    out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    # Rearrange to (B, H, N, N) so the third program‑id indexes head.
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N)

    out_perm = triton_batched_matmul(
        left_perm,
        right_perm,
        BLOCK_M=64,
        BLOCK_N=64,
        BLOCK_K=128,
    )   # (B, H, N, N) FP16

    out = out_perm.permute(0, 2, 3, 1).contiguous()    # (B,N,N,H) FP16

    # ------------------------------------------------------------------
    # 7️⃣ Output LayerNorm (FP32 – Triton)
    # ------------------------------------------------------------------
    out_fp32 = out.float()                               # cast for LN
    out_flat = out_fp32.view(M, hidden_dim)
    to_out_norm_weight = weights["to_out_norm.weight"].to(device, dtype=torch.float32)
    to_out_norm_bias   = weights["to_out_norm.bias"].to(device, dtype=torch.float32)

    out_norm_flat = triton_layernorm(
        out_flat,
        to_out_norm_weight,
        to_out_norm_bias,
        eps=1e-5,
        BLOCK_M=128,
        BLOCK_N=64,
    )
    out_norm = out_norm_flat.view(bs, seq_len, seq_len, hidden_dim)   # (B,N,N,H) FP32

    # ------------------------------------------------------------------
    # 8️⃣ Output gate + final projection back to model dimension
    # ------------------------------------------------------------------
    gated = out_norm * out_gate.float()                # (B,N,N,H) FP32

    # Final linear projection (FP16 weight)
    to_out_weight = weights["to_out.weight"].to(device, dtype=torch.float16)  # (C, H)
    gated_h = gated.half()
    final_fp16 = F.linear(gated_h, to_out_weight)      # (B,N,N,C) FP16
    final_fp32 = final_fp16.float()                    # back to FP32

    return final_fp32
