"""
TriMul (outgoing) – fast forward pass.

Algorithm
---------
1️⃣  Layer‑norm on the input tensor (flattened to 2‑D) – Triton kernel.
2️⃣  Cast to FP16 for the bulk of the work.
3️⃣  Fuse the four linear projections (left‑proj, left‑gate,
    right‑proj, right‑gate) into a single GEMM:
       [B·L·L, 4·H] = [B·L·L, D] @ Wᵀ  ,   W ∈ ℝ^{4·H×D}
   Split the result and apply a sigmoid to the two gate slices.
4️⃣  (Optional) apply the mask to the two projected tensors.
5️⃣  Multiply each projected tensor with its gate → final `left` and `right`
    (shape [B, L, L, H], FP16).
6️⃣  Compute the N³ contraction as a batched matrix‑multiply:
       outₕ = leftₕ @ rightₕᵀ   (for every hidden slice)
    Implemented with a single `torch.bmm` on the flattened batch
    dimension (B·H, L, L).  This exploits Tensor‑Cores on H100.
7️⃣  Row‑wise LayerNorm on the hidden dimension – same Triton kernel as (1).
8️⃣  Apply the output gate (computed by a second linear + sigmoid) and
    the final projection back to the model dimension.

Only the two LayerNorms are written in Triton; all dense linear algebra
uses the highly‑optimised cuBLAS kernels via `torch.nn.functional.linear`
and `torch.bmm`.  The fused projection/gate GEMM reduces the number of
large matrix multiplies from five to two, cutting memory traffic and
runtime dramatically while preserving exact numerical semantics
(the final output is returned in float32).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton row‑wise LayerNorm (works for any 2‑D tensor [M, N]).
# ----------------------------------------------------------------------
@triton.jit
def _layernorm_kernel(
    x_ptr,      # *float   input  [M, N]
    y_ptr,      # *float   output [M, N]
    w_ptr,      # *float   weight [N]
    b_ptr,      # *float   bias   [N]
    M, N, eps,
    stride_xm, stride_xn,
    stride_ym, stride_yn,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
):
    pid = tl.program_id(0)

    # ------------------------------------------------------------------
    # Row offset for this program
    # ------------------------------------------------------------------
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # ------------------------------------------------------------------
    # First pass: compute mean & variance (FP32 accumulation)
    # ------------------------------------------------------------------
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_n_blocks = tl.cdiv(N, BLOCK_N)
    for n in range(num_n_blocks):
        offs_n = n * BLOCK_N + tl.arange(0, BLOCK_N)
        mask_n = offs_n < N

        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        x_f32 = x.to(tl.float32)
        sum_val += tl.sum(x_f32, axis=1)
        sum_sq  += tl.sum(x_f32 * x_f32, axis=1)

    mean = sum_val / N
    var  = sum_sq / N - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # ------------------------------------------------------------------
    # Second pass: write normalized + affine‑transformed data
    # ------------------------------------------------------------------
    for n in range(num_n_blocks):
        offs_n = n * BLOCK_N + tl.arange(0, BLOCK_N)
        mask_n = offs_n < N

        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        x_f32 = x.to(tl.float32)
        norm_x = (x_f32 - mean[:, None]) * rstd[:, None]

        w = tl.load(w_ptr + offs_n, mask=mask_n, other=0.0)
        b = tl.load(b_ptr + offs_n, mask=mask_n, other=0.0)
        w_f32 = w.to(tl.float32)
        b_f32 = b.to(tl.float32)

        y_f32 = norm_x * w_f32 + b_f32
        y = y_f32.to(x.dtype)

        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn,
            y,
            mask=mask_m[:, None] & mask_n[None, :],
        )


def _triton_layernorm(x: torch.Tensor,
                      weight: torch.Tensor,
                      bias: torch.Tensor,
                      eps: float = 1e-5) -> torch.Tensor:
    """
    Row‑wise LayerNorm implemented in Triton.
    `x` must be 2‑D (M, N); output has the same shape and dtype as `x`.
    """
    assert x.dim() == 2, "LayerNorm kernel expects a 2‑D tensor"

    M, N = x.shape
    y = torch.empty_like(x)

    # Ensure contiguity – required for correct stride extraction
    x = x.contiguous()
    y = y.contiguous()
    weight = weight.contiguous()
    bias = bias.contiguous()

    stride_xm, stride_xn = x.stride()
    stride_ym, stride_yn = y.stride()

    grid = lambda meta: (triton.cdiv(M, meta["BLOCK_M"]),)

    _layernorm_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        w_ptr=weight,
        b_ptr=bias,
        M=M,
        N=N,
        eps=eps,
        stride_xm=stride_xm,
        stride_xn=stride_xn,
        stride_ym=stride_ym,
        stride_yn=stride_yn,
        BLOCK_M=128,   # rows per program (tuned for our workloads)
        BLOCK_N=64,    # columns per tile (covers hidden_dim up to 384)
    )
    return y


# ----------------------------------------------------------------------
# Entry point required by the benchmark harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor   [B, L, L, D]   (float32)
        - mask         : torch.Tensor   [B, L, L]   (float32) or ``None``
        - weights      : dict of torch tensors with keys:
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
        - config : dict with fields ``dim`` (D), ``hidden_dim`` (H) and optional
                   ``nomask`` (bool).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, L, L, D] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype  # expected to be torch.float32
    eps = 1e-5

    bs, L, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (Triton) – flatten to 2‑D
    # ------------------------------------------------------------------
    x_flat = input_tensor.reshape(bs * L * L, dim).contiguous()
    norm_w = weights["norm.weight"].to(device, dtype).contiguous()
    norm_b = weights["norm.bias"].to(device, dtype).contiguous()
    x_norm_flat = _triton_layernorm(x_flat, norm_w, norm_b, eps=eps)
    x_norm = x_norm_flat.view(bs, L, L, dim)

    # ------------------------------------------------------------------
    # 2️⃣ Cast to FP16 for the heavy computation
    # ------------------------------------------------------------------
    x_h = x_norm.to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Fuse four linear projections (proj + gate for both sides)
    # ------------------------------------------------------------------
    # Gather the four weight matrices (all bias‑free)
    left_proj_w = weights["left_proj.weight"].to(device, torch.float16).contiguous()
    left_gate_w = weights["left_gate.weight"].to(device, torch.float16).contiguous()
    right_proj_w = weights["right_proj.weight"].to(device, torch.float16).contiguous()
    right_gate_w = weights["right_gate.weight"].to(device, torch.float16).contiguous()

    # Concatenate along the out‑feature dimension → shape (4*H, D)
    combined_w = torch.cat(
        [left_proj_w, left_gate_w, right_proj_w, right_gate_w], dim=0
    )  # (4*H, D)

    # Single GEMM: (B·L·L, D) @ (D, 4·H).T → (B·L·L, 4·H)
    combined_out = F.linear(x_h, combined_w)  # FP16, no bias

    # Split the result
    h = hidden_dim
    left_proj   = combined_out[..., :h]                # (B, L, L, H)
    left_gate_pr = combined_out[..., h:2 * h]
    right_proj  = combined_out[..., 2 * h:3 * h]
    right_gate_pr = combined_out[..., 3 * h:4 * h]

    # Apply sigmoid to the gate tensors
    left_gate  = torch.sigmoid(left_gate_pr)
    right_gate = torch.sigmoid(right_gate_pr)

    # ------------------------------------------------------------------
    # 4️⃣ Optional mask
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_h = mask.to(device, torch.float16).unsqueeze(-1)  # (B, L, L, 1)
        left_proj = left_proj * mask_h
        right_proj = right_proj * mask_h

    # ------------------------------------------------------------------
    # 5️⃣ Multiply projections with their gates → final left/right tensors
    # ------------------------------------------------------------------
    left  = left_proj * left_gate   # (B, L, L, H)
    right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 6️⃣ Output gate (separate linear + sigmoid)
    # ------------------------------------------------------------------
    out_gate_w = weights["out_gate.weight"].to(device, torch.float16).contiguous()
    out_gate_pre = F.linear(x_h, out_gate_w)          # (B, L, L, H)
    out_gate = torch.sigmoid(out_gate_pre)            # (B, L, L, H)

    # ------------------------------------------------------------------
    # 7️⃣ Batched GEMM (the N³ contraction)
    #    out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    # reshape to (B*H, L, L) – each row of the batch will compute one
    # matrix multiplication; right is transposed on the second axis.
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(bs * hidden_dim, L, L)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(bs * hidden_dim, L, L)

    # (B*H, L, L) @ (B*H, L, L)ᵀ  →  (B*H, L, L)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # Reshape back to the original 4‑D layout
    out = (
        out_flat.view(bs, hidden_dim, L, L)
        .permute(0, 2, 3, 1)
        .contiguous()
    )  # (B, L, L, H), still FP16

    # ------------------------------------------------------------------
    # 8️⃣ Output LayerNorm over the hidden dimension (Triton)
    # ------------------------------------------------------------------
    out_flat2 = out.reshape(bs * L * L, hidden_dim).contiguous()
    to_out_norm_w = weights["to_out_norm.weight"].to(device, dtype).contiguous()
    to_out_norm_b = weights["to_out_norm.bias"].to(device, dtype).contiguous()
    out_norm_flat = _triton_layernorm(out_flat2, to_out_norm_w, to_out_norm_b, eps=eps)
    out_norm = out_norm_flat.view(bs, L, L, hidden_dim)   # (B, L, L, H), float32

    # ------------------------------------------------------------------
    # 9️⃣ Apply the output gate and final linear projection back to dim
    # ------------------------------------------------------------------
    out = out_norm * out_gate.to(dtype)                 # broadcast over H
    to_out_w = weights["to_out.weight"].to(device, dtype).contiguous()
    out = F.linear(out, to_out_w)                       # (B, L, L, D), float32

    return out
