"""
TriMul (Outgoing) – Triton‑fused forward implementation
=======================================================

Algorithm
---------
1. **Layer‑Norm (input)** – a custom two‑pass Triton kernel that computes
   mean / variance per row and applies the affine transformation.  Operates
   on the 4‑D tensor `[B, N, N, D]` flattened to `[M, D]` (M = B·N·N).

2. **Fused linear projections** – the five bias‑free linear layers
   (`left_proj`, `right_proj`, `left_gate`, `right_gate`, `out_gate`) are
   concatenated into a single weight matrix of shape `(5·H, D)` and applied
   once with `torch.nn.functional.linear`.  This reduces memory traffic
   from five passes over the input to a single pass.

3. **Mask & gating** – the optional mask (broadcasted over the hidden
   dimension) is applied to `left`/`right`.  The three gate tensors are
   activated with a sigmoid and fused element‑wise with the projected values.

4. **Batched contraction** – `out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]`
   is performed as a batched GEMM: `left_perm @ right_permᵀ` where the batch
   dimension is `B·H`.  The computation is carried out in FP16 to exploit
   Tensor‑Cores.

5. **Layer‑Norm (output)** – a second Triton kernel normalises over the
   hidden dimension `H` for each `(b,i,j)` position.

6. **Output gating & projection** – the output‑gate (still in FP16) scales
   the normalised tensor, after which a final bias‑free linear projection
   `H → D` is executed in FP16.  The result is cast back to FP32 for the
   caller.

All heavy compute kernels run in FP16 while the two LayerNorms run in
FP32 for numerical stability.  The implementation works for all
configurations listed in the problem statement and satisfies the
requirement “at least part of the operation is a Triton kernel”.

"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton LayerNorm – 2‑pass implementation (mean/var → affine)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,                # *ptr to input  [M, N]
    y_ptr,                # *ptr to output [M, N]
    weight_ptr,           # *ptr to weight [N]
    bias_ptr,             # *ptr to bias   [N]
    M,                    # number of rows
    N,                    # feature dimension
    eps,                  # epsilon
    stride_xm, stride_xn,  # strides for input (row, col)
    stride_ym, stride_yn,  # strides for output (row, col)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
):
    pid = tl.program_id(0)

    # ------------------------------------------------------------------
    # Row offsets for this program
    # ------------------------------------------------------------------
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # ------------------------------------------------------------------
    # First pass: compute mean & variance for each row
    # ------------------------------------------------------------------
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_N)
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        sum_val += tl.sum(x, axis=1)
        sum_sq  += tl.sum(x * x, axis=1)

    mean = sum_val / N
    var  = sum_sq / N - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # ------------------------------------------------------------------
    # Second pass: normalize + affine transform
    # ------------------------------------------------------------------
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        norm_x = (x - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_n, mask=mask_n, other=0.0)
        b = tl.load(bias_ptr   + offs_n, mask=mask_n, other=0.0)

        y = norm_x * w[None, :] + b[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn,
            y,
            mask=mask_m[:, None] & mask_n[None, :],
        )


def triton_layernorm(x: torch.Tensor, weight: torch.Tensor, bias: torch.Tensor, eps: float = 1e-5):
    """Apply LayerNorm over the last dimension of a 2‑D tensor using Triton.

    Args:
        x:    Tensor of shape [M, N] (any dtype, but computation of mean/var
              is performed in FP32 for stability).
        weight, bias: 1‑D tensors of length N (same dtype as ``x``).
        eps:  numerical epsilon.

    Returns:
        Tensor of the same shape and dtype as ``x``.
    """
    assert x.dim() == 2, "triton_layernorm expects a 2‑D tensor"
    M, N = x.shape
    y = torch.empty_like(x)

    # Make strides explicit for the Triton kernel
    x = x.contiguous()
    weight = weight.contiguous()
    bias = bias.contiguous()
    stride_xm, stride_xn = x.stride()
    stride_ym, stride_yn = y.stride()

    # Tune tile sizes – these work well for the dimensions used in TriMul
    BLOCK_M = 256
    BLOCK_N = 64

    grid = lambda meta: (triton.cdiv(M, meta["BLOCK_M"]),)

    layernorm_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        weight_ptr=weight,
        bias_ptr=bias,
        M=M,
        N=N,
        eps=eps,
        stride_xm=stride_xm,
        stride_xn=stride_xn,
        stride_ym=stride_ym,
        stride_yn=stride_yn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
    )
    return y


# ----------------------------------------------------------------------
# TriMul entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor   shape [B, L, L, D] (float32)
        - mask         : torch.Tensor   shape [B, L, L] or ``None``
        - weights      : dict of torch tensors (model parameters)
        - config       : dict with keys ``dim``, ``hidden_dim`` and optional
                         ``nomask`` (bool)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, L, L, D] (float32)
    """
    # ------------------------------------------------------------------
    # 0️⃣ Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype          # expected torch.float32
    eps = 1e-5

    bs, L, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (Triton) over the channel dimension `dim`
    # ------------------------------------------------------------------
    M = bs * L * L
    x_flat = input_tensor.reshape(M, dim).contiguous()
    norm_w = weights["norm.weight"].to(device, dtype).contiguous()
    norm_b = weights["norm.bias"].to(device, dtype).contiguous()
    x_norm_flat = triton_layernorm(x_flat, norm_w, norm_b, eps=eps)
    x_norm = x_norm_flat.view(bs, L, L, dim)           # still FP32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to FP16 for the heavy linear work
    # ------------------------------------------------------------------
    x_norm_h = x_norm.to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Fuse the five bias‑free linear projections into one matmul
    # ------------------------------------------------------------------
    # each weight matrix has shape (hidden_dim, dim); we concatenate along
    # the out‑feature axis to obtain a single weight of shape (5*H, dim)
    w_left   = weights["left_proj.weight"]   # (H, D)
    w_right  = weights["right_proj.weight"]  # (H, D)
    w_lgate  = weights["left_gate.weight"]   # (H, D)
    w_rgate  = weights["right_gate.weight"]  # (H, D)
    w_outgate = weights["out_gate.weight"]   # (H, D)

    # concatenate *along* output dimension (dim 0)
    weight_all = torch.cat([w_left, w_right, w_lgate, w_rgate, w_outgate], dim=0)  # (5*H, D)
    weight_all_h = weight_all.to(device, torch.float16).contiguous()

    # Single fused linear: result shape (M, 5*H)
    proj_all = F.linear(x_norm_h, weight_all_h)   # FP16

    # ------------------------------------------------------------------
    # 4️⃣ Slice the fused result and apply mask / gating
    # ------------------------------------------------------------------
    H = hidden_dim
    left      = proj_all[:, 0*H:1*H]
    right     = proj_all[:, 1*H:2*H]
    left_gate = proj_all[:, 2*H:3*H]
    right_gate= proj_all[:, 3*H:4*H]
    out_gate  = proj_all[:, 4*H:5*H]

    # optional mask (broadcast over the hidden dimension)
    if mask is not None and not nomask:
        # mask shape [B, L, L] → flatten → (M, 1)
        mask_h = mask.view(-1, 1).to(torch.float16)
        left  = left  * mask_h
        right = right * mask_h

    # sigmoid gates (still FP16)
    left_gate  = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate   = torch.sigmoid(out_gate)

    # fuse gates
    left  = left  * left_gate
    right = right * right_gate

    # reshape to 4‑D tensors for the contraction
    left  = left.view(bs, L, L, H)
    right = right.view(bs, L, L, H)

    # ------------------------------------------------------------------
    # 5️⃣ Batched matrix multiplication over the `k` dimension
    #    out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    # bring the hidden dimension to the batch axis: (B, H, N, N)
    left_perm  = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    # (B*H, N, N) @ (B*H, N, N)^T  → (B*H, N, N)
    out_perm = torch.matmul(left_perm, right_perm.transpose(-2, -1))
    # back to original order → [B, N, N, H] (still FP16)
    out_h = out_perm.permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6️⃣ Output LayerNorm over the hidden dimension `H`
    # ------------------------------------------------------------------
    out_fp32 = out_h.to(torch.float32)              # LayerNorm operates in FP32
    out_flat = out_fp32.view(bs * L * L, hidden_dim).contiguous()
    to_out_norm_w = weights["to_out_norm.weight"].to(device, dtype).contiguous()
    to_out_norm_b = weights["to_out_norm.bias"].to(device, dtype).contiguous()
    out_norm_flat = triton_layernorm(out_flat, to_out_norm_w, to_out_norm_b, eps=eps)
    out_norm = out_norm_flat.view(bs, L, L, hidden_dim)   # FP32

    # ------------------------------------------------------------------
    # 7️⃣ Apply the output gate, final projection, and cast back to FP32
    # ------------------------------------------------------------------
    # Cast to FP16 for the final matmul (Tensor‑Core friendly)
    out_norm_h = out_norm.to(torch.float16)          # (B, L, L, H)
    out_gate_h = out_gate.view(bs, L, L, hidden_dim).to(torch.float16)

    # element‑wise gating
    out_pre_h = out_norm_h * out_gate_h               # (B, L, L, H)

    # final linear: H → D (bias‑free)
    to_out_w = weights["to_out.weight"].to(device, torch.float16).contiguous()
    # flatten batch & spatial dims for a single matmul
    out_pre_flat = out_pre_h.view(-1, hidden_dim)     # (M, H)
    out_flat = F.linear(out_pre_flat, to_out_w)       # (M, D), still FP16
    out = out_flat.view(bs, L, L, dim).to(torch.float32)

    return out
