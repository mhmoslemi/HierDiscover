```python
@triton.jit
def mask_gate_fuse_kernel(
    left_ptr, # *fp16
    right_ptr,
    left_gate_ptr,
    right_gate_ptr,
    mask_ptr, # *fp16
    total_elems: tl.int64,
    hidden_dim: tl.int64,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems
    # load at indices
    # compute mask_idx = offs // hidden_dim
    mask_idx = offs // hidden_dim
    mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=tl.float16(0.0))
    left = tl.load(left_ptr + offs, mask=mask, other=tl.float16(0.0))
    right = tl.load(right_ptr + offs, mask=mask, other=tl.float16(0.0))
    left_gate_raw = tl.load(left_gate_ptr + offs, mask=mask, other=tl.float16(0.0))
    right_gate_raw = tl.load(right_gate_ptr + offs, mask=mask, other=tl.float16(0.0))
    # sigmoid
    left_gate = 1.0 / (1.0 + tl.exp(-left_gate_raw))
    right_gate = 1.0 / (1.0 + tl.exp(-right_gate_raw))
    left_out = left * left_gate * mask_val
    right_out = right * right_gate * mask_val
    tl.store(left_ptr + offs, left_out, mask=mask)
    tl.store(right_ptr + offs, right_out, mask=mask)
```

```python
@triton.jit
def mask_gate_fuse_kernel(
    left_ptr,
    right_ptr,
    left_gate_ptr,
    right_gate_ptr,
    mask_ptr,
    total_elements,
    hidden_dim,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elements

    # Compute mask index
    mask_idx = offs // hidden_dim

    # Load mask (broadcasted)
    mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=tl.float16(0.0))

    # Load data
    left = tl.load(left_ptr + offs, mask=mask, other=tl.float16(0.0))
    right = tl.load(right_ptr + offs, mask=mask, other=tl.float16(0.0))
    left_g = tl.load(left_gate_ptr + offs, mask=mask, other=tl.float16(0.0))
    right_g = tl.load(right_gate_ptr + offs, mask=mask, other=tl.float16(0.0))

    # Sigmoid
    left_g = 1.0 / (1.0 + tl.exp(-left_g))
    right_g = 1.0 / (1.0 + tl.exp(-right_g))

    # Apply mask and gates
    left_out = left * left_g * mask_val
    right_out = right * right_g * mask_val

    # Store results
    tl.store(left_ptr + offs, left_out, mask=mask)
    tl.store(right_ptr + offs, right_out, mask=mask)
```

"""
TriMul (outgoing) – fast Triton + cuBLAS implementation.

Key ideas
---------
1.  Fuse the four linear projections (left_proj, right_proj,
    left_gate, right_gate) into **one** large GEMM by concatenating the
    weight matrices along the output‑feature dimension.  This halves the
    number of cuBLAS launches (5 → 3) and reduces the total weight‑load
    traffic.

2.  Fuse the optional input mask together with the two sigmoid gates in
    a tiny Triton kernel.  The kernel loads a single element of the four
    tensors, computes the two sigmoids (`1/(1+exp(-x))`), multiplies by
    the mask (broadcast over the hidden dimension) and writes the
    result back in‑place – one memory pass instead of three.

3.  Perform the heavy O(N³) contraction with a single batched GEMM
    (`torch.bmm`).  After the permute the left tensor has shape
    `(B, H, i, k)` and the right tensor `(B, H, k, j)`.  The batched
    matrix multiplication yields the desired `out[i, j, d]`.

4.  The remaining lightweight ops (LayerNorm, final linear projection,
    final gating) stay in PyTorch – they are already highly tuned
    (cuDNN / cuBLAS tensor‑cores).

The implementation works for all test configurations, supports the
optional boolean/float mask, and returns a `float32` tensor as required.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr,          # fp16  [B*N*N*H]   – projected left (in‑place)
    right_ptr,         # fp16  [B*N*N*H]   – projected right (in‑place)
    left_gate_ptr,     # fp16  [B*N*N*H]   – raw left‑gate logits
    right_gate_ptr,    # fp16  [B*N*N*H]   – raw right‑gate logits
    mask_ptr,          # fp16  [B*N*N]     – broadcast mask (flattened)
    total_elems,       # i64   total number of elements = B*N*N*H
    hidden_dim,        # i64   H
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)          # offsets into the flat tensors
    mask = offs < total_elems                                    # mask for OOB threads

    # ----- mask index (broadcast over hidden dimension) -----
    mask_idx = offs // hidden_dim                                 # integer division
    mask_val = tl.load(mask_ptr + mask_idx,
                       mask=mask,
                       other=tl.float16(0.0))

    # ----- load tensors -----
    left      = tl.load(left_ptr      + offs, mask=mask, other=tl.float16(0.0))
    right     = tl.load(right_ptr     + offs, mask=mask, other=tl.float16(0.0))
    left_g    = tl.load(left_gate_ptr + offs, mask=mask, other=tl.float16(0.0))
    right_g   = tl.load(right_gate_ptr+ offs, mask=mask, other=tl.float16(0.0))

    # ----- sigmoid (in fp16) -----
    left_g  = tl.float16(1.0) / (tl.float16(1.0) + tl.exp(-left_g))
    right_g = tl.float16(1.0) / (tl.float16(1.0) + tl.exp(-right_g))

    # ----- fused element‑wise operation -----
    left_out  = left  * left_g  * mask_val
    right_out = right * right_g * mask_val

    # ----- store back (in‑place) -----
    tl.store(left_ptr  + offs, left_out,  mask=mask)
    tl.store(right_ptr + offs, right_out, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul block (no gradients).

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, D] (float32)
            - mask_tensor : torch.Tensor [B, N, N] (bool/float) – may be None
            - weights_dict : mapping of all module parameters (float32)
            - config_dict  : {"dim": D, "hidden_dim": H, "nomask": bool, ...}

    Returns:
        torch.Tensor [B, N, N, D] (float32)
    """
    # --------------------------------------------------------------------- #
    # 1) unpack arguments
    # --------------------------------------------------------------------- #
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    H = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5

    # --------------------------------------------------------------------- #
    # 2) Input LayerNorm (float32 for numerical stability)
    # --------------------------------------------------------------------- #
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )   # [B, N, N, dim]  (float32)

    # --------------------------------------------------------------------- #
    # 3) Cast to fp16 – the heavy kernels run in half precision
    # --------------------------------------------------------------------- #
    x_h = x.half()                               # [B,N,N,dim] (fp16)

    # --------------------------------------------------------------------- #
    # 4) Concatenate projection + gate weights and apply a single GEMM
    # --------------------------------------------------------------------- #
    # 4‑a) Load fp16 weights
    left_proj_w   = weights["left_proj.weight"].half()    # (H, dim)
    right_proj_w  = weights["right_proj.weight"].half()   # (H, dim)
    left_gate_w   = weights["left_gate.weight"].half()    # (H, dim)
    right_gate_w  = weights["right_gate.weight"].half()   # (H, dim)

    # concatenate 4 weight matrices along the output‑feature axis (dim=0)
    proj_gate_W = torch.cat(
        [left_proj_w, right_proj_w, left_gate_w, right_gate_w],
        dim=0,
    )   # shape (4*H, dim)

    # flatten input for the linear layer: (B*N*N, dim)
    x_flat = x_h.reshape(-1, dim)                 # fp16

    # one large GEMM: (B*N*N, dim) @ (4*H, dim).T -> (B*N*N, 4*H)
    combined = F.linear(x_flat, proj_gate_W)       # fp16
    # split the combined result back into the four tensors
    left_proj, right_proj, left_gate_logit, right_gate_logit = torch.split(
        combined, H, dim=1
    )   # each (B*N*N, H)

    # --------------------------------------------------------------------- #
    # 5) Compute the out‑gate (its own linear + sigmoid)
    # --------------------------------------------------------------------- #
    out_gate_w = weights["out_gate.weight"].half()      # (H, dim)
    out_gate_logit = F.linear(x_flat, out_gate_w)       # (B*N*N, H)
    out_gate = torch.sigmoid(out_gate_logit).half()     # (B*N*N, H)

    # --------------------------------------------------------------------- #
    # 6) Fuse mask (if present) and the two sigmoid gates onto the
    #    projected tensors – Triton kernel does the element‑wise work.
    # --------------------------------------------------------------------- #
    if nomask:
        # No mask → simple element‑wise multiplication
        left_gate  = torch.sigmoid(left_gate_logit).half()
        right_gate = torch.sigmoid(right_gate_logit).half()
        left  = left_proj  * left_gate
        right = right_proj * right_gate
    else:
        # Ensure contiguous storage for flat indexing
        left_proj      = left_proj.contiguous()
        right_proj     = right_proj.contiguous()
        left_gate_logit  = left_gate_logit.contiguous()
        right_gate_logit = right_gate_logit.contiguous()

        # Convert mask to fp16 and flatten (B*N*N)
        mask_fp16 = mask.to(dtype=torch.float16, device=device)
        mask_flat = mask_fp16.reshape(-1).contiguous()

        total_elements = left_proj.numel()                # B*N*N*H
        BLOCK_SIZE = 1024                                 # ≤ CUDA max threads per block
        grid = (total_elements + BLOCK_SIZE - 1) // BLOCK_SIZE

        _mask_gate_fuse_kernel[grid](
            left_proj,
            right_proj,
            left_gate_logit,
            right_gate_logit,
            mask_flat,
            total_elements,
            H,
            BLOCK_SIZE=BLOCK_SIZE,
            num_warps=4,          # 4 warps = 128 threads, good occupancy on H100
        )
        # The kernel overwrites left_proj/right_proj with the gated+masked values
        left, right = left_proj, right_proj

    # --------------------------------------------------------------------- #
    # 7) Reshape to 4‑D tensors ready for the O(N³) contraction
    # --------------------------------------------------------------------- #
    left  = left.reshape(B, N, N, H)                     # [B,N,N,H]
    right = right.reshape(B, N, N, H)                    # [B,N,N,H]

    # Permute to (B, H, i, k) for left and (B, H, k, j) for right
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, i, k)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, k, j)

    # Collapse the batch & head dimensions for batched GEMM
    batch = B * H
    left_flat  = left_perm.reshape(batch, N, N)          # (B*H, i, k)
    right_flat = right_perm.reshape(batch, N, N)         # (B*H, k, j)

    # --------------------------------------------------------------------- #
    # 8) Heavy O(N³) contraction via a single batched GEMM
    # --------------------------------------------------------------------- #
    out_flat = torch.bmm(left_flat, right_flat)          # (B*H, i, j)

    # Reshape back to [B, N, N, H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------------- #
    # 9) Hidden‑dim LayerNorm, out‑gate multiplication and final linear
    # --------------------------------------------------------------------- #
    out = F.layer_norm(
        out,
        normalized_shape=(H,),
        weight=weights["to_out_norm.weight"].half(),
        bias=weights["to_out_norm.bias"].half(),
        eps=eps,
    )
    # reshape out_gate to the same 4‑D layout and apply
    out_gate = out_gate.reshape(B, N, N, H)
    out = out * out_gate

    # final projection back to original dimension D
    out = F.linear(out, weights["to_out.weight"].half())   # (B,N,N,dim)

    # --------------------------------------------------------------------- #
    # 10) Cast back to float32 for downstream consumption
    # --------------------------------------------------------------------- #
    return out.float()
