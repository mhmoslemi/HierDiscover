@triton.jit
def layernorm_normalize_kernel(
    x_ptr,
    y_ptr,
    M,
    N,
    eps,
    stride_xm, stride_xn,
    stride_ym, stride_yn,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
):
    pid = tl.program_id(0)
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_N)
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N

        x = tl.load(x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0)
        sum_val += tl.sum(x, axis=1)
        sum_sq += tl.sum(x * x, axis=1)

    mean = sum_val / N
    var = sum_sq / N - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N

        x = tl.load(x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0)

"""
TriMul “outgoing” forward pass – concise Triton‑fused implementation.

Algorithm (high‑level)
----------------------
1️⃣  **Row‑wise layer‑norm without affine**  
    Compute mean/variance for every (B,N,N) row of the 4‑D tensor and
    emit the zero‑mean, unit‑variance tensor `v` in FP16.  The affine
    part (γ,β) is folded into the subsequent linear maps.

2️⃣  **Fold γ,β into the five bias‑free linear projections**  
    For each projection we scale the weight columns by γ and create an
    effective bias `β @ Wᵀ`.  All five projections are concatenated into a
    single weight matrix `W_all` (shape (5·H, D)), so that the huge
    matrix‑multiply `v @ W_allᵀ` reads `v` only once (≈256 MiB for the
    largest test‑case).

3️⃣  **Gating** – sigmoid on the three gate tensors, then element‑wise
    multiplication with the projected tensors.  The optional mask is
    applied *after* gating, which is mathematically equivalent but saves
    one extra broadcast‑multiply.

4️⃣  **Batched contraction**  
    `out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]` is a batched GEMM
    (`left @ rightᵀ`) for each head `h`.  The tensors are permuted to
    (B, H, N, N) and processed by `torch.matmul`, which dispatches to
    cuBLAS Tensor‑Core kernels.

5️⃣  **Layer‑norm on hidden dimension** (γ,β from `to_out_norm`).  
    Implemented with PyTorch’s fused `layer_norm` (half‑precision I/O).

6️⃣  **Output gate & final projection** – element‑wise gate, linear map
    back to the original channel dimension, and cast to `float32`.

The kernel required by the problem statement is the row‑wise layer‑norm
(Triton) that eliminates the expensive extra passes over the huge
input tensor.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: row‑wise normalization (no affine) → FP16 output
# ----------------------------------------------------------------------
@triton.jit
def row_norm_kernel(
    x_ptr,                # *ptr to input  [M, D]   (float32)
    y_ptr,                # *ptr to output [M, D]   (float16)
    M,                    # number of rows
    D,                    # feature dimension
    eps,                  # epsilon for std
    stride_xm, stride_xd,   # strides for input (row, col)
    stride_ym, stride_yd,   # strides for output (row, col)
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)

    # ------------------------------------------------------------------
    # Row indices processed by this program
    # ------------------------------------------------------------------
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # ------------------------------------------------------------------
    # First pass: sum and sum‑squares → mean & rstd
    # ------------------------------------------------------------------
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_d = tl.cdiv(D, BLOCK_D)
    for d in range(num_d):
        offs_d = tl.arange(0, BLOCK_D) + d * BLOCK_D
        mask_d = offs_d < D

        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_d[None, :] * stride_xd,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        )                               # float32
        sum_val += tl.sum(x, axis=1)   # sum over D
        sum_sq  += tl.sum(x * x, axis=1)

    mean = sum_val / D
    var  = sum_sq / D - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # ------------------------------------------------------------------
    # Second pass: write normalized (x-mean)/std  as FP16
    # ------------------------------------------------------------------
    for d in range(num_d):
        offs_d = tl.arange(0, BLOCK_D) + d * BLOCK_D
        mask_d = offs_d < D

        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_d[None, :] * stride_xd,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        )
        norm = (x - mean[:, None]) * rstd[:, None]     # float32
        y = norm.to(tl.float16)                       # cast to fp16

        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_d[None, :] * stride_yd,
            y,
            mask=mask_m[:, None] & mask_d[None, :],
        )


def triton_row_normalize(x_fp32: torch.Tensor, eps: float = 1e-5) -> torch.Tensor:
    """
    Row‑wise LayerNorm without affine transform.
    Input:  FP32 tensor of shape (M, D)
    Output: FP16 tensor of shape (M, D)  (standardised values)
    """
    assert x_fp32.dim() == 2
    M, D = x_fp32.shape
    y_fp16 = torch.empty((M, D), dtype=torch.float16, device=x_fp32.device)

    # ensure contiguous tensors for correct stride handling
    x_fp32 = x_fp32.contiguous()
    stride_xm, stride_xd = x_fp32.stride()
    stride_ym, stride_yd = y_fp16.stride()

    grid = lambda meta: (triton.cdiv(M, meta["BLOCK_M"]),)

    row_norm_kernel[grid](
        x_ptr=x_fp32,
        y_ptr=y_fp16,
        M=M,
        D=D,
        eps=eps,
        stride_xm=stride_xm,
        stride_xd=stride_xd,
        stride_ym=stride_ym,
        stride_yd=stride_yd,
        BLOCK_M=256,   # tuned for large M (≈B·N²)
        BLOCK_D=64,    # fine‑grained D‑tile
    )
    return y_fp16


# ----------------------------------------------------------------------
# Entry point used by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
            input_tensor : torch.Tensor   shape [B, L, L, D] (float32)
            mask         : torch.Tensor   shape [B, L, L] or ``None``
            weights      : dict of torch tensors (model parameters)
            config       : dict with keys ``dim``, ``hidden_dim`` and optional ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, L, L, D] (float32)
    """
    # ------------------------------------------------------------------
    # 0️⃣ unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype  = input_tensor.dtype         # expected float32

    bs, L, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask     = config.get("nomask", False)
    eps        = 1e-5

    # ------------------------------------------------------------------
    # 1️⃣  Row‑wise LayerNorm (no affine) → standardized tensor v (FP16)
    # ------------------------------------------------------------------
    M = bs * L * L                                 # number of rows
    x_flat = input_tensor.reshape(M, dim).contiguous()   # (M, D) float32
    v_fp16 = triton_row_normalize(x_flat, eps=eps)        # (M, D) FP16

    # ------------------------------------------------------------------
    # 2️⃣  Fold γ (norm.weight) / β (norm.bias) into all 5 linear maps
    # ------------------------------------------------------------------
    # γ and β from the initial LayerNorm
    gamma = weights["norm.weight"].to(device, torch.float32).contiguous()   # (D,)
    beta  = weights["norm.bias"].to(device, torch.float32).contiguous()    # (D,)

    # helper to scale a weight matrix and compute its fused bias
    def scale_weight_bias(w_key):
        w = weights[w_key].to(device, torch.float32)                # (H, D)
        w_scaled = (w * gamma[None, :]).to(torch.float16)           # (H, D) FP16
        b = (w * beta[None, :]).sum(dim=1).to(torch.float16)        # (H,)
        return w_scaled, b

    # five projections: left, right, left_gate, right_gate, out_gate
    left_w,    left_b   = scale_weight_bias("left_proj.weight")
    right_w,   right_b  = scale_weight_bias("right_proj.weight")
    left_gate_w, left_gate_b = scale_weight_bias("left_gate.weight")
    right_gate_w, right_gate_b = scale_weight_bias("right_gate.weight")
    out_gate_w, out_gate_b = scale_weight_bias("out_gate.weight")

    # concatenate into a single weight matrix / bias vector
    weight_all = torch.cat([left_w, right_w,
                            left_gate_w, right_gate_w,
                            out_gate_w], dim=0)                 # (5*H, D) FP16
    bias_all   = torch.cat([left_b, right_b,
                            left_gate_b, right_gate_b,
                            out_gate_b], dim=0)                  # (5*H,)   FP16

    # ------------------------------------------------------------------
    # 3️⃣  Single fused linear → obtain all five tensors in one pass
    # ------------------------------------------------------------------
    all_proj = F.linear(v_fp16, weight_all, bias_all)                # (M, 5*H) FP16

    # split back into individual tensors and reshape to (B, L, L, H)
    proj_chunks = torch.split(all_proj, hidden_dim, dim=1)
    left_proj, right_proj, left_gate, right_gate, out_gate = proj_chunks
    left_proj   = left_proj.view(bs, L, L, hidden_dim)
    right_proj  = right_proj.view(bs, L, L, hidden_dim)
    left_gate   = left_gate.view(bs, L, L, hidden_dim)
    right_gate  = right_gate.view(bs, L, L, hidden_dim)
    out_gate    = out_gate.view(bs, L, L, hidden_dim)

    # ------------------------------------------------------------------
    # 4️⃣  Gating (sigmoid) and optional mask
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(left_gate)        # (B, L, L, H) FP16
    right_gate = torch.sigmoid(right_gate)
    out_gate   = torch.sigmoid(out_gate)

    left  = left_proj * left_gate
    right = right_proj * right_gate

    if not nomask and mask is not None:
        # mask broadcasted over the hidden dimension
        mask_h = mask.to(device, torch.float16).unsqueeze(-1)   # (B, L, L, 1)
        left  = left * mask_h
        right = right * mask_h

    # ------------------------------------------------------------------
    # 5️⃣  Batched GEMM: out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    # Rearrange to (B, H, N, N) for cuBLAS batched GEMM
    left_perm  = left.permute(0, 3, 1, 2).contiguous()    # (B, H, N, N) FP16
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    # (B*H, N, N) @ (B*H, N, N)ᵀ   →   (B*H, N, N)
    out_perm = torch.matmul(left_perm, right_perm.transpose(-2, -1))
    # Restore original layout → (B, N, N, H)
    out_h = out_perm.permute(0, 2, 3, 1).contiguous()   # FP16

    # ------------------------------------------------------------------
    # 6️⃣  Output LayerNorm over the hidden dimension (affine)
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device, torch.float16).contiguous()
    to_out_norm_b = weights["to_out_norm.bias"].to(device, torch.float16).contiguous()
    out_norm = F.layer_norm(out_h,
                            normalized_shape=(hidden_dim,),
                            weight=to_out_norm_w,
                            bias=to_out_norm_b,
                            eps=eps)                               # FP16

    # ------------------------------------------------------------------
    # 7️⃣  Apply output gate and final projection back to original dim
    # ------------------------------------------------------------------
    out_norm = out_norm * out_gate                                    # FP16
    to_out_w = weights["to_out.weight"].to(device, torch.float16).contiguous()
    out = F.linear(out_norm, to_out_w)                                 # (B, L, L, D) FP16

    # Cast to float32 as required by the spec
    return out.to(torch.float32)
