"""
Highly‑optimized Triton‑accelerated implementation of the “outgoing” TriMul
operator (AlphaFold‑3).  The algorithm follows the reference PyTorch version
exactly but fuses the three linear projections (left_proj, right_proj,
out_gate) into a single GEMM, removes an extra pass over the input tensor,
and uses custom 2‑pass LayerNorm kernels (FP32 for the first norm,
FP16 for the second).  All heavy work is performed in FP16 while the final
output is returned in FP32.

Key steps
---------
1️⃣ LayerNorm over the model dimension (float‑32, custom Triton kernel).
2️⃣ Cast to FP16.
3️⃣ One fused linear layer:
   * concatenates left_proj, left_gate, right_proj, right_gate,
     out_gate weights → weight matrix (5·H × D).
   * produces a tensor of shape (B·N·N , 5·H) which is sliced into the
     five required components.
4️⃣ Apply optional mask, element‑wise sigmoid gates and combine projections.
5️⃣ Core N³ contraction: for each batch and hidden head a FP16 GEMM
   `out_{b,i,j,h}=∑_k left_{b,i,k,h}·right_{b,j,k,h}` realised as a
   batched `torch.bmm` on tensors of shape (B·H, N, N).
6️⃣ LayerNorm on the hidden dimension (FP16 kernel).
7️⃣ Multiply by the output gate and final linear projection back to `dim`.
8️⃣ Cast result to float‑32 and return.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------
# FP32 2‑pass LayerNorm (input: [M, D] float32) → output: [M, D]
# --------------------------------------------------------------
@triton.jit
def layernorm_fp32_kernel(
    x_ptr, y_ptr, weight_ptr, bias_ptr,
    M, D, eps,
    stride_xm, stride_xn,
    stride_ym, stride_yn,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr
):
    pid = tl.program_id(0)

    # Row range for this program
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # ------------------------------------------------------------------
    # First pass: accumulate sum and sum‑of‑squares (FP32)
    # ------------------------------------------------------------------
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_k = tl.cdiv(D, BLOCK_D)
    for k in range(num_k):
        offs_d = tl.arange(0, BLOCK_D) + k * BLOCK_D
        mask_d = offs_d < D
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_d[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0
        )
        sum_val += tl.sum(x, axis=1)
        sum_sq  += tl.sum(x * x, axis=1)

    mean = sum_val / D
    var  = sum_sq / D - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # ------------------------------------------------------------------
    # Second pass: write normalized + affine transformed values
    # ------------------------------------------------------------------
    for k in range(num_k):
        offs_d = tl.arange(0, BLOCK_D) + k * BLOCK_D
        mask_d = offs_d < D
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_d[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0
        )
        norm = (x - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)

        y = norm * w[None, :] + b[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_d[None, :] * stride_yn,
            y,
            mask=mask_m[:, None] & mask_d[None, :]
        )

def triton_layernorm_fp32(x: torch.Tensor,
                          weight: torch.Tensor,
                          bias: torch.Tensor,
                          eps: float = 1e-5) -> torch.Tensor:
    """LayerNorm on a 2‑D FP32 tensor using the Triton kernel above."""
    assert x.dim() == 2, "Input must be 2‑D [M, D]"
    M, D = x.shape
    y = torch.empty_like(x)

    x = x.contiguous()
    y = y.contiguous()
    weight = weight.contiguous()
    bias = bias.contiguous()

    stride_xm, stride_xn = x.stride()
    stride_ym, stride_yn = y.stride()

    grid = lambda meta: (triton.cdiv(M, meta["BLOCK_M"]),)

    layernorm_fp32_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        weight_ptr=weight,
        bias_ptr=bias,
        M=M,
        D=D,
        eps=eps,
        stride_xm=stride_xm,
        stride_xn=stride_xn,
        stride_ym=stride_ym,
        stride_yn=stride_yn,
        BLOCK_M=128,
        BLOCK_D=64,
    )
    return y

# --------------------------------------------------------------
# FP16 2‑pass LayerNorm (input: [M, H] float16) → output: [M, H]
# --------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr, y_ptr, weight_ptr, bias_ptr,
    M, H, eps,
    stride_xm, stride_xn,
    stride_ym, stride_yn,
    BLOCK_M: tl.constexpr, BLOCK_H: tl.constexpr
):
    pid = tl.program_id(0)

    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_k = tl.cdiv(H, BLOCK_H)
    for k in range(num_k):
        offs_h = tl.arange(0, BLOCK_H) + k * BLOCK_H
        mask_h = offs_h < H
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_h[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_h[None, :],
            other=0.0
        )
        xf = tl.cast(x, tl.float32)
        sum_val += tl.sum(xf, axis=1)
        sum_sq  += tl.sum(xf * xf, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    for k in range(num_k):
        offs_h = tl.arange(0, BLOCK_H) + k * BLOCK_H
        mask_h = offs_h < H
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_h[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_h[None, :],
            other=0.0
        )
        xf = tl.cast(x, tl.float32)
        norm_f32 = (xf - mean[:, None]) * rstd[:, None]
        norm = tl.cast(norm_f32, tl.float16)

        w = tl.load(weight_ptr + offs_h, mask=mask_h, other=0.0)
        b = tl.load(bias_ptr   + offs_h, mask=mask_h, other=0.0)

        y = norm * w[None, :] + b[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_h[None, :] * stride_yn,
            y,
            mask=mask_m[:, None] & mask_h[None, :]
        )

def triton_layernorm_fp16(x: torch.Tensor,
                          weight: torch.Tensor,
                          bias: torch.Tensor,
                          eps: float = 1e-5) -> torch.Tensor:
    """LayerNorm on a 2‑D FP16 tensor using the Triton kernel above."""
    assert x.dtype == torch.float16 and x.dim() == 2
    M, H = x.shape
    y = torch.empty_like(x)

    x = x.contiguous()
    y = y.contiguous()
    weight = weight.contiguous()
    bias = bias.contiguous()

    stride_xm, stride_xn = x.stride()
    stride_ym, stride_yn = y.stride()

    grid = lambda meta: (triton.cdiv(M, meta["BLOCK_M"]),)

    layernorm_fp16_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        weight_ptr=weight,
        bias_ptr=bias,
        M=M,
        H=H,
        eps=eps,
        stride_xm=stride_xm,
        stride_xn=stride_xn,
        stride_ym=stride_ym,
        stride_yn=stride_yn,
        BLOCK_M=128,
        BLOCK_H=64,
    )
    return y

# --------------------------------------------------------------
# Entry point – TriMul “outgoing” forward pass
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass for the TriMul “outgoing” operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor[B, N, N, D]  (float32)
        - mask         : torch.Tensor[B, N, N]    (bool/float) or None
        - weights      : dict of torch.Tensor (model parameters)
        - config       : dict with keys:
              * dim        – model dimension D
              * hidden_dim – hidden dimension H
              * nomask     – bool, optional (skip mask)

    Returns
    -------
    torch.Tensor[B, N, N, D] (float32)
    """
    # Unpack inputs
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype   # expected torch.float32

    bs, N, _, dim = input_tensor.shape
    H = config["hidden_dim"]
    nomask = config.get("nomask", False)
    eps = 1e-5

    # ----------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32) – flatten over (B, N, N)
    # ----------------------------------------------------------
    M = bs * N * N
    x_flat = input_tensor.reshape(M, dim)
    norm_weight = weights["norm.weight"].to(device, dtype)
    norm_bias   = weights["norm.bias"].to(device, dtype)
    x_norm_flat = triton_layernorm_fp32(x_flat, norm_weight, norm_bias, eps=eps)
    x_norm = x_norm_flat.view(bs, N, N, dim)  # (B, N, N, D)  FP32

    # ----------------------------------------------------------
    # 2️⃣ Cast to FP16 for the heavy part
    # ----------------------------------------------------------
    x_f16 = x_norm.to(torch.float16)

    # ----------------------------------------------------------
    # 3️⃣ Fused linear: left_proj, left_gate, right_proj,
    #    right_gate, out_gate → one GEMM
    # ----------------------------------------------------------
    # Gather weight tensors (all are shape [H, D] in the reference)
    w_left_proj = weights["left_proj.weight"]
    w_left_gate = weights["left_gate.weight"]
    w_right_proj = weights["right_proj.weight"]
    w_right_gate = weights["right_gate.weight"]
    w_out_gate = weights["out_gate.weight"]

    # Concatenate → (5*H, D) and cast to FP16
    combined_w = torch.cat(
        [w_left_proj, w_left_gate, w_right_proj, w_right_gate, w_out_gate],
        dim=0
    ).to(device, torch.float16)   # shape (5*H, dim)

    # One fused linear pass
    combined = F.linear(x_f16, combined_w)          # shape (M, 5*H)

    # Slice the fused result (all views, no extra allocation)
    left_proj   = combined[:, 0:H]                 # (M, H)
    left_gate_p = combined[:, H:2*H]
    right_proj  = combined[:, 2*H:3*H]
    right_gate_p= combined[:, 3*H:4*H]
    out_gate_p  = combined[:, 4*H:5*H]

    # ----------------------------------------------------------
    # 4️⃣ Sigmoid gates
    # ----------------------------------------------------------
    left_gate  = torch.sigmoid(left_gate_p)
    right_gate = torch.sigmoid(right_gate_p)
    out_gate   = torch.sigmoid(out_gate_p)        # (M, H)

    # ----------------------------------------------------------
    # 5️⃣ Optional mask (broadcast over hidden dim)
    # ----------------------------------------------------------
    if (not nomask) and mask is not None:
        # mask shape (B, N, N) → (M, 1) in FP16
        mask_f16 = mask.to(device, torch.float16).reshape(M, 1)
        left_proj = left_proj * mask_f16
        right_proj = right_proj * mask_f16

    # ----------------------------------------------------------
    # 6️⃣ Apply gates → gated left/right tensors
    # ----------------------------------------------------------
    left = left_proj * left_gate   # (M, H)
    right = right_proj * right_gate

    # Reshape back to 4‑D tensors for the N³ contraction
    left  = left.view(bs, N, N, H)   # (B, N, N, H)
    right = right.view(bs, N, N, H)

    # ----------------------------------------------------------
    # 7️⃣ Core N³ contraction – batched GEMM over the H heads
    # ----------------------------------------------------------
    # Move H to the batch dimension: (B, H, N, N)
    left_perm  = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    BH = bs * H
    left_mat  = left_perm.view(BH, N, N)          # (B*H, N, N)
    right_mat = right_perm.view(BH, N, N)         # (B*H, N, N)

    # out_mat = left_mat @ right_matᵀ  → (B*H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Restore (B, N, N, H) layout (still FP16)
    out = out_mat.view(bs, H, N, N).permute(0, 2, 3, 1).contiguous()

    # Release temporaries we no longer need to keep memory pressure low
    del left, right, left_perm, right_perm, left_mat, right_mat, out_mat, combined

    # ----------------------------------------------------------
    # 8️⃣ Second LayerNorm over the hidden dimension (FP16)
    # ----------------------------------------------------------
    out_flat = out.reshape(M, H)                     # (M, H)
    to_out_norm_w = weights["to_out_norm.weight"].to(device, torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device, torch.float16)
    out_norm = triton_layernorm_fp16(out_flat, to_out_norm_w, to_out_norm_b, eps=eps)

    # ----------------------------------------------------------
    # 9️⃣ Apply output gate and final linear projection (both FP16)
    # ----------------------------------------------------------
    out_gate_flat = out_gate                     # already (M, H) FP16
    gated = out_norm * out_gate_flat              # (M, H)

    to_out_w = weights["to_out.weight"].to(device, torch.float16)  # (dim, H)
    output_flat = F.linear(gated, to_out_w)       # (M, dim) FP16

    # Reshape to final 4‑D layout and cast back to FP32
    output = output_flat.view(bs, N, N, dim).to(torch.float32)

    return output
