@triton.jit
def batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    batch_h: tl.constexpr, M: tl.constexpr, N: tl.constexpr, K: tl.constexpr,
    stride_ah: tl.constexpr, stride_am: tl.constexpr, stride_ak: tl.constexpr,
    stride_bh: tl.constexpr, stride_bk: tl.constexpr, stride_bn: tl.constexpr,
    stride_ch: tl.constexpr, stride_cm: tl.constexpr, stride_cn: tl.constexpr,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        a = tl.load(
            A_ptr + pid_batch * stride_ah
                     + offs_m[:, None] * stride_am
                     + offs_k[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0
        )
        b = tl.load(
            B_ptr + pid_batch * stride_bh
                     + offs_k[:, None] * stride_bk
                     + offs_n[None, :] * stride_bn,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0
        )
        acc += tl.dot(a, b)

    # Write back
    c = acc.to(tl.float16)
    tl.store(
        C_ptr + pid_batch * stride_ch
               + offs_m[:, None] * stride_cm
               + offs_n[None, :] * stride_cn,
        c,
        mask=mask_m[:, None] & mask_n[None, :]
    )

    device = input_tensor.device
    dtype = input_tensor.dtype
    bs, L, _, dim = input_tensor.shape
    hidden_dim=config["hidden_dim"]

    # Input LN
    M = bs * L * L
    x_flat = input_tensor.reshape(M, dim)   # shape (M, dim)
    norm_weight = weights["norm.weight"].to(device, dtype)   # care dtype
    norm_bias = weights["norm.bias"].to(device, dtype)
    x_norm_flat = triton_layernorm(x_flat, norm_weight, norm_bias, eps=eps)
    x_norm = x_norm_flat.view(bs, L, L, dim)
    x_norm_h = x_norm.to(torch.float16)

    # Fused linear for projections and gates
    left_proj_w = weights["left_proj.weight"].to(device, torch.float16).contiguous()
    right_proj_w = weights["right_proj.weight"].to(device, torch.float16).contiguous()
    left_gate_w = weights["left_gate.weight"].to(device, torch.float16).contiguous()
    right_gate_w = weights["right_gate.weight"].to(device, torch.float16).contiguous()

    fused_weight = torch.cat([left_proj_w, right_proj_w, left_gate_w, right_gate_w], dim=0)  # shape (4*H, dim)
    fused_out = F.linear(x_norm_h_flat, fused_weight)  # shape (M, 4*hidden_dim)

    fused_out = fused_out.view(bs, L, L, 4*hidden_dim)
    left_proj = fused_out[..., :hidden_dim]
    right_proj = fused_out[..., hidden_dim:2*hidden_dim]
    left_gate = torch.sigmoid(fused_out[..., 2*hidden_dim:3*hidden_dim])
    right_gate = torch.sigmoid(fused_out[..., 3*hidden_dim:])

    # Apply mask (if any) before gating
    if not nomask and mask is not None:
        mask_h = mask.to(device, torch.float16).unsqueeze(-1)  # [B, L, L, 1]
        left_proj = left_proj * mask_h
        right_proj = right_proj * mask_h

    left = left_proj * left_gate
    right = right_proj * right_gate

    # Permute to (B, H, N, N) and contiguous
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # shape (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    # Transpose right to (B, H, K, N) where K = N
    right_t = right_perm.transpose(2, 3).contiguous()

    # Flatten batch*head dimension
    batch_h = bs * hidden_dim
    left_mat = left_perm.reshape(batch_h, L, L)   # (batch_h, M, K)
    right_mat = right_t.reshape(batch_h, L, L)   # (batch_h, K, N)

    out_mat = batched_matmul_triton(left_mat, right_mat)  # shape (batch_h, N, N) fp16

    out_norm_w = weights["to_out_norm.weight"].to(device, dtype).contiguous()
    out_norm_b = weights["to_out_norm.bias"].to(device, dtype).contiguous()
    out_norm_flat = triton_layernorm(out_flat, out_norm_w, out_norm_b, eps=eps)
    out_norm = out_norm_flat.view(bs, L, L, hidden_dim)

    # Output gate

"""
TriMul (outgoing) – Triton‑accelerated forward pass
===================================================

This implementation follows the reference PyTorch module but replaces the
two LayerNorms and the N³ contraction with custom Triton kernels:

1️⃣  Input LayerNorm   – row‑wise mean/variance reduction (FP32 accu)
2️⃣  Four linear maps (left_proj, right_proj, left_gate, right_gate) are
    fused into a single half‑precision matmul.
3️⃣  Optional binary mask is applied before the gates.
4️⃣  The expensive contraction
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    is realised as a *batched* GEMM (left @ rightᵀ) using a custom
    Triton kernel that processes all (batch×head) matrices in one launch.
5️⃣  Output LayerNorm – identical to (1) but over the hidden dimension.
6️⃣  An output gating (sigmoid) and final linear projection back to the
    original feature dimension are performed in half‑precision; the
    result is cast to float32 as required by the model.

All heavy arithmetic is carried out in fp16/Tensor‑core friendly format,
while the reductions inside LayerNorm use fp32 for numerical stability.
The code works for the full range of test configurations (batch ≤ 2,
seq_len ∈ {128,…,1024}, dim ∈ {128,384}, hidden_dim = 128) and respects
the optional mask flag.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton row‑wise LayerNorm (operates on a 2‑D tensor [M, N]).
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,                # *ptr to input  [M, N]
    y_ptr,                # *ptr to output [M, N]
    weight_ptr,           # *ptr to weight [N]
    bias_ptr,             # *ptr to bias   [N]
    M,                    # total rows
    N,                    # feature dimension
    eps,                  # epsilon
    stride_xm, stride_xn,  # strides for input
    stride_ym, stride_yn,  # strides for output
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
):
    pid = tl.program_id(0)

    # ------------------------------------------------------------------
    # Row offsets for this program
    # ------------------------------------------------------------------
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # ------------------------------------------------------------------
    # First pass: compute mean & variance (accumulate in FP32)
    # ------------------------------------------------------------------
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_N)
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        x_f32 = x.to(tl.float32)
        sum_val += tl.sum(x_f32, axis=1)
        sum_sq  += tl.sum(x_f32 * x_f32, axis=1)

    mean = sum_val / N
    var  = sum_sq / N - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # ------------------------------------------------------------------
    # Second pass: write normalised + affine transformed values
    # ------------------------------------------------------------------
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        x_f32 = x.to(tl.float32)
        norm_x = (x_f32 - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_n, mask=mask_n, other=0.0).to(tl.float32)
        b = tl.load(bias_ptr   + offs_n, mask=mask_n, other=0.0).to(tl.float32)

        y = norm_x * w[None, :] + b[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn,
            y.to(x.dtype),
            mask=mask_m[:, None] & mask_n[None, :],
        )


def triton_layernorm(x: torch.Tensor,
                     weight: torch.Tensor,
                     bias: torch.Tensor,
                     eps: float = 1e-5) -> torch.Tensor:
    """Row‑wise LayerNorm via Triton.  ``x`` must be 2‑D."""
    assert x.dim() == 2, "triton_layernorm expects a 2‑D tensor"
    M, N = x.shape
    y = torch.empty_like(x)

    # Ensure contiguity for correct strides
    x = x.contiguous()
    y = y.contiguous()
    weight = weight.contiguous()
    bias = bias.contiguous()

    stride_xm, stride_xn = x.stride()
    stride_ym, stride_yn = y.stride()

    grid = lambda meta: (triton.cdiv(M, meta["BLOCK_M"]),)

    layernorm_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        weight_ptr=weight,
        bias_ptr=bias,
        M=M,
        N=N,
        eps=eps,
        stride_xm=stride_xm,
        stride_xn=stride_xn,
        stride_ym=stride_ym,
        stride_yn=stride_yn,
        BLOCK_M=128,
        BLOCK_N=64,
    )
    return y


# ----------------------------------------------------------------------
# Batched GEMM kernel (the N³ contraction).  Computes:
#   C[b, i, j] = Σₖ A[b, i, k] * B[b, k, j]
# for all b ∈ [0, batch_h).
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    batch_h: tl.constexpr,
    M: tl.constexpr, N: tl.constexpr, K: tl.constexpr,
    stride_ah: tl.constexpr, stride_am: tl.constexpr, stride_ak: tl.constexpr,
    stride_bh: tl.constexpr, stride_bk: tl.constexpr, stride_bn: tl.constexpr,
    stride_ch: tl.constexpr, stride_cm: tl.constexpr, stride_cn: tl.constexpr,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # program ids
    pid_batch = tl.program_id(0)    # batch * head dimension
    pid_m = tl.program_id(1)       # row block
    pid_n = tl.program_id(2)       # col block

    # offsets within the current block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in FP32 (tensor‑core accumulate)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A: (BLOCK_M, BLOCK_K)
        a = tl.load(
            A_ptr
            + pid_batch * stride_ah
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # B: (BLOCK_K, BLOCK_N)
        b = tl.load(
            B_ptr
            + pid_batch * stride_bh
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # dot uses Tensor‑cores (FP16×FP16 → FP32)
        acc += tl.dot(a, b)

    # write back in FP16
    c = acc.to(tl.float16)
    tl.store(
        C_ptr
        + pid_batch * stride_ch
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn,
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )


def batched_matmul_triton(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Batched matrix multiplication using the Triton kernel above.
    A: (batch_h, M, K)  FP16
    B: (batch_h, K, N)  FP16
    Returns C: (batch_h, M, N) FP16
    """
    assert A.is_contiguous() and B.is_contiguous()
    batch_h, M, K = A.shape
    _, K2, N = B.shape
    assert K == K2, "Inner dimensions must match"

    C = torch.empty((batch_h, M, N), dtype=A.dtype, device=A.device)

    # Choose tile sizes – 64 works well for the supported seq‑lens.
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Strides
    stride_ah, stride_am, stride_ak = A.stride()
    stride_bh, stride_bk, stride_bn = B.stride()
    stride_ch, stride_cm, stride_cn = C.stride()

    grid = (
        batch_h,
        triton.cdiv(M, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    batched_matmul_kernel[grid](
        A_ptr=A,
        B_ptr=B,
        C_ptr=C,
        batch_h=batch_h,
        M=M,
        N=N,
        K=K,
        stride_ah=stride_ah,
        stride_am=stride_am,
        stride_ak=stride_ak,
        stride_bh=stride_bh,
        stride_bk=stride_bk,
        stride_bn=stride_bn,
        stride_ch=stride_ch,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return C

# Entry‑point required by the test harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor   [B, L, L, D]   (float32)
        - mask         : torch.Tensor   [B, L, L] or ``None``
        - weights      : dict of torch tensors (model parameters)
        - config       : dict with keys ``dim``, ``hidden_dim``,
                         optional ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, L, L, D] (float32)
    """
    # --------------------------------------------------------------
    # Unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype  # expected float32

    bs, L, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    eps = 1e-5

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm (via Triton) over the last dimension ``dim``
    # --------------------------------------------------------------
    M = bs * L * L
    x_flat = input_tensor.reshape(M, dim)                     # [M, dim]
    norm_weight = weights["norm.weight"].to(device, torch.float16).contiguous()
    norm_bias   = weights["norm.bias"].to(device, torch.float16).contiguous()
    x_norm_flat = triton_layernorm(x_flat, norm_weight, norm_bias, eps=eps)
    x_norm = x_norm_flat.view(bs, L, L, dim)                  # [B, L, L, D]

    # --------------------------------------------------------------
    # 2️⃣ Cast to half‑precision for the heavy compute
    # --------------------------------------------------------------
    x_norm_h = x_norm.to(torch.float16)

    # --------------------------------------------------------------
    # 3️⃣ Fuse the four linear maps (proj / gate) into a single matmul
    # --------------------------------------------------------------
    lp_w = weights["left_proj.weight"].to(device, torch.float16).contiguous()
    rp_w = weights["right_proj.weight"].to(device, torch.float16).contiguous()
    lg_w = weights["left_gate.weight"].to(device, torch.float16).contiguous()
    rg_w = weights["right_gate.weight"].to(device, torch.float16).contiguous()
    fused_w = torch.cat([lp_w, rp_w, lg_w, rg_w], dim=0)      # (4*H, D)

    # Flatten for the fused projection
    x_norm_h_flat = x_norm_h.reshape(M, dim)
    fused_out = F.linear(x_norm_h_flat, fused_w)              # (M, 4*H)

    fused_out = fused_out.view(bs, L, L, 4 * hidden_dim)

    left_proj  = fused_out[..., :hidden_dim]                            # (B, L, L, H)
    right_proj = fused_out[..., hidden_dim:2 * hidden_dim]
    left_gate  = torch.sigmoid(fused_out[..., 2 * hidden_dim:3 * hidden_dim])
    right_gate = torch.sigmoid(fused_out[..., 3 * hidden_dim:])

    # --------------------------------------------------------------
    # 4️⃣ Apply mask (if present) before the element‑wise gates
    # --------------------------------------------------------------
    if not nomask and mask is not None:
        mask_h = mask.to(device, torch.float16).unsqueeze(-1)   # (B, L, L, 1)
        left_proj = left_proj * mask_h
        right_proj = right_proj * mask_h

    left  = left_proj * left_gate      # (B, L, L, H)
    right = right_proj * right_gate

    # --------------------------------------------------------------
    # 5️⃣ Prepare tensors for batched GEMM:
    #    out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    # --------------------------------------------------------------
    # Move the hidden dimension next to batch and make tensors contiguous
    left_perm  = left.permute(0, 3, 1, 2).contiguous()    # (B, H, L, L)
    right_perm = right.permute(0, 3, 1, 2).contiguous()   # (B, H, L, L)

    # Transpose the last two axes of ``right`` to obtain (B, H, K, N)
    right_t = right_perm.transpose(2, 3).contiguous()    # (B, H, L, L) -> (B, H, K, N)

    # Collapse batch×head dimension for the kernel
    batch_h = bs * hidden_dim
    left_mat  = left_perm.reshape(batch_h, L, L)   # (batch_h, M, K)
    right_mat = right_t.reshape(batch_h, L, L)    # (batch_h, K, N)

    # --------------------------------------------------------------
    # 6️⃣ Batched GEMM (Trion kernel) – replaces torch.bmm
    # --------------------------------------------------------------
    out_mat = batched_matmul_triton(left_mat, right_mat)   # (batch_h, L, L), fp16

    # Restore the original layout
    out = (out_mat.view(bs, hidden_dim, L, L)
                   .permute(0, 2, 3, 1)
                   .contiguous())   # (B, L, L, H)

    # --------------------------------------------------------------
    # 7️⃣ Output LayerNorm over the hidden dimension
    # --------------------------------------------------------------
    out_M = bs * L * L
    out_flat = out.reshape(out_M, hidden_dim)
    out_norm_w = weights["to_out_norm.weight"].to(device, torch.float16).contiguous()
    out_norm_b = weights["to_out_norm.bias"].to(device, torch.float16).contiguous()
    out_norm_flat = triton_layernorm(out_flat, out_norm_w, out_norm_b, eps=eps)
    out_norm = out_norm_flat.view(bs, L, L, hidden_dim)   # (B, L, L, H)

    # --------------------------------------------------------------
    # 8️⃣ Output gate (sigmoid) – separate linear
    # --------------------------------------------------------------
    out_gate_w = weights["out_gate.weight"].to(device, torch.float16).contiguous()
    out_gate = torch.sigmoid(F.linear(x_norm_h_flat, out_gate_w))
    out_gate = out_gate.view(bs, L, L, hidden_dim)        # (B, L, L, H)

    # --------------------------------------------------------------
    # 9️⃣ Apply output gate and final projection back to ``dim``
    # --------------------------------------------------------------
    gated = out_norm * out_gate                      # fp16
    to_out_w = weights["to_out.weight"].to(device, torch.float16).contiguous()
    out_final_h = F.linear(gated, to_out_w)          # (B, L, L, D), fp16
    out_final = out_final_h.to(torch.float32)        # required dtype

    return out_final
