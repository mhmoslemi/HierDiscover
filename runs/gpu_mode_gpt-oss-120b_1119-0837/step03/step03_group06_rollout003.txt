assistantfinal**Solution Overview**

This implementation follows the “outgoing” version of the TriMul module described in the
AlphaFold‑3 paper.

1. **Layer‑Norm & Linear projections** – performed with `torch.nn.functional`
   (float32 → float16 conversion for the rest of the pipeline).

2. **Gating & optional masking** – element‑wise `sigmoid` and a broadcasted mask
   are applied in FP16.

3. **Core cubic multiplication** – the expensive
   ```
   out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   ```
   is expressed as a *batched GEMM*:
   ```
   out_c[M,N] = left_c[M,K] @ right_c[K,N]
   ```
   where the batch dimension `c = B·H` packs the original batch and hidden
   dimension.  The GEMM is executed by a custom Triton kernel
   `tri_mul_kernel`.  The kernel works on FP16 inputs, accumulates in FP32,
   and stores the result back as FP16.

4. **Post‑processing** – a second `LayerNorm`, an output gate, and the final
   linear projection back to the original channel dimension are performed in
   FP16.  The result is cast to `float32` before returning.

The Triton kernel is tuned for the target shapes:

* Block‑size `BLOCK_M = BLOCK_N = min(N, 128)` (one or a few blocks per
  sequence dimension, reducing launch overhead for small `N`).
* `BLOCK_K = 32` for `N ≤ 512`, otherwise `64`.
* `num_warps = 8` for larger sequences, `4` otherwise.
* `num_stages = 3` provides pipelined loads.

All tensors are kept on the CUDA device; no gradients are computed.

```python
"""
TriMul (outgoing) – fast half‑precision implementation.

The heavy “triangle” multiplication
    out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
is expressed as a batched GEMM per (batch,hidden) slice.
We pack the batch and hidden dimensions into a single batch dimension C = B*H
and launch a Triton kernel that computes
    out_c[M,N] = left_c[M,K] @ right_c[K,N]   (M = N = seq_len, K = seq_len).
All tensors (except the first LayerNorm) are kept in fp16, with
fp32 accumulation inside the kernel and the result stored as fp16.
The surrounding linear projections, gates, and the two LayerNorms are
performed with torch.nn.functional using half‑precision weights.
The final output is cast back to float32 for downstream consumption.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    out_ptr,                     # fp16*
    a_ptr,                       # fp16*  (C, M, K) – left tensor
    b_ptr,                       # fp16*  (C, K, N) – right tensor (logically transposed)
    M, N, K,                     # matrix dimensions (all = seq_len)
    stride_out_m, stride_out_n, stride_out_c,   # output strides
    stride_a_m, stride_a_k, stride_a_c,        # left strides
    stride_b_k, stride_b_n, stride_b_c,        # right strides (after logical transpose)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM for the outgoing TriMul.

    Computes for each packed batch channel `c`:
        out_c[M,N] = Σₖ left_c[M,K] * right_c[K,N]
    Inputs are fp16, accumulation is fp32, output stored as fp16.
    """
    pid_m = tl.program_id(0)   # block row index (i)
    pid_n = tl.program_id(1)   # block col index (j)
    pid_c = tl.program_id(2)   # packed (batch * hidden) channel

    # Tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator (fp32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over reduction dimension K
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Pointers to tiles of A and B
        a_ptrs = a_ptr + pid_c * stride_a_c \
                 + offs_m[:, None] * stride_a_m \
                 + offs_k[None, :] * stride_a_k

        b_ptrs = b_ptr + pid_c * stride_b_c \
                 + offs_k[:, None] * stride_b_k \
                 + offs_n[None, :] * stride_b_n

        # Load tiles (masked)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)                     # fp16
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)                     # fp16

        # GEMM accumulation (fp16 * fp16 → fp32)
        acc += tl.dot(a, b)

    # Write out tile (convert to fp16 on store)
    out_ptrs = out_ptr + pid_c * stride_out_c \
               + offs_m[:, None] * stride_out_m \
               + offs_n[None, :] * stride_out_n

    tl.store(out_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module (AlphaFold‑3 style).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor[B, N, N, C] (float32)
        - mask         : torch.Tensor[B, N, N] (bool/float) or None
        - weights      : dict of model parameters
        - config       : dict with keys
            * "dim"         – input channel dimension C
            * "hidden_dim" – hidden dimension H
            * "nomask"      – bool, True → ignore mask

    Returns
    -------
    torch.Tensor[B, N, N, C] (float32)
    """
    # -----------------------------------------------------------------
    # unpack
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    device = input_tensor.device

    # -----------------------------------------------------------------
    # 1) Input LayerNorm (fp32 for numeric stability)
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )                           # [B, N, N, C]  (fp32)
    x = x.to(torch.float16)    # cast to fp16 for the rest

    # -----------------------------------------------------------------
    # 2) Linear projections (no bias) → hidden_dim  (fp16)
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left = F.linear(x, left_proj_w)       # [B, N, N, H] fp16
    right = F.linear(x, right_proj_w)    # [B, N, N, H] fp16

    # -----------------------------------------------------------------
    # 3) Gates (sigmoid) – fp16
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))   # [B,N,N,H] fp16

    # -----------------------------------------------------------------
    # 4) Apply gates (and optional mask)
    left = left * left_gate
    right = right * right_gate

    if not nomask and mask is not None:
        # mask: [B,N,N] -> broadcast over hidden dim
        mask_h = mask.to(torch.float16).unsqueeze(-1)   # [B,N,N,1]
        left = left * mask_h
        right = right * mask_h

    # -----------------------------------------------------------------
    # 5) Prepare data for the Triton kernel
    #    Pack batch and hidden dimensions: (B, H, N, N) -> (C_total, N, N)
    B, N, _, _ = left.shape
    left = left.permute(0, 3, 1, 2).contiguous()     # [B, H, N, N]
    right = right.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]

    C_total = B * hidden_dim                         # = batch * hidden
    left_flat = left.reshape(C_total, N, N)          # (C, M, K) – K = N
    right_flat = right.reshape(C_total, N, N)        # (C, K, N) – will be read transposed

    # Output buffer (fp16)
    out_flat = torch.empty_like(left_flat, dtype=torch.float16, device=device)

    # -----------------------------------------------------------------
    # 6) Triton batched GEMM launch – tile sizes tuned per sequence length
    if N <= 128:
        BLOCK_M = BLOCK_N = N               # single block per dimension
        BLOCK_K = 32
    elif N <= 256:
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 32
    elif N <= 512:
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 32
    else:  # N up to 1024
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 64

    # Strides (in elements, not bytes)
    stride_out_c = out_flat.stride(0)
    stride_out_m = out_flat.stride(1)
    stride_out_n = out_flat.stride(2)

    stride_a_c = left_flat.stride(0)
    stride_a_m = left_flat.stride(1)
    stride_a_k = left_flat.stride(2)

    stride_b_c = right_flat.stride(0)
    stride_b_k = right_flat.stride(2)   # after logical transpose
    stride_b_n = right_flat.stride(1)   # after logical transpose

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # pid_m
        (N + BLOCK_N - 1) // BLOCK_N,   # pid_n
        C_total,                        # pid_c
    )

    # Choose number of warps (more warps for larger N)
    num_warps = 8 if N >= 512 else 4

    tri_mul_kernel[grid](
        out_flat,
        left_flat,
        right_flat,
        N, N, N,
        stride_out_m, stride_out_n, stride_out_c,
        stride_a_m, stride_a_k, stride_a_c,
        stride_b_k, stride_b_n, stride_b_c,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=num_warps,
        num_stages=3,                 # pipelined prefetch
    )

    # -----------------------------------------------------------------
    # 7) Reshape back to (B, N, N, H) in fp16
    out = out_flat.reshape(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,H]

    # Release large temporaries early
    del left, right, left_flat, right_flat, out_flat

    # -----------------------------------------------------------------
    # 8) Post‑processing: second LayerNorm → out‑gate → final linear
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)

    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    out = out * out_gate                 # apply output gate (already fp16)

    to_out_w = weights["to_out.weight"].to(torch.float16)
    out = F.linear(out, to_out_w)        # (B,N,N,dim)  fp16

    # -----------------------------------------------------------------
    # 9) Cast back to float32 for the expected return type
    return out.to(torch.float32)
```
