def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    …
    device = input_tensor.device
    dim = config["dim"]
    hidden = config["hidden_dim"]
    B,N,_,dim = input_tensor.shape

    # Layernorm on dim
    eps = 1e-5
    x = F.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=eps)
    x_h = x.half()

    # Combine linear weights
    left_proj_w = weights["left_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_proj_w = weights["right_proj.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]
    combined_w = torch.cat([left_proj_w, left_gate_w, right_proj_w, right_gate_w, out_gate_w], dim=0).half()

    # Linear with combined weights
    combined = F.linear(x_h, combined_w)  # shape [B,N,N,5*H] fp16

    # split
    slices = torch.split(combined, hidden, dim=-1)
    left_proj_raw, left_gate_raw, right_proj_raw, right_gate_raw, out_gate_raw = slices

    # Prepare mask (if none, allocate ones)
    if mask_tensor is None or mask_tensor.numel() == 0:
        mask_h = torch.ones((B, N, N), dtype=torch.float16, device=device)
    else:
        mask_h = mask_tensor.to(dtype=torch.float16, device=device)

@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr, left_gate_ptr, right_ptr, right_gate_ptr,
    mask_ptr,
    total_elements: tl.int64,
    hidden_dim: tl.int64,
    BLOCK_SIZE: tl.constexpr,
):

```
@triton.jit
def _mask_gate_fuse_combined_kernel(
    combined_ptr,   # FP16 pointer to combined tensor [B,N,N,5*H]
    mask_ptr,       # FP16 pointer [B,N,N] broadcasted
    total_elements: tl.int64,  # B*N*N*H
    hidden_dim: tl.int64,
    region_stride: tl.int64,   # 5*hidden
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elements

    # compute position indices
    d = offs % hidden_dim
    pos = offs // hidden_dim

    # mask value
    mask_val = tl.load(mask_ptr + pos, mask=mask, other=tl.float16(1.0))

    # compute base offset for this position in combined
    base = pos * region_stride

    # offsets for each region
    left_off = base + d
    left_gate_off = base + d + hidden_dim
    right_off = base + d + 2 * hidden_dim
    right_gate_off = base + d + 3 * hidden_dim

    # load
    left = tl.load(combined_ptr + left_off, mask=mask, other=tl.float16(0.0))
    left_gate = tl.load(combined_ptr + left_gate_off, mask=mask, other=tl.float16(0.0))
    right = tl.load(combined_ptr + right_off, mask=mask, other=tl.float16(0.0))
    right_gate = tl.load(combined_ptr + right_gate_off, mask=mask, other=tl.float16(0.0))

    # sigmoid
    left_gate_s = 1.0 / (1.0 + tl.exp(-left_gate))
    right_gate_s = 1.0 / (1.0 + tl.exp(-right_gate))

    # apply
    left_new = left * left_gate_s * mask_val
    right_new = right * right_gate_s * mask_val

    # store
    tl.store(combined_ptr + left_off, left_new, mask=mask)
    tl.store(combined_ptr + right_off, right_new, mask=mask)
```

"""
TriMul (outgoing) – fused Triton + cuBLAS implementation.

Algorithm
---------
1️⃣  Layer‑norm over the feature dimension (FP32 for stability).
2️⃣  Cast to FP16 and perform **one** large linear projection that merges the
    five original weight matrices (left_proj, left_gate, right_proj,
    right_gate, out_gate) into a single GEMM:  [B,N,N,D] × [5·H,D]ᵀ → [B,N,N,5·H].

3️⃣  A tiny Triton kernel fuses the element‑wise mask (or a tensor of ones
    when no mask is supplied), the sigmoid of the two gate tensors and the
    multiplication onto the projected left/right tensors, all in‑place.
    After the kernel we have:
        left  = left_proj * sigmoid(left_gate) * mask
        right = right_proj * sigmoid(right_gate) * mask
    The out‑gate tensor is left untouched (will be used later).

4️⃣  Heavy O(N³) contraction is a batched GEMM:
        out[i,j,d] = Σ_k left[i,k,d] * right[j,k,d]
    performed with `torch.bmm` on the reshaped tensors
    (batch = B·H, matrices of size N×N).

5️⃣  Hidden‑dim layer‑norm, out‑gate sigmoid, element‑wise gating and the final
    linear projection back to the original dimension.

All heavy matrix‑multiplications stay in cuBLAS (tensor‑cores on H100),
while the only custom Triton kernel eliminates two memory‑bound passes
(mask + gate) and works for both masked and un‑masked inputs.

The function returns a float‑32 tensor of shape [B, N, N, D].
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _mask_gate_fuse_combined_kernel(
    combined_ptr,               # FP16 * combined tensor [B,N,N,5*H]
    mask_ptr,                   # FP16 * mask flattened [B*N*N]
    total_elements: tl.int64,  # B * N * N * H
    hidden_dim: tl.int64,       # H
    stride: tl.int64,           # 5 * H  (distance between successive (i,k) blocks)
    BLOCK_SIZE: tl.constexpr,
):
    """Fuse mask, sigmoid‑gates and element‑wise multiplication in‑place.

    The kernel updates the left‑proj (region 0) and right‑proj (region 2)
    slices of the combined tensor.  Gate tensors (regions 1 & 3) stay
    unchanged for later use.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    active = offs < total_elements

    # hidden index inside the region and the (b,i,k) position
    d   = offs % hidden_dim          # 0 .. H-1
    pos = offs // hidden_dim         # linear index for (b,i,k)

    # mask broadcast over hidden dimension (value = 1.0 for nomask)
    mask_val = tl.load(mask_ptr + pos, mask=active, other=tl.float16(1.0))

    base = pos * stride               # start of the 5*H block for this (b,i,k)

    # offsets of the four tensors we need inside the combined buffer
    left_off       = base + d                     # region 0
    left_gate_off  = base + d + hidden_dim         # region 1
    right_off      = base + d + 2 * hidden_dim     # region 2
    right_gate_off = base + d + 3 * hidden_dim     # region 3

    # load raw values
    left       = tl.load(combined_ptr + left_off,      mask=active, other=tl.float16(0.0))
    left_gate  = tl.load(combined_ptr + left_gate_off, mask=active, other=tl.float16(0.0))
    right      = tl.load(combined_ptr + right_off,     mask=active, other=tl.float16(0.0))
    right_gate = tl.load(combined_ptr + right_gate_off,mask=active, other=tl.float16(0.0))

    # sigmoid(x) = 1 / (1 + exp(-x))  (FP16 version)
    left_gate_sig  = 1.0 / (1.0 + tl.exp(-left_gate))
    right_gate_sig = 1.0 / (1.0 + tl.exp(-right_gate))

    # apply gates and mask
    left_out  = left  * left_gate_sig  * mask_val
    right_out = right * right_gate_sig * mask_val

    # write the gated tensors back (in‑place)
    tl.store(combined_ptr + left_off,  left_out,  mask=active)
    tl.store(combined_ptr + right_off, right_out, mask=active)


def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul block (no gradients).

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, D] (float32)
            - mask_tensor : torch.Tensor [B, N, N] (bool/float) – may be None
            - weights_dict : mapping of all module parameters (float32)
            - config_dict : {"dim": D, "hidden_dim": H, "nomask": bool, ...}

    Returns:
        torch.Tensor [B, N, N, D] (float32)
    """
    # --------------------------------------------------------------------- #
    # unpack arguments
    # --------------------------------------------------------------------- #
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    H = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5

    # --------------------------------------------------------------------- #
    # 1) Input layer‑norm (FP32 for numerical stability)
    # --------------------------------------------------------------------- #
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )   # [B, N, N, D]  (float32)

    # --------------------------------------------------------------------- #
    # 2) Cast to FP16 for the heavy arithmetic
    # --------------------------------------------------------------------- #
    x_h = x.half()                                     # [B,N,N,D] (FP16)

    # --------------------------------------------------------------------- #
    # 3) Merge the five linear weight matrices into a single GEMM
    # --------------------------------------------------------------------- #
    combined_w = torch.cat(
        [
            weights["left_proj.weight"],   # (H, D)
            weights["left_gate.weight"],   # (H, D)
            weights["right_proj.weight"],  # (H, D)
            weights["right_gate.weight"],  # (H, D)
            weights["out_gate.weight"],    # (H, D)
        ],
        dim=0,
    ).half()                                            # (5*H, D)

    # shape → [B, N, N, 5*H]
    combined = F.linear(x_h, combined_w)

    # --------------------------------------------------------------------- #
    # 4) Prepare mask (flattened) – for nomask we use a tensor of ones
    # --------------------------------------------------------------------- #
    if nomask or mask_tensor is None:
        mask_h = torch.ones((B, N, N), dtype=torch.float16, device=device)
    else:
        mask_h = mask_tensor.to(dtype=torch.float16, device=device)
    mask_flat = mask_h.view(-1).contiguous()           # [B*N*N]

    # --------------------------------------------------------------------- #
    # 5) Fuse mask, sigmoid(gates) and element‑wise multiplication (Triton)
    # --------------------------------------------------------------------- #
    total_elements = B * N * N * H                     # number of hidden elements per region
    BLOCK_SIZE = 1024                                   # ≤ 1024 threads per CTA
    grid = (total_elements + BLOCK_SIZE - 1) // BLOCK_SIZE
    stride = H * 5                                      # distance between successive (b,i,k) blocks

    _mask_gate_fuse_combined_kernel[grid](
        combined,               # FP16 ptr
        mask_flat,              # FP16 mask ptr
        total_elements,
        H,
        stride,
        BLOCK_SIZE=BLOCK_SIZE,
        num_warps=8,            # 8 warps = 256 threads per CTA, good occupancy on H100
    )

    # --------------------------------------------------------------------- #
    # 6) Slice the gated tensors out of the combined buffer
    # --------------------------------------------------------------------- #
    left = combined[..., :H]                 # [B,N,N,H] – already gated
    right = combined[..., 2 * H : 3 * H]     # [B,N,N,H] – already gated
    out_gate_raw = combined[..., 4 * H : 5 * H]   # [B,N,N,H] – still raw
    out_gate = torch.sigmoid(out_gate_raw).half()

    # --------------------------------------------------------------------- #
    # 7) Heavy O(N³) contraction via batched GEMM (cuBLAS)
    # --------------------------------------------------------------------- #
    # Rearrange to (B, H, N, N) → (B*H, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    batch = B * H
    left_flat = left_perm.view(batch, N, N)                # (B*H, N, N)
    right_flat = right_perm.view(batch, N, N).transpose(1, 2)  # (B*H, N, N) -> (B*H, N, N)ᵀ

    # Batched matrix multiply:   (B*H, N, N) @ (B*H, N, N)ᵀ → (B*H, N, N)
    out_flat = torch.bmm(left_flat, right_flat)

    # Reshape back to [B, N, N, H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------------- #
    # 8) Hidden‑dim layer‑norm, out‑gate and final projection
    # --------------------------------------------------------------------- #
    out = F.layer_norm(
        out,
        normalized_shape=(H,),
        weight=weights["to_out_norm.weight"].half(),
        bias=weights["to_out_norm.bias"].half(),
        eps=eps,
    )
    out = out * out_gate                     # element‑wise gating (shape [B,N,N,H])
    out = F.linear(out, weights["to_out.weight"].half())   # back to original dim D

    # --------------------------------------------------------------------- #
    # 9) Cast result back to float32 for downstream consumption
    # --------------------------------------------------------------------- #
    return out.float()
