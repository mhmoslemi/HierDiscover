"""
TriMul “outgoing” – Triton‑accelerated forward pass (no gradients).

Algorithm
---------
1️⃣  Layer‑Norm on the model dimension (float16 I/O, FP32 statistics) using a
    custom Triton kernel – avoids the costly FP32 → FP16 → FP32 round‑trip.

2️⃣  Cast to FP16 for all heavy work.

3️⃣  Fuse each projection + its gate into a single matmul (2·H × D weight).
    The projection and the gate are split, the gate is passed through a sigmoid.

4️⃣  Apply the optional mask (broadcast over H).

5️⃣  Element‑wise gating of the projected tensors.

6️⃣  Compute the core N³ contraction:
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    – realised as a batched GEMM on (B·H) matrices of shape (N,N).
    This uses torch.bmm which internally maps to cuBLAS Tensor‑Core kernels.

7️⃣  Layer‑Norm on the hidden dimension (FP16) with a second Triton kernel.

8️⃣  Apply the output gate (sigmoid of a linear on the normalized input).

9️⃣  Final linear projection back to the model dimension (FP16) and cast to FP32.

The heavy linear layers (projections, out‑gate, final projection) and the two
LayerNorms stay in half‑precision, while FP32 accumulators guarantee numeric
stability.  The implementation works for all test configurations and respects
the optional mask flag.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 2‑pass LayerNorm (FP16 I/O, FP32 statistics)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,      # input  [M, N]  (float16)
    y_ptr,      # output [M, N]  (float16)
    weight_ptr, # weight [N]     (float16)
    bias_ptr,   # bias   [N]     (float16)
    M, N, eps,
    stride_xm, stride_xn,
    stride_ym, stride_yn,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr
):
    pid = tl.program_id(0)

    # ----- row range ---------------------------------------------------
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # ----- first pass – mean / variance (FP32) -------------------------
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_N)
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0
        )
        xf = tl.cast(x, tl.float32)
        sum_val += tl.sum(xf, axis=1)
        sum_sq  += tl.sum(xf * xf, axis=1)

    mean = sum_val / N
    var  = sum_sq / N - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # ----- second pass – write normalized + affine --------------------
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0
        )
        xf = tl.cast(x, tl.float32)
        norm = (xf - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_n, mask=mask_n, other=0.0)
        b = tl.load(bias_ptr   + offs_n, mask=mask_n, other=0.0)

        y = tl.cast(norm, tl.float16) * w[None, :] + b[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn,
            y,
            mask=mask_m[:, None] & mask_n[None, :]
        )


def triton_layernorm_fp16(x: torch.Tensor,
                          weight: torch.Tensor,
                          bias: torch.Tensor,
                          eps: float = 1e-5) -> torch.Tensor:
    """
    LayerNorm over the last dimension of a 2‑D tensor (M×N) in FP16.
    Uses a 2‑pass algorithm with FP32 accumulation.
    """
    assert x.dim() == 2, "LayerNorm expects a 2‑D tensor"
    M, N = x.shape
    y = torch.empty_like(x)

    x = x.contiguous()
    y = y.contiguous()
    weight = weight.contiguous()
    bias = bias.contiguous()

    stride_xm, stride_xn = x.stride()
    stride_ym, stride_yn = y.stride()

    grid = lambda meta: (triton.cdiv(M, meta["BLOCK_M"]),)

    layernorm_fp16_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        weight_ptr=weight,
        bias_ptr=bias,
        M=M,
        N=N,
        eps=eps,
        stride_xm=stride_xm,
        stride_xn=stride_xn,
        stride_ym=stride_ym,
        stride_yn=stride_yn,
        BLOCK_M=128,   # tuned for H100 – works well up to M≈1M
        BLOCK_N=64,
    )
    return y


# ----------------------------------------------------------------------
# Entry point – forward only
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Args
    ----
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor[B, N, N, D] (float32)
        - mask         : torch.Tensor[B, N, N]   (float32 or None)
        - weights      : dict of torch.Tensor parameters
        - config       : dict containing at least ``dim`` and ``hidden_dim``.
    Returns
    -------
    torch.Tensor[B, N, N, D] (float32)
    """
    # --------------------------------------------------------------
    # unpack
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype  = torch.float16      # we keep everything in fp16 after the first LN

    bs, N, _, dim = input_tensor.shape
    H = config["hidden_dim"]
    nomask = config.get("nomask", False)
    eps = 1e-5

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP16 I/O, FP32 stats)
    # --------------------------------------------------------------
    # Cast to FP16 first – saves a full‑precision copy later
    x_fp16 = input_tensor.to(torch.float16)

    # LayerNorm parameters
    norm_w = weights["norm.weight"].to(device, dtype)
    norm_b = weights["norm.bias"].to(device, dtype)

    # x_norm : [B, N, N, D]   fp16
    x_norm_flat = triton_layernorm_fp16(
        x_fp16.reshape(bs * N * N, dim), norm_w, norm_b, eps=eps
    )
    x_norm = x_norm_flat.view(bs, N, N, dim)

    # --------------------------------------------------------------
    # 2️⃣ Fuse projection + gate for left & right (single matmul each)
    # --------------------------------------------------------------
    # combine the linear weight for proj & gate (2·H × D)
    left_comb_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(device, dtype)                     # (2·H, D)

    right_comb_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(device, dtype)                     # (2·H, D)

    out_gate_w = weights["out_gate.weight"].to(device, dtype)  # (H, D)

    # Linear + split – all still fp16
    left_comb  = F.linear(x_norm, left_comb_w)   # (B,N,N,2H)
    right_comb = F.linear(x_norm, right_comb_w)  # (B,N,N,2H)

    left_proj      = left_comb[..., :H]           # (B,N,N,H)
    left_gate_pre  = left_comb[..., H:]           # (B,N,N,H)
    right_proj     = right_comb[..., :H]
    right_gate_pre = right_comb[..., H:]

    left_gate  = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)

    # --------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast over H)
    # --------------------------------------------------------------
    if not nomask and mask is not None:
        mask_f16 = mask.to(device, dtype).unsqueeze(-1)   # (B,N,N,1)
        left_proj  = left_proj * mask_f16
        right_proj = right_proj * mask_f16

    # --------------------------------------------------------------
    # 4️⃣ Apply element‑wise gates
    # --------------------------------------------------------------
    left  = left_proj * left_gate   # (B,N,N,H)
    right = right_proj * right_gate

    # --------------------------------------------------------------
    # 5️⃣ Output gate (independent of mask)
    # --------------------------------------------------------------
    out_gate = torch.sigmoid(F.linear(x_norm, out_gate_w))   # (B,N,N,H)

    # --------------------------------------------------------------
    # 6️⃣ Core N³ contraction via a batched half‑precision GEMM
    #    (B·H) independent matmuls of size N×N)
    # --------------------------------------------------------------
    # reshape to (B·H, N, N) – requires a contiguous copy (the dominant
    # memory movement for this step).  The copy is fast in FP16.
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B,H,N,N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    BH = bs * H
    left_mat  = left_perm.view(BH, N, N)                # (B·H, N, N)
    right_mat = right_perm.view(BH, N, N)

    # (BH, N, N) = (BH, N, N) @ (BH, N, N)^T   – Tensor‑Core fp16 GEMM
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # back to (B, N, N, H)
    out = out_mat.view(bs, H, N, N).permute(0, 2, 3, 1).contiguous()  # fp16

    # --------------------------------------------------------------
    # 7️⃣ Second LayerNorm on hidden dimension (FP16)
    # --------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device, dtype)
    to_out_norm_b = weights["to_out_norm.bias"].to(device, dtype)

    out_flat = out.reshape(bs * N * N, H)               # (M, H)
    out_norm = triton_layernorm_fp16(out_flat, to_out_norm_w, to_out_norm_b, eps=eps)  # fp16

    # --------------------------------------------------------------
    # 8️⃣ Apply the output gate and final projection (both FP16)
    # --------------------------------------------------------------
    out_gate_flat = out_gate.reshape(bs * N * N, H)    # (M, H)
    gated = out_norm * out_gate_flat                    # (M, H) fp16

    to_out_w = weights["to_out.weight"].to(device, dtype)     # (D, H)

    output_flat = F.linear(gated, to_out_w)            # (M, D) fp16
    output = output_flat.view(bs, N, N, dim).to(torch.float32)

    return output
