@triton.jit
def layernorm_gate_fuse_kernel(
    in_ptr, gate_ptr, norm_weight_ptr, norm_bias_ptr,
    out_ptr,
    total_positions, hidden_dim, BLOCK_POS: tl.constexpr, BLOCK_H: tl.constexpr, eps: tl.float32):
    pid = tl.program_id(0)
    pos = pid * BLOCK_POS + tl.arange(0, BLOCK_POS)
    mask = pos < total_positions

    # Compute sum and sum of squares across hidden dim
    sum_val = tl.zeros((BLOCK_POS,), dtype=tl.float32)
    sum_sq_val = tl.zeros((BLOCK_POS,), dtype=tl.float32)

    # Loop over hidden dim in chunks
    for offset in range(0, hidden_dim, BLOCK_H):
        cur_h = tl.minimum(hidden_dim - offset, BLOCK_H)
        offs_h = offset + tl.arange(0, BLOCK_H)

        # mask for the offset within hidden dim
        h_mask = offs_h < hidden_dim

        # compute addresses
        in_off = in_ptr + (pos * hidden_dim) + offs_h[None, :]
        # load values
        x = tl.load(in_off, mask=mask[:, None] & h_mask[None, :], other=tl.float16(0.0)).to(tl.float32)
        sum_val += tl.sum(x, axis=1)
        sum_sq_val += tl.sum(x * x, axis=1)

    # Compute mean and inv_std
    mean = sum_val / hidden_dim
    var = sum_sq_val / hidden_dim - mean * mean
    inv_std = 1.0 / tl.sqrt(var + eps)

    # Second pass: compute normalized, apply to_out_norm weight + bias, multiply gate, store
    # allocate accumulator for output? We'll write directly to out_ptr_out

    for offset in range(0, hidden_dim, BLOCK_H):
        cur_h = tl.minimum(hidden_dim - offset, BLOCK_H)
        offs_h = offset + tl.arange(0, BLOCK_H)
        h_mask = offs_h < hidden_dim

        # Pointers
        in_off = in_ptr + (pos * hidden_dim) + offs_h[None, :]
        gate_off = gate_ptr + (pos * hidden_dim) + offs_h[None, :]
        norm_w_off = norm_weight_ptr + offs_h[None, :]
        norm_b_off = norm_bias_ptr + offs_h[None, :]

        x = tl.load(in_off, mask=mask[:, None] & h_mask[None, :], other=tl.float16(0.0)).to(tl.float32)
        gate = tl.load(gate_off, mask=mask[:, None] & h_mask[None, :], other=tl.float16(0.0)).to(tl.float32)
        w = tl.load(norm_w_off, mask=h_mask, other=tl.float16(0.0)).to(tl.float32)
        b = tl.load(norm_b_off, mask=h_mask, other=tl.float16(0.0)).to(tl.float32)

        # normalized
        y = (x - mean[:, None]) * inv_std[:, None] * w + b
        y = y * gate

        # store to output
        out_off = out_ptr + (pos * hidden_dim) + offs_h[None, :]

"""
TriMul (outgoing) – fused Triton kernel for the final hidden‑dim LayerNorm + gating.
The implementation follows the reference PyTorch module but removes the costly
permute‑and‑contiguous copies needed for the N³ contraction.  The heavy contraction
is performed with a single ``torch.einsum`` which PyTorch back‑ends as a batched
matrix‑multiply without reshuffling memory.  A small Triton kernel fuses the
post‑contraction LayerNorm (over the hidden dimension), the learned scale/bias
(to_out_norm) and the out‑gate, reducing three separate memory passes to one.
The final projection back to the original dimension is a regular half‑precision
linear layer (cuBLAS).  All other lightweight ops (layer‑norm on the input,
linear projections, sigmoids) stay in PyTorch to leverage its highly‑tuned kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# -----------------------------------------------------------------------------#
# Triton kernel: fuse LayerNorm (over hidden_dim), the learned scale/bias
# and the out‑gate.  Operates on the flattened view of a tensor of shape
# [B, N, N, H] -> total_positions = B * N * N, hidden_dim = H.
# -----------------------------------------------------------------------------#
@triton.jit
def _layernorm_gate_fuse_kernel(
    in_ptr,          # FP16 *   input tensor from the N³ contraction [B,N,N,H]
    gate_ptr,        # FP16 *   out‑gate tensor                              [B,N,N,H]
    norm_w_ptr,      # FP16 *   to_out_norm.weight (H)
    norm_b_ptr,      # FP16 *   to_out_norm.bias   (H)
    out_ptr,         # FP16 *   output tensor (same shape as input)        [B,N,N,H]
    total_pos,       # i64      total number of (b,i,j) positions = B*N*N
    hidden_dim,      # i64      hidden dimension H
    eps,             # f32      epsilon for LayerNorm
    BLOCK_POS: tl.constexpr,   # compile‑time: how many positions per program
    BLOCK_H:   tl.constexpr,   # compile‑time: chunk size along hidden dim
):
    # --------------------------------------------------------------
    # 1) one‑dimensional index of the positions this program works on
    # --------------------------------------------------------------
    pid   = tl.program_id(0)
    pos   = pid * BLOCK_POS + tl.arange(0, BLOCK_POS)          # (BLOCK_POS,)
    pos_mask = pos < total_pos                                 # (BLOCK_POS,)

    # --------------------------------------------------------------
    # 2) first pass: compute mean and E[x²] for each position
    # --------------------------------------------------------------
    sum_val    = tl.zeros([BLOCK_POS], dtype=tl.float32)
    sum_sq_val = tl.zeros([BLOCK_POS], dtype=tl.float32)

    # loop over the hidden dimension in chunks of BLOCK_H
    for off in range(0, hidden_dim, BLOCK_H):
        cur_h   = tl.minimum(hidden_dim - off, BLOCK_H)       # actual chunk size
        h_idx   = off + tl.arange(0, BLOCK_H)                 # (BLOCK_H,)
        h_mask  = h_idx < hidden_dim                           # (BLOCK_H,)
        mask    = pos_mask[:, None] & h_mask[None, :]          # (BLOCK_POS, cur_h)

        # linear index into the flattened tensor: pos*H + h_idx
        idx = pos[:, None] * hidden_dim + h_idx[None, :]       # (BLOCK_POS, cur_h)

        x = tl.load(in_ptr + idx, mask=mask, other=tl.float16(0.0)).to(tl.float32)   # (BLOCK_POS, cur_h)
        sum_val    += tl.sum(x,          axis=1)               # (BLOCK_POS,)
        sum_sq_val += tl.sum(x * x,      axis=1)               # (BLOCK_POS,)

    mean = sum_val / hidden_dim
    var  = sum_sq_val / hidden_dim - mean * mean
    inv_std = 1.0 / tl.sqrt(var + eps)                         # (BLOCK_POS,)

    # --------------------------------------------------------------
    # 3) second pass: normalize, apply to_out_norm scale/bias,
    #                 multiply by out_gate and store result
    # --------------------------------------------------------------
    for off in range(0, hidden_dim, BLOCK_H):
        cur_h   = tl.minimum(hidden_dim - off, BLOCK_H)
        h_idx   = off + tl.arange(0, BLOCK_H)
        h_mask  = h_idx < hidden_dim
        mask    = pos_mask[:, None] & h_mask[None, :]

        idx = pos[:, None] * hidden_dim + h_idx[None, :]      # (BLOCK_POS, cur_h)

        # load the required tensors
        x      = tl.load(in_ptr   + idx, mask=mask, other=tl.float16(0.0)).to(tl.float32)
        g      = tl.load(gate_ptr + idx, mask=mask, other=tl.float16(0.0)).to(tl.float32)
        w      = tl.load(norm_w_ptr + h_idx, mask=h_mask, other=tl.float16(0.0)).to(tl.float32)
        b      = tl.load(norm_b_ptr + h_idx, mask=h_mask, other=tl.float16(0.0)).to(tl.float32)

        # (x - mean) * inv_std  -->  normalized value (float32)
        norm   = (x - mean[:, None]) * inv_std[:, None]       # (BLOCK_POS, cur_h)

        # apply learned scale & bias of the hidden‑dim LayerNorm
        norm   = norm * w + b                                # (BLOCK_POS, cur_h)

        # multiply by the out‑gate (still float32)
        norm   = norm * g

        # store back as FP16
        tl.store(out_ptr + idx, norm.to(tl.float16), mask=mask)


# -----------------------------------------------------------------------------#
# Public entry‑point used by the evaluator
# -----------------------------------------------------------------------------#
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul block (no gradients).

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, D] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) – may be None
            - weights_dict : mapping of all module parameters (float32)
            - config_dict  : {"dim": D, "hidden_dim": H, "nomask": bool, ...}

    Returns:
        torch.Tensor [B, N, N, D] (float32)
    """
    # --------------------------------------------------------------------- #
    # unpack arguments
    # --------------------------------------------------------------------- #
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    H = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5

    # --------------------------------------------------------------------- #
    # 1) Input LayerNorm (FP32 for stability)
    # --------------------------------------------------------------------- #
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )   # [B, N, N, dim]  (float32)

    # --------------------------------------------------------------------- #
    # 2) Cast to FP16 for the intensive ops
    # --------------------------------------------------------------------- #
    x_h = x.half()                                            # [B,N,N,dim] FP16

    # --------------------------------------------------------------------- #
    # 3) Load FP16 copies of the linear weights
    # --------------------------------------------------------------------- #
    left_proj_w   = weights["left_proj.weight"].half()
    right_proj_w  = weights["right_proj.weight"].half()
    left_gate_w   = weights["left_gate.weight"].half()
    right_gate_w  = weights["right_gate.weight"].half()
    out_gate_w    = weights["out_gate.weight"].half()
    to_out_norm_w = weights["to_out_norm.weight"].half()
    to_out_norm_b = weights["to_out_norm.bias"].half()
    to_out_w      = weights["to_out.weight"].half()

    # --------------------------------------------------------------------- #
    # 4) Linear projections (bias‑free) – all in half precision
    # --------------------------------------------------------------------- #
    left_proj  = F.linear(x_h, left_proj_w)   # [B, N, N, H]
    right_proj = F.linear(x_h, right_proj_w)  # [B, N, N, H]

    # --------------------------------------------------------------------- #
    # 5) Gating tensors (sigmoid) – keep them in FP16
    # --------------------------------------------------------------------- #
    left_gate  = torch.sigmoid(F.linear(x_h, left_gate_w)).half()   # [B,N,N,H]
    right_gate = torch.sigmoid(F.linear(x_h, right_gate_w)).half()
    out_gate   = torch.sigmoid(F.linear(x_h, out_gate_w)).half()    # [B,N,N,H]

    # --------------------------------------------------------------------- #
    # 6) Apply mask (if any) and element‑wise gate onto the projected tensors
    # --------------------------------------------------------------------- #
    if nomask:
        left  = left_proj * left_gate
        right = right_proj * right_gate
    else:
        # mask is broadcast over the hidden dimension
        mask_h = mask.to(dtype=torch.float16, device=device).unsqueeze(-1)   # [B,N,N,1]
        left  = left_proj * left_gate * mask_h
        right = right_proj * right_gate * mask_h

    # --------------------------------------------------------------------- #
    # 7) Heavy O(N³) contraction via a single einsum (recognised as batched GEMM)
    # --------------------------------------------------------------------- #
    # out shape: [B, N, N, H]
    out = torch.einsum('b i k d, b j k d -> b i j d', left, right)

    # --------------------------------------------------------------------- #
    # 8) Fuse hidden‑dim LayerNorm, to_out_norm affine, and out‑gate
    # --------------------------------------------------------------------- #
    total_positions = B * N * N
    BLOCK_POS = 128                     # positions per Triton program
    BLOCK_H   = 32                      # hidden‑dim chunk
    grid = (total_positions + BLOCK_POS - 1) // BLOCK_POS

    # allocate the tensor that will hold the fused result
    out_fused = torch.empty_like(out)   # FP16

    # ensure inputs are contiguous for simple linear indexing
    out      = out.contiguous()
    out_gate = out_gate.contiguous()
    to_out_norm_w = to_out_norm_w.contiguous()
    to_out_norm_b = to_out_norm_b.contiguous()

    _layernorm_gate_fuse_kernel[grid](
        out,
        out_gate,
        to_out_norm_w,
        to_out_norm_b,
        out_fused,
        total_positions,
        H,
        eps,
        BLOCK_POS=BLOCK_POS,
        BLOCK_H=BLOCK_H,
        num_warps=4,           # 4 warps (128 threads) works well on H100
    )

    # --------------------------------------------------------------------- #
    # 9) Final linear projection back to the original dimension
    # --------------------------------------------------------------------- #
    # flatten the first three dimensions for a single GEMM
    out_fused_flat = out_fused.view(B * N * N, H)          # (B*N*N, H)
    out_projected  = F.linear(out_fused_flat, to_out_w)    # (B*N*N, dim)   FP16
    out_final = out_projected.view(B, N, N, dim)           # (B, N, N, dim)

    # --------------------------------------------------------------------- #
    # 10) Cast back to FP32 for the downstream API
    # --------------------------------------------------------------------- #
    return out_final.float()
