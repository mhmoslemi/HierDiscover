"""
TriMul “outgoing” forward pass (AlphaFold‑3) – mixed‑precision Triton + PyTorch.

Algorithm
---------
1.  Layer‑norm over the last dimension (D) of the 4‑D input using a custom
    Triton kernel (two‑pass reduction → mean/variance → affine).  The kernel
    works on a 2‑D view [M, D] where M = B·N·N.

2.  Cast the normalized tensor to fp16 and compute *all* five bias‑free linear
    projections (left, right, left‑gate, right‑gate, out‑gate) in a *single*
    GEMM.  A concatenated weight matrix of shape (5·H, D) is used, then the
    result is split into the five tensors.

3.  If a mask is supplied, broadcast‑multiply it with the left/right tensors.

4.  Apply sigmoid to the three gate tensors and fuse the gating (element‑wise
    multiplication) with the projected values.

5.  The expensive contraction
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    is a batched matrix multiplication per head.  By permuting the tensors
    to shape (B·H, N, N) we can reuse cuBLAS/Tensor‑core batched GEMM
    (`torch.matmul`) which is much faster than a hand‑written Triton kernel.

6.  Layer‑norm over the hidden dimension (H) of the contracted result – again
    using the Triton kernel.

7.  Multiply by the out‑gate (now in fp32), then a final bias‑free linear
    projection H → D to obtain the output tensor in float32.

All tensors stay on the device; the only Triton code required is the
row‑wise LayerNorm kernel, satisfying the “at least one kernel” constraint.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton row‑wise LayerNorm (two‑pass: mean/var → affine)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,                 # *ptr to input  [M, N]
    y_ptr,                 # *ptr to output [M, N]
    weight_ptr,            # *ptr to weight [N]
    bias_ptr,              # *ptr to bias   [N]
    M,                     # number of rows
    N,                     # number of columns (feature dim)
    eps,                   # epsilon for numerical stability
    stride_xm, stride_xn,  # strides of input
    stride_ym, stride_yn,  # strides of output
    BLOCK_M: tl.constexpr, # rows per program
    BLOCK_N: tl.constexpr, # cols per program
):
    pid = tl.program_id(0)

    # ------------------------------------------------------------------
    # Row indices handled by this program
    # ------------------------------------------------------------------
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # ------------------------------------------------------------------
    # First pass: compute sum and sum‑of‑squares for each row
    # ------------------------------------------------------------------
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_N)
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N

        # Load a tile [BLOCK_M, BLOCK_N] with out‑of‑bounds filled with 0
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        sum_val += tl.sum(x, axis=1)
        sum_sq  += tl.sum(x * x, axis=1)

    mean = sum_val / N
    var  = sum_sq / N - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)  # 1/sqrt(var+eps)

    # ------------------------------------------------------------------
    # Second pass: normalize + affine transformation
    # ------------------------------------------------------------------
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N

        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        norm_x = (x - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_n, mask=mask_n, other=0.0)
        b = tl.load(bias_ptr   + offs_n, mask=mask_n, other=0.0)

        y = norm_x * w[None, :] + b[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn,
            y,
            mask=mask_m[:, None] & mask_n[None, :],
        )


def triton_layernorm(x: torch.Tensor,
                    weight: torch.Tensor,
                    bias: torch.Tensor,
                    eps: float = 1e-5) -> torch.Tensor:
    """
    Apply LayerNorm over the last dimension of ``x`` using the Triton kernel.
    ``x`` must be a 2‑D tensor [M, N]; the result has the same dtype as ``x``.
    """
    assert x.dim() == 2, "triton_layernorm expects a 2‑D tensor"
    M, N = x.shape
    y = torch.empty_like(x)

    # Ensure contiguous memory for correct stride handling
    x = x.contiguous()
    weight = weight.contiguous()
    bias = bias.contiguous()

    stride_xm, stride_xn = x.stride()
    stride_ym, stride_yn = y.stride()

    grid = lambda meta: (triton.cdiv(M, meta["BLOCK_M"]),)

    layernorm_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        weight_ptr=weight,
        bias_ptr=bias,
        M=M,
        N=N,
        eps=eps,
        stride_xm=stride_xm,
        stride_xn=stride_xn,
        stride_ym=stride_ym,
        stride_yn=stride_yn,
        BLOCK_M=256,   # rows per CTA (tuned for large M)
        BLOCK_N=64,    # columns per tile (multiple of 64 works well for fp16)
    )
    return y


# ----------------------------------------------------------------------
# Entry‑point used by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, L, L, D] (float32)
        - mask         : torch.Tensor, shape [B, L, L] or ``None``
        - weights      : dict of torch tensors (model parameters)
        - config       : dict with keys ``dim``, ``hidden_dim`` and optional ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, L, L, D] (float32)
    """
    # ------------------------------------------------------------------
    # 0️⃣ Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype          # expected float32
    bs, L, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (Triton) over the last dimension (D)
    # ------------------------------------------------------------------
    M = bs * L * L
    x_flat = input_tensor.reshape(M, dim).contiguous()
    norm_w = weights["norm.weight"].to(device, dtype).contiguous()
    norm_b = weights["norm.bias"].to(device, dtype).contiguous()
    x_norm_flat = triton_layernorm(x_flat, norm_w, norm_b, eps=eps)
    x_norm = x_norm_flat.view(bs, L, L, dim)          # [B, L, L, D] (float32)

    # ------------------------------------------------------------------
    # 2️⃣ FP16 work‑horse tensor
    # ------------------------------------------------------------------
    x_norm_h = x_norm.to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Fuse the five bias‑free linear projections into a single GEMM
    # ------------------------------------------------------------------
    # Assemble the concatenated weight matrix: (5*H, D)
    left_proj_w   = weights["left_proj.weight"]
    right_proj_w  = weights["right_proj.weight"]
    left_gate_w   = weights["left_gate.weight"]
    right_gate_w  = weights["right_gate.weight"]
    out_gate_w    = weights["out_gate.weight"]

    combined_w = torch.cat(
        [
            left_proj_w,
            right_proj_w,
            left_gate_w,
            right_gate_w,
            out_gate_w,
        ],
        dim=0,
    ).to(device, torch.float16).contiguous()   # shape (5*H, D)

    # Single linear projection
    combined = F.linear(x_norm_h, combined_w)   # shape (M, 5*H)
    combined = combined.view(bs, L, L, 5 * hidden_dim)

    # Split into dedicated tensors
    left, right, left_gate_pre, right_gate_pre, out_gate_pre = torch.split(
        combined, hidden_dim, dim=-1
    )   # each: [B, L, L, H] (fp16)

    # ------------------------------------------------------------------
    # 4️⃣ Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # mask is float‑like; promote to fp16 for element‑wise multiplication
        mask_h = mask.to(device, torch.float16).unsqueeze(-1)   # [B, L, L, 1]
        left = left * mask_h
        right = right * mask_h

    # ------------------------------------------------------------------
    # 5️⃣ Gating (sigmoid) + fusing with projected values
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(left_gate_pre)   # [B, L, L, H] (fp16)
    right_gate = torch.sigmoid(right_gate_pre)  # [B, L, L, H] (fp16)
    out_gate   = torch.sigmoid(out_gate_pre)    # [B, L, L, H] (fp16)

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6️⃣ Batched matrix multiplication: out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    # Rearrange to (B, H, N, N) – each head becomes an independent batch.
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    # Batched GEMM using cuBLAS/Tensor‑cores (fp16)
    out_perm = torch.matmul(left_perm, right_perm.transpose(-2, -1))  # [B, H, N, N]

    # Restore original axis order → [B, N, N, H]
    out_h = out_perm.permute(0, 2, 3, 1).contiguous()   # fp16

    # ------------------------------------------------------------------
    # 7️⃣ Output LayerNorm over hidden dimension (H) – Triton
    # ------------------------------------------------------------------
    out_fp32 = out_h.to(torch.float32)          # LayerNorm works in fp32
    out_M = bs * L * L
    out_flat = out_fp32.view(out_M, hidden_dim).contiguous()
    to_out_norm_w = weights["to_out_norm.weight"].to(device, dtype).contiguous()
    to_out_norm_b = weights["to_out_norm.bias"].to(device, dtype).contiguous()
    out_norm_flat = triton_layernorm(out_flat, to_out_norm_w, to_out_norm_b, eps=eps)
    out_norm = out_norm_flat.view(bs, L, L, hidden_dim)   # [B, L, L, H] (float32)

    # ------------------------------------------------------------------
    # 8️⃣ Apply output gate and final linear projection H → D
    # ------------------------------------------------------------------
    # Convert out_gate to fp32 for the scaling
    out_gate_f32 = out_gate.to(dtype)          # [B, L, L, H] (float32)
    out_norm = out_norm * out_gate_f32

    to_out_w = weights["to_out.weight"].to(device, dtype).contiguous()   # [D, H]
    out = F.linear(out_norm, to_out_w)        # [B, L, L, D] (float32)

    return out
