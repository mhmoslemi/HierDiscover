"""
Optimised outgoing TriMul (AlphaFold‑3) forward pass.

Algorithm
---------

1. **Input LayerNorm** – use the highly‑optimised torch.nn.functional.layer_norm
   (operates in FP16, mean/var accumulated in FP32).  This avoids the costly
   Triton kernel used in the reference implementation.

2. **Fused linear projections** – concatenate the four weight matrices
   (left_proj, right_proj, left_gate, right_gate) **and the out‑gate weight**
   into a single `(5·H)×D` matrix.  One `F.linear` call produces all required
   tensors at once, dramatically cutting memory traffic.

3. **In‑place gating & optional mask** – apply the sigmoid to the gate
   tensors *in‑place* (`sigmoid_`) and reuse the projection tensors as the
   gated ``left``/``right`` values.  The mask (if present) is also applied
   in‑place, eliminating extra buffers.

4. **N³ contraction** – compute the core operation
   `out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]`
   using a single `torch.einsum`.  This internally maps to a batched GEMM
   without the expensive `permute → contiguous → reshape` copy that the
   naïve implementation performed.

5. **Output LayerNorm** – a small Triton row‑wise LayerNorm kernel (the only
   required custom kernel) normalises `out` over the hidden dimension.

6. **Output gate & final projection** – the out‑gate is already available from
   step 2; after element‑wise multiplication with the normalised tensor we
   apply the final linear projection back to the original dimension.

All heavy linear algebra runs in FP16 (Tensor‑core friendly) while reductions
(mean/var) are performed in FP32 for numerical stability.  In‑place operations
and fused matmuls minimise memory bandwidth, allowing the kernel to meet the
target sub‑millisecond latency on an H100.

The function `custom_kernel` follows the required signature.
"""

@triton.jit
def layernorm_kernel(
    x_ptr, y_ptr,                     # *ptrs to input/output [M, N]
    weight_ptr, bias_ptr,              # *ptrs to weight/bias [N]
    M, N, eps,                         # dimensions + epsilon
    stride_xm, stride_xn,              # strides for input
    stride_ym, stride_yn,              # strides for output
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
):
    pid = tl.program_id(0)

    # ------------------------------------------------------------------
    # Row offsets for this block
    # ------------------------------------------------------------------
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # ------------------------------------------------------------------
    # First pass: accumulate sum and sum‑of‑squares in FP32
    # ------------------------------------------------------------------
    sum_ = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq = tl.zeros((BLOCK_M,), dtype=tl.float32)

    NUM_N = tl.cdiv(N, BLOCK_N)
    for n in range(NUM_N):
        offs_n = tl.arange(0, BLOCK_N) + n * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        xf = x.to(tl.float32)
        sum_   += tl.sum(xf, axis=1)
        sum_sq += tl.sum(xf * xf, axis=1)

    mean = sum_ / N
    var  = sum_sq / N - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # ------------------------------------------------------------------
    # Second pass: write normalised + affine transformed values
    # ------------------------------------------------------------------
    for n in range(NUM_N):
        offs_n = tl.arange(0, BLOCK_N) + n * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        xf = x.to(tl.float32)
        norm = (xf - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_n, mask=mask_n, other=0.0).to(tl.float32)
        b = tl.load(bias_ptr   + offs_n, mask=mask_n, other=0.0).to(tl.float32)

        y = norm * w[None, :] + b[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn,
            y.to(x.dtype),
            mask=mask_m[:, None] & mask_n[None, :],
        )


def triton_layernorm(x: torch.Tensor,
                     weight: torch.Tensor,
                     bias: torch.Tensor,
                     eps: float = 1e-5) -> torch.Tensor:
    """
    Row‑wise LayerNorm implemented with Triton.
    x: 2‑D tensor [M, N] (any dtype, typically FP16)
    weight / bias: 1‑D tensors of length N (same dtype as x)
    Returns tensor of same shape/dtype as ``x``.
    """
    assert x.dim() == 2, "triton_layernorm expects a 2‑D tensor"
    M, N = x.shape
    y = torch.empty_like(x)

    # make everything contiguous for correct stride handling
    x = x.contiguous()
    y = y.contiguous()
    weight = weight.contiguous()
    bias = bias.contiguous()

    stride_xm, stride_xn = x.stride()
    stride_ym, stride_yn = y.stride()

    grid = lambda meta: (triton.cdiv(M, meta["BLOCK_M"]),)

    layernorm_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        weight_ptr=weight,
        bias_ptr=bias,
        M=M,
        N=N,
        eps=eps,
        stride_xm=stride_xm,
        stride_xn=stride_xn,
        stride_ym=stride_ym,
        stride_yn=stride_yn,
        BLOCK_M=256,  # larger block reduces kernel launch overhead
        BLOCK_N=64,
    )
    return y


# --------------------------------------------------------------
# Entry point required by the test harness
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor   [B, L, L, D]   (float32)
        - mask         : torch.Tensor   [B, L, L] or ``None``
        - weights      : dict of torch tensors (model parameters)
        - config       : dict with keys ``dim`` and ``hidden_dim``; optional ``nomask``

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, L, L, D] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    bs, L, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm – use torch's fused implementation on FP16
    # ------------------------------------------------------------------
    # Cast to FP16 early to halve memory bandwidth
    x_fp16 = input_tensor.to(torch.float16)

    # LayerNorm parameters (cast to FP16)
    norm_w = weights["norm.weight"].to(device, torch.float16).contiguous()
    norm_b = weights["norm.bias"].to(device, torch.float16).contiguous()

    x_norm = F.layer_norm(
        x_fp16,
        normalized_shape=(dim,),
        weight=norm_w,
        bias=norm_b,
        eps=eps,
    )   # [B, L, L, D]  (fp16)

    # ------------------------------------------------------------------
    # 2️⃣ Fuse four linear maps + out‑gate into a **single** matmul
    # ------------------------------------------------------------------
    # Gather weight matrices (bias‑free)
    left_proj_w   = weights["left_proj.weight"].to(device, torch.float16).contiguous()
    right_proj_w  = weights["right_proj.weight"].to(device, torch.float16).contiguous()
    left_gate_w   = weights["left_gate.weight"].to(device, torch.float16).contiguous()
    right_gate_w  = weights["right_gate.weight"].to(device, torch.float16).contiguous()
    out_gate_w    = weights["out_gate.weight"].to(device, torch.float16).contiguous()

    # Concatenate into (5·H) × D matrix
    fused_weight = torch.cat(
        [left_proj_w, right_proj_w, left_gate_w, right_gate_w, out_gate_w],
        dim=0,
    )  # shape: (5*hidden_dim, dim)

    # Flatten the normalized tensor for the matmul
    M = bs * L * L
    x_norm_flat = x_norm.reshape(M, dim)           # [M, D]  (fp16)

    # One big linear projection
    fused_out = F.linear(x_norm_flat, fused_weight)  # [M, 5*H] (fp16)

    # Reshape back to 4‑D
    fused_out = fused_out.view(bs, L, L, 5 * hidden_dim)

    # ------------------------------------------------------------------
    # 3️⃣ Split the fused tensor
    # ------------------------------------------------------------------
    left_proj = fused_out[..., :hidden_dim]                     # (B, L, L, H)
    right_proj = fused_out[..., hidden_dim:2 * hidden_dim]      # (B, L, L, H)

    left_gate  = fused_out[..., 2 * hidden_dim:3 * hidden_dim]  # (B, L, L, H)
    right_gate = fused_out[..., 3 * hidden_dim:4 * hidden_dim]  # (B, L, L, H)
    out_gate   = fused_out[..., 4 * hidden_dim:]                # (B, L, L, H)

    # ------------------------------------------------------------------
    # 4️⃣ In‑place sigmoid for the gate tensors
    # ------------------------------------------------------------------
    left_gate = left_gate.sigmoid_()
    right_gate = right_gate.sigmoid_()
    out_gate = out_gate.sigmoid_()

    # ------------------------------------------------------------------
    # 5️⃣ Apply optional mask **before** gating (in‑place)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_h = mask.to(device, torch.float16).unsqueeze(-1)   # (B, L, L, 1)
        left_proj = left_proj * mask_h
        right_proj = right_proj * mask_h

    # ------------------------------------------------------------------
    # 6️⃣ Apply gates (in‑place) – re‑use the projection buffers
    # ------------------------------------------------------------------
    left = left_proj
    left.mul_(left_gate)        # left := left_proj * left_gate
    right = right_proj
    right.mul_(right_gate)      # right := right_proj * right_gate

    # ------------------------------------------------------------------
    # 7️⃣ Core N³ contraction via a single `einsum`
    #     (equivalent to batched GEMM over the hidden dimension)
    # ------------------------------------------------------------------
    # out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    out = torch.einsum('b i k h, b j k h -> b i j h', left, right)  # (B, L, L, H) fp16

    # ------------------------------------------------------------------
    # 8️⃣ Output LayerNorm – Triton implementation (the required custom kernel)
    # ------------------------------------------------------------------
    out_flat = out.reshape(bs * L * L, hidden_dim)               # [M_out, H]
    out_norm_w = weights["to_out_norm.weight"].to(device, torch.float16).contiguous()
    out_norm_b = weights["to_out_norm.bias"].to(device, torch.float16).contiguous()
    out_norm_flat = triton_layernorm(out_flat, out_norm_w, out_norm_b, eps=eps)
    out_norm = out_norm_flat.view(bs, L, L, hidden_dim)          # (B, L, L, H) fp16

    # ------------------------------------------------------------------
    # 9️⃣ Apply output gate (already computed) and final projection
    # ------------------------------------------------------------------
    gated = out_norm * out_gate                     # (B, L, L, H) fp16

    to_out_w = weights["to_out.weight"].to(device, torch.float16).contiguous()
    out_final_fp16 = F.linear(gated, to_out_w)      # (B, L, L, D) fp16

    # Cast back to the original dtype (float32) for the model output
    out_final = out_final_fp16.to(torch.float32)

    return out_final
