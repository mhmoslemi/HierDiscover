@triton.jit
def tri_mul_kernel(
    out_ptr, left_ptr, right_ptr,
    M, N, K,
    stride_out_c, stride_out_m, stride_out_n,
    stride_left_c, stride_left_m, stride_left_k,
    stride_right_c, stride_right_k, stride_right_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr, GROUP_M: tl.constexpr,
):
    pid_c = tl.program_id(2)
    pid = tl.program_id(0) # combined
    num_pid_m = tl.cdiv(M, BLOCK_M)
    num_pid_n = tl.cdiv(N, BLOCK_N)
    # Compute grouped m index
    pid_m = (pid // num_pid_n) % num_pid_m   # block row
    pid_n = pid % num_pid_n                  # block col

    # offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # left tile
        left_ptrs = left_ptr + pid_c * stride_left_c \
                    + offs_m[:, None] * stride_left_m \
                    + offs_k[None, :] * stride_left_k
        left = tl.load(left_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # right tile (transposed)
        right_ptrs = right_ptr + pid_c * stride_right_c \
                     + offs_k[:, None] * stride_right_k \
                     + offs_n[None, :] * stride_right_n
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # dot
        acc += tl.dot(left, right, out_dtype=tl.float32)

    # store
    out_ptrs = out_ptr + pid_c * stride_out_c \
               + offs_m[:, None] * stride_out_m \
               + offs_n[None, :] * stride_out_n
    tl.store(out_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])
```

"""
TriMul (outgoing) – fast Triton implementation.

The core of the module is the “triangle” multiplication
    out[b,i,j,h] = Σ_k  left[b,i,k,h] * right[b,j,k,h] .
For every hidden channel `h` (and batch element) this is a matrix
multiply `left @ rightᵀ` of size (N,N)·(N,N) → (N,N).  The kernel
processes all channels in parallel by folding the batch dimension
into the channel axis, i.e. we work on tensors of shape
(Ctot, N, N) where `Ctot = B * hidden_dim`.  The mat‑mul is performed
in FP16 (Tensor‑Core) with FP32 accumulation for numerical stability.
All other operations (LayerNorm, linear projections, sigmoid gates
and the final projection) are executed with regular PyTorch kernels.

Key optimisations
-----------------
*   FP16 mat‑mul with Tensor‑Core (`tl.dot(..., out_dtype=tl.float32)`);
    this avoids the extra FP16→FP32 cast in the accumulation loop.
*   Large 128×128 tiles + 8 warps per block – a good match for H100.
*   The hidden‑channel dimension is folded into the batch axis, yielding
    a single 3‑D GEMM where cuBLAS‑like memory reuse is achieved by the
    tile‑wise loads inside the kernel.
*   Optional mask handling is still performed in PyTorch (negligible cost)
    – the heavy O(N³·hidden) work stays on the Triton kernel.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiplication for the triangle product
#   out = left @ rightᵀ   (shape: (Ctot, N, N) = (B*H, N, N))
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    out_ptr,           # *float32   (Ctot, M, N)
    left_ptr,          # *float16   (Ctot, M, K)   – M == K == N
    right_ptr,         # *float16   (Ctot, K, N)   – K == N
    M, N, K,           # i32 matrix sizes (all equal to seq_len)
    # strides (in elements, not bytes)
    stride_out_c, stride_out_m, stride_out_n,
    stride_left_c, stride_left_m, stride_left_k,
    stride_right_c, stride_right_k, stride_right_n,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # --------------------------------------------------------------
    # program IDs – (pid_m, pid_n, pid_c) process a tile
    # --------------------------------------------------------------
    pid_m = tl.program_id(0)   # block row (i)
    pid_n = tl.program_id(1)   # block col (j)
    pid_c = tl.program_id(2)   # folded channel (B*H) index

    # --------------------------------------------------------------
    # tile offsets
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in FP32 (Tensor‑Core will write directly to FP32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # K‑loop: iterate over the reduction dimension
    # --------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # ---------- left tile (M × K) ----------
        left_ptrs = left_ptr + pid_c * stride_left_c \
                    + offs_m[:, None] * stride_left_m \
                    + offs_k[None, :] * stride_left_k
        left = tl.load(left_ptrs,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0)          # FP16

        # ---------- right tile (K × N) ----------
        # the right operand is used transposed on‑the‑fly
        right_ptrs = right_ptr + pid_c * stride_right_c \
                     + offs_k[:, None] * stride_right_k \
                     + offs_n[None, :] * stride_right_n
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_n[None, :],
                        other=0.0)          # FP16

        # ---------- MMA accumulation ----------
        # out_dtype=tl.float32 forces Tensor‑Core accumulation in FP32
        acc += tl.dot(left, right, out_dtype=tl.float32)

    # --------------------------------------------------------------
    # write back the result tile
    # --------------------------------------------------------------
    out_ptrs = out_ptr + pid_c * stride_out_c \
               + offs_m[:, None] * stride_out_m \
               + offs_n[None, :] * stride_out_n
    tl.store(out_ptrs, acc,
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Triton‑powered forward pass for the outgoing TriMul module (no grads)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """

    Args
    ----

        (input_tensor, mask, weights, config)

        - mask         : torch.Tensor[B, N, N] (bool/float) or None
        - weights      : dict of model parameters (see reference implementation)
        - config       : dict with keys "dim", "hidden_dim", optionally "nomask"

    Returns
    -------
    torch.Tensor[B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    B, N, _, C_in = input_tensor.shape

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )   # (B, N, N, C)  float32

    # ------------------------------------------------------------------
    # 2) Cast to half, linear projections and sigmoid gates
    # ------------------------------------------------------------------
    x_h = x.to(torch.float16)

    # Linear projections (no bias) – cast weights to half once
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    left = F.linear(x_h, left_proj_w)          # (B, N, N, H)
    right = F.linear(x_h, right_proj_w)       # (B, N, N, H)

    # optional mask (broadcast over hidden dim)
    if not nomask and mask is not None:
        mask_f = mask.to(torch.float16).unsqueeze(-1)   # (B, N, N, 1)
        left = left * mask_f
        right = right * mask_f

    # Sigmoid gates
    left_gate = torch.sigmoid(F.linear(x_h, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_h, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x_h, out_gate_w))   # (B, N, N, H)

    # apply gates
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 3) Rearrange tensors for the Triton kernel:
    #    (Ctot, N, N) where Ctot = B * hidden_dim
    # ------------------------------------------------------------------
    # move hidden dimension before the two spatial axes and make it contiguous
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    Ctot = B * hidden_dim
    left_flat = left.reshape(Ctot, N, N)          # (Ctot, M, K)
    right_flat = right.reshape(Ctot, N, N)        # (Ctot, K, N) – we will treat it as (K,N)

    # output buffer for the core multiplication (float32)
    out_flat = torch.empty((Ctot, N, N), dtype=torch.float32, device=device)

    # ------------------------------------------------------------------
    # 4) Triton kernel launch (batched matmul)
    # ------------------------------------------------------------------
    # Tune tile sizes – 128×128 works well for N≥512, 64×64 for smaller N
    if N <= 256:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 32
        num_warps = 4
    else:
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 32
        num_warps = 8

    # Strides in element units (not bytes)
    stride_out_c, stride_out_m, stride_out_n = out_flat.stride()
    stride_left_c, stride_left_m, stride_left_k = left_flat.stride()
    stride_right_c, stride_right_k, stride_right_n = (
        right_flat.stride(0), right_flat.stride(1), right_flat.stride(2)
    )

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,          # pid_m
        (N + BLOCK_N - 1) // BLOCK_N,          # pid_n
        Ctot,                                  # pid_c (folded batch*hidden)
    )

    tri_mul_kernel[grid](
        out_flat,
        left_flat,
        right_flat,
        N, N, N,
        stride_out_c, stride_out_m, stride_out_n,
        stride_left_c, stride_left_m, stride_left_k,
        stride_right_c, stride_right_k, stride_right_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=num_warps,
    )

    # ------------------------------------------------------------------
    # 5) Reshape back to (B, N, N, H)
    # ------------------------------------------------------------------
    out = out_flat.reshape(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6) Post‑processing: hidden‑dim LayerNorm → out_gate → final linear
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )   # (B, N, N, H) float32

    # out_gate is half‑precision – promote before the elementwise product
    out = out * out_gate.to(torch.float32)

    # final projection back to original dimensionality
    out = F.linear(out, weights["to_out.weight"])   # (B, N, N, dim) float32

    return out
```
