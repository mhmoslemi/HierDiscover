```python
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# Triton kernel for 4D TriMul outgoing reduction.
@triton.jit
def trilu_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # program IDs
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_head = tl.program_id(2)   # flattened batch * hidden

    # Determine batch and head indices
    b = pid_head // H
    h = pid_head % H

    # Offsets for i and j
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # Pointers to the start of the current head
    left_base = left_ptr + b * stride_left_b + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base = out_ptr + b * stride_out_b + h * stride_out_h

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over reduction dimension k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load a tile from left: shape (BLOCK_M, BLOCK_K)
        left = tl.load(
            left_base + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0
        )
        # Load a tile from right: shape (BLOCK_N, BLOCK_K)
        right = tl.load(
            right_base + offs_j[:, None] * stride_right_i + offs_k[None, :] * stride_right_k,
            mask=mask_j[:, None] & mask_k[None, :],
            other=0.0
        )
        # Accumulate product; right is transposed on the k dimension
        acc += tl.dot(left, right, trans_b=True)

    # Write block of output, cast to fp16
    out = acc.to(tl.float16)
    tl.store(
        out_base + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j,
        out,
        mask=mask_i[:, None] & mask_j[None, :]
    )
```

"""
Efficient Triton implementation of the “outgoing” TriMul update.

Algorithm
---------
1. Input LayerNorm (FP32) – performed with Torch’s native
   `torch.nn.functional.layer_norm` (fast fused kernel on H100).
2. A single fused half‑precision linear projection produces the five
   intermediate tensors: left_proj, right_proj, left_gate, right_gate,
   out_gate.
3. Mask (if any) and the three sigmoid gates are applied **in‑place**
   on the half‑precision tensors, avoiding extra temporaries.
4. Custom Triton kernel computes the O(N³) reduction
      out[b,i,j,h] = Σₖ left[b,i,k,h]·right[b,j,k,h]
   directly on the `[B,N,N,H]` layout; no transpose/contiguous copy is
   required.  FP32 accumulation is used for numerical stability, the
   result is written back as FP16.
5. Output LayerNorm (FP32) – again via the native Torch kernel.
6. Multiply by the output gate, a final half‑precision linear projection
   back to `dim`, and cast the result to FP32 for downstream use.

All heavy arithmetic runs in FP16 (Tensor‑core friendly) while the
normalisation steps stay in FP32, yielding a fast forward pass well
below the target latency.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel for the outgoing TriMul reduction
# ----------------------------------------------------------------------
@triton.jit
def tril_mul_outgoing_kernel(
    left_ptr,          # *ptr  left   [B, N, N, H]   (fp16)
    right_ptr,         # *ptr  right  [B, N, N, H]   (fp16)
    out_ptr,           # *ptr  output [B, N, N, H]   (fp16)
    B, N, H,           # int   batch, seq_len, hidden_dim
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Compute out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    using a standard GEMM tiling pattern, but with the batch*head
    dimension folded into the third program‑id axis.
    """
    pid_i = tl.program_id(0)          # tile over the i‑dimension
    pid_j = tl.program_id(1)          # tile over the j‑dimension
    pid_head = tl.program_id(2)       # flattened (batch, hidden) index

    # Derive batch and hidden indices from the flattened head index
    b = pid_head // H
    h = pid_head % H

    # Offsets inside the current tile
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Base pointers for the current (b, h) slice
    left_base  = left_ptr  + b * stride_left_b  + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base   = out_ptr   + b * stride_out_b   + h * stride_out_h

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load a tile of left: shape (BLOCK_M, BLOCK_K)
        left = tl.load(
            left_base + offs_i[:, None] * stride_left_i
                      + offs_k[None, :] * stride_left_k,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0
        )
        # Load a tile of right: shape (BLOCK_N, BLOCK_K)
        right = tl.load(
            right_base + offs_j[:, None] * stride_right_i
                       + offs_k[None, :] * stride_right_k,
            mask=mask_j[:, None] & mask_k[None, :],
            other=0.0
        )
        # Accumulate: right is transposed on the k‑axis
        acc += tl.dot(left, right, trans_b=True)

    # Write the result back as fp16
    out_tile = acc.to(tl.float16)
    tl.store(
        out_base + offs_i[:, None] * stride_out_i
                 + offs_j[None, :] * stride_out_j,
        out_tile,
        mask=mask_i[:, None] & mask_j[None, :]
    )


# ----------------------------------------------------------------------
# Entry point: forward pass of the outgoing TriMul
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, C] (float32)
        - mask         : torch.Tensor of shape [B, N, N] (float32) or None
        - weights      : dict of torch.Tensors containing model weights
        - config       : dict with keys ``dim``, ``hidden_dim`` and ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype        # expected torch.float32

    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32, native Torch kernel)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(device, dtype=torch.float32)
    norm_bias   = weights["norm.bias"].to(device, dtype=torch.float32)
    x_norm = F.layer_norm(input_tensor,
                          normalized_shape=(dim,),
                          weight=norm_weight,
                          bias=norm_bias,
                          eps=1e-5)                     # (B, N, N, C) FP32

    # ------------------------------------------------------------------
    # 2️⃣ Fused half‑precision linear that produces all five tensors
    # ------------------------------------------------------------------
    x_norm_h = x_norm.half()
    M = bs * N * N
    x_norm_h_flat = x_norm_h.view(M, dim)                 # (M, C)

    # Concatenate the five weight matrices once (shape (5·H, C))
    combined_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(device, dtype=torch.float16)                    # (5·H, C)

    proj_out = F.linear(x_norm_h_flat, combined_weight)  # (M, 5·H) FP16

    # ------------------------------------------------------------------
    # 3️⃣ Split and reshape the five tensors
    # ------------------------------------------------------------------
    left_proj, right_proj, left_gate, right_gate, out_gate = torch.split(
        proj_out, hidden_dim, dim=1
    )
    left_proj   = left_proj.view(bs, N, N, hidden_dim)
    right_proj  = right_proj.view(bs, N, N, hidden_dim)
    left_gate   = left_gate.view(bs, N, N, hidden_dim)
    right_gate  = right_gate.view(bs, N, N, hidden_dim)
    out_gate    = out_gate.view(bs, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 4️⃣ Optional mask (broadcast over hidden_dim)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_h = mask.to(device, dtype=torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left_proj.mul_(mask_h)
        right_proj.mul_(mask_h)

    # ------------------------------------------------------------------
    # 5️⃣ In‑place sigmoid gates and gate application
    # ------------------------------------------------------------------
    left_gate.sigmoid_()          # stay in fp16
    right_gate.sigmoid_()
    out_gate.sigmoid_()

    left_proj.mul_(left_gate)     # left = left_proj * left_gate
    right_proj.mul_(right_gate)   # right = right_proj * right_gate

    left  = left_proj   # aliases for clarity
    right = right_proj

    # ------------------------------------------------------------------
    # 6️⃣ Triton kernel: out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    out = torch.empty_like(left)   # (B,N,N,H) fp16

    # Strides of the tensors (all contiguous in B,N,N,H layout)
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_b, stride_r_i, stride_r_k, stride_r_h = right.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out.stride()

    grid = lambda meta: (
        triton.cdiv(N, meta["BLOCK_M"]),
        triton.cdiv(N, meta["BLOCK_N"]),
        bs * hidden_dim,               # batch * hidden dimension
    )

    tril_mul_outgoing_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=bs,
        N=N,
        H=hidden_dim,
        stride_left_b=stride_l_b,
        stride_left_i=stride_l_i,
        stride_left_k=stride_l_k,
        stride_left_h=stride_l_h,
        stride_right_b=stride_r_b,
        stride_right_i=stride_r_i,
        stride_right_k=stride_r_k,
        stride_right_h=stride_r_h,
        stride_out_b=stride_o_b,
        stride_out_i=stride_o_i,
        stride_out_j=stride_o_j,
        stride_out_h=stride_o_h,
        BLOCK_M=64,   # tuned for H100 – fits in shared memory with fp32 acc
        BLOCK_N=64,
        BLOCK_K=32,
    )

    # ------------------------------------------------------------------
    # 7️⃣ Output LayerNorm (FP32, native Torch kernel)
    # ------------------------------------------------------------------
    out_fp32 = out.float()
    to_out_norm_weight = weights["to_out_norm.weight"].to(device, dtype=torch.float32)
    to_out_norm_bias   = weights["to_out_norm.bias"].to(device, dtype=torch.float32)

    out_norm = F.layer_norm(out_fp32,
                            normalized_shape=(hidden_dim,),
                            weight=to_out_norm_weight,
                            bias=to_out_norm_bias,
                            eps=1e-5)                     # (B,N,N,H) FP32

    # ------------------------------------------------------------------
    # 8️⃣ Apply output gate and final projection back to model dim
    # ------------------------------------------------------------------
    # out_gate is still fp16 – broadcasting to fp32 works automatically
    gated = out_norm * out_gate                               # FP32

    to_out_weight = weights["to_out.weight"].to(device, dtype=torch.float16)  # (dim, H)

    final_fp16 = F.linear(gated_h, to_out_weight)            # (M, dim) fp16
    final = final_fp16.view(bs, N, N, dim).float()           # back to FP32

    return final
