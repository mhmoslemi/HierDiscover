"""
TriMul (outgoing) – fast Triton‑accelerated implementation
========================================================

The outgoing triangle multiplicative update computes, for each hidden channel
h, a batched matrix product:

    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

where `left` and `right` are linear projections of a layer‑normed input
tensor that are modulated by elementwise sigmoid gates (and an optional mask).
The bottleneck is the N×N matrix multiplication for every hidden channel.
The kernel below fuses the entire GEMM for a single (batch × hidden) slice:

* Input tensors `left` (M×K) and `right` (K×N) are FP16.
* Accumulation is performed in FP32 for numerical stability.
* A 128×128×64 tile (M, N, K) with double‑buffered loads (``num_stages=2``)
  and 8 warps per block is used, which is near‑optimal for H100 tensor‑cores.

All other operations (LayerNorm, linear projections, sigmoid gating,
final normalisation and projection) are done with regular PyTorch kernels.
The final output is returned in FP32 as required by downstream models.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    out_ptr,            # float32*   [C, M, N]   output
    a_ptr,              # float16*   [C, M, K]   left  matrix
    b_ptr,              # float16*   [C, K, N]   right matrix (transposed layout)
    M, N, K,            # int64     matrix sizes (all = seq_len)
    stride_out_c, stride_out_m, stride_out_n,   # strides for out
    stride_a_c, stride_a_m, stride_a_k,        # strides for a (left)
    stride_b_c, stride_b_k, stride_b_n,        # strides for b (right, K‑first)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM: out = left @ rightᵀ
    One program processes a tile (BLOCK_M, BLOCK_N) of a single (C‑index)
    matrix, i.e. a specific (batch * hidden_dim) slice.
    """
    pid_m = tl.program_id(0)   # block row   → i
    pid_n = tl.program_id(1)   # block col   → j
    pid_c = tl.program_id(2)   # hidden‑channel (batch * hidden_dim)

    # Tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator (FP32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in chunks of BLOCK_K
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = kk * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Pointers to the current tiles of left (M×K) and right (K×N)
        a_tile_ptr = (
            a_ptr
            + pid_c * stride_a_c
            + offs_m[:, None] * stride_a_m
            + offs_k[None, :] * stride_a_k
        )
        b_tile_ptr = (
            b_ptr
            + pid_c * stride_b_c
            + offs_k[:, None] * stride_b_k   # K dimension first (row)
            + offs_n[None, :] * stride_b_n   # N dimension (col)
        )

        # Load with masking; out‑of‑bounds values are zero.
        a = tl.load(
            a_tile_ptr,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        b = tl.load(
            b_tile_ptr,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # FP16×FP16 → FP32 dot product (tensor‑core)
        acc += tl.dot(a, b)

    # Write back the tile
    out_tile_ptr = (
        out_ptr
        + pid_c * stride_out_c
        + offs_m[:, None] * stride_out_m
        + offs_n[None, :] * stride_out_n
    )
    tl.store(
        out_tile_ptr,
        acc,
        mask=mask_m[:, None] & mask_n[None, :],
    )


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, C] (float32)
        - mask         : torch.Tensor [B, N, N] (bool/float) – may be None
        - weights      : dict of torch.Tensors (model parameters)
        - config       : dict with keys
            * "dim"        – input channel dimension C
            * "hidden_dim" – hidden dimension H
            * "nomask"     – bool, if True the mask is ignored

    Returns
    -------
    torch.Tensor [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # 1) Unpack inputs
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 2) Layer‑norm the input (float32)
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, C]

    # ------------------------------------------------------------------
    # 3) Cast to FP16 for all linear / elementwise ops
    x_h = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 4) Linear projections (no bias)
    left_proj_w  = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left  = F.linear(x_h, left_proj_w)   # [B, N, N, H]
    right = F.linear(x_h, right_proj_w)  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 5) Gates (sigmoid)
    left_gate_w  = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w   = weights["out_gate.weight"].to(torch.float16)

    left_gate  = torch.sigmoid(F.linear(x_h, left_gate_w))   # [B, N, N, H]
    right_gate = torch.sigmoid(F.linear(x_h, right_gate_w))  # [B, N, N, H]
    out_gate   = torch.sigmoid(F.linear(x_h, out_gate_w))   # [B, N, N, H]

    # ------------------------------------------------------------------
    # 6) Optional mask (applied before gating, as in reference)
    if not nomask and mask is not None:
        mask_f = mask.to(torch.float16).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # 7) Apply elementwise gates
    left = left * left_gate
    right = right * right_gate

    # Free intermediate tensors that are no longer needed
    del left_gate, right_gate, x_h, x

    # ------------------------------------------------------------------
    # 8) Rearrange for Triton GEMM
    B, N, _, _ = left.shape                      # B, N, N, H
    # Move hidden dimension before the two sequence axes for a contiguous layout
    left = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    C_total = B * hidden_dim                      # total number of independent GEMMs
    left_flat  = left.reshape(C_total, N, N)      # (C, M, K)
    right_flat = right.reshape(C_total, N, N)     # (C, K, N) – we will index it transposed

    # Output buffer (FP32 accumulation)
    out_flat = torch.empty_like(left_flat, dtype=torch.float32)

    # ------------------------------------------------------------------
    # 9) Triton kernel launch parameters
    # Tile sizes – chosen for H100 tensor‑core efficiency
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    stride_out_c = out_flat.stride(0)
    stride_out_m = out_flat.stride(1)
    stride_out_n = out_flat.stride(2)

    stride_a_c = left_flat.stride(0)
    stride_a_m = left_flat.stride(1)
    stride_a_k = left_flat.stride(2)

    stride_b_c = right_flat.stride(0)
    stride_b_k = right_flat.stride(2)   # last dim (K)
    stride_b_n = right_flat.stride(1)   # middle dim (N)

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # blocks over M (i)
        (N + BLOCK_N - 1) // BLOCK_N,   # blocks over N (j)
        C_total,                        # one program per (batch * hidden_dim)
    )

    # Launch the batched GEMM kernel
    tri_mul_kernel[grid](
        out_flat,
        left_flat,
        right_flat,
        N, N, N,
        stride_out_c,
        stride_out_m,
        stride_out_n,
        stride_a_c,
        stride_a_m,
        stride_a_k,
        stride_b_c,
        stride_b_k,
        stride_b_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=2,
    )

    # ------------------------------------------------------------------
    # 10) Reshape back to (B, N, N, H)
    out = out_flat.reshape(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 11) Post‑processing: final LayerNorm, output gate, output projection
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    # Multiply by the output gate (out_gate is still FP16; broadcast works)
    out = out * out_gate

    # Final linear projection back to original dimension (use FP16 for speed)
    to_out_w = weights["to_out.weight"].to(torch.float16)
    out = F.linear(out.to(torch.float16), to_out_w).to(torch.float32)

    return out
