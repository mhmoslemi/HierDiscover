    left_ptr, right_ptr, out_ptr,
    B, N, H,

    stride_right_b, stride_right_i, stride_right_k, stride_right_j, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr, eps: tl.constexpr
):
    pid_batch = tl.program_id(0)  # batch_h = b*H + h
    pid_i = tl.program_id(1)      # row tile
    pid_j = tl.program_id(2)      # col tile

    # Determine batch and hidden index
    b = pid_batch // H
    h = pid_batch % H

    # Offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Pointers base
    left_base = left_ptr + b * stride_left_b + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base = out_ptr + b * stride_out_b + h * stride_out_h

    # K loop
    for k in range(0, tl.cdiv(N, BLOCK_K)):

        mask_k = offs_k < N

            other=0.0,
        )

    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr

    pid_batch = tl.program_id(0) # batch*hidden
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Determine batch and hidden index
    b = pid_batch // H
    h = pid_batch % H

    # Offsets for i, j
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Compute base pointer offsets
    left_base = left_ptr + b * stride_left_b + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base = out_ptr + b * stride_out_b + h * stride_out_h

import torch.nn.functional as F
import triton
import triton.language as tl
import math

# Triton kernel for batched tri-multiplicative reduction (outgoing)
@triton.jit
def triu_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h] (FP16)."""
    pid_batch = tl.program_id(0)  # combined batch*hidden index
    pid_m = tl.program_id(1)      # row tile
    pid_n = tl.program_id(2)      # col tile

    # resolve batch and hidden id
    b = pid_batch // H
    h = pid_batch % H

    # tile offsets
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # base pointers for this (b,h)
    left_base = left_ptr + b * stride_left_b + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base = out_ptr + b * stride_out_b + h * stride_out_h

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    num_k = tl.cdiv(N, BLOCK_K)
    for k in range(num_k):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile  (M, K)   i -> row, k -> column
        left = tl.load(
            left_base
            + offs_i[:, None] * stride_left_i
            + offs_k[None, :] * stride_left_k,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )  # dtype matches input (fp16)

        # right tile transposed: (K, N). Treat rows = k, cols = j
        right = tl.load(
            right_base
            + offs_k[:, None] * stride_right_k
            + offs_j[None, :] * stride_right_j,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )
        # dot product accumulation
        acc += tl.dot(left, right)  # returns fp32

    # write result (fp16)
    out = acc.to(tl.float16)
    tl.store(
        out_base
        + offs_i[:, None] * stride_out_i
        + offs_j[None, :] * stride_out_j,
        out,
        mask=mask_i[:, None] & mask_j[None, :],
    )


def custom_kernel(data):
    """
    Fast forward implementation of the TriMul “outgoing” operator.

    The algorithm:
    1️⃣ Layer‑norm the input (FP32, torch.nn.functional.layer_norm).
    2️⃣ Fuse the five linear maps (left/right projections + three gates) into a
       single half‑precision GEMM.
    3️⃣ Apply the optional binary mask and the three sigmoid gates.
    4️⃣ Compute the core “TriMul” reduction
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h] using a custom
       Triton kernel that works on the original (B,N,N,H) layout, thus
       avoiding the costly permute+contiguous copies.
    5️⃣ Output LayerNorm (FP32, torch.nn.functional.layer_norm).
    6️⃣ Multiply by the output gate and project back to the model dimension.

    Returns a float‑32 tensor of shape [B, N, N, dim].
    """
    # -----------------------------------------------------------------
    # Unpack arguments
    # -----------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype  # we expect fp32 here
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32)
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(device, dtype=torch.float32)
    norm_bias = weights["norm.bias"].to(device, dtype=torch.float32)
    # torch.nn.functional.layer_norm works on the last dimension
    x_norm = F.layer_norm(
        input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=1e-5
    )  # (B, N, N, dim) FP32

    # --------------------------------------------------------------
    # 2️⃣ Combined linear projection (half‑precision)
    # --------------------------------------------------------------
    x_norm_h = x_norm.to(torch.float16)
    M = bs * seq_len * seq_len
    x_flat = x_norm_h.view(M, dim)  # (M, dim)

    combined_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(device, dtype=torch.float16)  # (5*H, dim)

    proj_out = F.linear(x_flat, combined_weight)  # (M, 5*H) FP16

    # --------------------------------------------------------------
    # 3️⃣ Split the five streams
    # --------------------------------------------------------------
    chunks = torch.split(proj_out, hidden_dim, dim=1)
    left_raw, right_raw, left_gate_raw, right_gate_raw, out_gate_raw = chunks

    # reshape to (B, N, N, H)
    left_raw = left_raw.view(bs, seq_len, seq_len, hidden_dim)
    right_raw = right_raw.view(bs, seq_len, seq_len, hidden_dim)
    left_gate_raw = left_gate_raw.view(bs, seq_len, seq_len, hidden_dim)
    right_gate_raw = right_gate_raw.view(bs, seq_len, seq_len, hidden_dim)
    out_gate_raw = out_gate_raw.view(bs, seq_len, seq_len, hidden_dim)

    # --------------------------------------------------------------
    # 4️⃣ Optional mask (broadcast over H)
    # --------------------------------------------------------------
    if not nomask and mask is not None:
        mask_h = mask.to(device, dtype=torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left_raw = left_raw * mask_h
        right_raw = right_raw * mask_h

    # --------------------------------------------------------------
    # 5️⃣ Gates (sigmoid) and gated projections
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_raw)   # (B,N,N,H)
    right_gate = torch.sigmoid(right_gate_raw) # (B,N,N,H)
    out_gate = torch.sigmoid(out_gate_raw)    # (B,N,N,H)

    left = left_raw * left_gate
    right = right_raw * right_gate

    # --------------------------------------------------------------
    # 6️⃣ Tri‑multiplicative reduction – Triton kernel
    # --------------------------------------------------------------
    out = torch.empty_like(left)  # (B,N,N,H) FP16

    # strides for left / right / out
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left.stride()
    stride_right_b, stride_right_j, stride_right_k, stride_right_h = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out.stride()

    # kernel launch configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    grid = (
        bs * hidden_dim,
        math.ceil(seq_len / BLOCK_M),
        math.ceil(seq_len / BLOCK_N),
    )
    triu_mul_kernel[grid](
        left,
        right,
        out,
        bs,
        seq_len,
        hidden_dim,
        stride_left_b,
        stride_left_i,
        stride_left_k,
        stride_left_h,
        stride_right_b,
        stride_right_j,
        stride_right_k,
        stride_right_h,
        stride_out_b,
        stride_out_i,
        stride_out_j,
        stride_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )
    # --------------------------------------------------------------
    # 7️⃣ Output LayerNorm (FP32)
    # --------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"].to(device, dtype=torch.float32)
    to_out_norm_bias = weights["to_out_norm.bias"].to(device, dtype=torch.float32)

    out_fp32 = out.to(torch.float32)
    out_norm = F.layer_norm(
        out_fp32,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )  # (B,N,N,H) FP32

    # --------------------------------------------------------------
    # 8️⃣ Apply output gate and final projection
    # --------------------------------------------------------------
    gated = out_norm * out_gate.to(torch.float32)  # (B,N,N,H) FP32
    to_out_weight = weights["to_out.weight"].to(device, dtype=torch.float16)  # (dim, H)

    # final linear (half precision)
    final_half = F.linear(gated.half(), to_out_weight)  # (B,N,N,dim) FP16
    return final_half.float()

"""
TriMul “outgoing” forward (AlphaFold‑3 style)

Algorithm
---------
1️⃣ Input LayerNorm (FP32, torch.nn.functional.layer_norm).
2️⃣ Fuse the five linear maps (left/right projections + three gates) into one
   half‑precision GEMM: (B·N·N, dim) @ (dim, 5·hidden) → (B·N·N, 5·hidden).
3️⃣ Reshape & split the result → left, right, three gate tensors of shape
   (B, N, N, hidden).  Apply the optional binary mask (broadcast over hidden)
   and the three sigmoid gates.
4️⃣ Core “TriMul” reduction:
   out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h].
   Implemented with a custom Triton kernel that works directly on the
   original (B,N,N,hidden) layout, thus eliminating the costly
   permute‑contiguous copies used in the naïve implementation.
5️⃣ Output LayerNorm (FP32, torch.nn.functional.layer_norm).
6️⃣ Multiply by the output gate and project back to the model dimension
   with a final half‑precision linear layer.  The function returns a
   float‑32 tensor of shape [B, N, N, dim].
"""

import math
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched mat‑mul with a reduction over the “k” axis.
# Computes out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h].
# ----------------------------------------------------------------------
@triton.jit
def triu_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)          # combined batch * hidden index
    pid_m = tl.program_id(1)              # row tile index
    pid_n = tl.program_id(2)              # column tile index

    # Resolve batch and hidden indices from the flattened id
    b = pid_batch // H
    h = pid_batch % H

    # Offsets for the current tile (i‑ and j‑dimension)
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Base pointers for the (b,h) slice
    left_base  = left_ptr  + b * stride_left_b  + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base   = out_ptr   + b * stride_out_b   + h * stride_out_h

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension k
    num_k = tl.cdiv(N, BLOCK_K)
    for k in range(num_k):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile – shape (BLOCK_M, BLOCK_K)
        left = tl.load(
            left_base
            + offs_i[:, None] * stride_left_i
            + offs_k[None, :] * stride_left_k,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )
        # right tile – transposed view (BLOCK_K, BLOCK_N)
        right = tl.load(
            right_base
            + offs_k[:, None] * stride_right_k
            + offs_j[None, :] * stride_right_j,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )
        acc += tl.dot(left, right)           # fp32 accumulation

    # Cast back to fp16 and store
    out = acc.to(tl.float16)
    tl.store(
        out_base
        + offs_i[:, None] * stride_out_i
        + offs_j[None, :] * stride_out_j,
        out,
        mask=mask_i[:, None] & mask_j[:, None],
    )

    torch.Tensor

    """

    # Unpack arguments
    # -----------------------------------------------------------------

    device = input_tensor.device
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # -----------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32)

    x_norm = F.layer_norm(

    left_raw, right_raw, left_gate_raw, right_gate_raw, out_gate_raw = torch.split(

    if not nomask and mask is not None:

        left_raw  = left_raw * mask_h
        right_raw = right_raw * mask_h

    # -----------------------------------------------------------------
    # 5️⃣ Gating (sigmoid) and gated projections
    # -----------------------------------------------------------------
    left_gate  = torch.sigmoid(left_gate_raw)      # (B,N,N,H) FP16

    out_gate   = torch.sigmoid(out_gate_raw)       # (B,N,N,H) FP16

    left  = left_raw * left_gate
    right = right_raw * right_gate

    # -----------------------------------------------------------------
    # 6️⃣ Tri‑multiplicative reduction – Triton kernel
    # -----------------------------------------------------------------
    out = torch.empty_like(left)   # (B,N,N,H) FP16

    # Strides for the three tensors (they are all contiguous in the
    # (B,N,N,H) layout)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left.stride()
    stride_right_b, stride_right_j, stride_right_k, stride_right_h = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out.stride()

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        bs * hidden_dim,                                 # combined batch·hidden
        math.ceil(seq_len / BLOCK_M),                    # rows

        right,
        out,

        seq_len,

        stride_right_b,
        stride_right_j,
        stride_right_k,
        stride_right_h,
        stride_out_b,
        stride_out_i,
        stride_out_j,
        stride_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,      # H100 benefits from more warps per block
    )

    # 7️⃣ Output LayerNorm (FP32)
    # -----------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"].to(device, dtype=torch.float32)
    to_out_norm_bias   = weights["to_out_norm.bias"].to(device, dtype=torch.float32)

    # -----------------------------------------------------------------
    gated = out_norm * out_gate.to(torch.float32)      # (B,N,N,H) FP32
    to_out_weight = weights["to_out.weight"].to(device, dtype=torch.float16)  # (dim, H)

    final_half = F.linear(gated.half(), to_out_weight)  # (B,N,N,dim) FP16
    return final_half.float()
