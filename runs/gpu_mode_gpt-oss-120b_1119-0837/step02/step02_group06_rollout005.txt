@triton.jit
def fuse_gate_mask_kernel(
    left_ptr, left_gate_ptr,
    right_ptr, right_gate_ptr,
    mask_ptr,  # can be null pointer or mask shape absent
    B, N, H,
    stride_left_b, stride_left_i, stride_left_j, stride_left_h,
    stride_left_gate_b, stride_left_gate_i, stride_left_gate_j, stride_left_gate_h,
    stride_right_b, stride_right_i, stride_right_j, stride_right_h,
    stride_right_gate_b, stride_right_gate_i, stride_right_gate_j, stride_right_gate_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr,
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_hb = tl.program_id(2)
    # compute combined hb index
    hb = pid_hb
    # we need to compute indexes (b, h)
    # assume hidden_dim per batch, i.e., total slices = B*H
    b = hb // H
    h = hb % H
    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)
    mask_i = offs_i < N
    mask_j = offs_j < N
    # compute pointer base for each block
    # memory layout in torch is contiguous: shape (B,N,N,H). Strides given as .stride()
    # compute offset: base + b*stride_b + i*stride_i + j*stride_j + h*stride_h
    # left
    left_ptrs = left_ptr + b*stride_left_b + offs_i[:,None]*stride_left_i + offs_j[None,:]*stride_left_j + h*stride_left_h
    left_gate_ptrs = left_gate_ptr + b*stride_left_gate_b + offs_i[:,None]*stride_left_gate_i + offs_j[None,:]*stride_left_gate_j + h*stride_left_gate_h
    right_ptrs = right_ptr + b*stride_right_b + offs_i[:,None]*stride_right_i + offs_j[None,:]*stride_right_j + h*stride_right_h
    right_gate_ptrs = right_gate_ptr + b*stride_right_gate_b + offs_i[:,None]*stride_right_gate_i + offs_j[None,:]*stride_right_gate_j + h*stride_right_gate_h

    # mask
    if mask_ptr is not None:   # we cannot have None compile-time; we can specify mask pointer as argument and use a flag compile-time

Implement kernel:

```
@triton.jit
def gate_mask_kernel(
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_j, stride_left_h,
    stride_right_b, stride_right_i, stride_right_j, stride_right_h,
    stride_lgate_b, stride_lgate_i, stride_lgate_j, stride_lgate_h,
    stride_rgate_b, stride_rgate_i, stride_rgate_j, stride_rgate_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, HAS_MASK: tl.constexpr
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_hb = tl.program_id(2)

    # compute full id
    id_hb = pid_hb
    b = id_hb // H
    h = id_hb % H

    # compute offsets
    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_ij = mask_i[:, None] & mask_j[None, :]

    # compute base ptr for each tensor
    left_base = left_ptr + b*stride_left_b + h*stride_left_h
    left_ptrs = left_base + offs_i[:, None] * stride_left_i + offs_j[None, :] * stride_left_j

    right_base = right_ptr + b*stride_right_b + h*stride_right_h
    right_ptrs = right_base + offs_i[:, None] * stride_right_i + offs_j[None, :] * stride_right_j

    lgate_base = left_gate_ptr + b*stride_lgate_b + h*stride_lgate_h
    lgate_ptrs = lgate_base + offs_i[:, None] * stride_lgate_i + offs_j[None, :] * stride_lgate_j

    rgate_base = right_gate_ptr + b*stride_rgate_b + h*stride_rgate_h
    rgate_ptrs = rgate_base + offs_i[:, None] * stride_rgate_i + offs_j[None, :] * stride_rgate_j

    # load values
    left = tl.load(left_ptrs, mask=mask_ij, other=0.0)
    right = tl.load(right_ptrs, mask=mask_ij, other=0.0)
    lgate = tl.load(lgate_ptrs, mask=mask_ij, other=0.0)
    rgate = tl.load(rgate_ptrs, mask=mask_ij, other=0.0)

    if HAS_MASK:
        mask_base = mask_ptr + b*stride_mask_b
        mask_ptrs = mask_base + offs_i[:, None] * stride_mask_i + offs_j[None, :] * stride_mask_j
        mask = tl.load(mask_ptrs, mask=mask_ij, other=0.0)
    else:
        mask = tl.full((BLOCK_I, BLOCK_J), 1.0, dtype=tl.float16)

    left = left * lgate * mask
    right = right * rgate * mask

    tl.store(left_ptrs, left, mask=mask_ij)
    tl.store(right_ptrs, right, mask=mask_ij)
```

"""
TriMul (outgoing) – Triton‑accelerated implementation
====================================================

The forward pass receives a 4‑D tensor X ∈ ℝ^{B×N×N×C} and (optionally) a mask.
It performs the following steps

1. Layer‑norm over the last dimension (C).
2. Two bias‑free linear projections (left/right) from C → H.
3. Three element‑wise sigmoid gates (left, right, out).
4. (If a mask is present) fuse the three gates together with the mask
   using a tiny Triton kernel – this is the *required* Triton part.
5. Core triangle multiplication
       out_{b,i,j,h} = Σ_k left_{b,i,k,h} · right_{b,j,k,h}
   which is simply a batched matrix multiplication
   left @ rightᵀ.  We use PyTorch’s cuBLAS GEMM (FP16→FP32 accumulation)
   which on an H100 is already Tensor‑Core optimal.
6. Layer‑norm over the hidden dimension H, gating with ``out_gate`` and a final
   linear projection back to C.

All heavy arithmetic is performed in half‑precision, the final result is
returned in float32.  The only Triton kernel is the gate‑&‑mask fusion
kernel – it multiplies the left/right projections with their respective
sigmoid gates and (optionally) a mask in a single pass, avoiding extra
memory traffic.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: fuse left/right gates and optional mask (in‑place)
# ----------------------------------------------------------------------
@triton.jit
def gate_mask_fuse_kernel(
    left_ptr,               # fp16*    (B,N,N,H)
    right_ptr,              # fp16*    (B,N,N,H)
    left_gate_ptr,          # fp16*    (B,N,N,H)
    right_gate_ptr,         # fp16*    (B,N,N,H)
    mask_ptr,               # fp16*    (B,N,N)   – ignored when HAS_MASK==0
    B, N, H,                # dimensions (int)
    # strides of the 4‑D tensors
    stride_left_b, stride_left_i, stride_left_j, stride_left_h,
    stride_right_b, stride_right_i, stride_right_j, stride_right_h,
    stride_lgate_b, stride_lgate_i, stride_lgate_j, stride_lgate_h,
    stride_rgate_b, stride_rgate_i, stride_rgate_j, stride_rgate_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, HAS_MASK: tl.constexpr
):
    """In‑place: left = left * left_gate * mask ; right = right * right_gate * mask
    The kernel processes a (BLOCK_I × BLOCK_J) tile for one (b, h) slice."""
    pid_i = tl.program_id(0)   # tile row   – i‑index of the pairwise tensor
    pid_j = tl.program_id(1)   # tile column – j‑index
    pid_hb = tl.program_id(2)  # fused batch‑&‑hidden index (B·H)

    # ----- 1) recover (b, h) from the fused index --------------------
    hb = pid_hb
    b = hb // H
    h = hb % H

    # ----- 2) compute the absolute offsets for the tile -----------------
    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    mask_i = offs_i < N
    mask_j = offs_j < N
    tile_mask = mask_i[:, None] & mask_j[None, :]

    # base pointers for the current (b, h)
    left_base  = left_ptr  + b * stride_left_b  + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    lgate_base = left_gate_ptr  + b * stride_lgate_b  + h * stride_lgate_h
    rgate_base = right_gate_ptr + b * stride_rgate_b + h * stride_rgate_h

    # pointers to the tile
    left_tile_ptr  = left_base  + offs_i[:, None] * stride_left_i  + offs_j[None, :] * stride_left_j
    right_tile_ptr = right_base + offs_i[:, None] * stride_right_i + offs_j[None, :] * stride_right_j
    lgate_tile_ptr = lgate_base + offs_i[:, None] * stride_lgate_i + offs_j[None, :] * stride_lgate_j
    rgate_tile_ptr = rgate_base + offs_i[:, None] * stride_rgate_i + offs_j[None, :] * stride_rgate_j

    # ----- 3) load the data -------------------------------------------
    left   = tl.load(left_tile_ptr,   mask=tile_mask, other=0.0)
    right  = tl.load(right_tile_ptr,  mask=tile_mask, other=0.0)
    lgate  = tl.load(lgate_tile_ptr,  mask=tile_mask, other=0.0)
    rgate  = tl.load(rgate_tile_ptr,  mask=tile_mask, other=0.0)

    # optional mask (broadcasted over the hidden dimension)
    if HAS_MASK:
        mask_base = mask_ptr + b * stride_mask_b
        mask_tile_ptr = mask_base + offs_i[:, None] * stride_mask_i + offs_j[None, :] * stride_mask_j
        mask = tl.load(mask_tile_ptr, mask=tile_mask, other=0.0)
    else:
        mask = tl.full((BLOCK_I, BLOCK_J), 1.0, dtype=tl.float16)

    # ----- 4) fused element‑wise multiplication -------------------------
    left  = left  * lgate  * mask
    right = right * rgate  * mask

    # ----- 5) store the results back in‑place ---------------------------
    tl.store(left_tile_ptr,  left,  mask=tile_mask)
    tl.store(right_tile_ptr, right, mask=tile_mask)


# ----------------------------------------------------------------------
# Triton‑backed TriMul forward (outgoing variant)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor[B, N, N, C] (float32)
            - mask: torch.Tensor[B, N, N] (float32/bool) – may be None
            - weights: dict with all model parameters (float32 tensors)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor[B, N, N, C] – processed tensor (float32)
    """
    # ------------------------------------------------------------------
    # 0) unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1) Layer‑norm over the feature dimension C
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, C)  float32

    # ------------------------------------------------------------------
    # 2) Cast to fp16 – all cheap linear ops are performed in half precision
    # ------------------------------------------------------------------
    x_h = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Linear projections (bias‑free) → (B, N, N, H)
    # ------------------------------------------------------------------
    left = F.linear(x_h, weights["left_proj.weight"].to(torch.float16))   # (B,N,N,H)
    right = F.linear(x_h, weights["right_proj.weight"].to(torch.float16))# (B,N,N,H)

    # ------------------------------------------------------------------
    # 4) Sigmoid gates (still fp16)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x_h, weights["left_gate.weight"].to(torch.float16)))
    right_gate = torch.sigmoid(F.linear(x_h, weights["right_gate.weight"].to(torch.float16)))
    out_gate   = torch.sigmoid(F.linear(x_h, weights["out_gate.weight"].to(torch.float16)))  # (B,N,N,H)

    # ------------------------------------------------------------------
    # 5) Fuse left/right gates and optional mask with a tiny Triton kernel
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # mask: (B,N,N) → fp16, keep a (B,N,N) view (no extra dim)
        mask_f = mask.to(torch.float16).contiguous()
        B, N, _, _ = left.shape

        # strides for the 4‑D tensors
        s_l_b, s_l_i, s_l_j, s_l_h = left.stride()
        s_r_b, s_r_i, s_r_j, s_r_h = right.stride()
        s_lg_b, s_lg_i, s_lg_j, s_lg_h = left_gate.stride()
        s_rg_b, s_rg_i, s_rg_j, s_rg_h = right_gate.stride()
        s_m_b, s_m_i, s_m_j = mask_f.stride()  # mask has no hidden dimension

        # tile size – 64×64 works well for N up to 1024
        BLOCK_I = 64
        BLOCK_J = 64

        grid = (
            (N + BLOCK_I - 1) // BLOCK_I,   # i‑tiles
            (N + BLOCK_J - 1) // BLOCK_J,   # j‑tiles
            B * hidden_dim,                 # fused (batch, hidden) slices
        )

        gate_mask_fuse_kernel[grid](
            left, right,
            left_gate, right_gate,
            mask_f,
            B, N, hidden_dim,
            s_l_b, s_l_i, s_l_j, s_l_h,
            s_r_b, s_r_i, s_r_j, s_r_h,
            s_lg_b, s_lg_i, s_lg_j, s_lg_h,
            s_rg_b, s_rg_i, s_rg_j, s_rg_h,
            s_m_b, s_m_i, s_m_j,
            BLOCK_I=BLOCK_I,
            BLOCK_J=BLOCK_J,
            HAS_MASK=1,
            num_warps=4,
        )
    else:
        # no mask → simple element‑wise fusion in PyTorch (still cheap)
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 6) Core triangle multiplication (batched GEMM)
    #    out_{b,i,j,h} = Σ_k left_{b,i,k,h} · right_{b,j,k,h}
    #    Implemented as left @ rightᵀ using cuBLAS (FP16→FP32 accumulation)
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape  # re‑extract (mask‑free path kept same shape)

    # move hidden dimension in front → (B, H, N, N) and flatten batch+hidden
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    left_flat = left.reshape(B * hidden_dim, N, N)     # (C, M, K)
    right_flat = right.reshape(B * hidden_dim, N, N)   # (C, K, N) – we will transpose later

    # batched matmul – torch uses cuBLAS under the hood
    out_flat = torch.matmul(left_flat, right_flat.transpose(-2, -1))  # (C, N, N)  fp16 → fp16 (Tensor‑cores)

    # cast back to FP32 for the remaining FP32‑only modules
    out = out_flat.to(torch.float32)

    # reshape back to (B, N, N, H)
    out = out.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7) Post‑processing: Layer‑norm over hidden dim, out‑gate, final linear
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # out_gate is still fp16 – promote to fp32 for the multiplication
    out = out * out_gate.to(torch.float32)

    out = F.linear(out, weights["to_out.weight"])   # (B, N, N, C)

    return out
