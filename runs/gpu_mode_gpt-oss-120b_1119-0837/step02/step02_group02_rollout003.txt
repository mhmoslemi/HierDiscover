"""
TriMul (outgoing) – high‑performance Triton implementation.

Algorithm
---------
1. Layer‑norm the input (float32 for stability).
2. Fuse the five linear projections (left_proj, left_gate, right_proj,
   right_gate, out_gate) into a *single* half‑precision GEMM:
       X_flat @ W_fused.T  →  [left_proj, left_gate,
                               right_proj, right_gate, out_gate]
3. Apply the sigmoid to the gate tensors and element‑wise mask (if any).
   Compute gated left/right tensors.
4. Core O(N³) contraction:
      out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   Implemented as a batched GEMM (left @ rightᵀ) with a custom Triton
   kernel that:
   * works in fp16 with fp32 accumulation,
   * uses 128×128 tiles and a 64‑wide K‑tile (no bounds checks – all
     test sizes are multiples of 128/64),
   * processes a flattened batch dimension (B·hidden_dim) in one launch.
5. Layer‑norm on the hidden dimension, out‑gate multiplication and final
   linear projection back to the original channel dimension.
6. Cast the result back to float32.

The fused linear projection reduces the number of cuBLAS calls from five
to one, and the Triton kernel avoids runtime bounds checks, giving a
large speed‑up over the reference implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# -----------------------------------------------------------------------------#
# Triton kernel: batched GEMM C = A @ Bᵀ   (A: M×K, B: N×K → Bᵀ: K×N)
# -----------------------------------------------------------------------------#
@triton.jit
def _trmul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,                     # fp16 pointers
    M, N, K,                                 # matrix dimensions (M=N=seq_len, K=seq_len)
    stride_am, stride_ak,                    # A row/col stride inside a batch
    stride_bk, stride_bn,                    # B (already transposed) row/col stride
    stride_cm, stride_cn,                    # C row/col stride
    stride_a_batch, stride_b_batch, stride_c_batch,  # strides between batch elements
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(axis=0)          # tile index over (M,N)
    batch_id = tl.program_id(axis=1)     # flattened (batch * hidden) dimension

    # --------------------------------------------------------------- #
    #   Compute tile coordinates
    # --------------------------------------------------------------- #
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    block_m = pid // num_blocks_n
    block_n = pid % num_blocks_n

    offs_m = block_m * BLOCK_M + tl.arange(0, BLOCK_M)      # [BLOCK_M]
    offs_n = block_n * BLOCK_N + tl.arange(0, BLOCK_N)      # [BLOCK_N]

    # --------------------------------------------------------------- #
    #   Pointers for the batch we are processing
    # --------------------------------------------------------------- #
    a_batch = a_ptr + batch_id * stride_a_batch
    b_batch = b_ptr + batch_id * stride_b_batch
    c_batch = c_ptr + batch_id * stride_c_batch

    # --------------------------------------------------------------- #
    #   Accumulator (fp32)
    # --------------------------------------------------------------- #
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------- #
    #   Main K‑loop
    # --------------------------------------------------------------- #
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_off = k * BLOCK_K + tl.arange(0, BLOCK_K)         # [BLOCK_K]

        # Load A tile (BLOCK_M × BLOCK_K)
        a_ptrs = a_batch + (offs_m[:, None] * stride_am) + (k_off[None, :] * stride_ak)
        a = tl.load(a_ptrs)                                 # fp16

        # Load B tile (BLOCK_K × BLOCK_N) – B is already transposed (K×N)
        b_ptrs = b_batch + (k_off[:, None] * stride_bk) + (offs_n[None, :] * stride_bn)
        b = tl.load(b_ptrs)                                 # fp16

        # Rank‑K update
        acc += tl.dot(a, b)                                 # fp16×fp16 → fp32

    # --------------------------------------------------------------- #
    #   Write back the result (fp16)
    # --------------------------------------------------------------- #
    c_ptrs = c_batch + (offs_m[:, None] * stride_cm) + (offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, acc.to(tl.float16))

        dim=0,
    )   # (5*hidden_dim, dim)

    # Perform the fused projection: reshape to (B*N*N, dim) → (B*N*N, 5*hidden_dim)
    x_flat = x_fp16.view(-1, dim)                     # (B*N*N, dim)
    fused_out = F.linear(x_flat, fused_weight)       # (B*N*N, 5*hidden_dim), fp16
    fused_out = fused_out.view(B, N, N, 5 * hidden_dim)

    # --------------------------------------------------------------------- #
    # 3) Split fused tensor and apply sigmoids to gates (keep fp32 for sigmoid)
    # --------------------------------------------------------------------- #
    split = torch.split(fused_out, hidden_dim, dim=-1)
    left_proj   = split[0]                              # (B,N,N,hidden_dim) fp16
    left_gate   = torch.sigmoid(split[1].float()).half()
    right_proj  = split[2]
    right_gate  = torch.sigmoid(split[3].float()).half()
    out_gate    = torch.sigmoid(split[4].float()).half()   # (B,N,N,hidden_dim)

    # --------------------------------------------------------------------- #
    # 4) Apply mask (if present) and gating
    # --------------------------------------------------------------------- #
    if not nomask and mask is not None:
        mask_f = mask.to(x_fp16.dtype).unsqueeze(-1)    # (B,N,N,1)
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f

    left = left_proj * left_gate      # fp16
    right = right_proj * right_gate   # fp16

    # --------------------------------------------------------------------- #
    # 5) Core contraction: batched GEMM left @ rightᵀ   (per hidden channel)
    # --------------------------------------------------------------------- #
    # Rearrange tensors to (B, hidden_dim, N, N) and make them contiguous
    left_perm = left.permute(0, 3, 1, 2).contiguous()      # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()    # (B, H, N, N)
    # Transpose the second N‑axis of right so that we can multiply A @ Bᵀ
    right_T = right_perm.permute(0, 1, 3, 2).contiguous()  # (B, H, N, N)

    total_batches = B * hidden_dim                         # flattened batch dimension
    M = N
    K = N
    # View as (total_batches, M, K) and (total_batches, K, N)
    a = left_perm.view(total_batches, M, K)                # fp16
    b = right_T.view(total_batches, K, N)                  # fp16
    c = torch.empty((total_batches, M, N), dtype=torch.float16, device=device)

    # --------------------------------------------------------------------- #
    # 6) Launch the Triton GEMM kernel
    # --------------------------------------------------------------------- #
    BLOCK_M, BLOCK_N, BLOCK_K = 128, 128, 64
    # All benchmarked N are multiples of 128/64, otherwise fall back to torch.einsum
    if (N % BLOCK_M == 0) and (N % BLOCK_N == 0) and (N % BLOCK_K == 0):
        grid = (
            ( (M + BLOCK_M - 1) // BLOCK_M ) * ( (N + BLOCK_N - 1) // BLOCK_N ),
            total_batches,
        )
        stride_a_batch = M * K
        stride_am = K
        stride_ak = 1
        stride_b_batch = K * N
        stride_bk = N
        stride_bn = 1
        stride_c_batch = M * N
        stride_cm = N
        stride_cn = 1

        _trmul_gemm_kernel[grid](
            a, b, c,
            M, N, K,
            stride_am, stride_ak,
            stride_bk, stride_bn,
            stride_cm, stride_cn,
            stride_a_batch, stride_b_batch, stride_c_batch,
            BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
            num_warps=8,
            num_stages=2,
        )
    else:
        # Fallback path for non‑divisible sizes (rare for the provided tests)
        out = torch.einsum('... i k d, ... j k d -> ... i j d', left, right)
        out = F.layer_norm(
            out,
            normalized_shape=(hidden_dim,),
            weight=weights["to_out_norm.weight"].half(),
            bias=weights["to_out_norm.bias"].half(),
            eps=eps,
        )
        out = out * out_gate
        out = F.linear(out, weights["to_out.weight"].half())
        return out.float()

    # --------------------------------------------------------------------- #
    # 7) Reshape the GEMM result back to [B, N, N, hidden_dim] (fp16)
    # --------------------------------------------------------------------- #
    out = c.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()   # (B,N,N,hidden_dim)

    # --------------------------------------------------------------------- #
    # 8) Hidden‑dim layer‑norm, out‑gate, and final linear projection
    # --------------------------------------------------------------------- #
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].half(),
        bias=weights["to_out_norm.bias"].half(),
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].half())   # back to original dim, fp16

    # --------------------------------------------------------------------- #
    # 9) Cast back to float32 for downstream use
    # --------------------------------------------------------------------- #
    return out.float()
