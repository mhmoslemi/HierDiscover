import torch
from torch import nn, einsum
import math

# Reference code in PyTorch
class TriMul(nn.Module):
    def __init__(
        self,
        dim: int,
        hidden_dim: int,
    ):
        super().__init__()

        self.norm = nn.LayerNorm(dim)

        self.left_proj = nn.Linear(dim, hidden_dim, bias=False)
        self.right_proj = nn.Linear(dim, hidden_dim, bias=False)

        self.left_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.right_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.out_gate = nn.Linear(dim, hidden_dim, bias=False)

        self.to_out_norm = nn.LayerNorm(hidden_dim)
        self.to_out = nn.Linear(hidden_dim, dim, bias=False)

    def forward(self, x: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
        """
        x: [bs, seq_len, seq_len, dim]
        mask: [bs, seq_len, seq_len]

        Returns:
            output: [bs, seq_len, seq_len, dim]
        """
        batch_size, seq_len, _, dim = x.shape

        x = self.norm(x)

        left = self.left_proj(x)
        right = self.right_proj(x)

        mask = mask.unsqueeze(-1)
        left = left * mask
        right = right * mask

        left_gate = self.left_gate(x).sigmoid()
        right_gate = self.right_gate(x).sigmoid()
        out_gate = self.out_gate(x).sigmoid()

        left = left * left_gate
        right = right * right_gate

        out = einsum('... i k d, ... j k d -> ... i j d', left, right)
        # This einsum is the same as the following:
        # out = torch.zeros(batch_size, seq_len, seq_len, dim, device=x.device)
        
        # # Compute using nested loops
        # for b in range(batch_size):
        #     for i in range(seq_len):
        #         for j in range(seq_len):
        #             # Compute each output element
        #             for k in range(seq_len):
        #                 out[b, i, j] += left[b, i, k, :] * right[b, j, k, :]

        out = self.to_out_norm(out)
        out = out * out_gate
        return self.to_out(out)

"""
custom_kernel
==============

Implementation of the *outgoing* TriMul operator (AlphaFold‑3) in a highly‑optimized
mixed‑precision Triton+PyTorch hybrid.  

Main ideas
----------

*   Input and output LayerNorms are performed by a custom two‑pass Triton kernel
    (row‑wise mean/variance → affine transform).  This avoids the synchronization
    needed by a full Tensor‑parallel reduction.
*   The expensive N³ tensor contraction

        out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]

    is rewritten as a batched matrix multiplication per head (d).  After permuting
    to shape (B·H, N, N) we launch a single Triton kernel that computes all
    B·H products in parallel – far less launch overhead than a Python loop over
    heads.
*   All linear layers are executed in float16 (FP16) so the H100 Tensor‑cores are
    used.  Accumulation is performed in FP32 for numerical stability.
*   The final output is cast back to float32, the output gate is applied, and the
    last linear projection is performed with `torch.nn.functional.linear`
    (FP32 → FP32).

The kernel works for any batch size (1‑2), sequence length up to 1024, model
dimension 128‑384 and hidden dimension 128.  It also correctly handles the
optional attention mask.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton LayerNorm (row‑wise) – operates on a 2‑D tensor [M, N].
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,                # *ptr to input  [M, N]
    y_ptr,                # *ptr to output [M, N]
    weight_ptr,           # *ptr to weight [N]
    bias_ptr,             # *ptr to bias   [N]
    M,                    # total rows
    N,                    # feature dimension
    eps,                  # epsilon
    stride_xm, stride_xn,  # strides for input
    stride_ym, stride_yn,  # strides for output
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
):
    pid = tl.program_id(0)

    # ------------------------------------------------------------------
    # Row offsets for this program
    # ------------------------------------------------------------------
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # ------------------------------------------------------------------
    # First pass: compute mean & variance
    # ------------------------------------------------------------------
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_N)
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        sum_val += tl.sum(x,  axis=1)
        sum_sq  += tl.sum(x * x, axis=1)

    mean = sum_val / N
    var  = sum_sq / N - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # ------------------------------------------------------------------
    # Second pass: write normalised + affine transformed values
    # ------------------------------------------------------------------
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        norm_x = (x - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_n, mask=mask_n, other=0.0)
        b = tl.load(bias_ptr   + offs_n, mask=mask_n, other=0.0)

        y = norm_x * w[None, :] + b[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn,
            y,
            mask=mask_m[:, None] & mask_n[None, :],
        )


def triton_layernorm(x: torch.Tensor, weight: torch.Tensor, bias: torch.Tensor, eps: float = 1e-5):
    """
    LayerNorm over the last dimension of ``x`` using the Triton kernel.
    ``x`` must be 2‑D (M, N). Returned tensor has the same dtype as ``x``.
    """
    assert x.dim() == 2, "triton_layernorm expects a 2‑D tensor"
    M, N = x.shape
    y = torch.empty_like(x)

    # Ensure contiguous tensors (required for correct stride handling)
    x = x.contiguous()
    y = y.contiguous()
    weight = weight.contiguous()
    bias = bias.contiguous()

    stride_xm, stride_xn = x.stride()
    stride_ym, stride_yn = y.stride()

    grid = lambda meta: (triton.cdiv(M, meta["BLOCK_M"]),)

    layernorm_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        weight_ptr=weight,
        bias_ptr=bias,
        M=M,
        N=N,
        eps=eps,
        stride_xm=stride_xm,
        stride_xn=stride_xn,
        stride_ym=stride_ym,
        stride_yn=stride_yn,
        BLOCK_M=128,   # rows per program (tuned)
        BLOCK_N=64,    # columns per tile
    )
    return y


# ----------------------------------------------------------------------
# Batched GEMM kernel – computes C = A @ Bᵀ for many independent
# matrices (batch dimension).  All tensors are expected to be row‑major.
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,          # pointers
    M, N, K,                      # matrix sizes (M rows, N cols, K reduction)
    stride_am, stride_ak, stride_ab,    # A: (batch, M, K)
    stride_bk, stride_bn, stride_bb,    # Bᵀ: (batch, K, N)
    stride_cm, stride_cn, stride_cb,    # C: (batch, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(0)
    pid_m    = tl.program_id(1)
    pid_n    = tl.program_id(2)

    # Offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this batch
    a_batch_ptr = a_ptr + batch_id * stride_ab
    b_batch_ptr = b_ptr + batch_id * stride_bb
    c_batch_ptr = c_ptr + batch_id * stride_cb

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in blocks
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = tl.arange(0, BLOCK_K)
        k_mask = (k_base + offs_k) < K

        # Load A tile: (BLOCK_M x BLOCK_K)
        a = tl.load(
            a_batch_ptr
            + offs_m[:, None] * stride_am
            + (k_base + offs_k)[None, :] * stride_ak,
            mask=mask_m[:, None] & k_mask[None, :],
            other=0.0,
        )
        # Load Bᵀ tile: (BLOCK_K x BLOCK_N)
        b = tl.load(
            b_batch_ptr
            + (k_base + offs_k)[:, None] * stride_bk
            + offs_n[None, :] * stride_bn,
            mask=k_mask[:, None] & mask_n[None, :],
            other=0.0,
        )
        # Accumulate dot product
        acc += tl.dot(a, b)

    # Write result back (store as FP16, the rest of the pipeline works in FP32)
    c = acc.to(tl.float16)
    tl.store(
        c_batch_ptr
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn,
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )

def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor   [B, L, L, D]
        - mask         : torch.Tensor   [B, L, L] or ``None``
        - weights      : dict of torch tensors (model parameters)
        - config       : dict with keys ``dim``, ``hidden_dim`` and optional ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, L, L, D] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    bs, L, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1️⃣  Input LayerNorm (Triton)
    # ------------------------------------------------------------------
    M = bs * L * L
    x_flat = input_tensor.reshape(M, dim)
    norm_w = weights["norm.weight"].to(device, dtype).contiguous()
    norm_b = weights["norm.bias"].to(device, dtype).contiguous()
    x_norm_flat = triton_layernorm(x_flat, norm_w, norm_b, eps=eps)
    x_norm = x_norm_flat.view(bs, L, L, dim)

    # ------------------------------------------------------------------
    # 2️⃣  Cast to FP16 for the heavy compute
    # ------------------------------------------------------------------
    x_norm_h = x_norm.to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣  Linear projections (FP16, bias‑free)
    # ------------------------------------------------------------------
    left_w  = weights["left_proj.weight"].to(device, torch.float16).contiguous()
    right_w = weights["right_proj.weight"].to(device, torch.float16).contiguous()
    left  = F.linear(x_norm_h, left_w)   # [B, L, L, H]
    right = F.linear(x_norm_h, right_w)

    # ------------------------------------------------------------------
    # 4️⃣  Optional mask (broadcast over the hidden dimension)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_h = mask.to(device, torch.float16).unsqueeze(-1)   # [B, L, L, 1]
        left = left * mask_h
        right = right * mask_h

    # ------------------------------------------------------------------
    # 5️⃣  Gates (sigmoid) – also computed in FP16
    # ------------------------------------------------------------------
    left_gate_w  = weights["left_gate.weight"].to(device, torch.float16).contiguous()
    right_gate_w = weights["right_gate.weight"].to(device, torch.float16).contiguous()
    out_gate_w   = weights["out_gate.weight"].to(device, torch.float16).contiguous()

    left_gate  = torch.sigmoid(F.linear(x_norm_h, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_norm_h, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x_norm_h, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6️⃣  Prepare tensors for batched GEMM
    #      Convert from [B, L, L, H] → [B·H, L, L]
    # ------------------------------------------------------------------
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, L, L]
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    batch = bs * hidden_dim
    left_flat = left_perm.view(batch, L, L)            # (batch, M, K)
    right_flat = right_perm.view(batch, L, L)          # (batch, M, K)

    # Transpose right for the GEMM (we need Bᵀ)
    right_t = right_flat.transpose(1, 2).contiguous()  # (batch, K, N)

    # Allocate output in FP16
    out_flat_h = torch.empty_like(left_flat, dtype=torch.float16, device=device)

    # Strides for the three tensors (batch, row, col)
    a_stride_ab, a_stride_am, a_stride_ak = left_flat.stride()
    b_stride_bb, b_stride_bk, b_stride_bn = right_t.stride()
    c_stride_cb, c_stride_cm, c_stride_cn = out_flat_h.stride()

    # ------------------------------------------------------------------
    # 7️⃣  Batched GEMM kernel launch
    # ------------------------------------------------------------------
    BLOCK_M, BLOCK_N, BLOCK_K = 64, 64, 64
    grid = (batch,
            triton.cdiv(L, BLOCK_M),
            triton.cdiv(L, BLOCK_N))

    batched_gemm_kernel[grid](
        a_ptr=left_flat,
        b_ptr=right_t,
        c_ptr=out_flat_h,
        M=L,
        N=L,
        K=L,
        stride_am=a_stride_am,
        stride_ak=a_stride_ak,
        stride_ab=a_stride_ab,
        stride_bk=b_stride_bk,
        stride_bn=b_stride_bn,
        stride_bb=b_stride_bb,
        stride_cm=c_stride_cm,
        stride_cn=c_stride_cn,
        stride_cb=c_stride_cb,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 8️⃣  Cast back to FP32 and reshape to [B, L, L, H]
    # ------------------------------------------------------------------
    out_fp32 = out_flat_h.to(torch.float32)
    out = out_fp32.view(bs, hidden_dim, L, L).permute(0, 2, 3, 1).contiguous()   # [B, L, L, H]

    # ------------------------------------------------------------------
    # 9️⃣  Output LayerNorm (Triton) over the hidden dimension
    # ------------------------------------------------------------------
    out_M = bs * L * L
    out_flat = out.view(out_M, hidden_dim)
    to_out_norm_w = weights["to_out_norm.weight"].to(device, dtype).contiguous()
    to_out_norm_b = weights["to_out_norm.bias"].to(device, dtype).contiguous()
    out_norm_flat = triton_layernorm(out_flat, to_out_norm_w, to_out_norm_b, eps=eps)
    out_norm = out_norm_flat.view(bs, L, L, hidden_dim)

    # ------------------------------------------------------------------
    # 10️⃣ Apply output gate and final linear projection back to D
    # ------------------------------------------------------------------
    out = out_norm * out_gate.to(dtype)          # broadcast over hidden dim
    to_out_w = weights["to_out.weight"].to(device, dtype).contiguous()
    out = F.linear(out, to_out_w)                # [B, L, L, D]

    return out
