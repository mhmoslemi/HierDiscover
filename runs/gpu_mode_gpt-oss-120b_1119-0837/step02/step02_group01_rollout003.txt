"""
TriMul “outgoing” forward pass (AlphaFold3 style) with heavy lifting in half‑precision.

Algorithm
---------
1️⃣  Input LayerNorm (Triton two‑pass kernel) on the last dimension (dim).
2️⃣  Fuse the four linear projections/gates (left/right proj + left/right gate)
    into a *single* half‑precision matmul using a concatenated weight matrix.
3️⃣  Apply the optional pairwise mask to the projected tensors.
4️⃣  Apply sigmoid gates and obtain gated left/right tensors.
5️⃣  Perform the N³ contraction:
       out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched half‑precision GEMM (torch.bmm) after permuting
   to shape (B·H, N, N).
6️⃣  Second LayerNorm (torch.nn.functional.layer_norm – highly optimized) on
   the hidden dimension H, with learnable weight/bias.
7️⃣  Multiply by the output gate (computed from step 2) and apply the final
   linear projection back to the model dimension (half‑precision).
8️⃣  Cast the result to float32.

All major compute kernels stay on‑device; only the first LayerNorm is
implemented in Triton as required.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton LayerNorm kernel (two‑pass, works for any dtype, accumulates in FP32)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr, y_ptr,                 # *ptrs to input and output   [M, D]
    w_ptr, b_ptr,                 # *ptrs to weight and bias    [D]
    M, D, eps,                    # dimensions
    stride_xm, stride_xn,         # strides of x
    stride_ym, stride_yn,         # strides of y
    BLOCK_M: tl.constexpr,        # rows per program
    BLOCK_D: tl.constexpr,        # columns per tile
):
    pid = tl.program_id(0)

    # ------------------------------------------------------------------
    # Row offset for this program
    # ------------------------------------------------------------------
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # ------------------------------------------------------------------
    # First pass: compute per‑row mean & variance
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    num_d = tl.cdiv(D, BLOCK_D)
    for d in range(num_d):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < D
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_d[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        )
        sum_val += tl.sum(x, axis=1)
        sum_sq  += tl.sum(x * x, axis=1)

    mean = sum_val / D
    var  = sum_sq / D - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # ------------------------------------------------------------------
    # Second pass: write normalized + affine transformed values
    # ------------------------------------------------------------------
    for d in range(num_d):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < D
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_d[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        )
        # (x - mean) * rstd
        norm = (x - mean[:, None]) * rstd[:, None]

        # affine transform
        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)

        y = norm * w[None, :] + b[None, :]

        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_d[None, :] * stride_yn,
            y,
            mask=mask_m[:, None] & mask_d[None, :],
        )


def triton_layernorm(x: torch.Tensor,
                     weight: torch.Tensor,
                     bias: torch.Tensor,
                     eps: float = 1e-5) -> torch.Tensor:
    """
    Triton implementation of LayerNorm over the last dimension.
    `x` must be a 2‑D contiguous tensor [M, D].
    Returns a tensor with the same shape and dtype as ``x``.
    """
    assert x.dim() == 2, "triton_layernorm expects a 2‑D tensor"
    M, D = x.shape
    y = torch.empty_like(x)

    # Ensure contiguous layout for correct stride handling
    x = x.contiguous()
    y = y.contiguous()
    weight = weight.contiguous()
    bias = bias.contiguous()

    stride_xm, stride_xn = x.stride()
    stride_ym, stride_yn = y.stride()

    grid = lambda meta: (triton.cdiv(M, meta["BLOCK_M"]),)

    layernorm_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        w_ptr=weight,
        b_ptr=bias,
        M=M,
        D=D,
        eps=eps,
        stride_xm=stride_xm,
        stride_xn=stride_xn,
        stride_ym=stride_ym,
        stride_yn=stride_yn,
        BLOCK_M=128,   # rows per program – good default on H100
        BLOCK_D=64,    # columns per tile
    )
    return y


# ----------------------------------------------------------------------
# Entrypoint: custom Triton‑accelerated TriMul “outgoing” operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass for the TriMul “outgoing” module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, D]  (float32)
        - mask         : torch.Tensor [B, N, N] or None
        - weights      : dict of torch.Tensors (model parameters)
        - config       : dict with keys ``dim`` and ``hidden_dim`` and optional
                         ``nomask`` (bool)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, D] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype               # expected float32
    eps = 1e-5

    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (Triton) over the model dimension
    # ------------------------------------------------------------------
    M = bs * seq_len * seq_len                         # rows for LN
    x_flat = input_tensor.reshape(M, dim)              # (M, D)
    norm_w = weights["norm.weight"].to(device, dtype)
    norm_b = weights["norm.bias"].to(device, dtype)
    x_norm_flat = triton_layernorm(x_flat, norm_w, norm_b, eps=eps)
    x_norm = x_norm_flat.view(bs, seq_len, seq_len, dim)   # (B,N,N,D) float32

    # ------------------------------------------------------------------
    # Cast to FP16 for the heavy part of the pipeline
    # ------------------------------------------------------------------
    x_norm_f16 = x_norm.to(torch.float16)

    # ------------------------------------------------------------------
    # Load and cast all linear weights once
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device, torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device, torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device, torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device, torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device, torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device, torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device, torch.float16)
    to_out_w      = weights["to_out.weight"].to(device, torch.float16)

    # ------------------------------------------------------------------
    # 2️⃣ Fuse left/right projections and their gates into ONE matmul
    #    (4 * hidden_dim) output channels
    # ------------------------------------------------------------------
    fused_weight = torch.cat(
        [left_proj_w, left_gate_w, right_proj_w, right_gate_w], dim=0
    )  # shape (4*H, D), dtype=fp16
    # fused output shape: (B, N, N, 4*H)
    fused = F.linear(x_norm_f16, fused_weight)   # half precision

    # split for readability
    split = torch.split(fused, hidden_dim, dim=-1)
    left_proj  = split[0]                         # (B,N,N,H) half
    left_gate_raw = split[1]                     # (B,N,N,H) half
    right_proj = split[2]                        # (B,N,N,H) half
    right_gate_raw = split[3]                    # (B,N,N,H) half

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast over hidden_dim)
    # ------------------------------------------------------------------
    if (mask is not None) and (not nomask):
        mask_f16 = mask.to(device, torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left_proj = left_proj * mask_f16
        right_proj = right_proj * mask_f16

    # ------------------------------------------------------------------
    # 4️⃣ Apply sigmoid gates
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(left_gate_raw)   # half
    right_gate = torch.sigmoid(right_gate_raw) # half

    # ------------------------------------------------------------------
    # 5️⃣ Gated left/right tensors
    # ------------------------------------------------------------------
    left  = left_proj * left_gate    # (B,N,N,H) half
    right = right_proj * right_gate  # (B,N,N,H) half

    # ------------------------------------------------------------------
    # 6️⃣ Output gate (computed from the same normalized input)
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(F.linear(x_norm_f16, out_gate_w))   # (B,N,N,H) half

    # ------------------------------------------------------------------
    # 7️⃣ Core N³ contraction via batched FP16 GEMM
    #    left[b,i,k,h] * right[b,j,k,h] → sum_k
    # ------------------------------------------------------------------
    # Move hidden dimension to batch axis for torch.bmm
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B,H,N,N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # (B,H,N,N)

    BH = bs * hidden_dim
    left_mat  = left_perm.view(BH, seq_len, seq_len)        # (B*H, N, N)
    right_mat = right_perm.view(BH, seq_len, seq_len)       # (B*H, N, N)

    # batched matrix multiplication: (i,k)·(j,k)^T
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B*H, N, N) half

    # restore (B,N,N,H)
    out = out_mat.view(bs, hidden_dim, seq_len, seq_len) \
                 .permute(0, 2, 3, 1).contiguous()           # (B,N,N,H) half

    # ------------------------------------------------------------------
    # 8️⃣ Second LayerNorm on hidden dimension (use torch's fast kernel)
    # ------------------------------------------------------------------
    # Convert to FP32 for numeric stability (torch.layer_norm is heavily
    # optimized and typically faster than a custom Triton kernel for this shape)
    out_fp32 = out.to(torch.float32)
    out_norm = F.layer_norm(
        out_fp32,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w.to(torch.float32),
        bias=to_out_norm_b.to(torch.float32),
        eps=eps,
    )   # (B,N,N,H) fp32

    # ------------------------------------------------------------------
    # 9️⃣ Apply output gate and final projection (all half‑precision)
    # ------------------------------------------------------------------
    out_norm_fp16 = out_norm.to(torch.float16)          # (B,N,N,H) half
    gated_fp16 = out_norm_fp16 * out_gate               # elementwise scale
    # final linear projection back to model dimension
    output_fp16 = F.linear(gated_fp16, to_out_w)        # (B,N,N,D) half
    output = output_fp16.to(torch.float32)             # cast back to FP32
    return output
