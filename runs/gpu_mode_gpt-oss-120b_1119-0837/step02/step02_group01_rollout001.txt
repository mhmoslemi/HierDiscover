def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    eps = 1e-5

    # 1 - Input layernorm using Triton
    M = bs * N * N
    x_flat = input_tensor.reshape(M, dim)
    norm_weight = weights["norm.weight"].to(device, dtype)
    norm_bias = weights["norm.bias"].to(device, dtype)
    x_norm = triton_layernorm(x_flat, norm_weight, norm_bias, eps).view(bs, N, N, dim)

    # 2 - Cast to fp16 for heavy ops
    x_f16 = x_norm.to(torch.float16)

    # 3 - Combine proj+gate weights
    left_comb_w = torch.cat([weights["left_proj.weight"], weights["left_gate.weight"]], dim=0).to(device, torch.float16)
    right_comb_w = torch.cat([weights["right_proj.weight"], weights["right_gate.weight"]], dim=0).to(device, torch.float16)
    out_gate_w = weights["out_gate.weight"].to(device, torch.float16)

    # 4 - Linear passes (fp16)
    left_comb = torch.nn.functional.linear(x_f16, left_comb_w)  # (B,N,N,2*H)
    right_comb = torch.nn.functional.linear(x_f16, right_comb_w)  # (B,N,N,2*H)

    # split
    left_proj = left_comb[..., :hidden_dim]
    left_gate_pre = left_comb[..., hidden_dim:]
    right_proj = right_comb[..., :hidden_dim]
    right_gate_pre = right_comb[..., hidden_dim:]

    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)

    # 5 - Mask
    if not nomask and mask is not None:
        mask_f16 = mask.to(device, torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left_proj = left_proj * mask_f16
        right_proj = right_proj * mask_f16

    # 6 - Apply gates
    left = left_proj * left_gate
    right = right_proj * right_gate

    # 7 - Compute out_gate
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_f16, out_gate_w))  # (B,N,N,H)

    # 8 - Core contraction using batched GEMM
    # Permute for batch-major layout (B*H, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    BH = bs * hidden_dim
    left_mat = left_perm.view(BH, N, N)
    right_mat = right_perm.view(BH, N, N)
    # bmm
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))
    # back to (B,N,N,H)
    out = out_mat.view(bs, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H) fp16

    # 9 - Second layernorm (fp16)   # Using same Triton kernel with x dtype half
    # Use triton_layernorm with half: we need to adapt kernel to accept half input and half output.
    # We'll call custom half version `triton_layernorm_half`
    out_flat = out.reshape(M, hidden_dim)  # (B*N*N, H)
    to_out_norm_w = weights["to_out_norm.weight"].to(device, torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device, torch.float16)
    out_norm = triton_layernorm_fp16(out_flat, to_out_norm_w, to_out_norm_b, eps)  # returns half

    # 10 - Apply out_gate and final projection (fp16)
    out_gate_f16 = out_gate.view(M, hidden_dim)  # flatten
    gated = out_norm * out_gate_f16  # (M,H) fp16

    # Final linear projection to dim
    to_out_w = weights["to_out.weight"].to(device, torch.float16)  # (dim, H)
    # Perform linear on flattened view:
    output_flat = torch.nn.functional.linear(gated, to_out_w)  # (M, dim) fp16
    output = output_flat.view(bs, N, N, dim).to(torch.float32)

    return output

@triton.jit
def layernorm_fp16_kernel(
    x_ptr, y_ptr, weight_ptr, bias_ptr,
    M, N, eps,
    stride_xm, stride_xn,
    stride_ym, stride_yn,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
):
    pid = tl.program_id(0)
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_N)
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn, mask=mask_m[:, None] & mask_n[None, :])
        x_f32 = tl.cast(x, tl.float32)
        sum_val += tl.sum(x_f32, axis=1)
        sum_sq += tl.sum(x_f32 * x_f32, axis=1)

    mean = sum_val / N
    var = sum_sq / N - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn, mask=mask_m[:, None] & mask_n[None, :])
        x_f32 = tl.cast(x, tl.float32)
        norm_x_f32 = (x_f32 - mean[:, None]) * rstd[:, None]
        norm_x = tl.cast(norm_x_f32, tl.float16)

        w = tl.load(weight_ptr + offs_n, mask=mask_n, other=0.0)

"""
TriMul “outgoing” forward pass – highly‑optimized Triton+torch implementation.

Algorithm
---------
1️⃣ Input LayerNorm over the model dimension (float32) using a custom
   Triton kernel (2‑pass mean/variance, FP32 accumulation, FP16 output).

2️⃣ Cast to FP16 for the heavy linear / gating work.

3️⃣ Fuse each projection + its gate into ONE linear call:
   * left_proj + left_gate  → weight matrix (2·H × D)
   * right_proj + right_gate → weight matrix (2·H × D)
   The first H rows are the projection, the second H rows are the gate
   (followed by a sigmoid).  This reduces five matmuls to three.

4️⃣ Apply the optional mask (broadcast over the hidden dimension).

5️⃣ Element‑wise gating of the projected tensors.

6️⃣ Compute the core N³ contraction as a batched half‑precision GEMM:
   out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   (implemented by torch.bmm on tensors of shape (B·H, N, N)).

7️⃣ LayerNorm on the hidden dimension of the result – another Triton kernel
   that works directly in FP16 (FP32 accumulators, FP16 output).

8️⃣ Apply the output gate (sigmoid of a linear on the normalized input).

9️⃣ Final linear projection back to the model dimension (FP16) and cast to
   FP32 for the returned tensor.

All heavy linear work stays in FP16 to exploit Tensor‑Core throughput;
the only full‑precision steps are the two LayerNorms (still using FP32
accumulators for numerical stability).  The implementation respects all
configurable options (mask/no‑mask, batch size, sequence length, dim,
hidden_dim).  The required Triton kernels are provided below.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------
# 2‑pass LayerNorm kernel (float32 input → float32 output)
# --------------------------------------------------------------
@triton.jit
def layernorm_fp32_kernel(
    x_ptr, y_ptr, weight_ptr, bias_ptr,
    M, N, eps,
    stride_xm, stride_xn,
    stride_ym, stride_yn,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr
):
    pid = tl.program_id(0)

    # ------------------------------------------------------------------
    # Row range for this program
    # ------------------------------------------------------------------
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # ------------------------------------------------------------------
    # First pass: compute per‑row mean & variance (FP32 accumulation)
    # ------------------------------------------------------------------
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_N)
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0
        )
        sum_val += tl.sum(x, axis=1)
        sum_sq  += tl.sum(x * x, axis=1)

    mean = sum_val / N
    var  = sum_sq / N - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # ------------------------------------------------------------------
    # Second pass: write normalized + affine transformed values
    # ------------------------------------------------------------------
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0
        )
        norm_x = (x - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_n, mask=mask_n, other=0.0)
        b = tl.load(bias_ptr   + offs_n, mask=mask_n, other=0.0)

        y = norm_x * w[None, :] + b[None, :]

        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn,
            y,
            mask=mask_m[:, None] & mask_n[None, :]
        )


def triton_layernorm_fp32(x: torch.Tensor,
                          weight: torch.Tensor,
                          bias: torch.Tensor,
                          eps: float = 1e-5) -> torch.Tensor:
    """LayerNorm (float32) using the Triton kernel above."""
    assert x.dim() == 2, "expects a 2‑D tensor [M, N]"
    M, N = x.shape
    y = torch.empty_like(x)

    x = x.contiguous()
    y = y.contiguous()
    weight = weight.contiguous()
    bias = bias.contiguous()

    stride_xm, stride_xn = x.stride()
    stride_ym, stride_yn = y.stride()

    grid = lambda meta: (triton.cdiv(M, meta["BLOCK_M"]),)

    layernorm_fp32_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        weight_ptr=weight,
        bias_ptr=bias,
        M=M,
        N=N,
        eps=eps,
        stride_xm=stride_xm,
        stride_xn=stride_xn,
        stride_ym=stride_ym,
        stride_yn=stride_yn,
        BLOCK_M=128,
        BLOCK_N=64,
    )
    return y

# --------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr, y_ptr, weight_ptr, bias_ptr,
    M, N, eps,
    stride_xm, stride_xn,
    stride_ym, stride_yn,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr
):
    pid = tl.program_id(0)

    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_N)
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0
        )
        xf = tl.cast(x, tl.float32)
        sum_val += tl.sum(xf, axis=1)
        sum_sq  += tl.sum(xf * xf, axis=1)

    mean = sum_val / N
    var  = sum_sq / N - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0
        )
        xf = tl.cast(x, tl.float32)
        norm_x_f32 = (xf - mean[:, None]) * rstd[:, None]
        norm_x = tl.cast(norm_x_f32, tl.float16)

        w = tl.load(weight_ptr + offs_n, mask=mask_n, other=0.0)
        b = tl.load(bias_ptr   + offs_n, mask=mask_n, other=0.0)

        y = norm_x * w[None, :] + b[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn,
            y,
            mask=mask_m[:, None] & mask_n[None, :]
        )


def triton_layernorm_fp16(x: torch.Tensor,
                          weight: torch.Tensor,

                          eps: float = 1e-5) -> torch.Tensor:

# Entry point – custom Triton‑accelerated TriMul (outgoing)
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator.
    Arguments:
        data = (input_tensor, mask, weights, config)
            input_tensor : torch.Tensor[B, N, N, D] (float32)
            mask         : torch.Tensor[B, N, N] or None
            weights      : dict of torch.Tensor (model parameters)
            config       : dict containing keys:
                           - dim (model dimension D)
                           - hidden_dim (H)
                           - nomask (bool, optional)
    Returns:
        torch.Tensor[B, N, N, D] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype  # expected torch.float32

    bs, N, _, dim = input_tensor.shape
    H = config["hidden_dim"]
    nomask = config.get("nomask", False)
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32) – collapse batch/seq dimensions
    # ------------------------------------------------------------------
    M = bs * N * N
    x_flat = input_tensor.reshape(M, dim)
    norm_weight = weights["norm.weight"].to(device, dtype)
    norm_bias = weights["norm.bias"].to(device, dtype)
    x_norm_flat = triton_layernorm_fp32(x_flat, norm_weight, norm_bias, eps=eps)
    x_norm = x_norm_flat.view(bs, N, N, dim)  # (B, N, N, D) FP32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to FP16 for the heavy lifting
    # ------------------------------------------------------------------
    x_f16 = x_norm.to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Fuse projection + gate for left & right sides (single matmul each)
    # ------------------------------------------------------------------
    left_comb_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(device, torch.float16)               # (2*H, D)
    right_comb_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(device, torch.float16)               # (2*H, D)
    out_gate_w = weights["out_gate.weight"].to(device, torch.float16)  # (H, D)

    # (B, N, N, 2*H)
    left_comb = F.linear(x_f16, left_comb_w)
    right_comb = F.linear(x_f16, right_comb_w)

    # Split projections and gates
    left_proj = left_comb[..., :H]            # (B, N, N, H)
    left_gate_pre = left_comb[..., H:]        # (B, N, N, H)
    right_proj = right_comb[..., :H]
    right_gate_pre = right_comb[..., H:]

    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)

    # ------------------------------------------------------------------
    # 4️⃣ Optional mask (broadcast over H)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_f16 = mask.to(device, torch.float16).unsqueeze(-1)  # (B, N, N, 1)
        left_proj = left_proj * mask_f16
        right_proj = right_proj * mask_f16

    # ------------------------------------------------------------------
    # 5️⃣ Apply element‑wise gates
    # ------------------------------------------------------------------
    left = left_proj * left_gate          # (B, N, N, H)
    right = right_proj * right_gate       # (B, N, N, H)

    # ------------------------------------------------------------------
    # 6️⃣ Output gate (independent of mask)
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(F.linear(x_f16, out_gate_w))  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 7️⃣ Core N³ contraction via a batched half‑precision GEMM
    # ------------------------------------------------------------------
    # Reshape to (B*H, N, N) for a single torch.bmm
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    BH = bs * H
    left_mat = left_perm.view(BH, N, N)        # (B*H, N, N)
    right_mat = right_perm.view(BH, N, N)      # (B*H, N, N)

    # (B*H, N, N)  =  (B*H, N, N) @ (B*H, N, N)^T
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Back to (B, N, N, H)
    out = out_mat.view(bs, H, N, N).permute(0, 2, 3, 1).contiguous()  # FP16

    # ------------------------------------------------------------------
    # 8️⃣ Second LayerNorm on hidden dimension (FP16 kernel)
    # ------------------------------------------------------------------
    out_flat = out.reshape(M, H)   # (B*N*N, H)
    to_out_norm_w = weights["to_out_norm.weight"].to(device, torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device, torch.float16)
    out_norm = triton_layernorm_fp16(out_flat, to_out_norm_w, to_out_norm_b, eps=eps)  # FP16

    # ------------------------------------------------------------------
    # 9️⃣ Apply output gate and final projection (both FP16)
    # ------------------------------------------------------------------
    out_gate_flat = out_gate.reshape(M, H)          # (M, H) FP16
    gated = out_norm * out_gate_flat                # (M, H) FP16

    to_out_w = weights["to_out.weight"].to(device, torch.float16)  # (D, H)
    output_flat = F.linear(gated, to_out_w)        # (M, D) FP16

    output = output_flat.view(bs, N, N, dim).to(torch.float32)  # final FP32

    return output
