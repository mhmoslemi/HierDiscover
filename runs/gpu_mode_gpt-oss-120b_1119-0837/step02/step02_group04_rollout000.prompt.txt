import torch
from torch import nn, einsum
import math

# Reference code in PyTorch
class TriMul(nn.Module):
    def __init__(
        self,
        dim: int,
        hidden_dim: int,
    ):
        super().__init__()

        self.norm = nn.LayerNorm(dim)

        self.left_proj = nn.Linear(dim, hidden_dim, bias=False)
        self.right_proj = nn.Linear(dim, hidden_dim, bias=False)

        self.left_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.right_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.out_gate = nn.Linear(dim, hidden_dim, bias=False)

        self.to_out_norm = nn.LayerNorm(hidden_dim)
        self.to_out = nn.Linear(hidden_dim, dim, bias=False)

    def forward(self, x: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
        """
        x: [bs, seq_len, seq_len, dim]
        mask: [bs, seq_len, seq_len]

        Returns:
            output: [bs, seq_len, seq_len, dim]
        """
        batch_size, seq_len, _, dim = x.shape

        x = self.norm(x)

        left = self.left_proj(x)
        right = self.right_proj(x)

        mask = mask.unsqueeze(-1)
        left = left * mask
        right = right * mask

        left_gate = self.left_gate(x).sigmoid()
        right_gate = self.right_gate(x).sigmoid()
        out_gate = self.out_gate(x).sigmoid()

        left = left * left_gate
        right = right * right_gate

        out = einsum('... i k d, ... j k d -> ... i j d', left, right)
        # This einsum is the same as the following:
        # out = torch.zeros(batch_size, seq_len, seq_len, dim, device=x.device)
        
        # # Compute using nested loops
        # for b in range(batch_size):
        #     for i in range(seq_len):
        #         for j in range(seq_len):
        #             # Compute each output element
        #             for k in range(seq_len):
        #                 out[b, i, j] += left[b, i, k, :] * right[b, j, k, :]

        out = self.to_out_norm(out)
        out = out * out_gate
        return self.to_out(out)

"""
TriMul (outgoing) – fast Triton implementation
================================================

The module implements the “outgoing” triangle multiplicative update from the
AlphaFold‑3 paper:

    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

where ``left`` and ``right`` are linear projections of the layer‑normed input
and are modulated by learned sigmoid gates (and an optional mask).  The
computational bottleneck is the batched GEMM over the sequence dimension N
(which can be up to 1024).  This code fuses the GEMM into a custom Triton
kernel that:

* works on half‑precision inputs (FP16) to exploit Tensor‑cores,
* accumulates in FP32 for numerical stability,
* uses a 128×128×64 tile (the sweet spot for H100) with 8 warps per block
  and double‑buffered loads (``num_stages=2``).

All remaining ops (LayerNorm, linear projections, sigmoid gates, final
normalisation and projection) are performed with regular PyTorch kernels.
The final tensor is returned in FP32 as required by the downstream model.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    out_ptr,                     # float32*   output  (C, M, N)
    a_ptr,                       # float16*   left    (C, M, K)
    b_ptr,                       # float16*   right   (C, K, N) – loaded in transposed order
    M, N, K,                     # int64      matrix dimensions (all = seq_len)
    stride_out_m, stride_out_n, stride_out_c,   # strides for out
    stride_a_m, stride_a_k, stride_a_c,        # strides for left
    stride_b_k, stride_b_n, stride_b_c,        # strides for right (K,N order)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM: out = left @ rightᵀ  (one program ID per hidden channel).

    Each program processes a tile of shape (BLOCK_M, BLOCK_N) for a given
    hidden channel (C = batch * hidden_dim).  Tensor‑core friendly
    FP16×FP16→FP32 dots are used, with double‑buffered loads.
    """
    pid_m = tl.program_id(0)   # block row   → i
    pid_n = tl.program_id(1)   # block col   → j
    pid_c = tl.program_id(2)   # hidden‑channel (batch * hidden_dim)

    # Tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K (seq_len) dimension in BLOCK_K‑sized chunks
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = kk * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Pointers to the current tile of left (M×K) and right (K×N)
        a_tile_ptr = (
            a_ptr
            + pid_c * stride_a_c
            + offs_m[:, None] * stride_a_m
            + offs_k[None, :] * stride_a_k
        )
        b_tile_ptr = (
            b_ptr
            + pid_c * stride_b_c
            + offs_k[:, None] * stride_b_k   # K dimension first (row)
            + offs_n[None, :] * stride_b_n   # N dimension (col)
        )

        # Load with masking; other=0.0 ensures out‑of‑bounds values are zero.
        a = tl.load(
            a_tile_ptr,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        b = tl.load(
            b_tile_ptr,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # GEMM accumulate (FP16·FP16 → FP32)
        acc += tl.dot(a, b)

    # Write the result tile back to global memory
    out_tile_ptr = (
        out_ptr
        + pid_c * stride_out_c
        + offs_m[:, None] * stride_out_m
        + offs_n[None, :] * stride_out_n
    )
    tl.store(
        out_tile_ptr,
        acc,
        mask=mask_m[:, None] & mask_n[None, :],
    )


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Args
    ----
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor[B, N, N, C] (float32)
        - mask         : torch.Tensor[B, N, N] (bool/float) – may be None
        - weights      : dict of torch.Tensors (model parameters)
        - config       : dict with keys
            * "dim"        – input channel dimension C
            * "hidden_dim" – hidden dimension H
            * "nomask"     – bool, if True the mask is ignored

    Returns
    -------
    torch.Tensor[B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1) Layer‑norm the input
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, C]  (fp32)

    # ------------------------------------------------------------------
    # 2) Cast to FP16 for all linear ops
    x_h = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Linear projections (no bias)
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left  = F.linear(x_h, left_proj_w)    # [B, N, N, H]
    right = F.linear(x_h, right_proj_w)   # [B, N, N, H]

    # ------------------------------------------------------------------
    # 4) Gates (sigmoid)
    left_gate_w  = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w   = weights["out_gate.weight"].to(torch.float16)

    left_gate  = torch.sigmoid(F.linear(x_h, left_gate_w))   # [B, N, N, H]
    right_gate = torch.sigmoid(F.linear(x_h, right_gate_w))  # [B, N, N, H]
    out_gate   = torch.sigmoid(F.linear(x_h, out_gate_w))   # [B, N, N, H]

    # ------------------------------------------------------------------
    # 5) Optional mask (applied before gating, like the reference)
    if not nomask and mask is not None:
        mask_f = mask.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left  = left * mask_f
        right = right * mask_f

    # 6) Apply gates
    left  = left  * left_gate
    right = right * right_gate

    # Release gate tensors that are no longer needed
    del left_gate, right_gate, x_h, x

    # ------------------------------------------------------------------
    # 7) Reshape for the Triton kernel
    B, N, _, _ = left.shape                          # B, N, N, H
    # move hidden dimension before the two seq dimensions for contiguous layout
    left  = left.permute(0, 3, 1, 2).contiguous()    # [B, H, N, N]
    right = right.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]

    C_total = B * hidden_dim                         # number of independent GEMMs
    left_flat  = left.reshape(C_total, N, N)         # (C, M, K)
    right_flat = right.reshape(C_total, N, N)        # (C, K, N)   (we will load it transposed)

    # Output buffer – FP32 accumulation
    out_flat = torch.empty_like(left_flat, dtype=torch.float32)

    # ------------------------------------------------------------------
    # 8) Triton kernel launch parameters
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    stride_out_c = out_flat.stride(0)
    stride_out_m = out_flat.stride(1)
    stride_out_n = out_flat.stride(2)

    stride_a_c = left_flat.stride(0)
    stride_a_m = left_flat.stride(1)
    stride_a_k = left_flat.stride(2)

    stride_b_c = right_flat.stride(0)
    stride_b_k = right_flat.stride(2)   # last dim (K)
    stride_b_n = right_flat.stride(1)   # middle dim (N)

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # blocks along M (i)
        (N + BLOCK_N - 1) // BLOCK_N,   # blocks along N (j)
        C_total,                        # one program per hidden channel
    )

    # Launch the kernel
    tri_mul_kernel[grid](
        out_flat,
        left_flat,
        right_flat,
        N, N, N,
        stride_out_m,
        stride_out_n,
        stride_out_c,
        stride_a_m,
        stride_a_k,
        stride_a_c,
        stride_b_k,
        stride_b_n,
        stride_b_c,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=2,
    )

    # ------------------------------------------------------------------
    # 9) Reshape back to (B, N, N, H)
    out = out_flat.reshape(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 10) Post‑processing: layer‑norm, output gate, final projection
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    # Gate in FP32 (automatic up‑cast)
    out = out * out_gate

    # Final linear projection back to the original dimension (use FP16 for speed)
    to_out_w = weights["to_out.weight"].to(torch.float16)
    out = F.linear(out.to(torch.float16), to_out_w).to(torch.float32)

    return out
