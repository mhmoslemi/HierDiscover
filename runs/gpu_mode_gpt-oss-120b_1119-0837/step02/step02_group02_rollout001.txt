"""
tri_mul_outgoing.py

Implementation of the “outgoing” TriMul block (AlphaFold‑3) in pure Triton + PyTorch.

Algorithm
---------
1. Layer‑norm the input tensor (float32) →  x  [B, N, N, D].
2. Cast to FP16 and apply three linear projections (left, right, three gates)
   – all operations stay in FP16 for bandwidth.
3. Optional pair‑wise mask is broadcast over the hidden dimension.
4. Apply sigmoid gates to left/right and mask‑gated tensors.
5. Core O(N³) contraction:
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   is a *batched* GEMM (left @ rightᵀ) for every hidden channel.
   A dedicated Triton kernel performs this GEMM in FP16 with FP32 accumulation,
   using 128×128 tiles and a 32‑element K‑tile (8 warps, double‑buffered).
6. Layer‑norm the hidden‑dim result, multiply by the output gate, and apply the
   final linear projection back to the original channel dimension.
7. Return a float32 tensor of shape [B, N, N, D].

The implementation keeps the heavyweight contraction on the GPU while the
light‑weight element‑wise work (layer‑norms, sigmoids, masking) stays in
PyTorch.  All operations are differentiable‑free – only the forward pass is
required.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# -----------------------------------------------------------------------------#
# Triton kernel: batched GEMM C = A @ Bᵀ   (A: M×K, B: N×K → Bᵀ: K×N)
# -----------------------------------------------------------------------------#
@triton.jit
def _trmul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,               # FP16 pointers
    M, N, K,                           # matrix shapes (M = N = seq_len, K = seq_len)
    stride_am, stride_ak,              # A row/col stride inside a batch (elements)
    stride_bk, stride_bn,              # B (already transposed) row/col stride
    stride_cm, stride_cn,              # C row/col stride
    stride_a_batch, stride_b_batch, stride_c_batch,   # stride between batches
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched half‑precision GEMM with FP32 accumulation.

    Each program works on a (BLOCK_M × BLOCK_N) tile of a given batch.
    """
    pid = tl.program_id(axis=0)          # tile index over (M,N)
    batch_id = tl.program_id(axis=1)     # flattened (batch * hidden) dimension

    # --------------------------------------------------------------- #
    #   Tile coordinates
    # --------------------------------------------------------------- #
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    block_m = pid // num_blocks_n
    block_n = pid % num_blocks_n

    offs_m = block_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = block_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # --------------------------------------------------------------- #
    #   Batch‑specific base pointers
    # --------------------------------------------------------------- #
    a_batch = a_ptr + batch_id * stride_a_batch
    b_batch = b_ptr + batch_id * stride_b_batch
    c_batch = c_ptr + batch_id * stride_c_batch

    # --------------------------------------------------------------- #
    #   Accumulator (FP32)
    # --------------------------------------------------------------- #
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------- #
    #   K‑loop
    # --------------------------------------------------------------- #
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_off = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_off < K

        # Load A tile (BLOCK_M × BLOCK_K)
        a_ptrs = a_batch + (offs_m[:, None] * stride_am) + (k_off[None, :] * stride_ak)
        a = tl.load(a_ptrs,
                    mask=(offs_m[:, None] < M) & mask_k[None, :],
                    other=0.0)

        # Load B tile (BLOCK_K × BLOCK_N) – B is already transposed
        b_ptrs = b_batch + (k_off[:, None] * stride_bk) + (offs_n[None, :] * stride_bn)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & (offs_n[None, :] < N),
                    other=0.0)

        # Rank‑K update
        acc += tl.dot(a, b)   # FP16 × FP16 → FP32

    # --------------------------------------------------------------- #
    #   Store result (cast back to FP16)
    # --------------------------------------------------------------- #
    c_ptrs = c_batch + (offs_m[:, None] * stride_cm) + (offs_n[None, :] * stride_cn)
    tl.store(c_ptrs,
             acc.to(tl.float16),
             mask=(offs_m[:, None] < M) & (offs_n[None, :] < N))


# -----------------------------------------------------------------------------#
# Public entry point
# -----------------------------------------------------------------------------#
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul block (no gradients).

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, D] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) – may be ignored
            - weights_dict : mapping of all module parameters (float32)
                Keys required:
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
            - config_dict  : {"dim": D, "hidden_dim": H, "nomask": bool, ...}
    Returns:
        torch.Tensor of shape [B, N, N, D] (float32)
    """
    # --------------------------------------------------------------------- #
    #   Unpack arguments
    # --------------------------------------------------------------------- #
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    eps = 1e-5

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------------- #
    #   1) Input layer‑norm (float32 for numerical stability)
    # --------------------------------------------------------------------- #
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )   # shape [B, N, N, dim] – float32

    # --------------------------------------------------------------------- #
    #   2) Cast to FP16 and fetch half‑precision weights
    # --------------------------------------------------------------------- #
    x_fp16 = x.half()

    left_proj_w   = weights["left_proj.weight"].half()
    right_proj_w  = weights["right_proj.weight"].half()
    left_gate_w   = weights["left_gate.weight"].half()
    right_gate_w  = weights["right_gate.weight"].half()
    out_gate_w    = weights["out_gate.weight"].half()
    to_out_norm_w = weights["to_out_norm.weight"].half()
    to_out_norm_b = weights["to_out_norm.bias"].half()
    to_out_w      = weights["to_out.weight"].half()

    # --------------------------------------------------------------------- #
    #   3) Linear projections (bias‑free) → FP16
    # --------------------------------------------------------------------- #
    left  = F.linear(x_fp16, left_proj_w)   # [B, N, N, H]
    right = F.linear(x_fp16, right_proj_w)  # [B, N, N, H]

    # --------------------------------------------------------------------- #
    #   4) Optional pair‑wise mask (broadcast over hidden dim)
    # --------------------------------------------------------------------- #
    if not nomask:
        # mask: [B, N, N] → [B, N, N, 1] (float16)
        mask_f = mask.to(x_fp16.dtype).unsqueeze(-1)
        left  = left * mask_f
        right = right * mask_f

    # --------------------------------------------------------------------- #
    #   5) Gating (sigmoid) – stay in FP16
    # --------------------------------------------------------------------- #
    left_gate  = torch.sigmoid(F.linear(x_fp16, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_fp16, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x_fp16, out_gate_w))

    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------------- #
    #   6) Core contraction via a batched GEMM (Triton)
    # --------------------------------------------------------------------- #
    #   Rearrange tensors to (batch, M, K) and (batch, K, N) layout.
    #   After permute, the hidden dimension becomes the second axis.
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    #   Transpose right for the GEMM (B, H, K, N)
    right_T = right_perm.permute(0, 1, 3, 2).contiguous()

    B_, H_, M, K = left_perm.shape          # M == K == N
    _, _, _, N_ = right_T.shape
    assert M == K == N_ == N, "Inconsistent sequence dimensions"

    total_batches = B_ * H_

    #   View as classical batched matrices
    a = left_perm.view(total_batches, M, K)     # (batch, M, K) – FP16
    b = right_T.view(total_batches, K, N_)      # (batch, K, N) – FP16
    c = torch.empty((total_batches, M, N_), dtype=torch.float16, device=device)

    # --------------------------------------------------------------------- #
    #   7) Compute strides (in number of elements) – no Python loops needed
    # --------------------------------------------------------------------- #
    stride_a_batch = a.stride(0)
    stride_am = a.stride(1)
    stride_ak = a.stride(2)

    stride_b_batch = b.stride(0)
    stride_bk = b.stride(1)
    stride_bn = b.stride(2)

    stride_c_batch = c.stride(0)
    stride_cm = c.stride(1)
    stride_cn = c.stride(2)

    # --------------------------------------------------------------------- #
    #   8) Launch Triton kernel
    # --------------------------------------------------------------------- #
    BLOCK_M, BLOCK_N, BLOCK_K = 128, 128, 32
    grid = (
        ((M + BLOCK_M - 1) // BLOCK_M) * ((N_ + BLOCK_N - 1) // BLOCK_N),
        total_batches,
    )

    _trmul_gemm_kernel[grid](
        a, b, c,
        M, N_, K,
        stride_am, stride_ak,
        stride_bk, stride_bn,
        stride_cm, stride_cn,
        stride_a_batch, stride_b_batch, stride_c_batch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,          # good occupancy for 128×128 tiles on H100
        num_stages=2,         # double‑buffering
    )

    # --------------------------------------------------------------------- #
    #   9) Reshape back to [B, N, N, H] (still FP16)
    # --------------------------------------------------------------------- #
    out_hidden = c.view(B_, hidden_dim, M, N_).permute(0, 2, 3, 1).contiguous()  # [B, N, N, H]

    # --------------------------------------------------------------------- #
    # 10) Hidden‑dim layer‑norm, output‑gate, final linear (all FP16)
    # --------------------------------------------------------------------- #
    out_hidden = F.layer_norm(
        out_hidden,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=eps,
    )
    out_hidden = out_hidden * out_gate                # element‑wise gating
    out = F.linear(out_hidden, to_out_w)              # back to original D

    # --------------------------------------------------------------------- #
    # 11) Cast back to float32 for downstream compatibility
    # --------------------------------------------------------------------- #
    return out.float()
