@triton.jit
def trmul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    N, H,  # seq_len, hidden_dim; K = N
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    stride_batch,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    hidden_dim: tl.constexpr,
):
    pid = tl.program_id(axis=0)
    batch_hidden_id = tl.program_id(axis=1)
    b = batch_hidden_id // hidden_dim
    h = batch_hidden_id % hidden_dim

    # base pointer for this (b,h) slice
    a_ptr_batch = a_ptr + b * stride_batch + h
    b_ptr_batch = b_ptr + b * stride_batch + h
    c_ptr_batch = c_ptr + b * stride_batch + h

    # tile offsets for M and N
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    block_m = pid // num_blocks_n
    block_n = pid % num_blocks_n

    offs_m = block_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = block_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # K loop
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_off = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_off < N

        # A tile: (M, K)
        a_ptrs = a_ptr_batch + (offs_m[:, None] * stride_am) + (k_off[None, :] * stride_ak)
        a = tl.load(a_ptrs,
                    mask=(offs_m[:, None] < N) & mask_k[None, :],
                    other=0.0)
        # B tile: (K, N)
        b_ptrs = b_ptr_batch + (k_off[:, None] * stride_bk) + (offs_n[None, :] * stride_bn)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & (offs_n[None, :] < N),
                    other=0.0)

        acc += tl.dot(a, b)   # FP16 * FP16 -> FP32

    # store results
    c_ptrs = c_ptr_batch + (offs_m[:, None] * stride_cm) + (offs_n[None, :] * stride_cn)

             mask=(offs_m[:, None] < N) & (offs_n[None, :] < N))

    HIDDEN_DIM: tl.constexpr,
):

    b = batch_hidden // HIDDEN_DIM
    h = batch_hidden % HIDDEN_DIM

    b_base = b_ptr + b * stride_batch + h
    c_base = c_ptr + b * stride_batch + h

    pid = tl.program_id(axis=0)
    # tile coordinates
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    block_m = pid // num_blocks_n
    block_n = pid % num_blocks_n

    offs_m = block_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = block_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # K loop
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_off = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_off < N

        # load A tile
        a_ptrs = a_base + (offs_m[:, None] * stride_am) + (k_off[None, :] * stride_ak)
        a = tl.load(a_ptrs,
                    mask=(offs_m[:, None] < N) & mask_k[None, :],
                    other=0.0)

        # load B tile
        b_ptrs = b_base + (k_off[:, None] * stride_bk) + (offs_n[None, :] * stride_bn)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & (offs_n[None, :] < N),
                    other=0.0)

        acc += tl.dot(a, b)

    # store result
    c_ptrs = c_base + (offs_m[:, None] * stride_cm) + (offs_n[None, :] * stride_cn)

             mask=(offs_m[:, None] < N) & (offs_n[None, :] < N))

"""
TriMul (outgoing) – optimized Triton implementation.

We implement the heavy O(N³) contraction
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
as a batched half‑precision GEMM (A @ Bᵀ) with fp32 accumulation.
Key optimisations over the reference implementation:

* No explicit permutes or copies – we operate directly on tensors in the
  native layout [B, N, N, H] (H = hidden_dim).  Strides are computed
  analytically and passed to the kernel.
* The hidden dimension is folded into the second grid axis.  Each program
  works on a (i‑tile, j‑tile) for a particular (batch, hidden) pair,
  eliminating the need for a large number of kernel launches.
* Tensor‑core friendly tile size (128×128×32) and double‑buffering
  (num_stages=3) with 8 warps give high occupancy on H100.
* All cheap operations (LayerNorm, linear projections, sigmoids,
  optional mask, output‐norm, final linear) stay in fp16, reducing
  traffic.  The final output is cast back to fp32 as required by the
  reference API.

The function `custom_kernel` follows the required signature:
    custom_kernel((input, mask, weights, config)) -> Tensor[float32]
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# -----------------------------------------------------------------------------#
# Triton kernel: batched GEMM C = A @ Bᵀ   (A: M×K, B: N×K → Bᵀ: K×N)
# -----------------------------------------------------------------------------#
@triton.jit
def _trmul_gemm_kernel(
    a_ptr,                     # FP16 left  [B,N,N,H]
    b_ptr,                     # FP16 right [B,N,N,H]
    c_ptr,                     # FP16 out   [B,N,N,H]
    N,                         # sequence length = M = K = N
    stride_am, stride_ak,      # A row/col stride (M‑stride, K‑stride)
    stride_bk, stride_bn,      # B row/col stride (K‑stride, N‑stride)
    stride_cm, stride_cn,      # C row/col stride (M‑stride, N‑stride)
    stride_batch,              # stride between consecutive batches (B‑axis)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    HIDDEN_DIM: tl.constexpr,  # hidden dimension (compile‑time constant)
):
    """
    Arguments:
        a_ptr, b_ptr, c_ptr : pointers to fp16 tensors in layout
                               [B, N, N, H] (H is innermost).
        N                   : sequence length.
        stride_*            : element‑wise strides (not bytes).
        stride_batch        : stride to jump from one batch to the next
                              (i.e. N * N * H).
        BLOCK_*             : tile sizes.
        HIDDEN_DIM          : hidden dimension (used to decode the second axis).
    """
    # --------------------------------------------------------------------- #
    #   Decode (batch, hidden) from the second program ID
    # --------------------------------------------------------------------- #
    batch_hidden = tl.program_id(axis=1)          # 0 … B*H‑1
    b = batch_hidden // HIDDEN_DIM                # batch index   ∈ [0, B)
    h = batch_hidden %  HIDDEN_DIM                # hidden index ∈ [0, H)

    # Base pointers for the current (b, h) slice
    a_base = a_ptr + b * stride_batch + h
    b_base = b_ptr + b * stride_batch + h
    c_base = c_ptr + b * stride_batch + h

    # --------------------------------------------------------------------- #
    #   Tile indices for the (i, j) plane
    # --------------------------------------------------------------------- #
    pid = tl.program_id(axis=0)                  # tile index over (M, N)
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    block_m = pid // num_blocks_n
    block_n = pid %  num_blocks_n

    offs_m = block_m * BLOCK_M + tl.arange(0, BLOCK_M)   # i‑offsets
    offs_n = block_n * BLOCK_N + tl.arange(0, BLOCK_N)   # j‑offsets

    # --------------------------------------------------------------------- #
    #   Accumulator (FP32) – Tensor‑core friendly
    # --------------------------------------------------------------------- #
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------------- #
    #   Main K‑loop (k is the contracted index)
    # --------------------------------------------------------------------- #
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_off = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_off < N                       # guard for partial tiles

        # ----------- Load A tile (M × K) -------------------------------- #
        a_ptrs = a_base + (offs_m[:, None] * stride_am) + (k_off[None, :] * stride_ak)
        a = tl.load(a_ptrs,
                    mask=(offs_m[:, None] < N) & mask_k[None, :],
                    other=0.0)                  # fp16 → fp32 via dot

        # ----------- Load B tile (K × N) -------------------------------- #
        b_ptrs = b_base + (k_off[:, None] * stride_bk) + (offs_n[None, :] * stride_bn)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & (offs_n[None, :] < N),
                    other=0.0)

        # ----------- Rank‑K update -------------------------------------- #
        acc += tl.dot(a, b)                     # fp16 × fp16 → fp32

    # --------------------------------------------------------------------- #
    #   Write back result (store as fp16)
    # --------------------------------------------------------------------- #
    c_ptrs = c_base + (offs_m[:, None] * stride_cm) + (offs_n[None, :] * stride_cn)
    tl.store(c_ptrs,
             acc.to(tl.float16),
             mask=(offs_m[:, None] < N) & (offs_n[None, :] < N))

# -----------------------------------------------------------------------------#

    """
    Forward pass of the outgoing TriMul block (no gradients).

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, D] (float32)
            - mask_tensor : torch.Tensor [B, N, N] (bool/float) – may be ignored
            - weights_dict : mapping of all module parameters (float32)
            - config_dict  : {"dim": D, "hidden_dim": H, "nomask": bool, ...}

    Returns:
        torch.Tensor of shape [B, N, N, D] (float32)
    """
    # --------------------------------------------------------------------- #
    #   Unpack inputs
    # --------------------------------------------------------------------- #
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    eps = 1e-5

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------------- #
    #   1) Layer‑norm on the feature dimension (full‑precision for stability)
    # --------------------------------------------------------------------- #
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim], float32

    # Work in fp16 for all subsequent cheap ops
    x_fp16 = x.half()

    # --------------------------------------------------------------------- #
    #   2) Cast model weights once (fp16)
    # --------------------------------------------------------------------- #
    left_proj_w   = weights["left_proj.weight"].half()
    right_proj_w  = weights["right_proj.weight"].half()
    left_gate_w   = weights["left_gate.weight"].half()
    right_gate_w  = weights["right_gate.weight"].half()
    out_gate_w    = weights["out_gate.weight"].half()
    to_out_norm_w = weights["to_out_norm.weight"].half()
    to_out_norm_b = weights["to_out_norm.bias"].half()
    to_out_w      = weights["to_out.weight"].half()

    # --------------------------------------------------------------------- #
    #   3) Linear projections (no bias) – stay in fp16
    # --------------------------------------------------------------------- #
    left  = F.linear(x_fp16, left_proj_w)   # [B, N, N, H]
    right = F.linear(x_fp16, right_proj_w)  # [B, N, N, H]

    # --------------------------------------------------------------------- #
    #   4) Gating (sigmoid) – fp16
    # --------------------------------------------------------------------- #
    left_gate  = torch.sigmoid(F.linear(x_fp16, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_fp16, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x_fp16, out_gate_w))

    # --------------------------------------------------------------------- #
    #   5) Optional mask (broadcast over hidden)
    # --------------------------------------------------------------------- #
    if not nomask:
        mask_f = mask.to(x_fp16.dtype).unsqueeze(-1)   # [B, N, N, 1]
        left  = left * mask_f
        right = right * mask_f

    # --------------------------------------------------------------------- #
    #   6) Apply per‑channel gates
    # --------------------------------------------------------------------- #
    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------------- #
    #   7) Core contraction via the Triton GEMM kernel
    # --------------------------------------------------------------------- #
    # Strides for tensors in native layout [B, N, N, H] (contiguous)
    stride_batch = N * N * hidden_dim            # distance between batches (axis 0)
    stride_i = N * hidden_dim                    # step when i (first N) increments
    stride_k = hidden_dim                        # step when k (second N) increments

    # A = left : rows=i, cols=k
    stride_am = stride_i
    stride_ak = stride_k

    # B = right : rows=k, cols=j (we treat it as K×N without transposition)
    stride_bk = stride_k
    stride_bn = stride_i

    # C = out : rows=i, cols=j
    stride_cm = stride_i
    stride_cn = stride_k

    # Output buffer (same layout as left/right, fp16)
    out_buf = torch.empty_like(left)

    # Tile sizes – tuned for H100 tensor‑core utilization
    BLOCK_M, BLOCK_N, BLOCK_K = 128, 128, 32

    # Grid: (tiles over (i,j) plane, total (batch * hidden) slices)
    num_m_blocks = (N + BLOCK_M - 1) // BLOCK_M
    num_n_blocks = (N + BLOCK_N - 1) // BLOCK_N
    grid = (num_m_blocks * num_n_blocks, B * hidden_dim)

    _trmul_gemm_kernel[grid](
        left,                               # a_ptr
        right,                              # b_ptr
        out_buf,                            # c_ptr
        N,                                  # seq_len
        stride_am, stride_ak,
        stride_bk, stride_bn,
        stride_cm, stride_cn,
        stride_batch,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        HIDDEN_DIM=hidden_dim,
        num_warps=8,                       # empirically good on H100
        num_stages=3,                      # double‑buffering + extra stage
    )

    # --------------------------------------------------------------------- #
    #   8) Hidden‑dim LayerNorm, out‑gate, final linear (still fp16)
    # --------------------------------------------------------------------- #
    out = F.layer_norm(
        out_buf,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)            # back to original dimension D (fp16)

    # --------------------------------------------------------------------- #
    #   9) Cast back to fp32 for downstream consumption
    # --------------------------------------------------------------------- #
    return out.float()
```
