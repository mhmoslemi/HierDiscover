"""
TriMul (outgoing) – fast Triton implementation
================================================

The forward pass consists of:

1.  Layer‑norm on the last dimension (use PyTorch’s highly‑optimised
    implementation – a single rubric, still a kernel is exercised later).

2.  Six linear layers (2 projections + 3 gates + final output projection).  
    To reduce memory traffic we fuse the two projection matrices
    (2 × hidden_dim) into one GEMM and the three gate matrices (3 × hidden_dim)
    into a second GEMM.  All computations are performed in FP16 and the
    results are kept in FP16 until they are needed in FP32.

3.  Optional binary mask is broadcast over the hidden dimension.

4.  Gated element‑wise products:
        left = left·σ(left_gate)   ,   right = right·σ(right_gate)

5.  The expensive N³ contraction  
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]  
    is a batched matrix‑multiplication per head:
        out_head = left_head @ right_headᵀ .
    We launch a custom Triton kernel that processes all
    (batch × head) matrices in parallel.  The kernel works on FP16
    inputs and accumulates in FP32, then stores the result back as FP16.

6.  Layer‑norm over the hidden dimension, element‑wise multiplication with the
    output gate, and the final linear projection back to the model dimension.

The only user‑defined kernel is the *batched GEMM* (step 5); everything else
uses fast PyTorch primitives.  Block sizes are chosen to minimise launch
overhead (128 × 128 tiles, K‑tile = 32) while fitting comfortably into the
64 KB shared‑memory budget of an H100 SM.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Batched GEMM kernel – computes C = A @ Bᵀ for many independent
# matrices (batch dimension).  All tensors are row‑major.
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    a_ptr,                   # left  matrix   [batch, M, K]   (FP16)
    b_ptr,                   # right matrix   [batch, K, N]   (FP16) – we pass the **un‑transposed** tensor and swap strides
    c_ptr,                   # output matrix  [batch, M, N]   (FP16)
    M, N, K,                 # matrix sizes
    # A strides
    stride_am, stride_ak, stride_ab,
    # B strides – note that stride_bk corresponds to the *K* (reduction) axis,
    #            stride_bn to the *N* (output‑column) axis.
    stride_bk, stride_bn, stride_bb,
    # C strides
    stride_cm, stride_cn, stride_cb,
    # tile sizes
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(0)
    pid_m    = tl.program_id(1)
    pid_n    = tl.program_id(2)

    # ------------------------------------------------------------------
    # Offsets for the current tile
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # ------------------------------------------------------------------
    # Base pointers for this batch
    # ------------------------------------------------------------------
    a_batch_ptr = a_ptr + batch_id * stride_ab
    b_batch_ptr = b_ptr + batch_id * stride_bb
    c_batch_ptr = c_ptr + batch_id * stride_cb

    # ------------------------------------------------------------------
    # Accumulator (FP32 for numeric stability)
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K dimension (blocked)
    # ------------------------------------------------------------------
    num_k_blocks = tl.cdiv(K, BLOCK_K)
    for k in range(num_k_blocks):
        k_base = k * BLOCK_K
        offs_k = tl.arange(0, BLOCK_K)
        mask_k = (k_base + offs_k) < K

        # Load A tile (M×K)
        a = tl.load(
            a_batch_ptr
            + offs_m[:, None] * stride_am
            + (k_base + offs_k)[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # Load B tile (K×N) – note swapped strides (see docstring)
        b = tl.load(
            b_batch_ptr
            + (k_base + offs_k)[:, None] * stride_bk
            + offs_n[None, :] * stride_bn,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # Accumulate
        acc += tl.dot(a, b)

    # ------------------------------------------------------------------
    # Write back (store as FP16)
    # ------------------------------------------------------------------
    c = acc.to(tl.float16)
    tl.store(
        c_batch_ptr
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn,
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )


# ----------------------------------------------------------------------
# Entry point required by the test harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (AlphaFold‑3).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor   [B, L, L, D]   (float32)
        - mask         : torch.Tensor   [B, L, L]   or None
        - weights      : dict of torch tensors (model parameters)
        - config       : dict with keys ``dim`` (D) and ``hidden_dim`` (H);
                        optional ``nomask`` (bool).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, L, L, D] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype  = input_tensor.dtype          # should be torch.float32

    bs, L, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1️⃣  Input LayerNorm – use the highly‑optimised torch implementation
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 2️⃣  Cast to FP16 for the dense work
    # ------------------------------------------------------------------
    x_h = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣  Fuse the two projection matrices (2 × hidden_dim)
    # ------------------------------------------------------------------
    proj_weight = torch.cat(
        [weights["left_proj.weight"], weights["right_proj.weight"]], dim=0
    ).to(device, torch.float16).contiguous()            # shape (2*H, D)
    proj_out = F.linear(x_h, proj_weight)               # shape [B, L, L, 2*H]
    left, right = torch.split(proj_out, hidden_dim, dim=-1)

    # ------------------------------------------------------------------
    # 4️⃣  Fuse the three gate matrices (3 × hidden_dim)
    # ------------------------------------------------------------------
    gate_weight = torch.cat(
        [
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(device, torch.float16).contiguous()          # shape (3*H, D)
    gate_out = F.linear(x_h, gate_weight)              # shape [B, L, L, 3*H]
    left_gate_raw, right_gate_raw, out_gate_raw = torch.split(
        gate_out, hidden_dim, dim=-1
    )
    left_gate = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)
    out_gate = torch.sigmoid(out_gate_raw)             # shape [B, L, L, H]

    # ------------------------------------------------------------------
    # 5️⃣  Apply optional binary mask (broadcast over the hidden dim)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_h = mask.to(device, torch.float16).unsqueeze(-1)   # [B, L, L, 1]
        left = left * mask_h
        right = right * mask_h

    # ------------------------------------------------------------------
    # 6️⃣  Apply per‑head gates
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 7️⃣  Prepare tensors for the batched GEMM
    # ------------------------------------------------------------------
    #   left, right : [B, L, L, H]   ->   (batch = B*H, M = L, K = L)
    left_perm = left.permute(0, 3, 1, 2).contiguous()    # [B, H, L, L]
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # [B, H, L, L]

    batch = bs * hidden_dim
    left_flat = left_perm.view(batch, L, L)              # (batch, M, K)
    right_flat = right_perm.view(batch, L, L)            # (batch, M, K)

    # --------------------------------------------------------------
    # No explicit transpose of `right`; we simply swap the row/col
    # strides when launching the kernel (see kernel comment).
    # --------------------------------------------------------------

    # Allocate output for the batched GEMM (FP16)
    out_flat_h = torch.empty_like(left_flat, dtype=torch.float16, device=device)

    # Strides (batch, row, col) for the three tensors
    a_stride_ab, a_stride_am, a_stride_ak = left_flat.stride()      # (batch, row, col)
    b_stride_bb, b_stride_bk, b_stride_bn = right_flat.stride()    # (batch, row, col)

    # For B we need:  stride_bk = stride over *K* (original col stride)
    #                stride_bn = stride over *N* (original row stride)
    stride_bk = b_stride_bn          # original col stride == 1
    stride_bn = b_stride_bk          # original row stride == L
    # (Note: after a contiguous view, right_flat.stride() == (L*L, L, 1))

    c_stride_cb, c_stride_cm, c_stride_cn = out_flat_h.stride()

    # ------------------------------------------------------------------
    # 8️⃣  Launch the batched GEMM Triton kernel
    # ------------------------------------------------------------------
    # Choose tile sizes – 128×128 works well up to L = 1024.
    BLOCK_M, BLOCK_N, BLOCK_K = 128, 128, 32
    grid = (
        batch,
        triton.cdiv(L, BLOCK_M),
        triton.cdiv(L, BLOCK_N),
    )
    batched_gemm_kernel[grid](
        a_ptr=left_flat,
        b_ptr=right_flat,
        c_ptr=out_flat_h,
        M=L,
        N=L,
        K=L,
        stride_am=a_stride_am,
        stride_ak=a_stride_ak,
        stride_ab=a_stride_ab,
        stride_bk=stride_bk,
        stride_bn=stride_bn,
        stride_bb=b_stride_bb,
        stride_cm=c_stride_cm,
        stride_cn=c_stride_cn,
        stride_cb=c_stride_cb,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 9️⃣  Reshape back to [B, L, L, H] and cast to FP32
    # ------------------------------------------------------------------
    out_fp32 = out_flat_h.to(torch.float32)
    out = out_fp32.view(bs, hidden_dim, L, L).permute(0, 2, 3, 1).contiguous()  # [B, L, L, H]

    # ------------------------------------------------------------------
    # 🔟  Output LayerNorm over the hidden dimension (torch impl)
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 1️⃣1️⃣  Apply output gate and final projection back to model dim
    # ------------------------------------------------------------------
    out = out * out_gate.to(dtype)                     # broadcast over H
    out = F.linear(out, weights["to_out.weight"])      # [B, L, L, D]  (FP32)

    return out
