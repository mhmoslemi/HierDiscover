def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    eps = 1e-5

    # Input LayerNorm using Triton (flatten)
    M = bs * N * N
    x_flat = input_tensor.reshape(M, dim)
    norm_weight = weights["norm.weight"].to(device, dtype)
    norm_bias = weights["norm.bias"].to(device, dtype)
    x_norm_flat = triton_layernorm(x_flat, norm_weight, norm_bias, eps=eps)
    x_norm = x_norm_flat.view(bs, N, N, dim)

    # Cast to fp16 for heavy parts
    x_norm_f16 = x_norm.to(torch.float16)

    # Concatenate linear weights for left/right + gates
    # We need left_proj, left_gate, right_proj, right_gate each weight shape (hidden_dim, dim)
    left_proj_w = weights["left_proj.weight"].to(device, torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device, torch.float16)
    left_gate_w = weights["left_gate.weight"].to(device, torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device, torch.float16)

    # concat order: left_proj, left_gate, right_proj, right_gate
    weight_all = torch.cat([left_proj_w, left_gate_w, right_proj_w, right_gate_w], dim=0)  # shape (4*H, dim)
    # Perform a single linear projection
    proj_all = torch.nn.functional.linear(x_norm_f16, weight_all)  # shape (B,N,N,4*H)

    # split
    H = hidden_dim
    left_proj = proj_all[..., :H]
    left_gate_pre = proj_all[..., H:2*H]
    right_proj = proj_all[..., 2*H:3*H]
    right_gate_pre = proj_all[..., 3*H:4*H]

    # apply sigmoid to gates
    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)

    # Mask if needed
    if not nomask and mask is not None:
        mask_f16 = mask.to(device, torch.float16).unsqueeze(-1)
        left_proj = left_proj * mask_f16
        right_proj = right_proj * mask_f16

    # Apply elementwise gating
    left = left_proj * left_gate
    right = right_proj * right_gate

    # Compute out_gate
    out_gate_w = weights["out_gate.weight"].to(device, torch.float16)
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm_f16, out_gate_w))
    # out_gate shape (B,N,N,H)

    # Permute for bmm
    left_perm = left.permute(0, 3, 1, 2).contiguous()  # (B,H,N,N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    BH = bs * hidden_dim
    left_mat = left_perm.view(BH, N, N)
    right_mat = right_perm.view(BH, N, N)

    # bmm: (B*H, N, N) x (B*H, N, N).transpose -> (B*H,N,N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1,2))

    # reshape back
    out = out_mat.view(bs, hidden_dim, N, N).permute(0,2,3,1).contiguous()  # shape (B,N,N,H) half
    # LayerNorm output (float32)
    out_fp32 = out.to(torch.float32)
    out_flat = out_fp32.reshape(M, hidden_dim)  # (M, H)
    to_out_norm_w = weights["to_out_norm.weight"].to(device, torch.float32)
    to_out_norm_b = weights["to_out_norm.bias"].to(device, torch.float32)
    out_norm_flat = triton_layernorm(out_flat, to_out_norm_w, to_out_norm_b, eps=eps)
    out_norm = out_norm_flat.view(bs, N, N, hidden_dim)  # float32

    # Apply out_gate (float32)
    gated = out_norm * out_gate.to(torch.float32)  # elementwise

    # Final projection
    to_out_w = weights["to_out.weight"].to(device, torch.float32)
    output = torch.nn.functional.linear(gated, to_out_w)  # shape (B,N,N,dim)
    return output

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# Triton kernel for layernorm (2-pass)
@triton.jit
def layernorm_kernel(
    x_ptr, y_ptr,
    weight_ptr, bias_ptr,
    M, N, eps,
    stride_xm, stride_xn,
    stride_ym, stride_yn,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
):
    pid = tl.program_id(0)
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # First pass: compute mean and variance for each row
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_N)
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        x_f32 = x.to(tl.float32)
        sum_val += tl.sum(x_f32, axis=1)
        sum_sq += tl.sum(x_f32 * x_f32, axis=1)

    mean = sum_val / N
    var = sum_sq / N - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # Second pass: write normalized output with affine parameters
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        x_f32 = x.to(tl.float32)
        norm = (x_f32 - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_n, mask=mask_n, other=0.0)
        b = tl.load(bias_ptr + offs_n, mask=mask_n, other=0.0)

        y = norm * w[None, :] + b[None, :]

        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn,
            y,
            mask=mask_m[:, None] & mask_n[None, :],
        )

def triton_layernorm(x: torch.Tensor, weight: torch.Tensor, bias: torch.Tensor, eps: float = 1e-5) -> torch.Tensor:
    """
    Apply LayerNorm over the last dimension of a 2‑D tensor (M, N) using Triton.
    """
    assert x.dim() == 2, "expect a 2‑D tensor"
    M, N = x.shape
    y = torch.empty_like(x)

    # Ensure contiguous for stride calculations
    x = x.contiguous()
    y = y.contiguous()
    weight = weight.contiguous()
    bias = bias.contiguous()

    stride_xm, stride_xn = x.stride()
    stride_ym, stride_yn = y.stride()

    grid = lambda meta: (triton.cdiv(M, meta["BLOCK_M"]),)

    layernorm_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        weight_ptr=weight,
        bias_ptr=bias,
        M=M,
        N=N,
        eps=eps,
        stride_xm=stride_xm,
        stride_xn=stride_xn,
        stride_ym=stride_ym,
        stride_yn=stride_yn,
        BLOCK_M=128,
        BLOCK_N=64,
    )
    return y

def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the TriMul “outgoing” operator.

    The implementation follows the reference PyTorch code but fuses the
    four linear projections (left/right + their gates) into a single GEMM,
    reduces memory traffic, and uses a custom Triton LayerNorm for the
    two normalisation steps.  All heavy arithmetic is performed in FP16
    (Tensor‑core friendly) while the final output remains FP32.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (Triton) – flatten the (B,N,N) leading axes
    # ------------------------------------------------------------------
    M = bs * N * N
    x_flat = input_tensor.reshape(M, dim)
    norm_w = weights["norm.weight"].to(device, input_tensor.dtype)
    norm_b = weights["norm.bias"].to(device, input_tensor.dtype)
    x_norm_flat = triton_layernorm(x_flat, norm_w, norm_b, eps=eps)
    x_norm = x_norm_flat.view(bs, N, N, dim)  # (B,N,N,D) float32

    # ------------------------------------------------------------------
    # Cast to FP16 for the compute‑intensive part
    # ------------------------------------------------------------------
    x_norm_f16 = x_norm.to(torch.float16)

    # ------------------------------------------------------------------
    # 2️⃣ Fuse the four linear projections (left/right + gates) into one GEMM
    # ------------------------------------------------------------------
    # Prepare concatenated weight matrix: [left_proj; left_gate; right_proj; right_gate]
    left_proj_w = weights["left_proj.weight"].to(device, torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device, torch.float16)
    left_gate_w = weights["left_gate.weight"].to(device, torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device, torch.float16)

    # Shape (4*H, dim)
    fused_weight = torch.cat([left_proj_w, left_gate_w, right_proj_w, right_gate_w], dim=0)

    # Single linear call – result shape (B,N,N,4*H)
    proj_all = torch.nn.functional.linear(x_norm_f16, fused_weight)

    H = hidden_dim
    left_proj = proj_all[..., :H]
    left_gate_pre = proj_all[..., H:2*H]
    right_proj = proj_all[..., 2*H:3*H]
    right_gate_pre = proj_all[..., 3*H:4*H]

    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)

    # ------------------------------------------------------------------
    # 3️⃣ Apply mask (if present) to the raw projections only
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_f16 = mask.to(device, torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left_proj = left_proj * mask_f16
        right_proj = right_proj * mask_f16

    # ------------------------------------------------------------------
    # 4️⃣ Element‑wise gate application
    # ------------------------------------------------------------------
    left = left_proj * left_gate
    right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Compute the out‑gate (separate linear)
    # ------------------------------------------------------------------
    out_gate_w = weights["out_gate.weight"].to(device, torch.float16)
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm_f16, out_gate_w))

    # ------------------------------------------------------------------
    # 6️⃣ Core O(N³·H) contraction via batched GEMM (Tensor Cores)
    # ------------------------------------------------------------------
    # (B,N,N,H) -> (B*H, N, N) layout required by torch.bmm
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B,H,N,N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    BH = bs * hidden_dim
    left_mat = left_perm.view(BH, N, N)                # (B*H, N, N)
    right_mat = right_perm.view(BH, N, N)

    # Batched matrix multiplication across the third axis (k)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*H, N, N)

    # Restore (B,N,N,H) layout; keep in FP16 for the next norm
    out = out_mat.view(bs, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # ------------------------------------------------------------------
    # 7️⃣ Output LayerNorm (Triton) – work in FP32 for numerical stability
    # ------------------------------------------------------------------
    out_fp32 = out.to(torch.float32)
    out_flat = out_fp32.reshape(M, hidden_dim)                # (B*N*N, H)

    to_out_norm_w = weights["to_out_norm.weight"].to(device, torch.float32)
    to_out_norm_b = weights["to_out_norm.bias"].to(device, torch.float32)

    out_norm_flat = triton_layernorm(out_flat, to_out_norm_w, to_out_norm_b, eps=eps)
    out_norm = out_norm_flat.view(bs, N, N, hidden_dim)      # (B,N,N,H) FP32

    # ------------------------------------------------------------------
    # 8️⃣ Apply out‑gate and final linear projection back to model dim
    # ------------------------------------------------------------------
    gated = out_norm * out_gate.to(torch.float32)            # (B,N,N,H) FP32

    to_out_w = weights["to_out.weight"].to(device, torch.float32)  # (dim, H)
    output = torch.nn.functional.linear(gated, to_out_w)            # (B,N,N,dim) FP32

    return output

"""
TriMul “outgoing” forward pass – Triton‑accelerated.

Algorithm
---------
1. Input LayerNorm is executed with a custom 2‑pass Triton kernel
   (mean/variance → normalisation → affine).
2. All five linear projections (left, right, their two gates and the
   output gate) are fused into a single FP16 GEMM:
        [left_proj; left_gate; right_proj; right_gate; out_gate] @ x
   This reduces memory traffic by reading the input only once.
3. The mask (if present) is applied only to the raw left/right
   projections, then element‑wise gates are applied.
4. The core O(N³·H) contraction is performed as a batched half‑precision
   GEMM (`torch.bmm`) which maps perfectly onto H100 Tensor‑Cores.
5. A second Triton LayerNorm normalises the contraction result in FP32.
6. The result is scaled by the output gate and projected back to the
   model dimension with a final FP32 linear layer.

All heavy arithmetic runs in FP16 (Tensor‑Core friendly) while the two
LayerNorms and the final projection stay in FP32 for numerical stability.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton 2‑pass LayerNorm kernel (operates on a 2‑D tensor)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr, y_ptr,               # *ptrs input / output [M, N]
    weight_ptr, bias_ptr,       # *ptrs affine parameters [N]
    M, N, eps,                  # dimensions + epsilon
    stride_xm, stride_xn,       # input strides
    stride_ym, stride_yn,       # output strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
):
    pid = tl.program_id(0)

    # ------------------------------------------------------------------
    # Row range for this program
    # ------------------------------------------------------------------
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # ------------------------------------------------------------------
    # First pass: compute per‑row mean & variance
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_N)
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N

        # Load a tile (as half) and up‑cast to fp32 for accumulation
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        x_f32 = x.to(tl.float32)

        sum_val += tl.sum(x_f32, axis=1)
        sum_sq  += tl.sum(x_f32 * x_f32, axis=1)

    mean = sum_val / N
    var  = sum_sq / N - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # ------------------------------------------------------------------
    # Second pass: write normalised + affine transformed values
    # ------------------------------------------------------------------
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N

        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        x_f32 = x.to(tl.float32)

        # normalise
        norm = (x_f32 - mean[:, None]) * rstd[:, None]

        # affine
        w = tl.load(weight_ptr + offs_n, mask=mask_n, other=0.0)
        b = tl.load(bias_ptr   + offs_n, mask=mask_n, other=0.0)

        y = norm * w[None, :] + b[None, :]

        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn,
            y,
            mask=mask_m[:, None] & mask_n[None, :],
        )


def triton_layernorm(x: torch.Tensor,
                     weight: torch.Tensor,
                     bias: torch.Tensor,
                     eps: float = 1e-5) -> torch.Tensor:
    """
    LayerNorm over the last dimension of a 2‑D tensor (M, N) using Triton.
    The input is expected to be contiguous.
    """
    assert x.dim() == 2, "triton_layernorm expects a 2‑D tensor"
    M, N = x.shape
    y = torch.empty_like(x)

    x = x.contiguous()
    y = y.contiguous()
    weight = weight.contiguous()
    bias = bias.contiguous()

    stride_xm, stride_xn = x.stride()
    stride_ym, stride_yn = y.stride()

    grid = lambda meta: (triton.cdiv(M, meta["BLOCK_M"]),)

    layernorm_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        weight_ptr=weight,
        bias_ptr=bias,
        M=M,
        N=N,
        eps=eps,
        stride_xm=stride_xm,
        stride_xn=stride_xn,
        stride_ym=stride_ym,
        stride_yn=stride_yn,
        BLOCK_M=128,          # rows per program (tuned for H100)
        BLOCK_N=64,           # columns per tile
    )
    return y


# ----------------------------------------------------------------------
# Triton‑accelerated TriMul “outgoing” entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, D] (float32)
        - mask         : torch.Tensor of shape [B, N, N] or None
        - weights      : dict of model parameters (LayerNorm weights,
                         linear weights, etc.)
        - config       : dict with keys ``dim``, ``hidden_dim``, ``nomask``

    Returns
    -------
    torch.Tensor of shape [B, N, N, D] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (Triton) – flatten leading axes
    # ------------------------------------------------------------------
    M = bs * N * N
    x_flat = input_tensor.reshape(M, dim)                       # (M, D)
    norm_w = weights["norm.weight"].to(device, input_tensor.dtype)
    norm_b = weights["norm.bias"].to(device, input_tensor.dtype)
    x_norm_flat = triton_layernorm(x_flat, norm_w, norm_b, eps=eps)
    x_norm = x_norm_flat.view(bs, N, N, dim)                    # (B, N, N, D) float32

    # ------------------------------------------------------------------
    # Cast to FP16 for the compute‑intensive part
    # ------------------------------------------------------------------
    x_norm_f16 = x_norm.to(torch.float16)

    # ------------------------------------------------------------------
    # 2️⃣ Fuse *all* linear projections (left/right + 2 gates + out_gate)
    # ------------------------------------------------------------------
    # Convert each weight to FP16 and concatenate along the output dim
    left_proj_w   = weights["left_proj.weight"].to(device, torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device, torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device, torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device, torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device, torch.float16)

    # order: left_proj, left_gate, right_proj, right_gate, out_gate
    fused_weight = torch.cat(
        [left_proj_w, left_gate_w, right_proj_w, right_gate_w, out_gate_w],
        dim=0,
    )                                                            # (5*H, D)

    # Single fused linear: (B,N,N,5*H)
    proj_all = F.linear(x_norm_f16, fused_weight)

    H = hidden_dim
    left_proj       = proj_all[..., :H]                         # (B,N,N,H)
    left_gate_pre   = proj_all[..., H:2*H]
    right_proj      = proj_all[..., 2*H:3*H]
    right_gate_pre  = proj_all[..., 3*H:4*H]
    out_gate_pre    = proj_all[..., 4*H:5*H]

    # Apply sigmoid to gate tensors
    left_gate  = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)
    out_gate   = torch.sigmoid(out_gate_pre)                     # (B,N,N,H)

    # ------------------------------------------------------------------
    # 3️⃣ Apply mask (if present) to the *raw* projections only
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_f16 = mask.to(device, torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left_proj  = left_proj * mask_f16
        right_proj = right_proj * mask_f16

    # ------------------------------------------------------------------
    # 4️⃣ Element‑wise gating
    # ------------------------------------------------------------------
    left  = left_proj * left_gate      # (B,N,N,H) fp16
    right = right_proj * right_gate    # (B,N,N,H) fp16

    # ------------------------------------------------------------------
    # 5️⃣ Core O(N³·H) contraction via batched half‑precision GEMM
    # ------------------------------------------------------------------
    # Rearrange to (B*H, N, N) layout required by torch.bmm
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B,H,N,N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # (B,H,N,N)

    BH = bs * hidden_dim
    left_mat  = left_perm.view(BH, N, N)                 # (B*H, N, N) fp16
    right_mat = right_perm.view(BH, N, N)                # (B*H, N, N) fp16

    # Batched matmul: (i,k)·(j,k)ᵀ → sum_k
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B*H, N, N) fp16

    # Restore (B,N,N,H) layout
    out = out_mat.view(bs, hidden_dim, N, N) \
                .permute(0, 2, 3, 1).contiguous()   # (B,N,N,H) fp16

    # ------------------------------------------------------------------
    # 6️⃣ Output LayerNorm (Triton) – work in FP32 for stability
    # ------------------------------------------------------------------
    out_fp32 = out.to(torch.float32)
    out_flat = out_fp32.reshape(M, hidden_dim)          # (B*N*N, H)

    to_out_norm_w = weights["to_out_norm.weight"].to(device, torch.float32)
    to_out_norm_b = weights["to_out_norm.bias"].to(device, torch.float32)

    out_norm_flat = triton_layernorm(out_flat, to_out_norm_w, to_out_norm_b, eps=eps)
    out_norm = out_norm_flat.view(bs, N, N, hidden_dim)  # (B,N,N,H) FP32

    # ------------------------------------------------------------------
    # 7️⃣ Apply output gate and final projection back to model dim
    # ------------------------------------------------------------------
    gated = out_norm * out_gate.to(torch.float32)        # (B,N,N,H) FP32

    to_out_w = weights["to_out.weight"].to(device, torch.float32)   # (D, H)
    output = F.linear(gated, to_out_w)                              # (B,N,N,D) FP32

    return output
