"""
TriMul (outgoing) – Triton‑accelerated implementation
====================================================

The outgoing triangle multiplicative update computes

    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

where *left* and *right* are linear projections of a LayerNormed input
tensor and are modulated by sigmoid gates (and an optional mask).  The
contraction over the sequence dimension *k* is a batched matrix
multiplication for every (batch, hidden) channel:

    out_c = left_c @ right_cᵀ        (c = b·hidden_dim)

A custom Triton kernel fuses the GEMM for all channels, works in
FP16→FP32 (Tensor‑core friendly) and is launched with a 128×128×64
tile (8 warps, 2 stages).  All remaining operations (LayerNorms,
linear projections, gates, final projection) are performed with regular
PyTorch primitives.

The function `custom_kernel` conforms to the required signature:
    (input_tensor, mask, weights, config) → output tensor of shape
    [B, N, N, dim] (float32).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def tri_mul_kernel(
    out_ptr,                     # float32*   output    (C, M, N)
    a_ptr,                       # float16*   left      (C, M, K)
    b_ptr,                       # float16*   rightᵀ    (C, K, N)
    M, N, K,                     # int64      matrix dimensions (all = seq_len)
    stride_out_m, stride_out_n, stride_out_c,   # strides for out
    stride_a_m, stride_a_k, stride_a_c,        # strides for left
    stride_b_k, stride_b_n, stride_b_c,        # strides for right (K,N order)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM: out_c = left_c @ right_cᵀ for every channel c.

    Each program instance processes a tile (BLOCK_M, BLOCK_N) of the
    output matrix for a given channel.  FP16 data are multiplied and
    accumulated in FP32.
    """
    pid_m = tl.program_id(0)   # tile row   → i
    pid_n = tl.program_id(1)   # tile col   → j
    pid_c = tl.program_id(2)   # channel id → batch*hidden_dim

    # ---------- tile offsets ----------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # ---------- accumulator ----------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---------- loop over K dimension ----------
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = kk * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # pointers for the current tiles of left (M×K) and right (K×N)
        a_tile_ptr = (
            a_ptr
            + pid_c * stride_a_c
            + offs_m[:, None] * stride_a_m
            + offs_k[None, :] * stride_a_k
        )
        b_tile_ptr = (
            b_ptr
            + pid_c * stride_b_c
            + offs_k[:, None] * stride_b_k   # K dimension (fast)
            + offs_n[None, :] * stride_b_n   # N dimension (slow)
        )

        a = tl.load(
            a_tile_ptr,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        b = tl.load(
            b_tile_ptr,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # GEMM accumulation (FP16·FP16 → FP32)
        acc += tl.dot(a, b)

    # ---------- write back ----------
    out_tile_ptr = (
        out_ptr
        + pid_c * stride_out_c
        + offs_m[:, None] * stride_out_m
        + offs_n[None, :] * stride_out_n
    )
    tl.store(
        out_tile_ptr,
        acc,
        mask=mask_m[:, None] & mask_n[None, :],
    )


def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor[B, N, N, dim] (float32)
        - mask         : torch.Tensor[B, N, N] (bool/float) or None
        - weights      : dict of torch.Tensors (model parameters)
        - config       : dict with keys
            * "dim"        – input channel dimension C
            * "hidden_dim" – hidden dimension H
            * "nomask"     – bool, True → ignore mask

    Returns
    -------
    torch.Tensor[B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # 1) unpack arguments
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # --------------------------------------------------------------
    # 2) Layer‑norm the input (float32)
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, C]  FP32

    # --------------------------------------------------------------
    # 3) Cast to FP16 for the cheap linear ops
    x_h = x.to(torch.float16)

    # --------------------------------------------------------------
    # 4) Linear projections (bias=False)
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left = F.linear(x_h, left_proj_w)   # [B, N, N, H]
    right = F.linear(x_h, right_proj_w)  # [B, N, N, H]

    # --------------------------------------------------------------
    # 5) Gates (sigmoid)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    left_gate = torch.sigmoid(F.linear(x_h, left_gate_w))   # [B, N, N, H]
    right_gate = torch.sigmoid(F.linear(x_h, right_gate_w))  # [B, N, N, H]
    out_gate = torch.sigmoid(F.linear(x_h, out_gate_w))   # [B, N, N, H]

    # --------------------------------------------------------------
    # 6) Optional mask (applied before gating as in the reference)
    if not nomask and mask is not None:
        # convert mask to {0,1} float16 and broadcast
        mask_f = mask.to(torch.float16).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # --------------------------------------------------------------
    # 7) Apply element‑wise gates
    left = left * left_gate
    right = right * right_gate

    # free intermediate tensors that will not be used again
    del left_gate, right_gate, x_h, x

    # --------------------------------------------------------------
    # 8) Prepare tensors for the Triton batched GEMM
    B, N, _, _ = left.shape                       # B, N, N, H
    # move hidden dim before the two sequence dims for a contiguous layout
    left = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    C_total = B * hidden_dim                       # number of independent GEMMs
    left_flat = left.reshape(C_total, N, N)        # (C, M, K)
    right_flat = right.reshape(C_total, N, N)      # (C, K, N) – we will load it transposed

    # Output buffer – FP32 accumulator as required by the reference
    out_flat = torch.empty_like(left_flat, dtype=torch.float32)

    # --------------------------------------------------------------
    # 9) Triton kernel launch configuration
    # Tile size: 128×128 with K‑tile 64 works well on H100.  We adapt
    # a smaller tile for very small N to avoid launching many empty threads.
    if N <= 128:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 32
    else:
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 64

    stride_out_c = out_flat.stride(0)
    stride_out_m = out_flat.stride(1)
    stride_out_n = out_flat.stride(2)

    stride_a_c = left_flat.stride(0)
    stride_a_m = left_flat.stride(1)
    stride_a_k = left_flat.stride(2)

    stride_b_c = right_flat.stride(0)
    stride_b_k = right_flat.stride(2)  # fast‑moving K dimension
    stride_b_n = right_flat.stride(1)  # slow‑moving N dimension

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # blocks along M (i)
        (N + BLOCK_N - 1) // BLOCK_N,   # blocks along N (j)
        C_total,                        # one program per channel
    )

    # Launch the Triton kernel
    tri_mul_kernel[grid](
        out_flat,
        left_flat,
        right_flat,
        N, N, N,                     # M, N, K
        stride_out_m,
        stride_out_n,
        stride_out_c,
        stride_a_m,
        stride_a_k,
        stride_a_c,
        stride_b_k,
        stride_b_n,
        stride_b_c,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=2,                # double‑buffered loads
    )

    # --------------------------------------------------------------
    # 10) Reshape back to (B, N, N, H)
    out = out_flat.reshape(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, H]

    # --------------------------------------------------------------
    # 11) Post‑processing: layer‑norm over hidden dim, out‑gate, final projection
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    # Multiply with the output gate (cast gate to FP32 for correct dtype)
    out = out * out_gate.to(out.dtype)

    # Final linear projection back to the original dimension (use FP16 for speed)
    to_out_w = weights["to_out.weight"].to(torch.float16)
    out = F.linear(out.to(torch.float16), to_out_w).to(torch.float32)

    return out
```

**Explanation of the workflow**

1. **LayerNorm** – normalises the input per channel (`dim`).
2. **Linear projections** – compute `left` and `right` (FP16) using the
   provided projection weights.
3. **Sigmoid gates** – compute `left_gate`, `right_gate`, `out_gate`.
4. **Mask** – if a mask is supplied it is applied to `left` and `right`
   before gating.
5. **Gate application** – element‑wise multiplication with the
   corresponding gates.
6. **Batched GEMM** – the costly contraction over the sequence dimension
   is performed by `tri_mul_kernel`, a fused GEMM for every
   (batch × hidden) channel.  The kernel works in FP16→FP32 for maximal
   Tensor‑core utilisation.
7. **Reshape** – bring the result back to the original 4‑D layout.
8. **Output normalisation & gating** – LayerNorm on the hidden dimension,
   multiplication by `out_gate`.
9. **Final projection** – linear layer back to the original `dim`
   (performed in FP16 then cast to FP32).

All heavy arithmetic (the O(N³) contraction) is executed inside Triton,
yielding a fast forward pass while preserving exact functional
behaviour of the reference PyTorch implementation.
