```python
"""TriMul outgoing implementation using Triton.

The kinetic part (the only N^3 operation) is a batched matrix multiplication:
    out[c, i, j] = Σ_k left[c, i, k] * right[c, j, k]
where c index folds the batch and hidden dimension (C_total = B * H).
All tensors are in FP16, accumulation in FP32, and the kernel uses
a 2‑D tile (BLOCK_M × BLOCK_N) and a 1‑D reduction tile (BLOCK_K). The
kernel is launched with a configurable number of warps and stages to
maximize occupancy on H100.

All other operations (LayerNorms, linear projections, sigmoid gates,
masking, final projection) are performed in PyTorch. This satisfies
the requirement of having at least part of the work done in a Triton
kernel while keeping the implementation straightforward and correct
for all supported shapes.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def tri_mul_gemm_kernel(
    out_ptr,           # *fp32 -> output tensor (C, M, N)
    a_ptr,             # *fp16 -> left tensor   (C, M, K)
    b_ptr,             # *fp16 -> right tensor  (C, K, N) (stored as (C, N, K))
    M, N, K,           # int64 sizes (all equal to seq_len)
    stride_out_c, stride_out_m, stride_out_n,
    stride_a_c, stride_a_m, stride_a_k,
    stride_b_c, stride_b_k, stride_b_n,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM used in TriMul.

    Computes:
        out_c[M,N] += left_c[M,K] @ right_c[K,N]
    for each channel `c` = batch * hidden.
    """
    # program ids
    pid_c = tl.program_id(2)  # channel
    pid_m = tl.program_id(0)  # block row
    pid_n = tl.program_id(1)  # block col

    # block offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for OOB rows/cols
    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator (FP32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -----------------------------------------------------------------
    # Loop over K dimension in tiles of size BLOCK_K
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # pointers
        a_tile_ptr = a_ptr + pid_c * stride_a_c \
                     + offs_m[:, None] * stride_a_m \
                     + offs_k[None, :] * stride_a_k      # shape (M, K)

        b_tile_ptr = b_ptr + pid_c * stride_b_c \
                     + offs_k[:, None] * stride_b_k \
                     + offs_n[None, :] * stride_b_n      # shape (K, N)

        # masked loads
        a = tl.load(a_tile_ptr,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)  # fp16
        b = tl.load(b_tile_ptr,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)  # fp16

        # matrix multiplication tile
        acc += tl.dot(a, b)  # fp16*fp16 -> fp32

    # -----------------------------------------------------------------
    # Store result
    out_tile_ptr = out_ptr + pid_c * stride_out_c \
                   + offs_m[:, None] * stride_out_m \
                   + offs_n[None, :] * stride_out_n

    tl.store(out_tile_ptr, acc,
             mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (AlphaFold3).

    Args:
        data (tuple): (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32)
            - mask: optional torch.Tensor of shape [B, N, N] (float32/bool) or None
            - weights: dict with the following keys
                * "norm.weight", "norm.bias"
                * "left_proj.weight", "right_proj.weight"
                * "left_gate.weight", "right_gate.weight", "out_gate.weight"
                * "to_out_norm.weight", "to_out_norm.bias"
                * "to_out.weight"
            - config: dict with keys
                * "dim"        – input channel dimension C
                * "hidden_dim"– hidden dimension H
                * "nomask"     – bool, if True ignore mask (default True)

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32).
    """
    # -----------------------------------------------------------------
    # Unpack inputs
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    device = input_tensor.device

    # -----------------------------------------------------------------
    # 1) Input LayerNorm (float32)
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, C] in float32

    # -----------------------------------------------------------------
    # 2) Cast to fp16 for projections / gates
    x_h = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 3) Linear projections (no bias) → hidden_dim (fp16)
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left = F.linear(x_h, left_proj_w)   # [B, N, N, H]
    right = F.linear(x_h, right_proj_w)  # [B, N, N, H]

    # -----------------------------------------------------------------
    # 4) Gates – sigmoid of linear projections (fp16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    left_gate = torch.sigmoid(F.linear(x_h, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_h, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x_h, out_gate_w))  # [B, N, N, H] (fp16)

    # -----------------------------------------------------------------
    # 5) Apply gates (and optional mask)
    left = left * left_gate
    right = right * right_gate

    if not nomask and mask is not None:
        # mask expected as 0/1; broadcast over hidden dim
        mask_h = mask.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_h
        right = right * mask_h

    # -----------------------------------------------------------------
    # 6) Prepare tensors for the Triton GEMM kernel
    #    Fold batch and hidden dimension into a single channel dimension.
    B, N, _, _ = left.shape

    # Permute to (B, H, N, N) and make contiguous
    left = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()

    C_total = B * hidden_dim  # channels for the kernel
    left_flat = left.reshape(C_total, N, N)   # (C, M, K)
    right_flat = right.reshape(C_total, N, N) # (C, N, K) stored as (C, N, K)

    # Output buffer (FP32 accumulation)
    out_flat = torch.empty((C_total, N, N), dtype=torch.float32, device=device)

    # -----------------------------------------------------------------
    # 7) Compute tile configuration
    if N <= 256:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 32
    elif N <= 512:
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 32
    else:
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 64

    # Strides (row‑major)
    stride_out_c, stride_out_m, stride_out_n = out_flat.stride()
    stride_a_c, stride_a_m, stride_a_k = left_flat.stride()
    stride_b_c, stride_b_n, stride_b_k = right_flat.stride()   # note: swapped for K,N

    # The `right` tensor should be accessed as (C, K, N). Its stride
    # over K is the stride of the innermost dimension (original N),
    # while the stride over N is the stride of the middle dimension.
    stride_b_k = stride_b_k   # already stride of dim 2 (K)
    stride_b_n = stride_b_n   # stride of dim 1 (N)

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        C_total,
    )

    # Number of warps: 4 works well for small N, 8 for larger.
    num_warps = 8 if N >= 512 else 4

    # -----------------------------------------------------------------
    # 8) Launch Triton kernel
    tri_mul_gemm_kernel[grid](
        out_flat,
        left_flat,
        right_flat,
        N, N, N,
        stride_out_c, stride_out_m, stride_out_n,
        stride_a_c, stride_a_m, stride_a_k,
        stride_b_c, stride_b_k, stride_b_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=num_warps,
        num_stages=2,   # double‑buffering
    )

"""
TriMul (outgoing) – Triton‑accelerated implementation.

The costly part of the module is the “triangle” multiplication

    out[c, i, j] = Σₖ left[c, i, k] * right[c, j, k]

where `c` folds the batch and hidden dimensions (C = B·H).
All tensors are stored in FP16, the accumulation is done in FP32, and the
kernel uses a 2‑D tile (BLOCK_M × BLOCK_N) together with a 1‑D reduction tile
(BLOCK_K).  The rest of the module (layer‑norms, linear projections, sigmoid
gates, final projection) is kept in PyTorch – this satisfies the requirement
that at least a part of the work is performed in a Triton kernel while keeping
the code easy to verify.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_gemm_kernel(
    out_ptr,          # *float32 → output  (C, M, N)
    a_ptr,            # *float16 → left   (C, M, K)
    b_ptr,            # *float16 → right  (C, K, N)  (stored as (C, N, K))
    M, N, K,          # matrix sizes (all equal to seq_len)
    stride_out_c, stride_out_m, stride_out_n,
    stride_a_c, stride_a_m, stride_a_k,
    stride_b_c, stride_b_k, stride_b_n,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM used inside the outgoing TriMul module.

    For every folded channel `c = b·h` we compute
        out_c[M,N] += left_c[M,K] @ right_c[K,N]
    where the right tensor is logically transposed (K,N) through pointer
    arithmetic.
    """
    pid_c = tl.program_id(2)                # channel id  (B·H)
    pid_m = tl.program_id(0)                # block row   → i
    pid_n = tl.program_id(1)                # block col   → j

    # Offsets of the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # OOB masks for the output tile
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator in FP32 (better numerical stability)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -----------------------------------------------------------------
    # Loop over the reduction dimension K in tiles of size BLOCK_K
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Pointers to the current A and B tiles
        a_tile = a_ptr + pid_c * stride_a_c \
                 + offs_m[:, None] * stride_a_m \
                 + offs_k[None, :] * stride_a_k          # (M, K)

        b_tile = b_ptr + pid_c * stride_b_c \
                 + offs_k[:, None] * stride_b_k \
                 + offs_n[None, :] * stride_b_n          # (K, N)

        # Load tiles with zero‑fill for OOB elements
        a = tl.load(a_tile,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)                     # fp16
        b = tl.load(b_tile,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)                     # fp16

        # Tile‑wise matrix multiplication (fp16·fp16 → fp32)
        acc += tl.dot(a, b)

    # -----------------------------------------------------------------
    # Write the tile back to global memory
    out_tile = out_ptr + pid_c * stride_out_c \
               + offs_m[:, None] * stride_out_m \
               + offs_n[None, :] * stride_out_n
    tl.store(out_tile, acc,
             mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module (AlphaFold3).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor[B, N, N, C] (float32)
        - mask         : torch.Tensor[B, N, N] (float32/bool) or None
        - weights      : dict of model parameters with keys
            * "norm.weight", "norm.bias"
            * "left_proj.weight", "right_proj.weight"
            * "left_gate.weight", "right_gate.weight", "out_gate.weight"
            * "to_out_norm.weight", "to_out_norm.bias"
            * "to_out.weight"
        - config       : dict with entries
            * "dim"         – input channel dimension C
            * "hidden_dim" – hidden dimension H
            * "nomask"      – bool, True → ignore mask (default True)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, C] (float32)
    """
    # -----------------------------------------------------------------
    # Unpack
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    device = input_tensor.device

    # -----------------------------------------------------------------
    # 1) Input LayerNorm (float32)
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, C] (FP32)

    # -----------------------------------------------------------------
    # 2) Cast to fp16 for all subsequent linear ops
    x_h = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 3) Linear projections (no bias) → hidden_dim (fp16)
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left = F.linear(x_h, left_proj_w)   # [B, N, N, H]  fp16
    right = F.linear(x_h, right_proj_w)  # [B, N, N, H] fp16

    # -----------------------------------------------------------------
    # 4) Gates (sigmoid) – also fp16
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    left_gate = torch.sigmoid(F.linear(x_h, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_h, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x_h, out_gate_w))      # fp16, shape [B,N,N,H]

    # -----------------------------------------------------------------
    # 5) Apply gates (and optional mask)
    left = left * left_gate
    right = right * right_gate

    if not nomask and mask is not None:
        # mask is 0/1 – broadcast over hidden dimension
        mask_h = mask.to(torch.float16).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask_h
        right = right * mask_h

    # -----------------------------------------------------------------
    # 6) Re‑order tensors so the batch * hidden dimension becomes one “channel”
    B, N, _, _ = left.shape

    #   (B, N, N, H) → (B, H, N, N) → (C, M, K) where C = B·H, M = N, K = N
    left = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()

    C_total = B * hidden_dim                      # folded batch‑hidden dimension
    left_flat = left.reshape(C_total, N, N)       # (C, M, K)
    right_flat = right.reshape(C_total, N, N)     # (C, N, K) – will be read as (K,N)

    # Output buffer – accumulation in FP32
    out_flat = torch.empty((C_total, N, N), dtype=torch.float32, device=device)

    # -----------------------------------------------------------------
    # 7) Choose tile sizes (empirically good for H100)
    if N <= 256:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 32
    elif N <= 512:
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 32
    else:  # N up to 1024
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 64

    # Strides for the three tensors (row‑major ordering)
    stride_out_c, stride_out_m, stride_out_n = out_flat.stride()
    stride_a_c, stride_a_m, stride_a_k = left_flat.stride()
    stride_b_c, stride_b_n, stride_b_k = right_flat.stride()   # note the swap for K,N

    # Grid: (row‑blocks, col‑blocks, channels)
    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        C_total,
    )

    # Use more warps for larger problems to keep the SM busy
    num_warps = 8 if N >= 512 else 4

    # -----------------------------------------------------------------
    # 8) Launch the Triton GEMM kernel
    tri_mul_gemm_kernel[grid](
        out_flat,
        left_flat,
        right_flat,
        N, N, N,
        stride_out_c, stride_out_m, stride_out_n,
        stride_a_c, stride_a_m, stride_a_k,
        stride_b_c, stride_b_k, stride_b_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=num_warps,
        num_stages=2,               # simple double‑buffering
    )

    # -----------------------------------------------------------------
    # 9) Reshape back to the original 4‑D layout (float32)
    out = out_flat.reshape(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 10) Final processing: LayerNorm → out‑gate → projection back to dim
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    # out_gate is still in fp16 – promote to fp32 for the multiplication
    out = out * out_gate.to(torch.float32)

    # Linear projection to the original channel dimension
    out = F.linear(out, weights["to_out.weight"])   # (B, N, N, C) FP32

    return out
```
