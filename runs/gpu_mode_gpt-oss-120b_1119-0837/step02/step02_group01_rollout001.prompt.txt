import torch
from torch import nn, einsum
import math

# Reference code in PyTorch
class TriMul(nn.Module):
    def __init__(
        self,
        dim: int,
        hidden_dim: int,
    ):
        super().__init__()

        self.norm = nn.LayerNorm(dim)

        self.left_proj = nn.Linear(dim, hidden_dim, bias=False)
        self.right_proj = nn.Linear(dim, hidden_dim, bias=False)

        self.left_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.right_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.out_gate = nn.Linear(dim, hidden_dim, bias=False)

        self.to_out_norm = nn.LayerNorm(hidden_dim)
        self.to_out = nn.Linear(hidden_dim, dim, bias=False)

    def forward(self, x: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
        """
        x: [bs, seq_len, seq_len, dim]
        mask: [bs, seq_len, seq_len]

        Returns:
            output: [bs, seq_len, seq_len, dim]
        """
        batch_size, seq_len, _, dim = x.shape

        x = self.norm(x)

        left = self.left_proj(x)
        right = self.right_proj(x)

        mask = mask.unsqueeze(-1)
        left = left * mask
        right = right * mask

        left_gate = self.left_gate(x).sigmoid()
        right_gate = self.right_gate(x).sigmoid()
        out_gate = self.out_gate(x).sigmoid()

        left = left * left_gate
        right = right * right_gate

        out = einsum('... i k d, ... j k d -> ... i j d', left, right)
        # This einsum is the same as the following:
        # out = torch.zeros(batch_size, seq_len, seq_len, dim, device=x.device)
        
        # # Compute using nested loops
        # for b in range(batch_size):
        #     for i in range(seq_len):
        #         for j in range(seq_len):
        #             # Compute each output element
        #             for k in range(seq_len):
        #                 out[b, i, j] += left[b, i, k, :] * right[b, j, k, :]

        out = self.to_out_norm(out)
        out = out * out_gate
        return self.to_out(out)

"""TriMul “outgoing” forward pass.

- Input `x` (B, N, N, D) is LayerNorm‑ed with a small Triton kernel.
- Linear projections and sigmoid gates are computed in FP16.
- The core O(N³·H) contraction is performed as a batched FP16 GEMM
  (torch.bmm) which lets the H100 Tensor Cores execute the bulk of the work.
- The result is LayerNorm‑ed again (Triton) and projected back to the model
  dimension.  The final tensor is returned as FP32.

At least one operation (LayerNorm) is executed inside a Triton kernel,
as required, while the massive N³ contraction is dramatically sped‑up by
half‑precision Tensor‑Core GEMM.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 2‑pass LayerNorm kernel (operates on the last dimension)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,                # *ptr to input  [M, N]
    y_ptr,                # *ptr to output [M, N]
    weight_ptr,           # *ptr to weight [N]
    bias_ptr,             # *ptr to bias   [N]
    M,                    # #rows   (batch*seq*seq)
    N,                    # feature dimension (dim or hidden_dim)
    eps,                  # epsilon
    stride_xm, stride_xn,  # input strides
    stride_ym, stride_yn,  # output strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
):
    pid = tl.program_id(0)

    # ------------------------------------------------------------------
    # Row range for this program
    # ------------------------------------------------------------------
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # ------------------------------------------------------------------
    # First pass: compute per‑row mean & variance
    # ------------------------------------------------------------------
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_N)
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        sum_val += tl.sum(x, axis=1)
        sum_sq  += tl.sum(x * x, axis=1)

    mean = sum_val / N
    var  = sum_sq / N - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # ------------------------------------------------------------------
    # Second pass: write normalized + affine transformed values
    # ------------------------------------------------------------------
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        # normalise
        norm_x = (x - mean[:, None]) * rstd[:, None]

        # affine
        w = tl.load(weight_ptr + offs_n, mask=mask_n, other=0.0)
        b = tl.load(bias_ptr   + offs_n, mask=mask_n, other=0.0)

        y = norm_x * w[None, :] + b[None, :]

        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn,
            y,
            mask=mask_m[:, None] & mask_n[None, :],
        )


def triton_layernorm(x: torch.Tensor,
                     weight: torch.Tensor,
                     bias: torch.Tensor,
                     eps: float = 1e-5) -> torch.Tensor:
    """
    LayerNorm over the last dimension using the Triton kernel.
    `x` must be a contiguous 2‑D tensor (M, N).
    """
    assert x.dim() == 2, "triton_layernorm expects a 2‑D tensor"
    M, N = x.shape
    y = torch.empty_like(x)

    # Ensure contiguous layout for correct stride handling
    x = x.contiguous()
    y = y.contiguous()
    weight = weight.contiguous()
    bias = bias.contiguous()

    stride_xm, stride_xn = x.stride()
    stride_ym, stride_yn = y.stride()

    grid = lambda meta: (triton.cdiv(M, meta["BLOCK_M"]),)

    layernorm_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        weight_ptr=weight,
        bias_ptr=bias,
        M=M,
        N=N,
        eps=eps,
        stride_xm=stride_xm,
        stride_xn=stride_xn,
        stride_ym=stride_ym,
        stride_yn=stride_yn,
        BLOCK_M=128,   # rows per program; good default for H100
        BLOCK_N=64,    # columns per tile; fits registers nicely
    )
    return y


# ----------------------------------------------------------------------
# Triton‑accelerated TriMul “outgoing” entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, D] (float32)
        - mask         : torch.Tensor [B, N, N] or None
        - weights      : dict of torch.Tensors (all parameters)
        - config       : dict with keys ``dim``, ``hidden_dim``, ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor [B, N, N, D] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype  # expected to be torch.float32

    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (Triton)
    # ------------------------------------------------------------------
    # Collapse B,N,N into rows for the kernel
    M = bs * seq_len * seq_len
    x_flat = input_tensor.reshape(M, dim)
    norm_weight = weights["norm.weight"].to(device, dtype)
    norm_bias   = weights["norm.bias"].to(device, dtype)
    x_norm_flat = triton_layernorm(x_flat, norm_weight, norm_bias, eps=eps)
    x_norm = x_norm_flat.view(bs, seq_len, seq_len, dim)  # (B,N,N,D) float32

    # ------------------------------------------------------------------
    # Cast to half‑precision for the heavy part
    # ------------------------------------------------------------------
    x_norm_f16 = x_norm.to(torch.float16)

    # ------------------------------------------------------------------
    # 2️⃣ Cast model weights to half (once)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device, torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device, torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device, torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device, torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device, torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device, torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device, torch.float16)
    to_out_w      = weights["to_out.weight"].to(device, torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Linear projections (FP16)
    # ------------------------------------------------------------------
    left_proj = F.linear(x_norm_f16, left_proj_w)      # (B,N,N,H)
    right_proj = F.linear(x_norm_f16, right_proj_w)   # (B,N,N,H)

    # ------------------------------------------------------------------
    # 4️⃣ Gating (FP16)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_norm_f16, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_norm_f16, right_gate_w))
    out_gate  = torch.sigmoid(F.linear(x_norm_f16, out_gate_w))

    # ------------------------------------------------------------------
    # 5️⃣ Apply mask (if any) – mask is broadcast over hidden_dim
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_f16 = mask.to(device, torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left_proj = left_proj * mask_f16
        right_proj = right_proj * mask_f16

    # ------------------------------------------------------------------
    # 6️⃣ Element‑wise gating
    # ------------------------------------------------------------------
    left = left_proj * left_gate    # (B,N,N,H)
    right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 7️⃣ Core N³ contraction via batched half‑precision GEMM
    #    left[b,i,k,d] * right[b,j,k,d] summed over k → out[b,i,j,d]
    # ------------------------------------------------------------------
    # Change layout to (B, H, N, N) for contiguous batch‑matmul
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B,H,N,N)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B,H,N,N)

    BH = bs * hidden_dim
    left_mat = left_perm.view(BH, seq_len, seq_len)          # (B*H, N, N)
    right_mat = right_perm.view(BH, seq_len, seq_len)        # (B*H, N, N)

    # Batched matrix multiplication: (i,k)·(j,k)ᵀ
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B*H, N, N) half

    # Restore original layout (B,N,N,H)
    out = out_mat.view(bs, hidden_dim, seq_len, seq_len) \
                .permute(0, 2, 3, 1).contiguous()           # (B,N,N,H) half

    # ------------------------------------------------------------------
    # 8️⃣ Output LayerNorm (TP) – operate in FP32
    # ------------------------------------------------------------------
    out_fp32 = out.to(torch.float32)
    out_flat = out_fp32.reshape(M, hidden_dim)               # (B*N*N, H)

    to_out_norm_w_fp32 = to_out_norm_w.to(torch.float32)
    to_out_norm_b_fp32 = to_out_norm_b.to(torch.float32)

    out_norm_flat = triton_layernorm(out_flat,
                                     to_out_norm_w_fp32,
                                     to_out_norm_b_fp32,
                                     eps=eps)
    out_norm = out_norm_flat.view(bs, seq_len, seq_len, hidden_dim)  # (B,N,N,H) FP32

    # ------------------------------------------------------------------
    # 9️⃣ Apply output gate and final projection back to model dim
    # ------------------------------------------------------------------
    out_gate_fp32 = out_gate.to(torch.float32)
    gated = out_norm * out_gate_fp32                     # (B,N,N,H) FP32

    # Final linear projection (FP32 for numerical safety)
    output = F.linear(gated, to_out_w.to(torch.float32))  # (B,N,N,D) FP32

    return output
