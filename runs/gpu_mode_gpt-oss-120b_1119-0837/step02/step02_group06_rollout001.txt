"""
TriMul (outgoing) – Triton‑accelerated core

The function implements the forward pass of the outgoing triangle multiplicative
update used in AlphaFold3.  The heavy part is the “tri‑mul”:

    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

which is a batched matrix multiplication per hidden channel:
    out_{c} = left_{c} @ right_{c}ᵀ   with   c = b·H + h

The kernel `tri_mul_kernel` computes this batched GEMM on FP16 inputs
(accumulating in FP32) using Tensor‑Core friendly `tl.dot`.  Block sizes are
chosen adaptively to keep register pressure low (64×64 tiles).  All other
operations (layer‑norms, linear projections, gating, final projection) are
performed with regular PyTorch for simplicity.

The implementation is fully self‑contained and works for all test
configurations (variable batch size, sequence length, feature dimensions and
optional masks).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------
# Triton kernel: batched matrix multiply (left @ rightᵀ)
# --------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    out_ptr,    # float32*   (C_total, M, N)
    a_ptr,      # float16*   (C_total, M, K)   left
    b_ptr,      # float16*   (C_total, K, N)   rightᵀ (interpreted)
    M, N, K,    # matrix dimensions (all == seq_len)
    stride_out_m, stride_out_n, stride_out_c,   # output strides
    stride_a_m, stride_a_k, stride_a_c,        # left strides
    stride_b_k, stride_b_n, stride_b_c,        # rightᵀ strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM: out[c, i, j] = Σ_k left[c, i, k] * right[c, j, k]"""

    pid_m = tl.program_id(0)   # block row (i)
    pid_n = tl.program_id(1)   # block col (j)
    pid_c = tl.program_id(2)   # fused (batch, hidden) slice

    # -----------------------------------------------
    # Offsets for this tile
    # -----------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)   # FP32 accumulator

    # -----------------------------------------------
    # Reduction over K
    # -----------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Pointers to the current tiles of left (M×K) and rightᵀ (K×N)
        a_tile_ptr = a_ptr + pid_c * stride_a_c \
                     + offs_m[:, None] * stride_a_m \
                     + offs_k[None, :] * stride_a_k

        b_tile_ptr = b_ptr + pid_c * stride_b_c \
                     + offs_k[:, None] * stride_b_k \
                     + offs_n[None, :] * stride_b_n

        # Masked loads (FP16)
        a = tl.load(a_tile_ptr,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)
        b = tl.load(b_tile_ptr,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # Tensor‑core friendly dot (FP16×FP16 → FP32)
        acc += tl.dot(a, b)

    # -----------------------------------------------
    # Write back the tile
    # -----------------------------------------------
    out_tile_ptr = out_ptr + pid_c * stride_out_c \
                   + offs_m[:, None] * stride_out_m \
                   + offs_n[None, :] * stride_out_n

    tl.store(out_tile_ptr,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])


# --------------------------------------------------------------
# Entrypoint used by the test harness
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Arguments
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor[B, N, N, C] (float32)
        - mask         : torch.Tensor[B, N, N] (bool/float) or None
        - weights      : dict of torch.Tensor containing model parameters
        - config       : dict with keys "dim", "hidden_dim", optional "nomask"

    Returns
    -------
    torch.Tensor[B, N, N, C]  (float32)
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1) Layer‑norm over the feature dimension C
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, C)  float32

    # ------------------------------------------------------------------
    # 2) Cast to FP16 for cheap linear layers and gates
    # ------------------------------------------------------------------
    x_h = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Linear projections (bias‑free) → shape (B, N, N, H)
    # ------------------------------------------------------------------
    left = F.linear(
        x_h,
        weights["left_proj.weight"].to(torch.float16),
    )
    right = F.linear(
        x_h,
        weights["right_proj.weight"].to(torch.float16),
    )

    # ------------------------------------------------------------------
    # 4) Gates (sigmoid) – still FP16
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(
        F.linear(x_h, weights["left_gate.weight"].to(torch.float16))
    )
    right_gate = torch.sigmoid(
        F.linear(x_h, weights["right_gate.weight"].to(torch.float16))
    )
    out_gate = torch.sigmoid(
        F.linear(x_h, weights["out_gate.weight"].to(torch.float16))
    )  # (B,N,N,H) FP16

    # ------------------------------------------------------------------
    # 5) Apply gates (and optional mask) – in‑place multiplication
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    if not nomask and mask is not None:
        # mask shape (B,N,N) → (B,N,N,1) and cast to FP16
        mask_f = mask.to(torch.float16).unsqueeze(-1)
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 6) Prepare tensors for the Triton kernel
    #    We fuse batch and hidden dimension: C_total = B * H,
    #    and keep the two sequence axes as the last dimensions.
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape  # left shape (B, N, N, H)

    # Move hidden dim before the two sequence dims → (B, H, N, N)
    left = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()

    C_total = B * hidden_dim
    left_flat = left.reshape(C_total, N, N)   # (C, M, K)  with M=K=N
    right_flat = right.reshape(C_total, N, N) # (C, K, N) – will be used transposed

    # Output buffer for the kernel (FP32 accumulation)
    out_flat = torch.empty_like(left_flat, dtype=torch.float32)

    # ------------------------------------------------------------------
    # 7) Triton kernel launch
    # ------------------------------------------------------------------
    # Choose tile sizes that keep register pressure moderate.
    # For N <= 256 we can afford 64×64 tiles; for larger N we keep the same
    # tile size (the number of tiles simply grows).
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # Strides expressed in elements (not bytes)
    stride_out_c = out_flat.stride(0)
    stride_out_m = out_flat.stride(1)
    stride_out_n = out_flat.stride(2)

    stride_a_c = left_flat.stride(0)
    stride_a_m = left_flat.stride(1)
    stride_a_k = left_flat.stride(2)

    stride_b_c = right_flat.stride(0)
    # right is interpreted as (C, K, N) where K is the original last dim
    stride_b_k = right_flat.stride(2)   # original last dimension
    stride_b_n = right_flat.stride(1)   # original middle dimension

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # blocks over i‑axis
        (N + BLOCK_N - 1) // BLOCK_N,   # blocks over j‑axis
        C_total,                        # one block per (batch, hidden) slice
    )

    tri_mul_kernel[grid](
        out_flat,
        left_flat,
        right_flat,
        N, N, N,                         # M, N, K (all equal seq_len)
        stride_out_m, stride_out_n, stride_out_c,
        stride_a_m, stride_a_k, stride_a_c,
        stride_b_k, stride_b_n, stride_b_c,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,                     # 4 warps per block (good for 64×64)
        num_stages=2,                    # 2‑stage pipeline
    )

    # ------------------------------------------------------------------
    # 8) Reshape kernel output back to (B, N, N, H)
    # ------------------------------------------------------------------
    out = out_flat.reshape(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 9) Final processing: layer‑norm over hidden dim, out‑gate, output linear
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    # out_gate was FP16; promote to FP32 for the multiplication
    out = out * out_gate.to(torch.float32)
    out = F.linear(out, weights["to_out.weight"])   # (B, N, N, C)

    return out
