    left_gate_w = weights["left_gate.weight"].to(torch.float16)

    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    right_comb = torch.cat([right_proj_w, right_gate_w], dim=0)

    left_comb_out = F.linear(x_h_flat, left_comb)   # (B*N*N, 2*H)

    left = left_comb_out[:, :hidden_dim]   # (P, H)
    left_gate = torch.sigmoid(left_comb_out[:, hidden_dim:]) # (P, H)
    right = right_comb_out[:, :hidden_dim]
    right_gate = torch.sigmoid(right_comb_out[:, hidden_dim:])

    left = left.view(B, N, N, hidden_dim)
    left_gate = left_gate.view(B, N, N, hidden_dim)

    left = left * left_gate
    right = right * right_gate

"""
TriMul (outgoing) – fast forward pass with fused linear+gate ops.

The heavy N³ part (triangle multiplication) is implemented in a custom Triton
kernel that treats each (batch × hidden) slice as an independent GEMM.
All linear projections and their corresponding sigmoid gates are fused into
just two matmul calls (one for left‑hand side + out‑gate, one for right‑hand
side + right‑gate). This halves the number of kernel launches and the amount
of memory traffic compared to the naïve implementation. The rest of the
operations (LayerNorms, mask, final projection) stay in PyTorch.

The kernel tiles the output matrix with a 2‑D tile (BLOCK_M×BLOCK_N) and a
1‑D tile over the reduction dimension (BLOCK_K).  FP16 inputs are accumulated
in FP32 using Tensor‑Core friendly `tl.dot`.  Tile sizes are chosen adaptively
to keep occupancy high on H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# -------------------------------------------------------------------------
# Triton kernel: batched GEMM for the "triangle" multiplication.
# -------------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    out_ptr,            # float32*  (C, M, N) output
    a_ptr,              # float16*  (C, M, K) left operand
    b_ptr,              # float16*  (C, K, N) right operand (logically transposed)
    M, N, K,            # matrix dimensions (all equal to seq_len)
    stride_out_c, stride_out_m, stride_out_n,   # output strides
    stride_a_c, stride_a_m, stride_a_k,        # left strides
    stride_b_c, stride_b_k, stride_b_n,        # right strides (K major)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_c = tl.program_id(2)   # channel idx = batch * hidden_dim
    pid_m = tl.program_id(0)   # block row   → i
    pid_n = tl.program_id(1)   # block col   → j

    # -----------------------------------------------------------------
    # Tile offsets (relative to the start of the tensor)
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for out‑of‑bounds (when seq_len not a multiple of BLOCK_*)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -----------------------------------------------------------------
    # Loop over K dimension (reduction)
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Pointers to current tile of A (left) and B (right)
        a_tile = a_ptr + pid_c * stride_a_c \
                 + offs_m[:, None] * stride_a_m \
                 + offs_k[None, :] * stride_a_k      # (M, K)

        b_tile = b_ptr + pid_c * stride_b_c \
                 + offs_k[:, None] * stride_b_k \
                 + offs_n[None, :] * stride_b_n      # (K, N)

        # Load with masking (zero for OOB)
        a = tl.load(a_tile,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)                         # fp16
        b = tl.load(b_tile,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)                         # fp16

        # GEMM accumulate: fp16 * fp16 → fp32
        acc += tl.dot(a, b)

    # -----------------------------------------------------------------
    # Write back the tile
    out_tile = out_ptr + pid_c * stride_out_c \
               + offs_m[:, None] * stride_out_m \
               + offs_n[None, :] * stride_out_n
    tl.store(out_tile, acc,
             mask=mask_m[:, None] & mask_n[None, :])


# -------------------------------------------------------------------------
# Entry‑point implementing the full forward pass.
# -------------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module (AlphaFold3 style).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor[B, N, N, C] (float32)
            - mask         : torch.Tensor[B, N, N] (float32/bool) or None
            - weights      : dict of model parameters (see keys below)
            - config       : dict with keys
                * "dim"        – input channel dimension C
                * "hidden_dim"– hidden dimension H
                * "nomask"     – bool, True → ignore mask

    Returns:
        torch.Tensor[B, N, N, C] (float32) – module output
    """
    # -----------------------------------------------------------------
    # 0) unpack ---------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -----------------------------------------------------------------
    # 1) Input LayerNorm (float32) --------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, C]  (FP32)

    # -----------------------------------------------------------------
    # 2) Cast to fp16 (the bulk of the computation uses fp16) ----------
    x_h = x.to(torch.float16)

    B, N, _, _ = x_h.shape
    P = B * N * N                     # total pairwise positions

    # -----------------------------------------------------------------
    # 3) Fuse linear projections + sigmoid gates (2 matmuls total) -----
    # Left side + out‑gate (3 * hidden_dim rows)
    left_out_gate_weight = torch.cat(
        [
            weights["left_proj.weight"].to(torch.float16),   # (H, C)
            weights["left_gate.weight"].to(torch.float16),   # (H, C)
            weights["out_gate.weight"].to(torch.float16),   # (H, C)
        ],
        dim=0,
    )  # shape (3*H, C)

    # Right side + right‑gate (2 * hidden_dim rows)
    right_gate_weight = torch.cat(
        [
            weights["right_proj.weight"].to(torch.float16),  # (H, C)
            weights["right_gate.weight"].to(torch.float16),  # (H, C)
        ],
        dim=0,
    )  # shape (2*H, C)

    # Flatten the sequence dimensions to a single matrix for matmul
    x_h_flat = x_h.view(P, dim)    # (P, C)

    # One matmul gives left_proj, left_gate, out_gate simultaneously
    left_out_gate = F.linear(x_h_flat, left_out_gate_weight)  # (P, 3H)

    # One matmul gives right_proj, right_gate
    right_gate = F.linear(x_h_flat, right_gate_weight)        # (P, 2H)

    # -----------------------------------------------------------------
    # 4) Split the fused outputs ----------------------------------------
    # left side
    left   = left_out_gate[:, :hidden_dim]                     # (P, H)
    left_gate = torch.sigmoid(left_out_gate[:, hidden_dim:2 * hidden_dim])
    out_gate = torch.sigmoid(left_out_gate[:, 2 * hidden_dim:])  # (P, H)

    # right side
    right = right_gate[:, :hidden_dim]                        # (P, H)
    right_gate = torch.sigmoid(right_gate[:, hidden_dim:])    # (P, H)

    # -----------------------------------------------------------------
    # 5) Reshape back to [B, N, N, H] ----------------------------------
    left   = left.view(B, N, N, hidden_dim)
    left_gate = left_gate.view(B, N, N, hidden_dim)
    right  = right.view(B, N, N, hidden_dim)
    right_gate = right_gate.view(B, N, N, hidden_dim)
    out_gate = out_gate.view(B, N, N, hidden_dim)            # Fp16 for now

    # -----------------------------------------------------------------
    # 6) Apply sigmoid gates (and optional mask) ------------------------
    left = left * left_gate
    right = right * right_gate

    if not nomask and mask is not None:
        # mask broadcast over hidden dim (0/1 values)
        mask_h = mask.to(torch.float16).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask_h
        right = right * mask_h

    # -----------------------------------------------------------------
    # 7) Prepare tensors for the triangle‑multiplication kernel ----------
    # Move hidden dim before the two sequence dim to get layout (C, M, K)
    left = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    C_total = B * hidden_dim                         # flattened batch*hidden
    left_flat = left.reshape(C_total, N, N)          # (C, M, K)   with K=N
    right_flat = right.reshape(C_total, N, N)        # (C, K, N)   (will be read as K-major)

    # output buffer for the GEMM (FP32 accumulations)
    out_flat = torch.empty_like(left_flat, dtype=torch.float32, device=device)

    # -----------------------------------------------------------------
    # 8) Choose tiling parameters adaptively ---------------------------
    if N <= 256:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 32
    elif N <= 512:
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 32
    else:
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 64

    # Strides (in number of elements, not bytes)
    stride_out_c = out_flat.stride(0)
    stride_out_m = out_flat.stride(1)
    stride_out_n = out_flat.stride(2)

    stride_a_c = left_flat.stride(0)
    stride_a_m = left_flat.stride(1)
    stride_a_k = left_flat.stride(2)

    stride_b_c = right_flat.stride(0)
    stride_b_k = right_flat.stride(2)   # reduction dim (k) is innermost
    stride_b_n = right_flat.stride(1)   # output‑col dim (j)

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        C_total,
    )

    # More warps give higher occupancy on H100; 8 works well for large N.
    num_warps = 8 if N >= 512 else 4

    # -----------------------------------------------------------------
    # 9) Launch the triangular multiplication kernel --------------------
    tri_mul_kernel[grid](
        out_flat,
        left_flat,
        right_flat,
        N, N, N,
        stride_out_m, stride_out_n, stride_out_c,
        stride_a_m, stride_a_k, stride_a_c,
        stride_b_k, stride_b_n, stride_b_c,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=num_warps,
    )

    # -----------------------------------------------------------------
    # 10) Reshape back to [B, N, N, H] (FP32) ---------------------------
    out = out_flat.reshape(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, H] FP32

    # -----------------------------------------------------------------
    # 11) Post‑processing: LayerNorm → out‑gate → final linear ---------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    # out_gate was computed in fp16; cast to fp32 for the elementwise product
    out = out * out_gate.to(torch.float32)

    # Final linear projection back to the original channel dimension
    out = F.linear(out, weights["to_out.weight"])

    return out
