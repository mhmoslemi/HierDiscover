import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.autotune(
    configs=[
        triton.Config(
            {"BLOCK_M": 64, "BLOCK_N": 64, "BLOCK_K": 32},
            num_warps=4,
        ),
        triton.Config(
            {"BLOCK_M": 128, "BLOCK_N": 128, "BLOCK_K": 64},
            num_warps=8,
        ),
    ],
    key=["M", "N", "K"],
)
@triton.jit
def tri_mul_kernel(
    out_ptr,
    a_ptr,
    b_ptr,
    M, N, K,
    stride_out_m, stride_out_n, stride_out_c,
    stride_a_m, stride_a_k, stride_a_c,
    stride_b_k, stride_b_n, stride_b_c,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM for outgoing TriMul: out = left @ rightᵀ.

    Parameters
    ----------
    out_ptr : *float32
        Output tensor of shape (C, M, N). C = batch * hidden_dim.
    a_ptr : *float16
        Left input of shape (C, M, K).
    b_ptr : *float16
        Right input of shape (C, N, K) stored as (C, K, N) (the K‑dimension is the
        inner index and is loaded column‑wise by the kernel).
    M, N, K : int64
        Matrix dimensions (all equal to seq_len in our use‑case).
    stride_* : int64
        Strides for each tensor in element units.
    BLOCK_M/N/K : constexpr int
        Tile sizes – selected by Triton autotuning.
    """
    pid_m = tl.program_id(0)  # tile row index (i)
    pid_n = tl.program_id(1)  # tile col index (j)
    pid_c = tl.program_id(2)  # channel (batch * hidden_dim)

    # Offsets for this tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for out‑of‑bounds (required for non‑divisible sizes)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator in FP32 – Tensor‑cores compute FP16×FP16 → FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in chunks of BLOCK_K.
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = kk * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Pointers to current block of left (M×K) and right (K×N)
        a_tile_ptr = (
            a_ptr
            + pid_c * stride_a_c
            + offs_m[:, None] * stride_a_m
            + offs_k[None, :] * stride_a_k
        )
        b_tile_ptr = (
            b_ptr
            + pid_c * stride_b_c
            + offs_k[:, None] * stride_b_k
            + offs_n[None, :] * stride_b_n
        )

        # Load tiles; out‑of‑range entries are zeroed.
        a = tl.load(a_tile_ptr, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_tile_ptr, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # Core GEMM – FP16·FP16 → FP32
        acc += tl.dot(a, b)

    # Store tile to output
    out_tile_ptr = (
        out_ptr
        + pid_c * stride_out_c
        + offs_m[:, None] * stride_out_m
        + offs_n[None, :] * stride_out_n
    )
    tl.store(out_tile_ptr, acc, mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    """Outgoing TriMul forward pass (no gradients).

    This implementation follows the AlphaFold‑3 definition:
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    where ``left``/``right`` are linear projections of a layer‑normed input
    and are modulated by per‑position sigmoid gates (and an optional mask).
    The heavy N³ fused multiplication is performed by a custom Triton kernel
    that works in FP16 and accumulates in FP32. All other lightweight operations
    (layer‑norm, linear projections, gates, final normalisation) are kept in
    PyTorch for simplicity but are executed in half‑precision where possible.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        * input_tensor : torch.Tensor[B, N, N, C] (float32)
        * mask         : torch.Tensor[B, N, N] (bool/float) – may be None
        * weights      : dict of torch.Tensors containing module parameters
        * config       : dict with keys ``dim`` (C) and ``hidden_dim`` (H)

    Returns
    -------
    torch.Tensor[B, N, N, C] (float32)
    """
    with torch.no_grad():
        # ------------------------------------------------------------------
        # Unpack arguments
        input_tensor, mask, weights, config = data
        dim = config["dim"]
        hidden_dim = config["hidden_dim"]
        nomask = config.get("nomask", True)

        device = input_tensor.device

        # ------------------------------------------------------------------
        # 1) Layer‑norm the input (FP32)
        x = F.layer_norm(
            input_tensor,
            (dim,),
            weight=weights["norm.weight"],
            bias=weights["norm.bias"],
        )  # [B, N, N, C]

        # ------------------------------------------------------------------
        # 2) Cast to FP16 for all subsequent linear ops
        x_h = x.to(torch.float16)

        # ------------------------------------------------------------------
        # 3) Projections (no bias)
        left_proj_w = weights["left_proj.weight"].to(torch.float16)
        right_proj_w = weights["right_proj.weight"].to(torch.float16)

        left = F.linear(x_h, left_proj_w)   # [B, N, N, H]
        right = F.linear(x_h, right_proj_w)  # [B, N, N, H]

        # ------------------------------------------------------------------
        # 4) Gating (sigmoid)
        left_gate_w = weights["left_gate.weight"].to(torch.float16)
        right_gate_w = weights["right_gate.weight"].to(torch.float16)
        out_gate_w = weights["out_gate.weight"].to(torch.float16)

        left_gate = torch.sigmoid(F.linear(x_h, left_gate_w))   # [B, N, N, H]
        right_gate = torch.sigmoid(F.linear(x_h, right_gate_w))  # [B, N, N, H]
        out_gate = torch.sigmoid(F.linear(x_h, out_gate_w))   # [B, N, N, H]

        # ------------------------------------------------------------------
        # 5) Optional mask (applied before gating, replicating the reference)
        if not nomask and mask is not None:
            # Mask is broadcasted over the hidden dimension.
            mask_f = mask.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
            left = left * mask_f
            right = right * mask_f

        # 6) Apply per‑position gates
        left = left * left_gate
        right = right * right_gate

        # ------------------------------------------------------------------
        # 7) Prepare data for the batched GEMM kernel
        B, N, _, _ = left.shape  # B, N, N, H
        # Move hidden dimension to the second axis for a contiguous (B, H, N, N) layout.
        left = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
        right = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

        # Collapse batch and hidden dimension – each program processes one GEMM.
        C_total = B * hidden_dim
        left_flat = left.reshape(C_total, N, N)   # (C, M, K)
        # Right will be accessed as (C, K, N) inside the kernel.
        right_flat = right.reshape(C_total, N, N) # (C, K, N) after pointer arithmetic

        # Allocate output buffer (float32 accumulation)
        out_flat = torch.empty_like(left_flat, dtype=torch.float32)

        # ------------------------------------------------------------------
        # 8) Kernel launch configuration
        stride_out_c = out_flat.stride(0)
        stride_out_m = out_flat.stride(1)
        stride_out_n = out_flat.stride(2)

        stride_a_c = left_flat.stride(0)
        stride_a_m = left_flat.stride(1)
        stride_a_k = left_flat.stride(2)

        stride_b_c = right_flat.stride(0)
        stride_b_k = right_flat.stride(2)   # dimension K (the last axis after reshaping)
        stride_b_n = right_flat.stride(1)   # dimension N (the middle axis)

        # Grid: one block per output tile (M, N) and one program per channel.
        grid = (
            (N + 128 - 1) // 128,  # blocks along M (i)
            (N + 128 - 1) // 128,  # blocks along N (j)
            C_total,                # channels (batch * hidden_dim)
        )

        # Launch the GEMM kernel
        tri_mul_kernel[grid](
            out_flat,
            left_flat,
            right_flat,
            N, N, N,
            stride_out_m,
            stride_out_n,
            stride_out_c,
            stride_a_m,
            stride_a_k,
            stride_a_c,
            stride_b_k,
            stride_b_n,
            stride_b_c,
            BLOCK_M=128,
            BLOCK_N=128,
            BLOCK_K=64,
            num_warps=8,
            num_stages=2,
        )

        # ------------------------------------------------------------------
        # 9) Reshape back to original layout: (B, N, N, H)
        out = out_flat.reshape(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

        # ------------------------------------------------------------------
        # 10) Post‑processing: hidden‑dim layer‑norm, output gate, final projection
        out = F.layer_norm(
            out,
            (hidden_dim,),
            weight=weights["to_out_norm.weight"],
            bias=weights["to_out_norm.bias"],
        )
        out = out * out_gate  # out_gate is already FP32‑compatible

        # Final linear projection back to original dim (use FP16 for speed)
        to_out_w = weights["to_out.weight"].to(torch.float16)
        out = F.linear(out.to(torch.float16), to_out_w).to(torch.float32)

        return out

        out_ptr
        + pid_c * stride_out_c
        + offs_m[:, None] * stride_out_m
        + offs_n[None, :] * stride_out_n
    )
    tl.store(
        out_tile_ptr,
        acc,
        mask=mask_m[:, None] & mask_n[None, :],
    )


# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """Forward pass of the outgoing TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor[B, N, N, C] (float32)
        - mask         : torch.Tensor[B, N, N] (bool/float) – may be None
        - weights      : dict of torch.Tensors (module parameters)
        - config       : dict with keys
            * "dim"        – input channel dimension C
            * "hidden_dim" – hidden dimension H
            * "nomask"     – bool, True = ignore mask

    Returns
    -------
    torch.Tensor[B, N, N, C] (float32)
    """
    # --------------------------------------------------------------
    # Disable autograd – we only need the forward pass.
    # --------------------------------------------------------------
    with torch.no_grad():
        # ----------------------------------------------------------
        # 0) Unpack inputs
        # ----------------------------------------------------------
        input_tensor, mask, weights, config = data
        dim = config["dim"]
        hidden_dim = config["hidden_dim"]
        nomask = config.get("nomask", True)

        device = input_tensor.device

        # ----------------------------------------------------------
        # 1) Layer‑norm the raw input (FP32)
        # ----------------------------------------------------------
        x = F.layer_norm(
            input_tensor,
            (dim,),
            weight=weights["norm.weight"],
            bias=weights["norm.bias"],
        )  # [B, N, N, C]

        # ----------------------------------------------------------
        # 2) Cast to FP16 for all subsequent linear ops
        # ----------------------------------------------------------
        x_h = x.to(torch.float16)

        # ----------------------------------------------------------
        # 3) Linear projections (no bias) → [B, N, N, H]
        # ----------------------------------------------------------
        left_proj_w = weights["left_proj.weight"].to(torch.float16)
        right_proj_w = weights["right_proj.weight"].to(torch.float16)

        left = F.linear(x_h, left_proj_w)    # [B, N, N, H]
        right = F.linear(x_h, right_proj_w)  # [B, N, N, H]

        # ----------------------------------------------------------
        # 4) Gating (sigmoid) → same shape
        # ----------------------------------------------------------
        left_gate_w = weights["left_gate.weight"].to(torch.float16)
        right_gate_w = weights["right_gate.weight"].to(torch.float16)
        out_gate_w = weights["out_gate.weight"].to(torch.float16)

        left_gate = torch.sigmoid(F.linear(x_h, left_gate_w))   # [B, N, N, H]
        right_gate = torch.sigmoid(F.linear(x_h, right_gate_w))  # [B, N, N, H]
        out_gate = torch.sigmoid(F.linear(x_h, out_gate_w))   # [B, N, N, H]

        # ----------------------------------------------------------
        # 5) Optional mask (applied before gating, identical to reference)
        # ----------------------------------------------------------
        if not nomask and mask is not None:
            mask_f = mask.to(torch.float16).unsqueeze(-1)   # [B, N, N, 1]
            left = left * mask_f
            right = right * mask_f

        # ----------------------------------------------------------
        # 6) Apply per‑position gates
        # ----------------------------------------------------------
        left = left * left_gate
        right = right * right_gate

        # ----------------------------------------------------------
        # 7) Reshape for the Triton batched GEMM
        # ----------------------------------------------------------
        B, N, _, _ = left.shape          # B, N, N, H
        # Move hidden dimension before the two sequence dimensions for a
        # contiguous (B, H, N, N) layout.
        left = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
        right = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

        # Collapse batch and hidden dimension → one independent GEMM per channel
        C_total = B * hidden_dim                       # = batch * hidden_dim
        left_flat = left.reshape(C_total, N, N)        # (C, M, K)
        right_flat = right.reshape(C_total, N, N)      # (C, K, N) – accessed transposed

        # Output buffer – FP32 accumulation
        out_flat = torch.empty_like(left_flat, dtype=torch.float32)

        # ----------------------------------------------------------
        # 8) Compute strides (in element units)
        # ----------------------------------------------------------
        stride_out_c = out_flat.stride(0)
        stride_out_m = out_flat.stride(1)
        stride_out_n = out_flat.stride(2)

        stride_a_c = left_flat.stride(0)
        stride_a_m = left_flat.stride(1)
        stride_a_k = left_flat.stride(2)

        stride_b_c = right_flat.stride(0)
        stride_b_k = right_flat.stride(2)   # K‑dimension (last axis after reshape)
        stride_b_n = right_flat.stride(1)   # N‑dimension (middle axis)

        # ----------------------------------------------------------
        # 9) Kernel launch configuration
        # ----------------------------------------------------------
        # Tile size constants must match the ones we will pass at launch.
        BLOCK_M = 128
        BLOCK_N = 128
        BLOCK_K = 64

        grid = (
            (N + BLOCK_M - 1) // BLOCK_M,   # blocks along the i‑axis
            (N + BLOCK_N - 1) // BLOCK_N,   # blocks along the j‑axis
            C_total,                        # one program per (batch × hidden) channel
        )

        # ----------------------------------------------------------
        # 10) Launch the Triton kernel
        # ----------------------------------------------------------
        tri_mul_kernel[grid](
            out_flat,
            left_flat,
            right_flat,
            N, N, N,
            stride_out_m,
            stride_out_n,
            stride_out_c,
            stride_a_m,
            stride_a_k,
            stride_a_c,
            stride_b_k,
            stride_b_n,
            stride_b_c,
            BLOCK_M=BLOCK_M,
            BLOCK_N=BLOCK_N,
            BLOCK_K=BLOCK_K,
            num_warps=8,
            num_stages=2,
        )

        # ----------------------------------------------------------
        # 11) Reshape the GEMM result back to (B, N, N, H)
        # ----------------------------------------------------------
        out = out_flat.reshape(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

        # ----------------------------------------------------------
        # 12) Post‑processing: hidden‑dim layer‑norm, output gate, final proj.
        # ----------------------------------------------------------
        out = F.layer_norm(
            out,
            (hidden_dim,),
            weight=weights["to_out_norm.weight"],
            bias=weights["to_out_norm.bias"],
        )
        # Multiply by the output gate (cast to FP32 automatically)
        out = out * out_gate

        # Final projection back to the original dimension (use FP16 for speed)
        to_out_w = weights["to_out.weight"].to(torch.float16)
        out = F.linear(out.to(torch.float16), to_out_w).to(torch.float32)

        return out
