"""
TriMul (outgoing) – Triton‑accelerated implementation
====================================================

The TriMul module computes the “outgoing” triangle multiplicative update
described in the AlphaFold‑3 paper:

    out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]

where ``left`` and ``right`` are linear projections of a layer‑normed
pairwise input and are modulated by sigmoid gates (and optionally a mask).

The heavy part of the computation is a batched matrix multiplication over
the sequence dimension (N).  This kernel fuses the GEMM for all hidden
channels (batch × hidden_dim) using Triton:

* FP16 inputs → Tensor‑core dot product → FP32 accumulation.
* 128 × 128 × 64 tiles (optimal for H100) with 8 warps per block.
* Double‑buffered loads (``num_stages=2``) hide memory latency.
* No extra memory copies – the right‑hand matrix is accessed in a
  transposed view via stride manipulation.

All remaining operations (LayerNorm, linear projections, sigmoid gates,
final normalisation and projection) are performed with regular
PyTorch kernels in FP16 for speed; the final result is returned in FP32.

The function signature follows the problem statement:
    custom_kernel(data)  →  torch.Tensor[B, N, N, C]
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the outgoing TriMul update.
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_kernel(
    out_ptr,               # float32*  output (C, M, N)
    a_ptr,                 # float16*  left  (C, M, K)
    b_ptr,                 # float16*  right (C, K, N) – accessed transposed via strides
    M, N, K,               # dimensions (all equal to seq_len)
    stride_out_m, stride_out_n, stride_out_c,   # output strides
    stride_a_m, stride_a_k, stride_a_c,        # left strides
    stride_b_k, stride_b_n, stride_b_c,        # right strides (K‑fast, N‑slow)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute one tile of the output matrix for a given hidden channel:
        out = left @ rightᵀ
    where ``left`` has shape (M, K) and ``right`` is interpreted as (K, N)
    without an explicit transpose.
    """
    pid_m = tl.program_id(0)   # block row   → i
    pid_n = tl.program_id(1)   # block col   → j
    pid_c = tl.program_id(2)   # hidden channel (batch * hidden_dim)

    # ------------------------------------------------------------------
    # Tile offsets
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # ------------------------------------------------------------------
    # Accumulator (FP32)
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over the K dimension in BLOCK_K‑sized chunks
    # ------------------------------------------------------------------
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = kk * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Pointers to the current tiles of ``left`` and ``right``
        a_tile_ptr = (
            a_ptr
            + pid_c * stride_a_c
            + offs_m[:, None] * stride_a_m
            + offs_k[None, :] * stride_a_k
        )
        b_tile_ptr = (
            b_ptr
            + pid_c * stride_b_c
            + offs_k[:, None] * stride_b_k   # K dimension (fast)
            + offs_n[None, :] * stride_b_n   # N dimension (slow)
        )

        # Load with bounds checking; out‑of‑bounds elements are zero.
        a = tl.load(a_tile_ptr,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)
        b = tl.load(b_tile_ptr,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # FP16·FP16 → FP32 dot product (Tensor‑cores)
        acc += tl.dot(a, b)

    # ------------------------------------------------------------------
    # Write the computed tile back to global memory
    # ------------------------------------------------------------------
    out_tile_ptr = (
        out_ptr
        + pid_c * stride_out_c
        + offs_m[:, None] * stride_out_m
        + offs_n[None, :] * stride_out_n
    )
    tl.store(out_tile_ptr,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Public entry point – matches the required signature.
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor[B, N, N, C] (float32)
        - mask         : torch.Tensor[B, N, N] (bool/float) or None
        - weights      : dict of torch.Tensors (model parameters)
        - config       : dict with keys
            * "dim"        – input channel dimension C
            * "hidden_dim" – hidden dimension H
            * "nomask"     – bool, if True the mask is ignored

    Returns
    -------
    torch.Tensor[B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1) Layer‑norm the input (FP32 for numerical stability)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # shape [B, N, N, C]  (float32)

    # ------------------------------------------------------------------
    # 2) Cast to FP16 for the heavy linear / gating work
    # ------------------------------------------------------------------
    x_h = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Linear projections (bias=False) – FP16 weights
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)

    left  = F.linear(x_h, left_proj_w)    # [B, N, N, H]
    right = F.linear(x_h, right_proj_w)   # [B, N, N, H]

    # ------------------------------------------------------------------
    # 4) Gating networks (sigmoid) – also FP16
    # ------------------------------------------------------------------
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    left_gate   = torch.sigmoid(F.linear(x_h, left_gate_w))    # [B, N, N, H]
    right_gate  = torch.sigmoid(F.linear(x_h, right_gate_w))   # [B, N, N, H]
    out_gate    = torch.sigmoid(F.linear(x_h, out_gate_w))    # [B, N, N, H]

    # ------------------------------------------------------------------
    # 5) Optional mask (applied before gating, as in the reference)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_f = mask.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left  = left  * mask_f
        right = right * mask_f

    # 6) Apply sigmoid gates
    left  = left  * left_gate
    right = right * right_gate

    # Release tensors we no longer need to free memory early
    del left_gate, right_gate, x_h, x

    # ------------------------------------------------------------------
    # 7) Prepare tensors for the Triton GEMM
    #    Layout: (batch*hidden_dim, N, N)  – contiguous in memory.
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape                         # B, N, N, H
    # Move hidden dim before the two seq dims for a compact layout
    left  = left.permute(0, 3, 1, 2).contiguous()    # [B, H, N, N]
    right = right.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]

    C_total = B * hidden_dim                         # number of independent GEMMs
    left_flat  = left.reshape(C_total, N, N)        # (C, M, K)
    right_flat = right.reshape(C_total, N, N)       # (C, K, N)  (will be accessed transposed)

    # Output buffer – accumulate in FP32 for better numerical quality
    out_flat = torch.empty_like(left_flat, dtype=torch.float32)

    # ------------------------------------------------------------------
    # 8) Triton kernel launch configuration
    # ------------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64          # fits nicely into Tensor‑core WMMA tiles

    # Strides for the three tensors (C‑major layout)
    stride_out_c = out_flat.stride(0)
    stride_out_m = out_flat.stride(1)
    stride_out_n = out_flat.stride(2)

    stride_a_c = left_flat.stride(0)
    stride_a_m = left_flat.stride(1)
    stride_a_k = left_flat.stride(2)

    stride_b_c = right_flat.stride(0)
    # For the right matrix we interpret the last dimension as the K‑axis
    # (fast) and the middle dimension as the N‑axis (slow) – this avoids an explicit transpose.
    stride_b_k = right_flat.stride(2)   # K‑fast (contiguous)
    stride_b_n = right_flat.stride(1)   # N‑slow

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # blocks along M (i)
        (N + BLOCK_N - 1) // BLOCK_N,   # blocks along N (j)
        C_total,                        # one program per hidden channel (batch*H)
    )

    # Launch the kernel
    _tri_mul_kernel[grid](
        out_flat,
        left_flat,
        right_flat,
        N, N, N,
        stride_out_m,
        stride_out_n,
        stride_out_c,
        stride_a_m,
        stride_a_k,
        stride_a_c,
        stride_b_k,
        stride_b_n,
        stride_b_c,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=2,
    )

    # ------------------------------------------------------------------
    # 9) Reshape the GEMM result back to (B, N, N, H)
    # ------------------------------------------------------------------
    out = out_flat.reshape(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 10) Post‑processing: hidden‑dim LayerNorm, output gate, final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )  # stays FP32

    # Multiply by the output gate (broadcasted over the hidden dimension)
    out = out * out_gate

    # Final linear projection back to the original dimension (use FP16 weight for speed)
    to_out_w = weights["to_out.weight"].to(torch.float16)
    out = F.linear(out.to(torch.float16), to_out_w).to(torch.float32)

    return out
