@triton.jit
def trimul_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # program ids
    pid_bh = tl.program_id(0)  # combined batch*head
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # compute b and h
    b = pid_bh // H
    h = pid_bh % H

    # compute offsets for output
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for bounds
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Initialize accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # compute number of k blocks
    num_k_blocks = tl.cdiv(N, BLOCK_K)

    for kb in range(num_k_blocks):
        # k offset for this tile
        offs_k = kb * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Pointers
        # left: shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_ptr \
            + b * stride_left_b \
            + h * stride_left_h \
            + offs_i[:, None] * stride_left_i \
            + offs_k[None, :] * stride_left_k

        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # right: we need transposed tile (BLOCK_K, BLOCK_N)
        # right original layout: (B, N, N, H) index (b, j, k, h)
        # for transposed, we load with k as first, j as second.
        right_ptrs = right_ptr \
            + b * stride_right_b \
            + h * stride_right_h \
            + offs_k[:, None] * stride_right_k \
            + offs_j[None, :] * stride_right_j

        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)

        # do dot: left (M,K) * right (K,N) -> (M,N)
        acc += tl.dot(left, right)

    # Write output
    out_ptrs = out_ptr \
        + b * stride_out_b \
        + h * stride_out_h \
        + offs_i[:, None] * stride_out_i \
        + offs_j[None, :] * stride_out_j

    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])

@triton.jit
def trimul_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0) # combined batch*head
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    total_heads = H
    b = pid // total_heads
    h = pid % total_heads

    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_K)

    for k in range(num_k):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left tile (M, K)
        left_ptrs = left_ptr \
            + b * stride_left_b \
            + h * stride_left_h \
            + offs_i[:, None] * stride_left_i \
            + offs_k[None, :] * stride_left_k

        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Load right tile transposed (K, N)
        right_ptrs = right_ptr \
            + b * stride_right_b \
            + h * stride_right_h \
            + offs_k[:, None] * stride_right_k \
            + offs_j[None, :] * stride_right_j

        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)

        acc += tl.dot(left, right)

    out_ptrs = out_ptr \
        + b * stride_out_b \
        + h * stride_out_h \
        + offs_i[:, None] * stride_out_i \
        + offs_j[None, :] * stride_out_j

    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])

"""
TriMul “outgoing” – fast Triton‑only forward.

Algorithm
---------
1️⃣ Input LayerNorm  – two‑pass Triton kernel (FP32 accumulation).
2️⃣ Fuse the five linear layers (left/right projections + 3 gates) into one
   half‑precision GEMM by concatenating the weight matrices.
3️⃣ Apply the optional binary mask and the three sigmoid gates (still FP16).
4️⃣ **Core N³ reduction** – a custom Triton GEMM kernel computes
        out[b,i,j,h] = Σₖ left[b,i,k,h]·right[b,j,k,h]
   for every head *h* without any data transposes or copies.
5️⃣ Output LayerNorm – same Triton kernel on the hidden dimension.
6️⃣ Multiply by the output gate and a final half‑precision linear projection
   back to the model dimension.  The result is returned as FP32.

The heavy N³ step is now a single fused GEMM that runs on the H100’s
Tensor‑cores, avoiding the costly permute+torch.bmm used in the reference.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# 2‑pass LayerNorm (FP32 accumulation, FP32 output)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,                # *ptr to input  [M, N]
    y_ptr,                # *ptr to output [M, N]
    weight_ptr,           # *ptr to weight [N]
    bias_ptr,             # *ptr to bias   [N]
    M,                    # total rows
    N,                    # feature dimension
    eps,                  # epsilon
    stride_xm, stride_xn,  # strides for input
    stride_ym, stride_yn,  # strides for output
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
):
    pid = tl.program_id(0)

    # ------------------------------------------------------------------
    # Row offsets for this program
    # ------------------------------------------------------------------
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # ------------------------------------------------------------------
    # First pass – mean & variance (FP32)
    # ------------------------------------------------------------------
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    NUM_K = tl.cdiv(N, BLOCK_N)
    for k in range(NUM_K):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        x_f = x.to(tl.float32)
        sum_val += tl.sum(x_f, axis=1)
        sum_sq  += tl.sum(x_f * x_f, axis=1)

    mean = sum_val / N
    var  = sum_sq / N - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # ------------------------------------------------------------------
    # Second pass – normalise + affine
    # ------------------------------------------------------------------
    for k in range(NUM_K):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N

        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        x_f = x.to(tl.float32)

        norm_x = (x_f - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_n, mask=mask_n, other=0.0).to(tl.float32)
        b = tl.load(bias_ptr   + offs_n, mask=mask_n, other=0.0).to(tl.float32)

        y = norm_x * w[None, :] + b[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn,
            y,
            mask=mask_m[:, None] & mask_n[None, :],
        )


def triton_layernorm(x: torch.Tensor,
                     weight: torch.Tensor,
                     bias: torch.Tensor,
                     eps: float = 1e-5,
                     BLOCK_M: int = 128,
                     BLOCK_N: int = 64) -> torch.Tensor:
    """
    Apply LayerNorm over the last dimension of a 2‑D tensor ``x``.
    """
    assert x.dim() == 2, "triton_layernorm expects a 2‑D tensor"
    M, N = x.shape
    y = torch.empty_like(x)

    x = x.contiguous()
    y = y.contiguous()
    weight = weight.contiguous()
    bias = bias.contiguous()

    stride_xm, stride_xn = x.stride()
    stride_ym, stride_yn = y.stride()

    grid = lambda meta: (triton.cdiv(M, meta["BLOCK_M"]),)

    layernorm_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        weight_ptr=weight,
        bias_ptr=bias,
        M=M,
        N=N,
        eps=eps,
        stride_xm=stride_xm,
        stride_xn=stride_xn,
        stride_ym=stride_ym,
        stride_yn=stride_yn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
    )
    return y


# ----------------------------------------------------------------------
# Core N³ reduction – fused GEMM over the hidden dimension
# ----------------------------------------------------------------------
@triton.jit
def trimul_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,                              # batch, seq_len, hidden_dim
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    All tensors are in FP16; accumulation in FP32.
    Grid dimensions:
        (B*H, ceil(N/BLOCK_M), ceil(N/BLOCK_N))
    """
    pid0 = tl.program_id(0)      # combined batch*head index
    pid_i = tl.program_id(1)     # i‑tile
    pid_j = tl.program_id(2)     # j‑tile

    # de‑compose combined index
    b = pid0 // H
    h = pid0 % H

    # block offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator (FP32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # number of K‑tiles
    num_k = tl.cdiv(N, BLOCK_K)

    for k_tile in range(num_k):
        offs_k = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ----------- load left tile (M x K) -----------------
        left_ptrs = left_ptr \
            + b * stride_left_b \
            + h * stride_left_h \
            + offs_i[:, None] * stride_left_i \
            + offs_k[None, :] * stride_left_k

        left = tl.load(
            left_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)                     # (M, K)

        # ----------- load right tile transposed (K x N) -----
        # original layout right[b, j, k, h]; we need (k, j)
        right_ptrs = right_ptr \
            + b * stride_right_b \
            + h * stride_right_h \
            + offs_k[:, None] * stride_right_k \
            + offs_j[None, :] * stride_right_j

        right = tl.load(
            right_ptrs,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        ).to(tl.float32)                     # (K, N)

        # matrix multiply for this K‑tile
        acc += tl.dot(left, right)            # (M, N)

    # ----------- store result -----------------------------
    out_ptrs = out_ptr \
        + b * stride_out_b \
        + h * stride_out_h \
        + offs_i[:, None] * stride_out_i \
        + offs_j[None, :] * stride_out_j

    tl.store(
        out_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


# ----------------------------------------------------------------------
# Entry point – custom Triton TriMul forward (outgoing)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator using Triton kernels.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, C] (float32)
        - mask         : torch.Tensor of shape [B, N, N] (float32) or None
        - weights      : dict of model weight tensors
        - config       : dict with keys ``dim``, ``hidden_dim``, ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, C] (float32)
    """
    # --------------------------------------------------------------
    # unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype          # expected float32
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32 via Triton)
    # ------------------------------------------------------------------
    M = bs * seq_len * seq_len
    x_flat = input_tensor.reshape(M, dim)               # (M, C)
    norm_weight = weights["norm.weight"].to(device, dtype=torch.float32)
    norm_bias   = weights["norm.bias"].to(device, dtype=torch.float32)
    x_norm_fp32 = triton_layernorm(x_flat, norm_weight, norm_bias)
    x_norm = x_norm_fp32.reshape(bs, seq_len, seq_len, dim)   # (B,N,N,C)

    # ------------------------------------------------------------------
    # 2️⃣ Combined linear projection (FP16)
    # ------------------------------------------------------------------
    x_norm_h = x_norm.half()                            # (B,N,N,C) fp16
    x_norm_flat = x_norm_h.reshape(M, dim)               # (M, C)

    combined_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(device, dtype=torch.float16)                   # (5*H, C)

    proj_out = F.linear(x_norm_flat, combined_weight)   # (M, 5*H) fp16

    # ------------------------------------------------------------------
    # 3️⃣ Split the fused projection
    # ------------------------------------------------------------------
    parts = torch.split(proj_out, hidden_dim, dim=1)
    left_proj_raw, right_proj_raw, left_gate_raw, right_gate_raw, out_gate_raw = parts

    left_proj_raw = left_proj_raw.view(bs, seq_len, seq_len, hidden_dim)
    right_proj_raw = right_proj_raw.view(bs, seq_len, seq_len, hidden_dim)
    left_gate_raw = left_gate_raw.view(bs, seq_len, seq_len, hidden_dim)
    right_gate_raw = right_gate_raw.view(bs, seq_len, seq_len, hidden_dim)
    out_gate_raw = out_gate_raw.view(bs, seq_len, seq_len, hidden_dim)

    # ------------------------------------------------------------------
    # 4️⃣ Optional binary mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_h = mask.to(device, dtype=torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left_proj_raw = left_proj_raw * mask_h
        right_proj_raw = right_proj_raw * mask_h

    # ------------------------------------------------------------------
    # 5️⃣ Gates (sigmoid) and gating of the projections
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_raw)          # (B,N,N,H) fp16
    right_gate = torch.sigmoid(right_gate_raw)        # (B,N,N,H) fp16
    out_gate = torch.sigmoid(out_gate_raw)            # (B,N,N,H) fp16

    left = left_proj_raw * left_gate
    right = right_proj_raw * right_gate

    # ------------------------------------------------------------------
    # 6️⃣ Core N³ reduction – custom Triton GEMM
    # ------------------------------------------------------------------
    out_hidden = torch.empty_like(left)               # (B,N,N,H) fp16

    # strides of the 4‑D tensors
    s_left_b, s_left_i, s_left_k, s_left_h = left.stride()
    s_right_b, s_right_j, s_right_k, s_right_h = right.stride()
    s_out_b, s_out_i, s_out_j, s_out_h = out_hidden.stride()

    # block sizes – chosen to keep BLOCK_M*BLOCK_N ≤ 1024 (GPU launch limit)
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32

    grid = (
        bs * hidden_dim,                       # combined batch*head dimension
        triton.cdiv(seq_len, BLOCK_M),        # i‑tiles
        triton.cdiv(seq_len, BLOCK_N),        # j‑tiles
    )

    trimul_gemm_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out_hidden,
        B=bs,
        N=seq_len,
        H=hidden_dim,
        stride_left_b=s_left_b,
        stride_left_i=s_left_i,
        stride_left_k=s_left_k,
        stride_left_h=s_left_h,
        stride_right_b=s_right_b,
        stride_right_j=s_right_j,
        stride_right_k=s_right_k,
        stride_right_h=s_right_h,
        stride_out_b=s_out_b,
        stride_out_i=s_out_i,
        stride_out_j=s_out_j,
        stride_out_h=s_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 7️⃣ Output LayerNorm (FP32 via Triton)
    # ------------------------------------------------------------------
    out_hidden_fp32 = out_hidden.float()               # cast for LN
    out_flat = out_hidden_fp32.reshape(M, hidden_dim)

    to_out_norm_weight = weights["to_out_norm.weight"].to(device, dtype=torch.float32)
    to_out_norm_bias   = weights["to_out_norm.bias"].to(device, dtype=torch.float32)

    out_norm_flat = triton_layernorm(
        out_flat,
        to_out_norm_weight,
        to_out_norm_bias,
        eps=1e-5,
        BLOCK_M=128,
        BLOCK_N=64,
    )
    out_norm = out_norm_flat.view(bs, seq_len, seq_len, hidden_dim)   # (B,N,N,H) FP32

    # ------------------------------------------------------------------
    # 8️⃣ Apply output gate and final projection back to model dim
    # ------------------------------------------------------------------
    gated = out_norm * out_gate.float()                # (B,N,N,H) FP32

    to_out_weight_fp16 = weights["to_out.weight"].to(device, dtype=torch.float16)  # (C, H)
    gated_h = gated.half()                             # (B,N,N,H) FP16
    final_fp16 = F.linear(gated_h, to_out_weight_fp16) # (B,N,N,C) FP16

    return final_fp16.float()
