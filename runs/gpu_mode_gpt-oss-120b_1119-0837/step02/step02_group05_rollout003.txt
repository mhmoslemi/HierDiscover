```python
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def tri_mul_kernel(
    out_ptr, left_ptr, right_ptr,
    M, N, K,
    stride_out_c, stride_out_m, stride_out_n,
    stride_left_c, stride_left_m, stride_left_k,
    stride_right_c, stride_right_k, stride_right_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # block indices
    pid_c = tl.program_id(2)  # channel index (B * hidden_dim)
    pid_m = tl.program_id(0)  # tile row
    pid_n = tl.program_id(1)  # tile col

    # offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator, FP32 for numeric stability
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # load left tile (M x K)
        left_ptrs = left_ptr \
            + pid_c * stride_left_c \
            + offs_m[:, None] * stride_left_m \
            + offs_k[None, :] * stride_left_k
        left = tl.load(left_ptrs,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0)   # half

        # load right tile (K x N) – we read it as K rows, N columns directly
        right_ptrs = right_ptr \
            + pid_c * stride_right_c \
            + offs_k[:, None] * stride_right_k \
            + offs_n[None, :] * stride_right_n
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_n[None, :],
                        other=0.0)   # half

        # tensor‑core dot
        acc += tl.dot(left, right)

    # write result, casting to float16 to halve output traffic
    out_ptrs = out_ptr \
        + pid_c * stride_out_c \
        + offs_m[:, None] * stride_out_m \
        + offs_n[None, :] * stride_out_n
    tl.store(out_ptrs,
             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    """
    Outgoing TriMul forward‐pass. Core O(N³·C) “triangle”
    multiplication is off‑loaded to a Triton kernel that computes
    per‑channel matrix‑multiply  out[b,i,j,c] = Σ_k left[b,i,k,c]·right[b,j,k,c].
    All other ops (layer‑norms, linear projections, sigmoid gates and the final
    linear) are performed with PyTorch in FP16/F32 for simplicity.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (FP32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )   # (B,N,N,dim) FP32

    # ------------------------------------------------------------------
    # 2) Cast to half and compute linear projections + gates
    # ------------------------------------------------------------------
    x_h = x.to(torch.float16)

    # linear projections (bias‑free)
    left = F.linear(x_h, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x_h, weights["right_proj.weight"].to(torch.float16))

    # sigmoid gates – still half precision
    left_gate = torch.sigmoid(F.linear(x_h, weights["left_gate.weight"].to(torch.float16)))
    right_gate = torch.sigmoid(F.linear(x_h, weights["right_gate.weight"].to(torch.float16)))
    out_gate = torch.sigmoid(F.linear(x_h, weights["out_gate.weight"].to(torch.float16)))  # (B,N,N,hidden)

    # apply gates
    left = left * left_gate
    right = right * right_gate

    # optional mask (mask shape [B,N,N], broadcast over hidden dim)
    if not nomask and mask is not None:
        mask_f = mask.to(torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 3) Flatten batch and hidden dim for Triton kernel
    #    layout: (C_total, N, N) where C_total = B * hidden_dim
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous()

    C_total = B * hidden_dim
    left_flat = left.reshape(C_total, N, N)    # (C_total, M, K)
    right_flat = right.reshape(C_total, N, N)  # (C_total, N, K) – we will read transposed

    # allocate output buffer (FP16 to reduce traffic)
    out_flat_fp16 = torch.empty_like(left_flat, dtype=torch.float16, device=device)

    # ------------------------------------------------------------------
    # 4) Triton kernel launch
    # ------------------------------------------------------------------
    # Tiling parameters – tuned for H100
    BLOCK_M = 256
    BLOCK_N = 256
    BLOCK_K = 128

    stride_out_c, stride_out_m, stride_out_n = out_flat_fp16.stride()
    stride_left_c, stride_left_m, stride_left_k = left_flat.stride()
    # right is accessed as (C, K, N) – swap K and N strides
    stride_right_c, stride_right_k, stride_right_n = (
        right_flat.stride(0), right_flat.stride(2), right_flat.stride(1)
    )

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # pid_m
        (N + BLOCK_N - 1) // BLOCK_N,   # pid_n
        C_total,                        # pid_c (folded batch*hidden)
    )

    # launch kernel
    tri_mul_kernel[grid](
        out_flat_fp16,
        left_flat,
        right_flat,
        N, N, N,
        stride_out_c, stride_out_m, stride_out_n,
        stride_left_c, stride_left_m, stride_left_k,
        stride_right_c, stride_right_k, stride_right_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )

    # ------------------------------------------------------------------
    # 5) Reshape back to (B,N,N,hidden) and cast to FP32 for post‑proc
    # ------------------------------------------------------------------
    out_hidden_fp16 = out_flat_fp16.reshape(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out_hidden = out_hidden_fp16.to(torch.float32)

    # ------------------------------------------------------------------
    # 6) Post‑processing: layer‑norm → out_gate → final linear
    # ------------------------------------------------------------------
    out_hidden = F.layer_norm(
        out_hidden,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )   # (B,N,N,hidden) float32

    # out_gate is half → cast before multiplication
    out = out_hidden * out_gate.to(torch.float32)

    # final projection to original dim (float32)
    out = F.linear(out, weights["to_out.weight"])

    return out

"""
Outgoing Triangle‑Multiplicative Update (TriMul) – forward only.
Core O(N³·C) “triangle” multiplication (for each hidden channel h):
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

All cheap ops (LayerNorms, Linear projections, sigmoid gates,
final projection) are performed with PyTorch.
The heavy multiplication is off‑loaded to a Triton kernel that:
  • works on FP16 inputs, accumulates in FP32,
  • stores the result in FP16 (halves output traffic),
  • uses a 128×128 tile and a 128‑wide K‑tile (fewer K‑loops),
  • launches one CTA per (tile‑row, tile‑col, hidden‑channel) – i.e. the
    batch dimension is folded into the hidden‑channel dimension.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: out = left @ rightᵀ   (per hidden channel)
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    out_ptr,               # *float16  (C_total, M, N) – result
    left_ptr,              # *float16  (C_total, M, K) – left matrix
    right_ptr,             # *float16  (C_total, K, N) – right matrix (read transposed)
    M, N, K,               # matrix sizes (all = seq_len)
    # strides (in elements, not bytes)
    stride_out_c, stride_out_m, stride_out_n,
    stride_left_c, stride_left_m, stride_left_k,
    stride_right_c, stride_right_k, stride_right_n,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Matrix‑multiply a single hidden‑channel slice:
       out[c, i, j] = Σ_k left[c, i, k] * right[c, k, j]"""

    pid_c = tl.program_id(2)          # hidden‑channel (B*hidden_dim) index
    pid_m = tl.program_id(0)          # tile row (i)
    pid_n = tl.program_id(1)          # tile col (j)

    # ------------------------------------------------------------------
    # Offsets for the current tile
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # ------------------------------------------------------------------
    # Accumulator (FP32 for numerics)
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K dimension
    # ------------------------------------------------------------------
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # left tile: (M, K)
        left_ptrs = left_ptr + pid_c * stride_left_c \
                    + offs_m[:, None] * stride_left_m \
                    + offs_k[None, :] * stride_left_k
        left = tl.load(left_ptrs,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0)                     # FP16

        # right tile (K, N) – note we read it transposed on the fly
        right_ptrs = right_ptr + pid_c * stride_right_c \
                     + offs_k[:, None] * stride_right_k \
                     + offs_n[None, :] * stride_right_n
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_n[None, :],
                        other=0.0)                    # FP16

        # Tensor‑core dot (M×K)·(K×N) → (M×N)
        acc += tl.dot(left, right)

    # ------------------------------------------------------------------
    # Write result (cast to FP16 to halve output traffic)
    # ------------------------------------------------------------------
    out_ptrs = out_ptr + pid_c * stride_out_c \
               + offs_m[:, None] * stride_out_m \
               + offs_n[None, :] * stride_out_n
    tl.store(out_ptrs,
             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Public entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        • input_tensor : torch.Tensor[B, N, N, C]   (float32)
        • mask         : torch.Tensor[B, N, N] or None
        • weights      : dict of model parameters (see reference)
        • config       : dict with keys "dim", "hidden_dim" and optional "nomask"

    Returns
    -------
    torch.Tensor[B, N, N, C]   (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (FP32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)  float32

    # ------------------------------------------------------------------
    # 2) Cast to half and compute projections + gates
    # ------------------------------------------------------------------
    x_h = x.to(torch.float16)

    # linear projections (bias‑free)
    left = F.linear(x_h, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x_h, weights["right_proj.weight"].to(torch.float16))

    # sigmoid gates (half‑precision)
    left_gate = torch.sigmoid(
        F.linear(x_h, weights["left_gate.weight"].to(torch.float16)))
    right_gate = torch.sigmoid(
        F.linear(x_h, weights["right_gate.weight"].to(torch.float16)))
    out_gate = torch.sigmoid(
        F.linear(x_h, weights["out_gate.weight"].to(torch.float16)))   # (B,N,N,hidden)

    # apply element‑wise gates
    left = left * left_gate
    right = right * right_gate

    # optional mask (broadcast over hidden dimension)
    if not nomask and mask is not None:
        mask_f = mask.to(torch.float16).unsqueeze(-1)   # (B, N, N, 1)
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 3) Rearrange to (C_total, N, N) for the Triton kernel
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape                # B, N, N, hidden_dim
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    C_total = B * hidden_dim
    left_flat = left.reshape(C_total, N, N)   # (C_total, M, K)
    right_flat = right.reshape(C_total, N, N) # (C_total, K, N) – we will read transposed

    # Output buffer – store in FP16 to reduce traffic
    out_flat_fp16 = torch.empty_like(left_flat, dtype=torch.float16, device=device)

    # ------------------------------------------------------------------
    # 4) Triton kernel launch
    # ------------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 128          # larger K‑tile ⇒ fewer loop iterations

    # strides in *elements* (not bytes)
    stride_out_c, stride_out_m, stride_out_n = out_flat_fp16.stride()
    stride_left_c, stride_left_m, stride_left_k = left_flat.stride()
    stride_right_c, stride_right_k, stride_right_n = right_flat.stride()

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # pid_m
        (N + BLOCK_N - 1) // BLOCK_N,   # pid_n
        C_total,                        # pid_c (folded batch × hidden)
    )

    tri_mul_kernel[grid](
        out_flat_fp16,
        left_flat,
        right_flat,
        N, N, N,
        stride_out_c, stride_out_m, stride_out_n,
        stride_left_c, stride_left_m, stride_left_k,
        stride_right_c, stride_right_k, stride_right_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )

    # ------------------------------------------------------------------
    # 5) Reshape back to (B, N, N, hidden) and convert to FP32
    # ------------------------------------------------------------------
    out_hidden_fp16 = out_flat_fp16.reshape(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out_hidden = out_hidden_fp16.to(torch.float32)   # (B, N, N, hidden)

    # ------------------------------------------------------------------
    # 6) Post‑processing: hidden‑dim LayerNorm → out_gate → final linear
    # ------------------------------------------------------------------
    out_hidden = F.layer_norm(
        out_hidden,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )   # (B, N, N, hidden)  float32

    # out_gate is half‑precision → cast before multiplication
    out = out_hidden * out_gate.to(torch.float32)

    # Final projection back to original dimensionality (float32)
    out = F.linear(out, weights["to_out.weight"])   # (B, N, N, dim)

    return out
