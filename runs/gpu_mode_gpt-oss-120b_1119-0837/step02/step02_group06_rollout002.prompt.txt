import torch
from torch import nn, einsum
import math

# Reference code in PyTorch
class TriMul(nn.Module):
    def __init__(
        self,
        dim: int,
        hidden_dim: int,
    ):
        super().__init__()

        self.norm = nn.LayerNorm(dim)

        self.left_proj = nn.Linear(dim, hidden_dim, bias=False)
        self.right_proj = nn.Linear(dim, hidden_dim, bias=False)

        self.left_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.right_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.out_gate = nn.Linear(dim, hidden_dim, bias=False)

        self.to_out_norm = nn.LayerNorm(hidden_dim)
        self.to_out = nn.Linear(hidden_dim, dim, bias=False)

    def forward(self, x: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
        """
        x: [bs, seq_len, seq_len, dim]
        mask: [bs, seq_len, seq_len]

        Returns:
            output: [bs, seq_len, seq_len, dim]
        """
        batch_size, seq_len, _, dim = x.shape

        x = self.norm(x)

        left = self.left_proj(x)
        right = self.right_proj(x)

        mask = mask.unsqueeze(-1)
        left = left * mask
        right = right * mask

        left_gate = self.left_gate(x).sigmoid()
        right_gate = self.right_gate(x).sigmoid()
        out_gate = self.out_gate(x).sigmoid()

        left = left * left_gate
        right = right * right_gate

        out = einsum('... i k d, ... j k d -> ... i j d', left, right)
        # This einsum is the same as the following:
        # out = torch.zeros(batch_size, seq_len, seq_len, dim, device=x.device)
        
        # # Compute using nested loops
        # for b in range(batch_size):
        #     for i in range(seq_len):
        #         for j in range(seq_len):
        #             # Compute each output element
        #             for k in range(seq_len):
        #                 out[b, i, j] += left[b, i, k, :] * right[b, j, k, :]

        out = self.to_out_norm(out)
        out = out * out_gate
        return self.to_out(out)

"""
TriMul (outgoing) – fast Triton implementation
================================================

The module receives a 4‑D tensor X ∈ ℝ^{B×N×N×C} and a (optional) mask.
The forward pass consists of

1. Layer‑norm over the feature dimension C.
2. Two bias‑free linear projections to a hidden dimension H (left/right).
3. Three element‑wise sigmoid gates (left, right, out).
4. Optional masking of the left/right tensors.
5. Core triangle multiplication (the expensive step):
       out_{b,i,j,h} = Σ_k left_{b,i,k,h} ⋅ right_{b,j,k,h}
   which is exactly a batched matrix multiply
       out_{b,:, :,h} = left_{b,:, :,h} @ right_{b,:, :,h}ᵀ .
   This is performed by the Triton kernel `tri_mul_kernel`.  All
   arithmetic inside the kernel is FP16 → FP32 accumulation (tensor‑core
   friendly).  The kernel works on a fused batch‑&‑channel dimension
   C_total = B·H, i.e. each program‑ID processes one (B,H) slice.
6. Layer‑norm over the hidden dimension, gating with ``out_gate`` and a
   final linear projection back to C.

The kernel is tuned for H100:
  * BLOCK_M = BLOCK_N = 128, BLOCK_K = 32 (full‑warp tiles).
  * 8 warps per block, 3 pipeline stages.
  * Half‑precision loads, FP32 accumulation.
  * No shared memory shuffling – `tl.dot` already emits Tensor‑Core
    instructions.

The surrounding Python code uses only regular PyTorch ops (they are
already highly optimized) and keeps the memory footprint low by performing
in‑place gating and by discarding the float‑32 copy of the input as soon
as the half‑precision tensors are ready.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------
# Triton kernel: batched GEMM (left @ rightᵀ) for the triangle mul
# --------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    out_ptr,            # float32*   (C_total, M, N)
    a_ptr,              # float16*   (C_total, M, K)  left
    b_ptr,              # float16*   (C_total, K, N)  rightᵀ (loaded transposed)
    M, N, K,            # matrix sizes (all = seq_len)
    stride_out_m, stride_out_n, stride_out_c,   # output strides
    stride_a_m, stride_a_k, stride_a_c,        # left strides
    stride_b_k, stride_b_n, stride_b_c,        # rightᵀ strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # program IDs
    pid_m = tl.program_id(0)   # block row (i)
    pid_n = tl.program_id(1)   # block col (j)
    pid_c = tl.program_id(2)   # fused batch‑&‑hidden channel

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # tile‑level masks
    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the reduction dimension K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # pointers for the current tile of left (M×K) and rightᵀ (K×N)
        a_tile_ptr = a_ptr + pid_c * stride_a_c \
                     + offs_m[:, None] * stride_a_m \
                     + offs_k[None, :] * stride_a_k                # (M,K)

        b_tile_ptr = b_ptr + pid_c * stride_b_c \
                     + offs_k[:, None] * stride_b_k \
                     + offs_n[None, :] * stride_b_n                # (K,N)

        # masked loads (FP16)
        a = tl.load(a_tile_ptr,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)                                    # (M,K)
        b = tl.load(b_tile_ptr,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)                                    # (K,N)

        # GEMM accumulation – Tensor‑Core friendly
        acc += tl.dot(a, b)      # (M,N) FP32

    # write result tile back to global memory
    out_tile_ptr = out_ptr + pid_c * stride_out_c \
                   + offs_m[:, None] * stride_out_m \
                   + offs_n[None, :] * stride_out_n
    tl.store(out_tile_ptr,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])


# --------------------------------------------------------------
# Entry‑point used by the test harness
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor[B, N, N, C] (float32)
            - mask: torch.Tensor[B, N, N] (float32/bool) – may be None
            - weights: dict with all model parameters
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor[B, N, N, C] – processed tensor (float32)
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1) Layer‑norm over the feature dimension C
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, C)  float32

    # ------------------------------------------------------------------
    # 2) Cast to FP16 for the cheap linear layers
    # ------------------------------------------------------------------
    x_h = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Linear projections (bias‑free)
    # ------------------------------------------------------------------
    left = F.linear(x_h, weights["left_proj.weight"].to(torch.float16))   # (B,N,N,H)
    right = F.linear(x_h, weights["right_proj.weight"].to(torch.float16))# (B,N,N,H)

    # ------------------------------------------------------------------
    # 4) Gates (sigmoid) – also FP16, will be cast back later if needed
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x_h, weights["left_gate.weight"].to(torch.float16)))
    right_gate = torch.sigmoid(F.linear(x_h, weights["right_gate.weight"].to(torch.float16)))
    out_gate   = torch.sigmoid(F.linear(x_h, weights["out_gate.weight"].to(torch.float16)))

    # ------------------------------------------------------------------
    # 5) Apply gates (and optional mask) – in‑place to save memory
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    if not nomask and mask is not None:
        # mask shape (B,N,N) → (B,N,N,1)
        mask_f = mask.to(torch.float16).unsqueeze(-1)
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 6) Prepare tensors for the Triton kernel
    #    We want shape (C_total, N, N) where C_total = B * hidden_dim.
    #    We keep the sequence dimensions as the last two axes so that the
    #    memory layout is contiguous for the tiles.
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape

    # Move hidden dim before the two seq dims → (B, H, N, N)
    left = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()

    C_total = B * hidden_dim
    left_flat = left.reshape(C_total, N, N)    # (C, M, K) with M=K=N
    # right will be accessed transposed inside the kernel, so we keep the
    # original layout (C, N, N) and let the kernel reinterpret it as (C, K, N)
    right_flat = right.reshape(C_total, N, N)

    # Output buffer – fp32 accumulation
    out_flat = torch.empty_like(left_flat, dtype=torch.float32)

    # ------------------------------------------------------------------
    # 7) Triton kernel launch
    # ------------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    # Strides are expressed in **elements** (not bytes)
    stride_out_c = out_flat.stride(0)
    stride_out_m = out_flat.stride(1)
    stride_out_n = out_flat.stride(2)

    stride_a_c = left_flat.stride(0)
    stride_a_m = left_flat.stride(1)
    stride_a_k = left_flat.stride(2)

    stride_b_c = right_flat.stride(0)
    # right will be treated as (C, K, N) → K dimension corresponds to dim 2,
    # N dimension corresponds to dim 1 after transposition.
    stride_b_k = right_flat.stride(2)   # original last dimension (N)
    stride_b_n = right_flat.stride(1)   # original middle dimension (N)

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # blocks over i‑axis
        (N + BLOCK_N - 1) // BLOCK_N,   # blocks over j‑axis
        C_total,                        # one block per (batch, hidden) slice
    )

    tri_mul_kernel[grid](
        out_flat,
        left_flat,
        right_flat,
        N, N, N,                     # M, N, K (all equal seq_len)
        stride_out_m, stride_out_n, stride_out_c,
        stride_a_m, stride_a_k, stride_a_c,
        stride_b_k, stride_b_n, stride_b_c,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,                 # H100 can handle many warps per block
        num_stages=3,                # pipelining for better occupancy
    )

    # ------------------------------------------------------------------
    # 8) Reshape kernel output back to (B, N, N, H)
    # ------------------------------------------------------------------
    out = out_flat.reshape(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 9) Final post‑processing (layer‑norm, out‑gate, final linear)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    # out_gate was FP16 – promote to fp32 for the multiplication
    out = out * out_gate.to(torch.float32)
    out = F.linear(out, weights["to_out.weight"])   # (B, N, N, C)

    return out
