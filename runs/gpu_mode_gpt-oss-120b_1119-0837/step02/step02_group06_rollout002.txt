"""
Custom Triton kernel for the outgoing TriMul operation.

The operator:
- LayerNorm over the input channel dimension.
- Two bias‑free linear projections (left/right) to a hidden dimension H.
- Three sigmoid gates (left, right, out) applied element‑wise.
- Optional mask applied to left/right after gating.
- Core triangle multiplication:
     out_{b,i,j,h} = Σ_k left_{b,i,k,h} * right_{b,j,k,h}
  performed as a batched GEMM left @ rightᵀ with a fused batch‑&‑hidden C dimension.
- LayerNorm + out‑gate over the hidden dimension followed by a final linear
  projection back to the original channel dimension.

The heavy part (the O(N³) loop) is implemented as a Triton kernel
`tri_mul_kernel`.  The kernel works on half‑precision inputs, accumulates in
FP32, and processes a fused (batch × hidden) slice per program‑ID.  Block
sizes are tuned for H100 (128×128 tiles with K‑tile of 32) but automatically
fallback to smaller tiles for tiny sequences.

All other operations are performed with highly‑optimized PyTorch kernels.
The final output is returned as FP32.

"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM (left @ rightᵀ) for the triangle multiplication
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    out_ptr,   # float32*   (C_total, M, N)
    a_ptr,     # float16*   (C_total, M, K)  left
    b_ptr,     # float16*   (C_total, K, N)  rightᵀ (loaded transposed)
    M, N, K,   # integer sizes (all = seq_len)
    stride_out_c, stride_out_m, stride_out_n,  # output strides
    stride_a_c,  stride_a_m, stride_a_k,       # left strides
    stride_b_c,  stride_b_k, stride_b_n,       # rightᵀ strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM: out = left @ rightᵀ for each (batch, hidden) slice.

    Arguments are similar to a typical blocked matrix multiplication.
    Accumulation is performed in FP32 for numerical stability.
    """

    pid_m = tl.program_id(0)   # block row of output (i)
    pid_n = tl.program_id(1)   # block column of output (j)
    pid_c = tl.program_id(2)   # fused (batch * hidden) index

    # Compute global offsets for this tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for boundary handling
    mask_m = offs_m < M
    mask_n = offs_n < N

        mask_k = offs_k < K

        a_tile_ptr = a_ptr + pid_c * stride_a_c \
                     + offs_m[:, None] * stride_a_m \
                     + offs_k[None, :] * stride_a_k      # (M, K)

        b_tile_ptr = b_ptr + pid_c * stride_b_c \
                     + offs_k[:, None] * stride_b_k \
                     + offs_n[None, :] * stride_b_n      # (K, N)

        a = tl.load(a_tile_ptr,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)
        b = tl.load(b_tile_ptr,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        acc += tl.dot(a, b)

    out_tile_ptr = out_ptr + pid_c * stride_out_c \
                   + offs_m[:, None] * stride_out_m \
                   + offs_n[None, :] * stride_out_n
    tl.store(out_tile_ptr,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])

"""
TriMul (outgoing) – Triton‑accelerated implementation
====================================================

The operator works on a 4‑D tensor X ∈ ℝ^{B×N×N×C} and (optionally) a mask.
The forward pass is:

1. LayerNorm over the last dimension C.
2. Two bias‑free linear projections (left/right) to a hidden dimension H.
3. Three element‑wise sigmoid gates (left, right, out).
4. Optional masking of the left/right tensors.
5. Core triangle multiplication (the heavy O(N³) part):
       out_{b,i,j,h} = Σ_k left_{b,i,k,h} · right_{b,j,k,h}
   which is exactly a batched GEMM
       out_{b,:, :,h} = left_{b,:, :,h} @ right_{b,:, :,h}ᵀ .
   The GEMM is performed by the Triton kernel `tri_mul_kernel`.  The kernel
   fuses the batch and hidden dimensions (C_total = B·H) so each program‑ID
   processes one (batch, hidden) slice.  Inputs are FP16, accumulation is
   FP32 (Tensor‑Core friendly).
6. LayerNorm over the hidden dimension, gating with ``out_gate`` and a final
   linear projection back to the original channel dimension.

All non‑heavy operations are done with highly‑optimized PyTorch kernels;
the final output is returned as FP32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM (left @ rightᵀ) for the triangle multiplication
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    out_ptr,   # float32*   (C_total, M, N)
    a_ptr,     # float16*   (C_total, M, K)  left
    b_ptr,     # float16*   (C_total, K, N)  rightᵀ (loaded transposed)
    M, N, K,   # integer sizes (all = seq_len)
    stride_out_c, stride_out_m, stride_out_n,  # output strides
    stride_a_c,  stride_a_m, stride_a_k,       # left strides
    stride_b_c,  stride_b_k, stride_b_n,       # rightᵀ strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched GEMM: out = left @ rightᵀ for each (batch, hidden) slice.

    Arguments are analogous to a standard blocked matrix multiplication.  The
    kernel accumulates in FP32 for numerical stability while all loads are
    FP16.  Tile sizes (BLOCK_*) are passed as compile‑time constants.
    """
    pid_m = tl.program_id(0)  # block row (i)
    pid_n = tl.program_id(1)  # block col (j)
    pid_c = tl.program_id(2)  # fused (batch * hidden) index

    # Global offsets for this tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for boundary handling
    mask_m = offs_m < M
    mask_n = offs_n < N

    # FP32 accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over reduction dimension K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Pointer to the current tile of left (M×K) and rightᵀ (K×N)
        a_tile_ptr = a_ptr + pid_c * stride_a_c \
                     + offs_m[:, None] * stride_a_m \
                     + offs_k[None, :] * stride_a_k

        b_tile_ptr = b_ptr + pid_c * stride_b_c \
                     + offs_k[:, None] * stride_b_k \
                     + offs_n[None, :] * stride_b_n

        # Masked loads (outside region => 0)
        a = tl.load(a_tile_ptr,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)
        b = tl.load(b_tile_ptr,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # Tensor‑core friendly accumulation
        acc += tl.dot(a, b)   # (M,N) FP32

    # Write tile back
    out_tile_ptr = out_ptr + pid_c * stride_out_c \
                   + offs_m[:, None] * stride_out_m \
                   + offs_n[None, :] * stride_out_n
    tl.store(out_tile_ptr,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])

    left_gate = torch.sigmoid(

    left = left.permute(0, 3, 1, 2).contiguous()   # (B, H, i, k)

    stride_out_c = out_flat.stride(0)
    stride_out_m = out_flat.stride(1)
    stride_out_n = out_flat.stride(2)

    stride_a_c = left_flat.stride(0)
    stride_a_m = left_flat.stride(1)
    stride_a_k = left_flat.stride(2)

    stride_b_c = right_flat.stride(0)
    # right is accessed as (C_total, K, N) – K is the last axis after reshape
    stride_b_k = right_flat.stride(2)   # reduction dimension (k)
    stride_b_n = right_flat.stride(1)   # column dimension (j)

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        C_total,                        # one block per (batch, hidden)
    )

    tri_mul_kernel[grid](

        left_flat,
        right_flat,
        N, N, N,                       # M, N, K (all = seq_len)

        stride_a_c, stride_a_m, stride_a_k,
        stride_b_c, stride_b_k, stride_b_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=3,
    )

    # --------------------------------------------------------------
    # 8) Reshape kernel output back to (B, N, N, hidden_dim)
    # --------------------------------------------------------------
    out = out_flat.reshape(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 9) Final post‑processing: hidden‑dim LayerNorm, out‑gate, final proj
    # --------------------------------------------------------------
    out = F.layer_norm(

        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    # out_gate was computed in FP16 → promote to FP32 for the element‑wise product
    out = out * out_gate.to(torch.float32)
