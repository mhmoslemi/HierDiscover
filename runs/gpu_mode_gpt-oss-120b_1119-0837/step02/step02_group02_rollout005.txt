@triton.jit
def _tri_mul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    batch_stride, # stride between batches
    hidden_dim: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(axis=0)  # tile id
    batch_id = tl.program_id(axis=1)  # batch (b*H + h)
   
    # Determine b, h
    b = batch_id // hidden_dim
    h = batch_id % hidden_dim

    # Compute base pointer for this batch
    a_base = a_ptr + b * batch_stride + h
    b_base = b_ptr + b * batch_stride + h
    c_base = c_ptr + b * batch_stride + h

    # offsets for tile positions as before
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    block_m = pid // num_blocks_n
    block_n = pid % num_blocks_n

    # offsets for rows and columns
    offs_m = block_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = block_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Create accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = a_base + (offs_m[:, None] * stride_am) + (offs_k[None, :] * stride_ak)
        a = tl.load(a_ptrs, mask=(offs_m[:, None] < M) & mask_k[None, :], other=0.0)

        # B tile: B_T shape (BLOCK_K, BLOCK_N)
        b_ptrs = b_base + (offs_k[:, None] * stride_bk) + (offs_n[None, :] * stride_bn)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & (offs_n[None, :] < N), other=0.0)

        acc += tl.dot(a, b)
    
    # write to C
    c_ptrs = c_base + (offs_m[:, None] * stride_cm) + (offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, acc.to(tl.float16), mask=(offs_m[:, None] < M) & (offs_n[None, :] < N))

"""
Implementation of TriMul outgoing operator using Triton for the
batched GEMM `C = A @ B^T` where A and B are 4‑D tensors of shape
[batch, seq_len, seq_len, hidden_dim]. The kernel works on the FP16
data, accumulates in FP32 and writes back in FP16. All other
operations (layer‑norm, linear projections, gates, masking) stay in
PyTorch but operate in half precision to keep memory traffic low.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def _tri_mul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    batch_stride,
    HIDDEN: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched GEMM for TriMul.

    For each (b, h) pair we compute:
        out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]
    """
    pid = tl.program_id(axis=0)   # tile over (i, j)
    batch_id = tl.program_id(axis=1)  # flattened (batch, hidden)

    # Decompose batch_id into batch and hidden channel
    b = batch_id // HIDDEN
    h = batch_id % HIDDEN

    # Base pointers for this (b, h)
    a_batch = a_ptr + b * batch_stride + h
    b_batch = b_ptr + b * batch_stride + h
    c_batch = c_ptr + b * batch_stride + h

    # --------------------------------------------------------------
    # Tile coordinates
    # --------------------------------------------------------------
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    block_m = pid // num_blocks_n
    block_n = pid % num_blocks_n

    offs_m = block_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = block_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Accumulator in FP32
    # --------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Main K loop
    # --------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_off = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_off < K

        # Load A tile: [BLOCK_M, BLOCK_K] (i, k)
        a_ptrs = a_batch + (offs_m[:, None] * stride_am) + (k_off[None, :] * stride_ak)
        a = tl.load(
            a_ptrs,
            mask=(offs_m[:, None] < M) & mask_k[None, :],
            other=0.0,
        )

        # Load B tile (already transposed): [BLOCK_K, BLOCK_N] (k, j)
        b_ptrs = b_batch + (k_off[:, None] * stride_bk) + (offs_n[None, :] * stride_bn)
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & (offs_n[None, :] < N),
            other=0.0,
        )

        # Rank‑K update
        acc += tl.dot(a, b)   # FP16 × FP16 → FP32

    # --------------------------------------------------------------
    # Write the result back (store as FP16)
    # --------------------------------------------------------------
    c_ptrs = c_batch + (offs_m[:, None] * stride_cm) + (offs_n[None, :] * stride_cn)
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=(offs_m[:, None] < M) & (offs_n[None, :] < N),
    )

    # --------------------------------------------------------------

    right_gate = torch.sigmoid(F.linear(x_fp16, weights["right_gate.weight"].half()))
    out_gate = torch.sigmoid(F.linear(x_fp16, weights["out_gate.weight"].half()))

    right = right * right_gate

    # --------------------------------------------------------------

    out = torch.empty_like(left, dtype=torch.float16, device=device)  # [B, N, N, H]

    stride_ak = left.stride(2)      # H
    stride_cm = out.stride(1)       # N * H
    stride_cn = out.stride(2)       # H

    stride_bk = right.stride(2)     # H
    stride_bn = right.stride(1)     # N * H

    batch_stride = left.stride(0)   # N * N * H

    BLOCK_N = 128
    BLOCK_K = 32

"""
TriMul (outgoing) – fast Triton implementation.

The expensive O(N³) contraction

    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

is a batched GEMM (A @ Bᵀ) where each hidden channel h is an
independent batch.  The kernel works completely in FP16, accumulates
in FP32 (tensor‑core friendly) and writes the result back in FP16.
All light‑weight operations (layer‑norm, linear projections,
sigmoids, optional masking) stay in PyTorch but are also performed
in half‑precision to keep memory traffic low.

Key ideas
---------
* No tensor transposes – we compute directly on the original
  [B, N, N, H] layout.  This eliminates a costly O(B·N²·H) copy that
  the naïve implementation performed.
* The hidden dimension is folded into the batch axis: each (batch,
  hidden) pair is processed by one program instance.
* Tile sizes (128×128×32) match the FP16 tensor‑core granularity of an
  H100 and give high occupancy with 8 warps and double‑buffering.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _tri_mul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,          # FP16 pointers
    M, N, K,                      # matrix dimensions (M=N=seq_len, K=seq_len)
    stride_am, stride_ak,         # A row/col strides
    stride_bk, stride_bn,         # B (already transposed) row/col strides
    stride_cm, stride_cn,         # C row/col strides
    batch_stride,                 # stride between successive batches (B dimension)
    HIDDEN: tl.constexpr,         # hidden dimension (compile‑time)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute, for each (b, h) pair,
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    A is left, B is rightᵀ.
    """
    pid = tl.program_id(axis=0)          # tile over (i,j)
    batch_id = tl.program_id(axis=1)     # flattened (batch, hidden)

    # Decompose flattened batch id
    b = batch_id // HIDDEN
    h = batch_id % HIDDEN

    # Base pointers for this (b, h)
    a_batch = a_ptr + b * batch_stride + h
    b_batch = b_ptr + b * batch_stride + h
    c_batch = c_ptr + b * batch_stride + h

    # --------------------------------------------------------------
    # Tile coordinates
    # --------------------------------------------------------------
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    block_m = pid // num_blocks_n
    block_n = pid % num_blocks_n

    offs_m = block_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = block_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # --------------------------------------------------------------
    # Accumulator (FP32)
    # --------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Main K loop
    # --------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_off = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_off < K

        # Load A tile: shape (BLOCK_M, BLOCK_K)  → (i, k)
        a_ptrs = a_batch + (offs_m[:, None] * stride_am) + (k_off[None, :] * stride_ak)
        a = tl.load(
            a_ptrs,
            mask=(offs_m[:, None] < M) & mask_k[None, :],
            other=0.0,
        )

        # Load B tile (already transposed): shape (BLOCK_K, BLOCK_N) → (k, j)
        b_ptrs = b_batch + (k_off[:, None] * stride_bk) + (offs_n[None, :] * stride_bn)
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & (offs_n[None, :] < N),
            other=0.0,
        )

        # Rank‑K update
        acc += tl.dot(a, b)   # FP16×FP16 → FP32

    # --------------------------------------------------------------
    # Write back (store as FP16)
    # --------------------------------------------------------------
    c_ptrs = c_batch + (offs_m[:, None] * stride_cm) + (offs_n[None, :] * stride_cn)
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=(offs_m[:, None] < M) & (offs_n[None, :] < N),
    )

    """
    Forward pass of the “outgoing” TriMul block (no gradients).

    device = input_tensor.device
    eps = 1e-5

    hidden_dim = config["hidden_dim"]

    # 1) Input LayerNorm (keep FP32 for numeric stability)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,

        bias=weights["norm.bias"],
        eps=eps,

    left = left * left_gate
    right = right * right_gate

    stride_ak = left.stride(2)          # H      (col  k   stride)

    BLOCK_M, BLOCK_N, BLOCK_K = 128, 128, 32

        total_batches,

        left,
        right,
        out,
        N, N, N,
        stride_am, stride_ak,
        stride_bk, stride_bn,
        stride_cm, stride_cn,
        batch_stride,
        hidden_dim,                 # HIDDEN (compile‑time constant)
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,                # H100 performs well with 8 warps
        num_stages=2,               # double‑buffering
    )

    # --------------------------------------------------------------
    # 6) Hidden‑dim layer‑norm, out‑gate, final projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].half(),
        bias=weights["to_out_norm.bias"].half(),
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].half())

    # --------------------------------------------------------------
    # 7) Cast back to float32 for downstream consumption
    # --------------------------------------------------------------
    return out.float()
