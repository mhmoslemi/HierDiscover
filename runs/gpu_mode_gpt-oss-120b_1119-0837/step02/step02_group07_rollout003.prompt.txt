import torch
from torch import nn, einsum
import math

# Reference code in PyTorch
class TriMul(nn.Module):
    def __init__(
        self,
        dim: int,
        hidden_dim: int,
    ):
        super().__init__()

        self.norm = nn.LayerNorm(dim)

        self.left_proj = nn.Linear(dim, hidden_dim, bias=False)
        self.right_proj = nn.Linear(dim, hidden_dim, bias=False)

        self.left_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.right_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.out_gate = nn.Linear(dim, hidden_dim, bias=False)

        self.to_out_norm = nn.LayerNorm(hidden_dim)
        self.to_out = nn.Linear(hidden_dim, dim, bias=False)

    def forward(self, x: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
        """
        x: [bs, seq_len, seq_len, dim]
        mask: [bs, seq_len, seq_len]

        Returns:
            output: [bs, seq_len, seq_len, dim]
        """
        batch_size, seq_len, _, dim = x.shape

        x = self.norm(x)

        left = self.left_proj(x)
        right = self.right_proj(x)

        mask = mask.unsqueeze(-1)
        left = left * mask
        right = right * mask

        left_gate = self.left_gate(x).sigmoid()
        right_gate = self.right_gate(x).sigmoid()
        out_gate = self.out_gate(x).sigmoid()

        left = left * left_gate
        right = right * right_gate

        out = einsum('... i k d, ... j k d -> ... i j d', left, right)
        # This einsum is the same as the following:
        # out = torch.zeros(batch_size, seq_len, seq_len, dim, device=x.device)
        
        # # Compute using nested loops
        # for b in range(batch_size):
        #     for i in range(seq_len):
        #         for j in range(seq_len):
        #             # Compute each output element
        #             for k in range(seq_len):
        #                 out[b, i, j] += left[b, i, k, :] * right[b, j, k, :]

        out = self.to_out_norm(out)
        out = out * out_gate
        return self.to_out(out)

"""
TriMul (outgoing) – fast forward pass.

The only N³‑cost part is the “triangle” multiplication
    out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h] .
We treat each (batch × hidden) slice as an independent GEMM and
implement it with a custom Triton kernel that uses fp16 inputs,
fp32 accumulation and Tensor‑Core friendly tiling.
All other ops (layer‑norms, linear projections, gates, final
projection) stay in PyTorch for simplicity.

The kernel uses a 2‑D tile (BLOCK_M × BLOCK_N) over the
output matrix and a 1‑D tile (BLOCK_K) over the reduction dimension.
Tile sizes are chosen adaptively to the sequence length to keep
register/SM‑memory usage low while maximising occupancy.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    out_ptr,       # float32*
    a_ptr,         # float16*   (C, M, K)   – left  tensor
    b_ptr,         # float16*   (C, K, N)   – right tensor (K major)
    M, N, K,       # matrix sizes (all = seq_len)
    stride_out_m, stride_out_n, stride_out_c,   # output strides
    stride_a_m, stride_a_k, stride_a_c,        # left strides
    stride_b_k, stride_b_n, stride_b_c,        # right strides (after “transpose”)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched GEMM: for each channel `c` (folded batch*hidden) compute
        out_c[M,N] += left_c[M,K] @ right_c[K,N]
    All tensors are stored in row‑major order.  The right tensor is
    logically transposed (K,N) by the pointer arithmetic.
    """
    pid_m = tl.program_id(0)  # block row   → i
    pid_n = tl.program_id(1)  # block col   → j
    pid_c = tl.program_id(2)  # channel id  → (b,h)

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # masks for out‑of‑bounds (required for non‑multiple of BLOCK sizes)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -----------------------------------------------------------------
    # Loop over K dimension in BLOCK_K steps
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # pointers to the current tiles
        a_tile = a_ptr + pid_c * stride_a_c \
                 + offs_m[:, None] * stride_a_m \
                 + offs_k[None, :] * stride_a_k          # (M,K)

        b_tile = b_ptr + pid_c * stride_b_c \
                 + offs_k[:, None] * stride_b_k \
                 + offs_n[None, :] * stride_b_n          # (K,N)

        # load with masking (zero for out‑of‑bounds)
        a = tl.load(a_tile,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)                             # FP16
        b = tl.load(b_tile,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)                             # FP16

        # GEMM accumulate: FP16 * FP16 → FP32
        acc += tl.dot(a, b)

    # -----------------------------------------------------------------
    # write out tile
    out_tile = out_ptr + pid_c * stride_out_c \
               + offs_m[:, None] * stride_out_m \
               + offs_n[None, :] * stride_out_n
    tl.store(out_tile, acc,
             mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module (AlphaFold3 style).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor[B, N, N, C] (float32)
            - mask         : torch.Tensor[B, N, N] (float32/bool) or None
            - weights      : dict of model parameters (see keys below)
            - config       : dict with keys
                * "dim"        – input channel dimension C
                * "hidden_dim"– hidden dimension H
                * "nomask"     – bool, True → ignore mask

    Returns:
        torch.Tensor[B, N, N, C] (float32) – module output
    """
    # -----------------------------------------------------------------
    # unpack
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    device = input_tensor.device

    # -----------------------------------------------------------------
    # 1) Input LayerNorm (float32)
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, C]  (FP32)

    # -----------------------------------------------------------------
    # 2) Cast to fp16 for all linear ops
    x_h = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 3) Linear projections (no bias) → hidden_dim
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left = F.linear(x_h, left_proj_w)   # [B, N, N, H]  fp16
    right = F.linear(x_h, right_proj_w)  # [B, N, N, H]  fp16

    # -----------------------------------------------------------------
    # 4) Gates (sigmoid) – also fp16
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    left_gate = torch.sigmoid(F.linear(x_h, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_h, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x_h, out_gate_w))   # will be cast later

    # -----------------------------------------------------------------
    # 5) Apply gates (and optional mask)
    left = left * left_gate
    right = right * right_gate

    if not nomask and mask is not None:
        # mask is 0/1 – broadcast over hidden dim
        mask_h = mask.to(torch.float16).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_h
        right = right * mask_h

    # -----------------------------------------------------------------
    # 6) Prepare tensors for the Triton kernel
    #    Move hidden dimension before the two sequence dims:
    #    (B, N, N, H) → (B, H, N, N) → flatten B*H as first dim.
    B, N, _, _ = left.shape
    left = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    C_total = B * hidden_dim                       # = batch * hidden_dim
    left_flat = left.reshape(C_total, N, N)        # (C, M, K)   – K = N
    right_flat = right.reshape(C_total, N, N)      # (C, N, K) – will be read as (K,N) in kernel

    # output buffer (FP32)
    out_flat = torch.empty_like(left_flat, dtype=torch.float32, device=device)

    # -----------------------------------------------------------------
    # 7) Triton GEMM kernel launch
    #    Choose tile sizes adaptively (empirically good for H100)
    if N <= 256:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 32
    elif N <= 512:
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 32
    else:
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 64

    stride_out_c = out_flat.stride(0)
    stride_out_m = out_flat.stride(1)
    stride_out_n = out_flat.stride(2)

    stride_a_c = left_flat.stride(0)
    stride_a_m = left_flat.stride(1)
    stride_a_k = left_flat.stride(2)

    stride_b_c = right_flat.stride(0)
    stride_b_k = right_flat.stride(2)   # innermost dim → K
    stride_b_n = right_flat.stride(1)   # middle dim → N

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        C_total,
    )

    # More warps give higher occupancy on H100; 8 works well for large N.
    num_warps = 8 if N >= 512 else 4

    tri_mul_kernel[grid](
        out_flat,
        left_flat,
        right_flat,
        N, N, N,
        stride_out_m, stride_out_n, stride_out_c,
        stride_a_m, stride_a_k, stride_a_c,
        stride_b_k, stride_b_n, stride_b_c,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=num_warps,
    )

    # -----------------------------------------------------------------
    # 8) Reshape back to (B, N, N, H)
    out = out_flat.reshape(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,H] FP32

    # -----------------------------------------------------------------
    # 9) Post‑processing (LayerNorm → out‑gate → final linear)
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    # out_gate was computed in fp16; cast to fp32 for final multiplication
    out = out * out_gate.to(torch.float32)

    # final projection back to original dim
    out = F.linear(out, weights["to_out.weight"])   # (B,N,N,dim) FP32

    return out
