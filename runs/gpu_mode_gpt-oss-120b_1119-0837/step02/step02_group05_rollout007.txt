import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def tri_mul_kernel(
    out_ptr, left_ptr, right_ptr,
    M, N, K,
    # strides (in elements)
    stride_out_batch,
    stride_out_i,
    stride_out_j,
    stride_left_batch,
    stride_left_i,
    stride_left_k,
    stride_right_batch,
    stride_right_i,
    stride_right_k,
    # compile‑time constants
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    HIDDEN_DIM: tl.constexpr,
):
    """Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]."""
    pid_c = tl.program_id(2)   # combined batch*hidden index
    pid_m = tl.program_id(0)   # block row i
    pid_n = tl.program_id(1)   # block col j

    # decompose the combined channel index
    b_idx = pid_c // HIDDEN_DIM
    h_idx = pid_c % HIDDEN_DIM
    # base offset for this (b,h) slice
    base = b_idx * stride_left_batch + h_idx

    # coordinate offsets within the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator (FP32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # left tile: shape (M, K)
        left_tile_ptr = left_ptr + base \
            + offs_m[:, None] * stride_left_i \
            + offs_k[None, :] * stride_left_k
        left = tl.load(left_tile_ptr,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0)          # fp16

        # right tile (transposed on‑the‑fly): shape (K, N)
        # we read right at (k, j)
        right_tile_ptr = right_ptr + base \
            + offs_k[:, None] * stride_right_k \
            + offs_n[None, :] * stride_right_i
        right = tl.load(right_tile_ptr,
                        mask=mask_k[:, None] & mask_n[None, :],
                        other=0.0)          # fp16

        # GEMM accumulate
        acc += tl.dot(left, right)   # (BLOCK_M, BLOCK_N) ← (BLOCK_M, BLOCK_K)·(BLOCK_K, BLOCK_N)

    # write result
    out_tile_ptr = out_ptr + base \
        + offs_m[:, None] * stride_out_i \
        + offs_n[None, :] * stride_out_j
    tl.store(out_tile_ptr, acc,
             mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    """
    Efficient Triton implementation of the outgoing TriMul operator.

    The heavy tensor‑tensor multiplication
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    is performed in a batched FP16 GEMM using a custom Triton kernel.
    Input‑side layer‑norm, linear projections, sigmoid gates and the final
    projection stay in PyTorch (cuBLAS/cuDNN).  No gradient is computed.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    # ------------------------------------------------------------------
    # 1) Input layer‑norm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)  float32

    # ------------------------------------------------------------------
    # 2) Cast to half‑precision and apply the linear projections / gates
    # ------------------------------------------------------------------
    x_h = x.to(torch.float16)

    # linear projections (bias=False)
    left = F.linear(x_h, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x_h, weights["right_proj.weight"].to(torch.float16))

    # sigmoid gates
    left_gate = torch.sigmoid(
        F.linear(x_h, weights["left_gate.weight"].to(torch.float16)))
    right_gate = torch.sigmoid(
        F.linear(x_h, weights["right_gate.weight"].to(torch.float16)))
    out_gate = torch.sigmoid(
        F.linear(x_h, weights["out_gate.weight"].to(torch.float16)))   # (B,N,N,hidden)

    # apply gates
    left = left * left_gate
    right = right * right_gate

    # optional mask (broadcast over hidden dim)
    if not nomask and mask is not None:
        mask_f = mask.to(torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 3) Triton kernel: batched mat‑mul for every hidden channel
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape  # left has shape (B, N, N, hidden_dim)

    # output buffer (float32)
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # strides (in elements)
    stride_out_batch, stride_out_i, stride_out_j, stride_out_h = out.stride()
    stride_left_batch, stride_left_i, stride_left_k, stride_left_h = left.stride()
    stride_right_batch, stride_right_i, stride_right_k, stride_right_h = right.stride()
    # sanity: stride_*_h should be 1
    assert stride_left_h == 1 and stride_right_h == 1

    # Kernel launch configuration
    BLOCK_M = 128   # can be tuned; 256 also works on H100
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,    # pid_m
        (N + BLOCK_N - 1) // BLOCK_N,    # pid_n
        B * hidden_dim,                  # pid_c  (folded batch & channel)
    )

    tri_mul_kernel[grid](
        out,
        left,
        right,
        N, N, N,
        # output strides
        stride_out_batch,
        stride_out_i,
        stride_out_j,
        # left strides
        stride_left_batch,
        stride_left_i,
        stride_left_k,
        # right strides
        stride_right_batch,
        stride_right_i,
        stride_right_k,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        HIDDEN_DIM=hidden_dim,
        num_warps=8,
    )
    # ------------------------------------------------------------------
    # 4) Post‑processing (LayerNorm → out‑gate → Linear)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )  # (B,N,N,hidden) float32

    out = out * out_gate.to(torch.float32)

    out = F.linear(out, weights["to_out.weight"])   # (B,N,N,dim) float32

    return out

"""
Fast outgoing TriMul (Triangle Multiplicative Update) implementation.

The heavy computation
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
is a batched matrix multiplication for each hidden channel h (and each batch b).
We keep the input tensors in their native layout (B, N, N, H) to avoid the
expensive transpose+contiguous copy performed by the reference implementation.
A custom Triton kernel loads strided tiles directly from this layout,
accumulates in FP32 (Tensor‑Core friendly FP16 dot) and writes the result back.
All other operations (LayerNorms, Linear projections, sigmoid gates, final
projection) remain in PyTorch and use cuBLAS/cuDNN.  This reduces data movement
and improves runtime while preserving exact numerical results.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    out_ptr,                # *float32  (B, N, N, H) output buffer
    left_ptr,               # *float16  (B, N, N, H) left tensor
    right_ptr,              # *float16  (B, N, N, H) right tensor
    M, N, K,                # matrix dimensions (all equal to seq_len)
    # output strides (elements)
    stride_out_batch,       # stride for the batch dimension
    stride_out_i,           # stride for the first N dimension (i)
    stride_out_j,           # stride for the second N dimension (j)
    # left tensor strides (elements)
    stride_left_batch,      # stride for the batch dimension
    stride_left_i,          # stride for the first N dimension (i)
    stride_left_k,          # stride for the second N dimension (k)
    # right tensor strides (elements)
    stride_right_batch,     # stride for the batch dimension
    stride_right_i,         # stride for the first N dimension (i) – used as column stride after transposition
    stride_right_k,         # stride for the second N dimension (k)
    # compile‑time constants
    BLOCK_M: tl.constexpr,  # tile size in the i dimension
    BLOCK_N: tl.constexpr,  # tile size in the j dimension
    BLOCK_K: tl.constexpr,  # tile size in the reduction (k) dimension
    HIDDEN_DIM: tl.constexpr,   # hidden dimension (size of the channel axis)
):
    """
    Compute one tile of the batched GEMM:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    The combined channel index pid_c encodes (batch, hidden) → fast
    decomposition inside the kernel.
    """
    pid_c = tl.program_id(2)      # combined batch*hidden index
    pid_m = tl.program_id(0)      # block row (i)
    pid_n = tl.program_id(1)      # block column (j)

    # ------------------------------------------------------------------
    # 1) Recover batch and hidden indices from the flattened pid_c
    # ------------------------------------------------------------------
    b_idx = pid_c // HIDDEN_DIM
    h_idx = pid_c % HIDDEN_DIM
    # Base offset for this (b, h) slice (the hidden stride is 1)
    base = b_idx * stride_left_batch + h_idx

    # ------------------------------------------------------------------
    # 2) Tile offsets and masks
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator (FP32) – Tensor‑Core dot on FP16 accumulates in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # 3) Loop over the reduction dimension K
    # ------------------------------------------------------------------
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # left tile: shape (M, K)  -> left[b,i,k,h]
        left_tile_ptr = left_ptr + base \
            + offs_m[:, None] * stride_left_i \
            + offs_k[None, :] * stride_left_k
        left = tl.load(left_tile_ptr,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0)          # FP16

        # right tile (transposed on‑the‑fly): shape (K, N) -> right[b,k,j,h]
        right_tile_ptr = right_ptr + base \
            + offs_k[:, None] * stride_right_k \
            + offs_n[None, :] * stride_right_i
        right = tl.load(right_tile_ptr,
                        mask=mask_k[:, None] & mask_n[None, :],
                        other=0.0)          # FP16

        # GEMM accumulate (Tensor‑Core dot)
        acc += tl.dot(left, right)     # (BLOCK_M, BLOCK_N)

    # ------------------------------------------------------------------
    # 4) Write the result tile back to global memory
    # ------------------------------------------------------------------
    out_tile_ptr = out_ptr + base \
        + offs_m[:, None] * stride_out_i \
        + offs_n[None, :] * stride_out_j
    tl.store(out_tile_ptr, acc,
             mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator (no gradients).

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config) where
        - input_tensor : torch.Tensor[B, N, N, C]   (float32)
        - mask         : torch.Tensor[B, N, N] or None
        - weights      : dict of model parameters
        - config       : dict with keys "dim", "hidden_dim", optional "nomask"

    Returns
    -------
    torch.Tensor[B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )                                   # (B, N, N, dim)  float32

    # ------------------------------------------------------------------
    # 2) Cast to FP16 and apply linear projections + sigmoid gates
    # ------------------------------------------------------------------
    x_h = x.to(torch.float16)

    # Linear projections (bias=False)
    left = F.linear(x_h, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x_h, weights["right_proj.weight"].to(torch.float16))

    # Sigmoid gates
    left_gate  = torch.sigmoid(
        F.linear(x_h, weights["left_gate.weight"].to(torch.float16)))
    right_gate = torch.sigmoid(
        F.linear(x_h, weights["right_gate.weight"].to(torch.float16)))
    out_gate   = torch.sigmoid(
        F.linear(x_h, weights["out_gate.weight"].to(torch.float16)))   # (B,N,N,hidden_dim)

    # Apply multiplicative gates
    left = left * left_gate
    right = right * right_gate

    # Optional masking (broadcast over hidden dimension)
    if not nomask and mask is not None:
        mask_f = mask.to(torch.float16).unsqueeze(-1)   # (B, N, N, 1)
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 3) Core TriMul via custom Triton kernel
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape   # left shape is (B, N, N, hidden_dim)

    # Output buffer for the core multiplication (float32)
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # Strides in element units (not bytes)
    stride_out_batch, stride_out_i, stride_out_j, _ = out.stride()
    stride_left_batch, stride_left_i, stride_left_k, _ = left.stride()
    stride_right_batch, stride_right_i, stride_right_k, _ = right.stride()

    # Triton launch configuration
    # Tile sizes have been tuned for H100; 256×256 tiles also work but 128×128
    # gives a good balance between shared‑memory usage and occupancy.
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,            # pid_m (i‑tiles)
        (N + BLOCK_N - 1) // BLOCK_N,            # pid_n (j‑tiles)
        B * hidden_dim,                          # pid_c (combined batch & channel)
    )

    tri_mul_kernel[grid](
        out,
        left,
        right,
        N, N, N,
        # output strides
        stride_out_batch,
        stride_out_i,
        stride_out_j,
        # left strides
        stride_left_batch,
        stride_left_i,
        stride_left_k,
        # right strides
        stride_right_batch,
        stride_right_i,
        stride_right_k,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        HIDDEN_DIM=hidden_dim,
        num_warps=8,
    )

    # ------------------------------------------------------------------
    # 4) Post‑processing: hidden‑dim LayerNorm → gate → final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )                                   # (B, N, N, hidden_dim)  float32

    # out_gate is FP16 – promote before multiplication
    out = out * out_gate.to(torch.float32)

    # Final linear back to the original dimensionality
    out = F.linear(out, weights["to_out.weight"])   # (B, N, N, dim)  float32

    return out
