def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # 1. LayerNorm
    x = F.layer_norm(input_tensor,
                     (dim,),
                     weight=weights["norm.weight"],
                     bias=weights["norm.bias"])

    # 2. Cast to fp16
    x_h = x.to(torch.float16)

    # 3. Combined Linear for left_proj, right_proj, left_gate, right_gate, out_gate
    combined_weights = torch.cat([
        weights["left_proj.weight"],
        weights["right_proj.weight"],
        weights["left_gate.weight"],
        weights["right_gate.weight"],
        weights["out_gate.weight"]
    ], dim=0).to(torch.float16)  # shape (5*hidden_dim, dim)

    combined = F.linear(x_h, combined_weights)  # shape [B, N, N, 5*hidden_dim]

    hs = hidden_dim
    left_proj = combined[..., :hs]
    right_proj = combined[..., hs:2*hs]
    left_gate = torch.sigmoid(combined[..., 2*hs:3*hs])
    right_gate = torch.sigmoid(combined[..., 3*hs:4*hs])
    out_gate = torch.sigmoid(combined[..., 4*hs:])  # shape [B,N,N,hidden_dim]

    # 4. Mask
    if not nomask and mask is not None:
        mask_h = mask.to(torch.float16).unsqueeze(-1)
        left_proj = left_proj * mask_h
        right_proj = right_proj * mask_h

    # 5. Apply gates
    left = left_proj * left_gate
    right = right_proj * right_gate

    # 6. TriMul kernel: prepare
    B, N, _, _ = left.shape  # shape [B,N,N,hidden_dim]
    # Permute to (B, hidden_dim, N, N)
    left = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()
    C_total = B * hidden_dim
    left_flat = left.reshape(C_total, N, N)
    right_flat = right.reshape(C_total, N, N)

    # Output buffer
    out_flat = torch.empty_like(left_flat, dtype=torch.float32, device=device)

    # Choose tile sizes based on N
    if N <= 256:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 32
    elif N <= 512:
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 32
    else:  # N > 512
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 64

    # Strides
    stride_out_c = out_flat.stride(0)
    stride_out_m = out_flat.stride(1)
    stride_out_n = out_flat.stride(2)

    stride_a_c = left_flat.stride(0)
    stride_a_m = left_flat.stride(1)
    stride_a_k = left_flat.stride(2)

    stride_b_c = right_flat.stride(0)
    stride_b_k = right_flat.stride(2)
    stride_b_n = right_flat.stride(1)

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        C_total,
    )
    num_warps = 8 if N >= 512 else 4

    tri_mul_kernel[grid](
        out_flat,
        left_flat,
        right_flat,
        N, N, N,
        stride_out_m, stride_out_n, stride_out_c,
        stride_a_m, stride_a_k, stride_a_c,
        stride_b_k, stride_b_n, stride_b_c,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=num_warps,
    )

    # Reshape back
    out = out_flat.reshape(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # Post-process

                       (hidden_dim,),

                       bias=weights["to_out_norm.bias"])

    out = F.linear(out, weights["to_out.weight"])

"""
TriMul (outgoing) – fast forward implementation.

The operator computes
    out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]

where `left` and `right` are linear projections of the input tensor
followed by element‑wise gating. The N³ reduction for each hidden
channel is expressed as a batched GEMM and is evaluated by a custom
Triton kernel that:
* works on fp16 inputs,
* accumulates in fp32,
* uses a 2‑D (BLOCK_M×BLOCK_N) tile for the output matrix and a
  1‑D (BLOCK_K) tile for the reduction,
* launches a separate program for each (batch × hidden) channel – a
  pattern that maps well to H100 Tensor‑Cores.

All remaining operations (layer‑norms, gating, final projection) are
performed in PyTorch.  To minimise memory traffic the four linear
projections and three gate‐linear layers are fused into a single
matrix multiplication, thus reading the large input tensor only once.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    out_ptr,          # float32*
    a_ptr,            # float16*  (C, M, K)      – left tensor
    b_ptr,            # float16*  (C, K, N)      – right tensor (logically transposed)
    M, N, K,          # matrix sizes (all = seq_len)
    stride_out_m, stride_out_n, stride_out_c,   # output strides
    stride_a_m, stride_a_k, stride_a_c,        # left strides
    stride_b_k, stride_b_n, stride_b_c,        # right strides (after “transpose”)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched GEMM: for each channel `c` (folded batch*hidden) compute
        out_c[M,N] += left_c[M,K] @ right_c[K,N]
    """
    pid_m = tl.program_id(0)  # block row   → i
    pid_n = tl.program_id(1)  # block col   → j
    pid_c = tl.program_id(2)  # channel id  → (b, h)

    # --------------------- tile offsets ---------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ----------------------- reduction ----------------------
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # pointer to left tile  (M,K)
        a_tile = a_ptr + pid_c * stride_a_c \
                 + offs_m[:, None] * stride_a_m \
                 + offs_k[None, :] * stride_a_k

        # pointer to right tile (K,N) – note the logical transpose
        b_tile = b_ptr + pid_c * stride_b_c \
                 + offs_k[:, None] * stride_b_k \
                 + offs_n[None, :] * stride_b_n

        a = tl.load(a_tile,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)
        b = tl.load(b_tile,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # fp16 * fp16 → fp32 accumulation
        acc += tl.dot(a, b)

    # ----------------------- store -------------------------
    out_tile = out_ptr + pid_c * stride_out_c \
               + offs_m[:, None] * stride_out_m \
               + offs_n[None, :] * stride_out_n

    tl.store(out_tile, acc,
             mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module (AlphaFold3 style).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor[B, N, N, C] (float32)
            - mask         : torch.Tensor[B, N, N] (bool/float) or None
            - weights      : dict of model parameters (see keys below)
            - config       : dict with keys:
                * "dim"        – input channel dimension C
                * "hidden_dim"– hidden dimension H
                * "nomask"     – bool, True → ignore mask (default: True)

    Returns:
        torch.Tensor[B, N, N, C] (float32) – module output
    """
    # -----------------------------------------------------------------
    # unpack
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -----------------------------------------------------------------
    # 1) Input LayerNorm (float32, stable)
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, C]  (FP32)

    # -----------------------------------------------------------------
    # 2) Cast to fp16 for all dense ops
    x_h = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 3) Fuse all linear layers (proj + gates + out_gate) into one matmul
    #    Order: left_proj, right_proj, left_gate, right_gate, out_gate
    #    Each weight has shape (hidden_dim, dim)
    fused_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)  # (5*H, C)

    fused_out = F.linear(x_h, fused_weight)  # [B, N, N, 5*H] (fp16)

    H = hidden_dim
    left_proj = fused_out[..., :H]                # (B, N, N, H)
    right_proj = fused_out[..., H : 2 * H]        # (B, N, N, H)
    left_gate = torch.sigmoid(fused_out[..., 2 * H : 3 * H])
    right_gate = torch.sigmoid(fused_out[..., 3 * H : 4 * H])
    out_gate = torch.sigmoid(fused_out[..., 4 * H :])  # (B, N, N, H)

    # -----------------------------------------------------------------
    # 4) Apply mask (if any) **before** gating – mirrors reference code
    if not nomask and mask is not None:
        # mask is 0/1, broadcast over hidden dim
        mask_h = mask.to(torch.float16).unsqueeze(-1)  # (B, N, N, 1)
        left_proj = left_proj * mask_h
        right_proj = right_proj * mask_h

    # -----------------------------------------------------------------
    # 5) Apply gates
    left = left_proj * left_gate   # (B, N, N, H) fp16
    right = right_proj * right_gate  # (B, N, N, H) fp16

    # -----------------------------------------------------------------
    # 6) Prepare tensors for the Triton GEMM kernel
    B, N, _, _ = left.shape
    # move hidden dimension before the two sequence axes for easy flattening
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    C_total = B * hidden_dim                         # = batch * hidden
    left_flat = left.reshape(C_total, N, N)          # (C, M, K)  where M=K=N
    right_flat = right.reshape(C_total, N, N)        # (C, N, K)  (logically transposed later)

    # output buffer in FP32 (required by layer‑norm downstream)
    out_flat = torch.empty_like(left_flat, dtype=torch.float32, device=device)

    # -----------------------------------------------------------------
    # 7) Choose a tile size that balances register pressure & occupancy
    if N <= 256:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 32
    elif N <= 512:
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 32
    else:  # N > 512
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 64

    # strides for the kernel (PyTorch uses row‑major layout)
    stride_out_c = out_flat.stride(0)
    stride_out_m = out_flat.stride(1)
    stride_out_n = out_flat.stride(2)

    stride_a_c = left_flat.stride(0)
    stride_a_m = left_flat.stride(1)
    stride_a_k = left_flat.stride(2)

    stride_b_c = right_flat.stride(0)
    stride_b_k = right_flat.stride(2)   # innermost dimension (K)
    stride_b_n = right_flat.stride(1)   # second dimension   (N)

    # launch grid: (tile rows, tile cols, channels)
    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        C_total,
    )

    # More warps for larger N improves occupancy on H100
    num_warps = 8 if N >= 512 else 4

    tri_mul_kernel[grid](
        out_flat,
        left_flat,
        right_flat,
        N, N, N,
        stride_out_m, stride_out_n, stride_out_c,
        stride_a_m, stride_a_k, stride_a_c,
        stride_b_k, stride_b_n, stride_b_c,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=num_warps,
    )

    # -----------------------------------------------------------------
    # 8) Reshape back to (B, N, N, H) – still FP32
    out = out_flat.reshape(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 9) Post‑processing: LayerNorm → out‑gate → final linear
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    # out_gate was computed in fp16; promote to fp32 for the final element‑wise mult
    out = out * out_gate.to(torch.float32)

    # Final projection back to original dimensionality
    out = F.linear(out, weights["to_out.weight"])

    return out
