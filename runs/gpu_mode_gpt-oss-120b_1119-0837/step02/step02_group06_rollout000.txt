"""
TriMul (outgoing) – Triton‑accelerated forward pass
=================================================

The kernel implements the expensive “triangle multiplicative update”

    out_{b,i,j,h} = Σ_k left_{b,i,k,h} · right_{b,j,k,h}

which is exactly a batched matrix multiplication
    left  (B·H, N, N)  @  rightᵀ (B·H, N, N) → out (B·H, N, N)

All heavy arithmetic is performed in half‑precision (fp16) with
fp32 accumulation (Tensor‑Core friendly).  The surrounding Python code
handles LayerNorms, bias‑free projections, sigmoid gates, optional
masking and the final post‑processing.  An `autotune` decorator picks a
good tile size for the given sequence length.

The final output has dtype torch.float32, as required by AlphaFold3.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------
# Triton kernel: batched GEMM (left @ rightᵀ) for the triangle mul
# --------------------------------------------------------------
@triton.autotune(
    configs=[
        # Good defaults for most sequence lengths on H100
        triton.Config({'BLOCK_M': 128, 'BLOCK_N': 128, 'BLOCK_K': 32},
                       num_warps=8, num_stages=3),
        # Fallback for very small N (e.g. N=256) to improve occupancy
        triton.Config({'BLOCK_M': 64, 'BLOCK_N': 64, 'BLOCK_K': 32},
                       num_warps=4, num_stages=2),
    ],
    key=['M', 'N', 'K'],
)
@triton.jit
def tri_mul_kernel(
    out_ptr,            # float32*   (C_total, M, N)
    a_ptr,              # float16*   (C_total, M, K)   – left
    b_ptr,              # float16*   (C_total, K, N)   – rightᵀ (loaded transposed)
    M, N, K,            # matrix sizes (all equal to seq_len)
    stride_out_m, stride_out_n, stride_out_c,   # output strides
    stride_a_m, stride_a_k, stride_a_c,        # left strides
    stride_b_k, stride_b_n, stride_b_c,        # rightᵀ strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched GEMM for the outgoing TriMul.
    Each program (block) processes one tile (BLOCK_M×BLOCK_N) of one
    (batch, hidden) slice (pid_c).  A‑tile = left[i,k], B‑tile = rightᵀ[k,j].
    """
    pid_m = tl.program_id(0)   # tile row  (i)
    pid_n = tl.program_id(1)   # tile col  (j)
    pid_c = tl.program_id(2)   # fused batch‑&‑hidden index (C_total)

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # bounds (in case N is not a multiple of the block size)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over reduction dimension K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # pointers for the current K‑tile
        a_tile_ptr = a_ptr + pid_c * stride_a_c \
                     + offs_m[:, None] * stride_a_m \
                     + offs_k[None, :] * stride_a_k        # (M,K)

        b_tile_ptr = b_ptr + pid_c * stride_b_c \
                     + offs_k[:, None] * stride_b_k \
                     + offs_n[None, :] * stride_b_n        # (K,N)

        # load with masking
        a = tl.load(a_tile_ptr,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)                               # (M,K) fp16
        b = tl.load(b_tile_ptr,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)                               # (K,N) fp16

        # matrix‑multiply accumulate (fp16·fp16 → fp32)
        acc += tl.dot(a, b)                               # (M,N) fp32

    # write the tile back to global memory
    out_tile_ptr = out_ptr + pid_c * stride_out_c \
                   + offs_m[:, None] * stride_out_m \
                   + offs_n[None, :] * stride_out_n
    tl.store(out_tile_ptr,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])


# --------------------------------------------------------------
# Entry‑point: forward pass of the outgoing TriMul module
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul (no gradients).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor[B, N, N, C] (float32)
            - mask         : torch.Tensor[B, N, N] (bool/float) or None
            - weights      : dict of model parameters (see above)
            - config       : dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor[B, N, N, C] – processed tensor (float32)
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1) Layer‑norm over the feature dimension C
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, C)  float32

    # ------------------------------------------------------------------
    # 2) Cast to FP16 – all cheap linear ops can work in half‑precision
    # ------------------------------------------------------------------
    x_h = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Linear projections (bias‑free) to hidden dimension H
    # ------------------------------------------------------------------
    left = F.linear(
        x_h,
        weights["left_proj.weight"].to(torch.float16)
    )  # (B, N, N, H)
    right = F.linear(
        x_h,
        weights["right_proj.weight"].to(torch.float16)
    )  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 4) Gates (sigmoid) – keep them in FP16
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(
        F.linear(x_h, weights["left_gate.weight"].to(torch.float16))
    )
    right_gate = torch.sigmoid(
        F.linear(x_h, weights["right_gate.weight"].to(torch.float16))
    )
    out_gate   = torch.sigmoid(
        F.linear(x_h, weights["out_gate.weight"].to(torch.float16))
    )  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 5) Apply gates (and optional mask) – in‑place to reduce memory traffic
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    if (not nomask) and (mask is not None):
        # mask shape (B,N,N) → (B,N,N,1) in fp16
        mask_f = mask.to(torch.float16).unsqueeze(-1)
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 6) Prepare tensors for the Triton kernel
    #    We want (C_total, N, N) where C_total = B * H.
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    # bring hidden dimension before the two sequence dims → (B, H, N, N)
    left = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()

    C_total = B * hidden_dim
    # reshape to (C_total, N, N)
    left_flat  = left.reshape(C_total, N, N)   # (C, M, K)
    right_flat = right.reshape(C_total, N, N)  # (C, N, N) – will be read as (C, K, N)

    # output buffer – fp32 accumulation (final result of the triangle mul)
    out_flat = torch.empty_like(left_flat, dtype=torch.float32)

    # ------------------------------------------------------------------
    # 7) Triton kernel launch
    # ------------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    # strides in elements (not bytes)
    stride_out_c = out_flat.stride(0)
    stride_out_m = out_flat.stride(1)
    stride_out_n = out_flat.stride(2)

    stride_a_c = left_flat.stride(0)
    stride_a_m = left_flat.stride(1)
    stride_a_k = left_flat.stride(2)

    stride_b_c = right_flat.stride(0)
    # we treat right as (C, K, N) inside the kernel
    stride_b_k = right_flat.stride(2)   # original last dimension → K
    stride_b_n = right_flat.stride(1)   # original middle dimension → N

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # blocks over i (rows)
        (N + BLOCK_N - 1) // BLOCK_N,   # blocks over j (cols)
        C_total,                        # one block per (batch, hidden) slice
    )

    tri_mul_kernel[grid](
        out_flat,
        left_flat,
        right_flat,
        N, N, N,                       # M, N, K
        stride_out_m, stride_out_n, stride_out_c,
        stride_a_m,   stride_a_k,   stride_a_c,
        stride_b_k,   stride_b_n,   stride_b_c,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=3,
    )

    # ------------------------------------------------------------------
    # 8) Reshape kernel output back to (B, N, N, H)
    # ------------------------------------------------------------------
    out = out_flat.reshape(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 9) Final post‑processing
    #       - LayerNorm over hidden dim
    #       - Gating with out_gate
    #       - Linear projection back to original dim C
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    # out_gate was fp16; promote to fp32 for the multiplication
    out = out * out_gate.to(torch.float32)
    out = F.linear(out, weights["to_out.weight"])   # (B, N, N, C)

    return out
