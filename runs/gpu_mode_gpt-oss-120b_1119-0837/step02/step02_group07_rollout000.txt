"""
Outgoing TriMul (AlphaFold3 style) – Triton‑accelerated forward pass.

The heavy part of the module is the “triangle” multiplication
    out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]                (1)

For each hidden channel `h` (and batch `b`) this is a dense matrix multiply
between `left[b,:, :,h] ∈ ℝ^{N×N}` and the transpose of
`right[b,:, :,h]`.  Equation (1) can therefore be expressed as a batched GEMM
over `C = B * hidden_dim` independent matrices of size `N×N`.

The implementation below:
* Runs the LayerNorms, linear projections and gates in PyTorch (fp16/fp32).
* Packs the `left` and `right` tensors so that each `(b,h)` pair becomes one
  GEMM of shape `(N,N)`.
* Launches a custom Triton kernel that computes all GEMMs in one call.
  The kernel tiles the output with `BLOCK_M × BLOCK_N` and the reduction
  dimension with `BLOCK_K`, uses fp16 inputs with fp32 accumulation
  (Tensor‑Core friendly), and adapts its tile sizes to the sequence length.
* Post‑processes the result with a second LayerNorm, an output gate and a
  final linear projection back to the original dimension.

The kernel returns a float‑32 tensor of shape `[B, N, N, dim]`.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    out_ptr,            # float32*   output (C, M, N)
    a_ptr,              # float16*   left  (C, M, K)   – M=N=K
    b_ptr,              # float16*   right (C, K, N)   – stored as (C, N, N) but read transposed
    M, N, K,            # matrix sizes (all = seq_len)
    stride_out_c, stride_out_m, stride_out_n,   # output strides
    stride_a_c, stride_a_m, stride_a_k,         # left  strides
    stride_b_c, stride_b_k, stride_b_n,         # right strides (K dimension is innermost)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched GEMM for the TriMul operation.

    For each channel `c = 0 .. C‑1` (where C = batch * hidden_dim) we compute
        out_c[M,N] += left_c[M,K] @ right_c[K,N]

    The right tensor is stored in the original layout (C, N, N) and is
    accessed as its transpose by swapping the K/N strides.
    """
    pid_m = tl.program_id(0)  # block row (i)
    pid_n = tl.program_id(1)  # block col (j)
    pid_c = tl.program_id(2)  # channel index (b * hidden_dim + h)

    # Tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Bounds (for non‑multiple of block size)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -----------------------------------------------------------------
    # Loop over K in BLOCK_K steps
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Pointers to current tiles (row‑major)
        a_tile = a_ptr + pid_c * stride_a_c \
                 + offs_m[:, None] * stride_a_m \
                 + offs_k[None, :] * stride_a_k        # (M, K)

        b_tile = b_ptr + pid_c * stride_b_c \
                 + offs_k[:, None] * stride_b_k \
                 + offs_n[None, :] * stride_b_n        # (K, N)

        # Load with masking (zero‑pad out‑of‑bounds)
        a = tl.load(a_tile,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)                        # fp16
        b = tl.load(b_tile,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)                        # fp16

        # GEMM accumulation: fp16 * fp16 → fp32 (Tensor‑Core friendly)
        acc += tl.dot(a, b)

    # -----------------------------------------------------------------
    # Write the accumulated tile
    out_tile = out_ptr + pid_c * stride_out_c \
               + offs_m[:, None] * stride_out_m \
               + offs_n[None, :] * stride_out_n
    tl.store(out_tile, acc,
             mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module (AlphaFold3 style).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor[B, N, N, C] (float32)
        - mask         : torch.Tensor[B, N, N] (float32/bool) or None
        - weights      : dict of model parameters (see expected keys)
        - config       : dict with keys
            * "dim"        – input channel dimension C
            * "hidden_dim"– hidden dimension H
            * "nomask"     – bool, True → ignore mask

    Returns
    -------
    torch.Tensor[B, N, N, C] (float32)
        The module output.
    """
    # -----------------------------------------------------------------
    # Unpack arguments
    input_tensor, mask, weights, config = data
    dim = config["dim"]                     # input / output channel dim (C)
    hidden_dim = config["hidden_dim"]       # hidden dimension (H)
    nomask = config.get("nomask", True)
    device = input_tensor.device

    # -----------------------------------------------------------------
    # 1) Input LayerNorm (float32)
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, C]  (FP32)

    # -----------------------------------------------------------------
    # 2) Cast to fp16 for all subsequent linear ops
    x_h = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 3) Linear projections (no bias) → hidden_dim
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    # F.linear works on the last dimension; it automatically flattens the
    # preceding dims, which is exactly what we need.
    left = F.linear(x_h, left_proj_w)   # [B, N, N, H]  fp16
    right = F.linear(x_h, right_proj_w) # [B, N, N, H]  fp16

    # -----------------------------------------------------------------
    # 4) Gates (sigmoid) – also fp16
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    left_gate = torch.sigmoid(F.linear(x_h, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_h, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x_h, out_gate_w))   # later cast to fp32

    # -----------------------------------------------------------------
    # 5) Apply gates (and optional mask)
    left = left * left_gate
    right = right * right_gate

    if not nomask and mask is not None:
        # Mask is element‑wise 0/1; broadcast over hidden dim.
        mask_h = mask.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_h
        right = right * mask_h

    # -----------------------------------------------------------------
    # 6) Prepare tensors for the Triton kernel.
    #    Move the hidden dimension before the two sequence axes and flatten
    #    the (batch, hidden) pair into a single channel dimension C_total.
    B, N, _, _ = left.shape
    left = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    C_total = B * hidden_dim                        # = batch * hidden_dim

    left_flat = left.reshape(C_total, N, N)    # (C, M, K) where M=K=N
    right_flat = right.reshape(C_total, N, N)  # (C, N, N) – will be read as (C, K, N)

    # Output buffer (FP32 accumulator)
    out_flat = torch.empty_like(left_flat, dtype=torch.float32, device=device)

    # -----------------------------------------------------------------
    # 7) Kernel launch – tile sizes adapt to sequence length.
    if N <= 256:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 32
    elif N <= 512:
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 32
    else:
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 64

    # Strides (in element units, not bytes)
    stride_out_c = out_flat.stride(0)
    stride_out_m = out_flat.stride(1)
    stride_out_n = out_flat.stride(2)

    stride_a_c = left_flat.stride(0)
    stride_a_m = left_flat.stride(1)
    stride_a_k = left_flat.stride(2)

    stride_b_c = right_flat.stride(0)
    # Right is accessed as its transpose: K stride = innermost dim, N stride = middle dim
    stride_b_k = right_flat.stride(2)   # K dimension (original column)
    stride_b_n = right_flat.stride(1)   # N dimension (original row)

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # #blocks along M
        (N + BLOCK_N - 1) // BLOCK_N,   # #blocks along N
        C_total,                        # one program per channel
    )

    # More warps give higher occupancy on H100; 8 works well for large N.
    num_warps = 8 if N >= 512 else 4

    # Launch the Triton kernel
    tri_mul_kernel[grid](
        out_flat,
        left_flat,
        right_flat,
        N, N, N,
        stride_out_c, stride_out_m, stride_out_n,
        stride_a_c, stride_a_m, stride_a_k,
        stride_b_c, stride_b_k, stride_b_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=num_warps,
    )

    # -----------------------------------------------------------------
    # 8) Reshape back to (B, N, N, H) – currently in FP32
    out = out_flat.reshape(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,H]

    # -----------------------------------------------------------------
    # 9) Post‑processing: LayerNorm → out‑gate → final projection
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # out_gate is still fp16; cast before multiplication
    out = out * out_gate.to(torch.float32)

    # Final linear back to the original dimension (float32)
    out = F.linear(out, weights["to_out.weight"])   # (B, N, N, dim)

    return out
