def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    eps = 1e-5

    # 1) LayerNorm (input)
    M = bs * seq_len * seq_len
    x_flat = input_tensor.reshape(M, dim)
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm_flat = triton_layernorm(x_flat, norm_weight, norm_bias, eps)  # outputs same dtype (float32)
    x_norm = x_norm_flat.view(bs, seq_len, seq_len, dim)

    # 2) Cast to fp16
    x_norm_f16 = x_norm.to(torch.float16)

    # 3) Cast model weights to fp16 (except final projection maybe fp16)
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float32)  # keep float32 for stability
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float32)
    to_out_w = weights["to_out.weight"].to(torch.float16)

    # 4) Linear projections
    left_proj = F.linear(x_norm_f16, left_proj_w)   # (B,N,N,H)
    right_proj = F.linear(x_norm_f16, right_proj_w)

    # 5) Gating
    left_gate = torch.sigmoid(F.linear(x_norm_f16, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_norm_f16, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x_norm_f16, out_gate_w))

    # 6) Apply mask if needed
    if not nomask and mask is not None:
        mask_f16 = mask.to(torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left_proj = left_proj * mask_f16
        right_proj = right_proj * mask_f16

    # 7) Apply elementwise gating
    left = left_proj * left_gate   # (B,N,N,H)
    right = right_proj * right_gate

    # 8) Core contraction using Triton kernel
    out = torch.empty_like(left)  # fp16

    # Compute strides
    l_stride_b, l_stride_i, l_stride_k, l_stride_h = left.stride()
    # right stride identical
    r_stride_b, r_stride_j, r_stride_k, r_stride_h = right.stride()
    o_stride_b, o_stride_i, o_stride_j, o_stride_h = out.stride()

    # Block sizes – tune for H100
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    BLOCK_H = 8

    # Compute grid dimensions
    num_h_slices = (hidden_dim + BLOCK_H - 1) // BLOCK_H
    grid = (
        (seq_len + BLOCK_M - 1) // BLOCK_M,
        (seq_len + BLOCK_N - 1) // BLOCK_N,
        bs * num_h_slices,
    )

    trimul_contraction_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=bs,
        N=seq_len,
        H=hidden_dim,
        stride_left_b=l_stride_b,
        stride_left_i=l_stride_i,
        stride_left_k=l_stride_k,
        stride_left_h=l_stride_h,
        stride_right_b=r_stride_b,
        stride_right_j=r_stride_j,
        stride_right_k=r_stride_k,
        stride_right_h=r_stride_h,
        stride_out_b=o_stride_b,
        stride_out_i=o_stride_i,
        stride_out_j=o_stride_j,
        stride_out_h=o_stride_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        BLOCK_H=BLOCK_H,
    )

    # 9) Output layernorm (use fp16 input)
    out_fp16 = out  # already fp16
    out_2d = out_fp16.reshape(-1, hidden_dim)  # (B*N*N, H)

    out_norm_2d = triton_layernorm(out_2d, to_out_norm_w, to_out_norm_b, eps)  # fp32 output
    out_norm = out_norm_2d.view(bs, seq_len, seq_len, hidden_dim).to(torch.float16)

    # 10) Apply out gate and final projection
    out_gate_fp16 = out_gate.to(torch.float16)
    gated = out_norm * out_gate_fp16
    output = F.linear(gated, to_out_w)  # (B,N,N,dim) fp16

    # Cast final output to float32
    return output.to(torch.float32)

"""
Triton‑accelerated “outgoing” TriMul.

Algorithm
---------
1️⃣  Layer‑Norm of the input  (B,N,N,D)  –  Triton kernel.
2️⃣  FP16 linear projections and sigmoid gates.
3️⃣  Optional binary mask is applied in‑place (broadcast over H).
4️⃣  Element‑wise gating → tensors  L,B  of shape (B,N,N,H).
5️⃣  Core O(N³·H) contraction:
        out[b,i,j,h] = Σₖ L[b,i,k,h] * R[b,j,k,h]
   Implemented as a single Triton kernel that works on the native
   layout (B,N,N,H) – no permute + .contiguous() copy.  The kernel
   tiles the i‑ and j‑dimensions (BLOCK_M×BLOCK_N) and the hidden
   dimension (BLOCK_H) while looping over K‑tiles (BLOCK_K).  Each
   hidden slice is reduced with a small GEMM (tl.dot) giving a true
   batched mat‑mul without extra memory traffic.
6️⃣  Layer‑Norm of the result (B,N,N,H) – Triton kernel (FP32
   accumulator, FP16 output).
7️⃣  Output gate (sigmoid) is applied.
8️⃣  Final linear projection back to the model dimension (FP16
   mat‑mul, cast to float32 for the returned tensor).

The implementation keeps the heavy‑weight contraction on‑device,
removes the expensive permute/contiguous shuffles and stays in
half‑precision for the bulk of the work, giving a large speed‑up
over the reference “torch.bmm” version.

# ----------------------------------------------------------------------
# 2‑pass LayerNorm (last dimension) – works for FP16 or FP32 input.
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,                # *ptr to input  [M, N]
    y_ptr,                # *ptr to output [M, N]
    weight_ptr,           # *ptr to weight [N]
    bias_ptr,             # *ptr to bias   [N]
    M,                    # rows (= B*N*N)
    N,                    # feature dimension (D or H)
    eps,                  # epsilon
    stride_xm, stride_xn,  # input strides
    stride_ym, stride_yn,  # output strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
):
    pid = tl.program_id(0)

    # ------------------------------------------------------------------
    # Row range for this program
    # ------------------------------------------------------------------
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # ------------------------------------------------------------------
    # First pass: compute mean & variance (float32 accumulator)
    # ------------------------------------------------------------------
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_N)
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        # load as fp16/fp32, promote to fp32 for reduction
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        x_f32 = x.to(tl.float32)
        sum_val += tl.sum(x_f32, axis=1)
        sum_sq  += tl.sum(x_f32 * x_f32, axis=1)

    mean = sum_val / N
    var  = sum_sq / N - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # ------------------------------------------------------------------
    # Second pass: write normalised + affine‑transformed values
    # ------------------------------------------------------------------
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        x_f32 = x.to(tl.float32)

        norm_x = (x_f32 - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_n, mask=mask_n, other=0.0)
        b = tl.load(bias_ptr   + offs_n, mask=mask_n, other=0.0)

        y = norm_x * w[None, :] + b[None, :]
        # Cast back to the input dtype (fp16 or fp32)
        y = y.to(x.dtype)

        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn,
            y,
            mask=mask_m[:, None] & mask_n[None, :],
        )


def triton_layernorm(x: torch.Tensor,
                     weight: torch.Tensor,
                     bias: torch.Tensor,
                     eps: float = 1e-5) -> torch.Tensor:
    """LayerNorm on the last dimension using the Triton kernel above."""
    assert x.dim() == 2, "`triton_layernorm` expects a 2‑D tensor."
    M, N = x.shape
    y = torch.empty_like(x)

    x = x.contiguous()
    y = y.contiguous()
    weight = weight.contiguous()
    bias = bias.contiguous()

    stride_xm, stride_xn = x.stride()
    stride_ym, stride_yn = y.stride()

    grid = lambda meta: (triton.cdiv(M, meta["BLOCK_M"]),)

    layernorm_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        weight_ptr=weight,
        bias_ptr=bias,
        M=M,
        N=N,
        eps=eps,
        stride_xm=stride_xm,
        stride_xn=stride_xn,
        stride_ym=stride_ym,
        stride_yn=stride_yn,
        BLOCK_M=128,   # rows per program – a good default for H100
        BLOCK_N=64,    # cols per tile – fits registers nicely
    )
    return y

    BLOCK_H: tl.constexpr,

    pid_n = tl.program_id(1)   # j‑tile
    pid_z = tl.program_id(2)   # batch × hidden‑slice

    # ------------------------------------------------------------------

    b = pid_z // total_h_slices
    h_slice = pid_z % total_h_slices
    h_start = h_slice * BLOCK_H

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------

    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

            right_ptrs,
            mask=mask_k[:, None, None] &

            other=0.0,
        )

        # ------------------------------------------------------------------

    # ------------------------------------------------------------------

        + offs_h[None, None, :] * stride_out_h

        acc.to(tl.float16),
        mask=mask_i[:, None, None] &
             mask_j[None, :, None] &
             mask_h[None, None, :],

    # Final linear projection back to model dim (use FP16 weight for Tensor‑Cores)
    output = F.linear(gated.to(torch.float16), to_out_w)  # (B,N,N,D) FP16
    return output.to(torch.float32)                       # cast to the required dtype
```
