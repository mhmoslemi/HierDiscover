"""
TriMul (outgoing) – fast half‑precision implementation.

The heavy “triangle” multiplication
    out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
is expressed as a batched matrix multiplication per (batch,hidden) slice.
We pack the batch and hidden dimensions into a single GEMM batch dimension
C = B * H and launch a Triton kernel that computes
    out_c[M,N] = left_c[M,K] @ right_c[K,N]   (M = N = seq_len, K = seq_len).
All tensors (except the very first LayerNorm) are kept in fp16, with
fp32 accumulation inside the kernel and the result stored as fp16.
The surrounding linear projections, gates, and the two LayerNorms are
performed with torch.nn.functional using half‑precision weights.
The final output is cast back to float32 for downstream consumption.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    out_ptr,                     # fp16*
    a_ptr,                       # fp16*  (C, M, K)  – left tensor
    b_ptr,                       # fp16*  (C, K, N)  – right tensor (logically transposed)
    M, N, K,                     # matrix dimensions (all = seq_len)
    stride_out_m, stride_out_n, stride_out_c,   # output strides
    stride_a_m, stride_a_k, stride_a_c,        # left tensor strides
    stride_b_k, stride_b_n, stride_b_c,        # right tensor strides (after transposition)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM: for each channel `c` (packed batch*hidden) compute
    out_c[M,N] += left_c[M,K] @ right_c[K,N].  FP16 inputs, FP32
    accumulation, result stored as FP16.
    """
    pid_m = tl.program_id(0)   # block row index (i)
    pid_n = tl.program_id(1)   # block col index (j)
    pid_c = tl.program_id(2)   # channel index = batch*hidden

    # Tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator, FP32 for numerical stability
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over reduction dimension K
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Pointers to the current tiles of A and B
        a_tile = a_ptr + pid_c * stride_a_c \
                 + offs_m[:, None] * stride_a_m \
                 + offs_k[None, :] * stride_a_k          # (M, K)

        b_tile = b_ptr + pid_c * stride_b_c \
                 + offs_k[:, None] * stride_b_k \
                 + offs_n[None, :] * stride_b_n          # (K, N)

        # Load with masking, zero out‑of‑bounds
        a = tl.load(a_tile,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)                     # fp16
        b = tl.load(b_tile,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)                     # fp16

        # GEMM accumulate: fp16 * fp16 → fp32
        acc += tl.dot(a, b)

    # Write out tile (convert to fp16 on store to reduce bandwidth)
    out_tile = out_ptr + pid_c * stride_out_c \
               + offs_m[:, None] * stride_out_m \
               + offs_n[None, :] * stride_out_n
    tl.store(out_tile, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module (AlphaFold3 style).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor[B, N, N, C] (float32)
            - mask         : torch.Tensor[B, N, N] (bool/float) or None
            - weights      : dict of model parameters (see keys below)
            - config       : dict with keys
                * "dim"         – input channel dimension C
                * "hidden_dim" – hidden dimension H
                * "nomask"      – bool, True → ignore mask

    Returns:
        torch.Tensor[B, N, N, C] (float32) – module output
    """
    # -----------------------------------------------------------------
    # unpack
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    device = input_tensor.device

    # -----------------------------------------------------------------
    # 1) Input LayerNorm (still in fp32 for numerical safety)
    #    -> then cast to fp16 for the rest of the pipeline.
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, C] fp32
    x = x.to(torch.float16)  # now fp16

    # -----------------------------------------------------------------
    # 2) Linear projections (no bias) → hidden_dim   (fp16)
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left = F.linear(x, left_proj_w)   # [B, N, N, H] fp16
    right = F.linear(x, right_proj_w)  # [B, N, N, H] fp16

    # -----------------------------------------------------------------
    # 3) Gates (sigmoid) – fp16
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))   # [B,N,N,H] fp16

    # -----------------------------------------------------------------
    # 4) Apply gates (and optional mask)
    left = left * left_gate
    right = right * right_gate

    if not nomask and mask is not None:
        # mask is 0/1 – broadcast over hidden dim
        mask_h = mask.to(torch.float16).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_h
        right = right * mask_h

    # -----------------------------------------------------------------
    # 5) Prepare tensors for the Triton kernel.
    #    Bring hidden dim before the two sequence dims and flatten
    #    batch*hidden into a single GEMM batch dimension.
    B, N, _, _ = left.shape
    left = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    C_total = B * hidden_dim                       # = batch * hidden_dim
    left_flat = left.reshape(C_total, N, N)        # (C, M, K)   – K = N
    right_flat = right.reshape(C_total, N, N)      # (C, K, N)   – will be interpreted as (K,N)

    # Output buffer (fp16)
    out_flat = torch.empty_like(left_flat, dtype=torch.float16, device=device)

    # -----------------------------------------------------------------
    # 6) Triton GEMM kernel launch – tile sizes adapted to seq_len.
    if N <= 256:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 32
    elif N <= 512:
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 32
    else:  # N up to 1024
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 64

    # Strides (in elements, not bytes)
    stride_out_c = out_flat.stride(0)
    stride_out_m = out_flat.stride(1)
    stride_out_n = out_flat.stride(2)

    stride_a_c = left_flat.stride(0)
    stride_a_m = left_flat.stride(1)
    stride_a_k = left_flat.stride(2)

    stride_b_c = right_flat.stride(0)
    stride_b_k = right_flat.stride(2)   # innermost dim → K
    stride_b_n = right_flat.stride(1)   # middle dim → N

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # pid_m
        (N + BLOCK_N - 1) // BLOCK_N,   # pid_n
        C_total,                        # pid_c
    )

    # More warps improve occupancy for larger N
    num_warps = 8 if N >= 512 else 4

    tri_mul_kernel[grid](
        out_flat,
        left_flat,
        right_flat,
        N, N, N,
        stride_out_m, stride_out_n, stride_out_c,
        stride_a_m, stride_a_k, stride_a_c,
        stride_b_k, stride_b_n, stride_b_c,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=num_warps,
    )

    # -----------------------------------------------------------------
    # 7) Reshape back to (B, N, N, H)  (still fp16)
    out = out_flat.reshape(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,H] fp16

    # -----------------------------------------------------------------
    # 8) Post‑processing: LayerNorm → out‑gate → final linear.
    #    All kept in fp16 – weights are cast once.
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)

    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # Apply out‑gate (already fp16)
    out = out * out_gate

    # Final projection back to original dim
    to_out_w = weights["to_out.weight"].to(torch.float16)
    out = F.linear(out, to_out_w)   # (B,N,N,dim) fp16

    # -----------------------------------------------------------------
    # 9) Cast to float32 for the expected return type
    return out.to(torch.float32)
