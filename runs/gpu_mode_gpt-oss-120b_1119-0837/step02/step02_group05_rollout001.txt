TriMul (outgoing) – fast Triton + PyTorch implementation.

The heavy part of the TriMul forward is the “triangle” multiplication

    out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]

For each batch element and hidden channel `h` this is exactly a matrix‑multiply
`left @ rightᵀ` of shape (N,N)·(N,N) → (N,N).  The implementation fuses the five
linear projections (left_proj, left_gate, right_proj, right_gate, out_gate) into a
single matrix‑multiply (`F.linear`) so the input tensor `x` is read only once.
The two gated tensors `left` and `right` are then packed as (C_total,N,N) with
`C_total = B·hidden_dim` and a custom Triton kernel performs a batched GEMM for
all channels in parallel (FP16 inputs, FP32 accumulation).  Finally a layer‑norm,
the output gate, and a linear projection back to the original dimension are
applied in FP32.

The kernel uses 128×128 tiles, K‑tile = 64 and 8 warps – a good fit for the
H100 tensor‑core pipeline.  All operations are pure forward‑only (no gradients).

"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: out = left @ rightᵀ for every hidden channel (C_total)
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    out_ptr,          # *float32  (C_total, M, N)
    left_ptr,         # *float16  (C_total, M, K) – left tensor
    right_ptr,        # *float16  (C_total, N, K) – right tensor (will be read transposed)
    M, N, K,          # int32 matrix dimensions (all equal to seq_len)
    # strides (in elements, not bytes)
    stride_out_c, stride_out_m, stride_out_n,
    stride_left_c, stride_left_m, stride_left_k,
    stride_right_c, stride_right_k, stride_right_n,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # --------------------------------------------------------------
    # program IDs – each (pid_c, pid_m, pid_n) processes a tile
    # --------------------------------------------------------------
    pid_c = tl.program_id(2)   # hidden‑channel (B*hidden_dim) index
    pid_m = tl.program_id(0)   # block row index (i)
    pid_n = tl.program_id(1)   # block column index (j)

    # --------------------------------------------------------------
    # tile offsets
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in FP32 (Tensor‑Core accumulation)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # iterate over K dimension
    # --------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # left tile: (M, K)
        left_tile_ptr = left_ptr + pid_c * stride_left_c \
                         + offs_m[:, None] * stride_left_m \
                         + offs_k[None, :] * stride_left_k
        left = tl.load(
            left_tile_ptr,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )  # fp16

        # right tile (transposed on‑the‑fly): (K, N)
        right_tile_ptr = right_ptr + pid_c * stride_right_c \
                          + offs_k[:, None] * stride_right_k \
                          + offs_n[None, :] * stride_right_n
        right = tl.load(
            right_tile_ptr,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )  # fp16

        # GEMM accumulate – Tensor‑core dot, output in FP32
        acc += tl.dot(left, right, out_dtype=tl.float32)

    # --------------------------------------------------------------
    # write result
    # --------------------------------------------------------------
    out_tile_ptr = out_ptr + pid_c * stride_out_c \
                    + offs_m[:, None] * stride_out_m \
                    + offs_n[None, :] * stride_out_n
    tl.store(out_tile_ptr, acc,
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor[B, N, N, C]      (float32)
            - mask:        torch.Tensor[B, N, N]          (bool/float) – may be None
            - weights: dict of model parameters (see reference implementation)
            - config:  dict with keys "dim", "hidden_dim", optionally "nomask"

    Returns:
        torch.Tensor[B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1) Input Layer‑Norm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, C)  float32

    # ------------------------------------------------------------------
    # 2) Fuse all linear projections (proj + gates) into one matmul
    # ------------------------------------------------------------------
    # cast to half‑precision once
    x_h = x.to(torch.float16)

    # concatenate all projection/gate weight matrices:
    # order: left_proj, left_gate, right_proj, right_gate, out_gate
    combined_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["left_gate.weight"],
            weights["right_proj.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)  # (5*hidden_dim, dim)

    # single batched linear: (B·N·N, dim) @ (dim, 5·hidden_dim) → (B·N·N, 5·hidden_dim)
    all_proj = F.linear(x_h, combined_weight)  # (B, N, N, 5*hidden_dim)

    # split into the five tensors
    left_proj, left_gate, right_proj, right_gate, out_gate = torch.split(
        all_proj, hidden_dim, dim=-1
    )
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)

    # ------------------------------------------------------------------
    # 3) Optional mask (broadcast over hidden dim) – applied before gating
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_f = mask.to(torch.float16).unsqueeze(-1)  # (B, N, N, 1)
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f

    # ------------------------------------------------------------------
    # 4) Apply the gating scalars
    # ------------------------------------------------------------------
    left = left_proj * left_gate      # (B, N, N, H)  fp16
    right = right_proj * right_gate   # (B, N, N, H)  fp16

    # ------------------------------------------------------------------
    # 5) Prepare tensors for the Triton kernel:
    #    reshape to (C_total, N, N) where C_total = B * hidden_dim
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    # move hidden dimension before the two sequence axes → contiguous layout
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    C_total = B * hidden_dim
    left_flat = left.reshape(C_total, N, N)   # (C_total, M, K)
    right_flat = right.reshape(C_total, N, N) # (C_total, N, K) – will be read transposed

    # output buffer for the core multiplication (FP32)
    out_flat = torch.empty(
        C_total, N, N, dtype=torch.float32, device=left_flat.device
    )  # (C_total, M, N)

    # ------------------------------------------------------------------
    # 6) Triton kernel launch (batched GEMM)
    # ------------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    # strides in element units (not bytes)
    stride_out_c, stride_out_m, stride_out_n = out_flat.stride()
    stride_left_c, stride_left_m, stride_left_k = left_flat.stride()
    stride_right_c, stride_right_k, stride_right_n = right_flat.stride()

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # pid_m
        (N + BLOCK_N - 1) // BLOCK_N,   # pid_n
        C_total,                        # pid_c (batch*hidden)
    )

    tri_mul_kernel[grid](
        out_flat,
        left_flat,
        right_flat,
        N,
        N,
        N,
        stride_out_c,
        stride_out_m,
        stride_out_n,
        stride_left_c,
        stride_left_m,
        stride_left_k,
        stride_right_c,
        stride_right_k,
        stride_right_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=2,      # double‑buffering for the K‑loop
    )

    # ------------------------------------------------------------------
    # 7) Reshape back to (B, N, N, H)
    # ------------------------------------------------------------------
    out = out_flat.reshape(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8) Post‑processing: hidden‑dim LayerNorm → out_gate → final linear
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )  # (B, N, N, H)  float32

    # out_gate is still half‑precision → promote before multiplication
    out = out * out_gate.to(torch.float32)

    # final projection back to the original dimension (float32)
    out = F.linear(out, weights["to_out.weight"])   # (B, N, N, dim)

    return out
