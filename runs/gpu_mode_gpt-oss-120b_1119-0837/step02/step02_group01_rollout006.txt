"""
Highly‑optimized Triton implementation of the “outgoing” TriMul operator.

Algorithm
---------
1️⃣ Layer‑Norm over the last dimension of the input 4‑D tensor using a
   custom Triton kernel (operates on a flattened (M, D) view where
   M = B·N·N).

2️⃣ Fuse the four linear projections (left/right + their gates) into a
   single half‑precision GEMM:
   * Concatenate the four weight matrices (left_proj, left_gate,
     right_proj, right_gate) → shape (4·H, D).
   * A single `F.linear` yields a tensor `(M, 4·H)` which is then split
     into the four components.
   * Apply the optional mask *before* the gates, then apply sigmoid
     gates element‑wise.  The result (`left`, `right`) stays in fp16.

3️⃣ Compute the output‑gate (`out_gate`) with a second half‑precision
   GEMM and sigmoid.

4️⃣ Core O(N³·H) contraction:
   * Reshape `left` / `right` to (B, H, N, N) and fuse the batch‑ and
     hidden‑dimension into one large batched GEMM (`torch.bmm`) that
     executes on Tensor‑Cores.
   * The result has shape (B, N, N, H) in fp16.

5️⃣ Second Layer‑Norm (over the hidden dimension) implemented with the
   same Triton kernel.  Inputs are cast to fp32 for numeric stability,
   accumulation is performed in fp32, and the output is fp32.

6️⃣ Apply the output‑gate (`out_gate`), then a final linear projection
   back to the model dimension (`to_out`).  The final result is returned
   as fp32.

The implementation reduces passes over the large tensor `x` from five
to two by fusing the projections and gates, keeps the heavy N³ part
in half‑precision on Tensor‑Cores, and uses custom Triton kernels for
both LayerNorms as required.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# 2‑pass LayerNorm kernel (operates on the last dimension)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,                # *ptr to input  [M, N]
    y_ptr,                # *ptr to output [M, N]
    weight_ptr,           # *ptr to weight [N]
    bias_ptr,             # *ptr to bias   [N]
    M,                    # number of rows    (B·N·N)
    N,                    # channel dimension (D or H)
    eps,                  # epsilon
    stride_xm, stride_xn,  # input strides
    stride_ym, stride_yn,  # output strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
):
    pid = tl.program_id(0)

    # ------------------------------------------------------------------
    # Row range for this program
    # ------------------------------------------------------------------
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # ------------------------------------------------------------------
    # First pass: compute per‑row mean & variance (in FP32)
    # ------------------------------------------------------------------
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_N)
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        # Load as FP16 (or FP32) then cast to FP32 for accumulation
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        x_f32 = x.to(tl.float32)
        sum_val += tl.sum(x_f32, axis=1)
        sum_sq  += tl.sum(x_f32 * x_f32, axis=1)

    mean = sum_val / N
    var  = sum_sq / N - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # ------------------------------------------------------------------
    # Second pass: write normalized + affine transformed values
    # ------------------------------------------------------------------
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        x_f32 = x.to(tl.float32)
        norm_x = (x_f32 - mean[:, None]) * rstd[:, None]  # (BLOCK_M, BLOCK_N)

        # Load weight & bias (both FP32)
        w = tl.load(weight_ptr + offs_n, mask=mask_n, other=0.0)
        b = tl.load(bias_ptr   + offs_n, mask=mask_n, other=0.0)

        y = norm_x * w[None, :] + b[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn,
            y,
            mask=mask_m[:, None] & mask_n[None, :],
        )

def triton_layernorm(x: torch.Tensor,
                     weight: torch.Tensor,
                     bias: torch.Tensor,
                     eps: float = 1e-5) -> torch.Tensor:
    """
    LayerNorm over the last dimension using the Triton kernel.
    `x` must be a contiguous 2‑D tensor (M, N) and can be FP16 or FP32.
    The kernel accumulates in FP32 and writes the output in the dtype of `x`.
    """
    assert x.dim() == 2, "triton_layernorm expects a 2‑D tensor"
    M, N = x.shape
    y = torch.empty_like(x)

    # Ensure contiguous layout for correct stride handling
    x = x.contiguous()
    y = y.contiguous()
    weight = weight.contiguous()
    bias = bias.contiguous()

    stride_xm, stride_xn = x.stride()
    stride_ym, stride_yn = y.stride()

    grid = lambda meta: (triton.cdiv(M, meta["BLOCK_M"]),)

    layernorm_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        weight_ptr=weight,
        bias_ptr=bias,
        M=M,
        N=N,
        eps=eps,
        stride_xm=stride_xm,
        stride_xn=stride_xn,
        stride_ym=stride_ym,
        stride_yn=stride_yn,
        BLOCK_M=128,   # rows per program – good default for H100
        BLOCK_N=64,    # columns per tile
    )
    return y

# ----------------------------------------------------------------------
# Triton‑accelerated TriMul “outgoing” entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, D] (float32)
        - mask         : torch.Tensor [B, N, N] (bool/float) or None
        - weights      : dict of torch.Tensors (model parameters)
        - config       : dict with keys ``dim``, ``hidden_dim``, ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor [B, N, N, D] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype  # expected torch.float32

    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (Triton)
    # ------------------------------------------------------------------
    M = bs * seq_len * seq_len                     # rows for LN
    x_flat = input_tensor.reshape(M, dim)          # (M, D)
    norm_weight = weights["norm.weight"].to(device, dtype)
    norm_bias   = weights["norm.bias"].to(device, dtype)
    x_norm_flat = triton_layernorm(x_flat, norm_weight, norm_bias, eps=eps)
    x_norm = x_norm_flat.view(bs, seq_len, seq_len, dim)   # (B,N,N,D)

    # ------------------------------------------------------------------
    # Cast to half‑precision for the heavy part
    # ------------------------------------------------------------------
    x_norm_f16 = x_norm.to(torch.float16)

    # ------------------------------------------------------------------
    # 2️⃣ Fuse left/right projections and their gates (single GEMM)
    # ------------------------------------------------------------------
    # Load and cast weights to fp16
    left_proj_w   = weights["left_proj.weight"].to(device, torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device, torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device, torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device, torch.float16)

    # Concatenate the four matrices: [Lproj, Lgate, Rproj, Rgate] → (4H, D)
    concat_w = torch.cat([left_proj_w, left_gate_w, right_proj_w, right_gate_w], dim=0)
    # (M, 4H) half‑precision
    proj_gate = F.linear(x_norm_f16, concat_w)   # fused linear

    H = hidden_dim
    # Slice the concatenated result
    left_proj   = proj_gate[:, 0:H]               # (M, H)
    left_gate_p = proj_gate[:, H:2*H]             # (M, H)
    right_proj  = proj_gate[:, 2*H:3*H]           # (M, H)
    right_gate_p = proj_gate[:, 3*H:4*H]          # (M, H)

    # ------------------------------------------------------------------
    # 3️⃣ Mask (if present) – broadcast over hidden dimension
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_f16 = mask.to(device, torch.float16).reshape(M, 1)   # (M,1)
        left_proj = left_proj * mask_f16
        right_proj = right_proj * mask_f16

    # ------------------------------------------------------------------
    # 4️⃣ Apply sigmoid gates
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(left_gate_p)
    right_gate = torch.sigmoid(right_gate_p)

    left  = left_proj * left_gate    # (M, H)
    right = right_proj * right_gate  # (M, H)

    # ------------------------------------------------------------------
    # 5️⃣ Output gate (separate linear)
    # ------------------------------------------------------------------
    out_gate_w = weights["out_gate.weight"].to(device, torch.float16)
    out_gate_pre = F.linear(x_norm_f16, out_gate_w)   # (M, H)
    out_gate = torch.sigmoid(out_gate_pre)

    # Reshape to 4‑D tensors
    left  = left.view(bs, seq_len, seq_len, H)   # (B,N,N,H)
    right = right.view(bs, seq_len, seq_len, H)  # (B,N,N,H)
    out_gate = out_gate.view(bs, seq_len, seq_len, H)  # (B,N,N,H)

    # ------------------------------------------------------------------
    # 6️⃣ Core N³ contraction: batched fp16 GEMM (Tensor‑Core)
    # ------------------------------------------------------------------
    # Rearrange to (B, H, N, N) → (B*H, N, N) for torch.bmm
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B,H,N,N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # (B,H,N,N)

    BH = bs * H
    left_mat  = left_perm.view(BH, seq_len, seq_len)   # (B*H, N, N)
    right_mat = right_perm.view(BH, seq_len, seq_len)  # (B*H, N, N)

    # Batched matrix multiply: (i,k)·(j,k)ᵀ → (i,j) for each hidden dim
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B*H, N, N) half

    # Restore (B,N,N,H)
    out = out_mat.view(bs, H, seq_len, seq_len).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # ------------------------------------------------------------------
    # 7️⃣ Second LayerNorm (after contraction)
    #    Operates in FP32 for numerical stability
    # ------------------------------------------------------------------
    out_fp32 = out.to(torch.float32)
    out_flat = out_fp32.reshape(M, H)                # (B*N*N, H)

    to_out_norm_w = weights["to_out_norm.weight"].to(device, torch.float32)
    to_out_norm_b = weights["to_out_norm.bias"].to(device, torch.float32)
    out_norm_flat = triton_layernorm(out_flat, to_out_norm_w, to_out_norm_b, eps=eps)
    out_norm = out_norm_flat.view(bs, seq_len, seq_len, H)   # (B,N,N,H) fp32

    # ------------------------------------------------------------------
    # 8️⃣ Apply output gate and final projection back to model dim
    # ------------------------------------------------------------------
    out_gate_fp32 = out_gate.to(torch.float32)
    gated = out_norm * out_gate_fp32                # (B,N,N,H) fp32

    to_out_w = weights["to_out.weight"].to(device, torch.float32)   # (D, H)
    output = F.linear(gated, to_out_w)                             # (B,N,N,D) fp32

    return output
