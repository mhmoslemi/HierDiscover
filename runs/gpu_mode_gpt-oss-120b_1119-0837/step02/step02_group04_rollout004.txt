import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def tri_mul_kernel(out_ptr, a_ptr, b_ptr,
                   M, N, K,
                   stride_out_m, stride_out_n, stride_out_c,
                   stride_a_m, stride_a_k, stride_a_c,
                   stride_b_k, stride_b_n, stride_b_c,
                   BLOCK_M: tl.constexpr,
                   BLOCK_N: tl.constexpr,
                   BLOCK_K: tl.constexpr):
    pid_m = tl.program_id(0) # i block
    pid_n = tl.program_id(1) # j block
    pid_c = tl.program_id(2) # channel

    # compute offets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = kk * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        a_ptrs = a_ptr + pid_c * stride_a_c + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
        b_ptrs = b_ptr + pid_c * stride_b_c + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n

        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        acc += tl.dot(a, b)

    out_ptrs = out_ptr + pid_c * stride_out_c + offs_m[:, None] * stride_out_m + offs_n[None, :] * stride_out_n
    tl.store(out_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])
```

TriMul (outgoing) – fast Triton‑fused batched GEMM

The core of the outgoing triangle‑multiplicative update is

    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

where ``left`` and ``right`` are linear projections (with learned sigmoid gates)
of a layer‑normed input tensor.  The heavy N³ work is a batched matrix‑multiply:
for each (batch , hidden) channel we need a GEMM of shape (N,N)·(N,N)ᵀ.
The implementation below

* performs the LayerNorm and the four linear projections (including gating) in
  regular PyTorch – cuBLAS‑accelerated – which is already optimal for those
  small‑batch matmuls;
* computes the large batched GEMM with a custom Triton kernel that
  – works on FP16 inputs,
  – accumulates in FP32,
  – uses a 128 × 128 × 64 tile (the sweet spot on Hopper GPUs),
  – double‑buffers loads (``num_stages=2``) to hide global‑memory latency,
  – launches one program per hidden channel (C = batch × hidden_dim).

All tensors remain on the device, the final result is returned in FP32
as required by downstream layers.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    out_ptr,               # float32*   output  (C,M,N)
    a_ptr,                 # float16*   left    (C,M,K)
    b_ptr,                 # float16*   rightᵀ  (C,K,N) – loaded transposed
    M, N, K,               # matrix dimensions (all = seq_len)
    stride_out_m, stride_out_n, stride_out_c,   # strides for out
    stride_a_m, stride_a_k, stride_a_c,        # strides for left
    stride_b_k, stride_b_n, stride_b_c,        # strides for rightᵀ
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched GEMM: out = left @ rightᵀ.

    One program (block) works on a tile (BLOCK_M, BLOCK_N) of a single hidden
    channel (identified by pid_c).  The implementation follows the classic
    tiled GEMM pattern with double‑buffered loads and FP16×FP16 → FP32
    tensor‑core dot products.
    """
    pid_m = tl.program_id(0)   # block row   → i‑index of the output
    pid_n = tl.program_id(1)   # block col   → j‑index of the output
    pid_c = tl.program_id(2)   # hidden channel (batch * hidden_dim)

    # --------------------------------------------------------------
    # Tile offsets in the M and N dimensions
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator in FP32 (more stable than FP16)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over K in BLOCK_K‑sized chunks
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = kk * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Pointers to the current blocks of left (M×K) and rightᵀ (K×N)
        a_tile_ptr = (
            a_ptr
            + pid_c * stride_a_c
            + offs_m[:, None] * stride_a_m
            + offs_k[None, :] * stride_a_k
        )
        b_tile_ptr = (
            b_ptr
            + pid_c * stride_b_c
            + offs_k[:, None] * stride_b_k   # K dimension first (row)
            + offs_n[None, :] * stride_b_n   # N dimension (col)
        )

        # Load tiles (masked loads return 0.0 outside the matrix)
        a = tl.load(a_tile_ptr,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)
        b = tl.load(b_tile_ptr,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # FP16·FP16 → FP32 dot‑product
        acc += tl.dot(a, b)

    # --------------------------------------------------------------
    # Write the result tile back to global memory
    out_tile_ptr = (
        out_ptr
        + pid_c * stride_out_c
        + offs_m[:, None] * stride_out_m
        + offs_n[None, :] * stride_out_n
    )
    tl.store(out_tile_ptr,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward‑pass of the outgoing TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor[bs, N, N, dim] (float32)
        - mask         : torch.Tensor[bs, N, N] (bool/float) – may be None
        - weights      : dict of torch.Tensors (model parameters)
        - config       : dict with keys
            * "dim"        – input channel dimension `C`
            * "hidden_dim" – hidden dimension `H`
            * "nomask"     – bool, if True the mask is ignored

    Returns
    -------
    torch.Tensor[bs, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # 0) unpack
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1) Layer‑norm the input (FP32 for numerical stability)
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )                                                   # [B, N, N, C] (fp32)

    # ------------------------------------------------------------------
    # 2) Cast to FP16 – all linear work will use half‑precision
    x_h = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Linear projections (no bias)
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)

    left  = F.linear(x_h, left_proj_w)      # [B, N, N, H]
    right = F.linear(x_h, right_proj_w)    # [B, N, N, H]

    # ------------------------------------------------------------------
    # 4) Gate logits → sigmoid (still FP16)
    left_gate_w  = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w   = weights["out_gate.weight"].to(torch.float16)

    left_gate  = torch.sigmoid(F.linear(x_h, left_gate_w))   # [B, N, N, H]
    right_gate = torch.sigmoid(F.linear(x_h, right_gate_w)) # [B, N, N, H]
    out_gate   = torch.sigmoid(F.linear(x_h, out_gate_w))   # [B, N, N, H]

    # ------------------------------------------------------------------
    # 5) Optional mask (applied before gates, as in the reference)
    if not nomask and mask is not None:
        mask_f = mask.to(torch.float16).unsqueeze(-1)        # [B, N, N, 1]
        left  = left * mask_f
        right = right * mask_f

    # 6) Apply gates
    left  = left * left_gate
    right = right * right_gate

    # Release temporary tensors that are no longer needed
    del left_gate, right_gate, x_h, x

    # ------------------------------------------------------------------
    # 7) Prepare data for the Triton GEMM
    B, N, _, _ = left.shape                       # (B, N, N, H)
    # Move hidden dimension before the two sequence dims → contiguous layout
    left  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    C_total = B * hidden_dim                       # number of independent GEMMs
    left_flat  = left.reshape(C_total, N, N)       # (C, M, K)
    right_flat = right.reshape(C_total, N, N)      # (C, K, N)  (we will read it transposed)

    # Output buffer – accumulate in FP32
    out_flat = torch.empty_like(left_flat, dtype=torch.float32)

    # ------------------------------------------------------------------
    # 8) Triton kernel launch configuration
    #   Block size 128×128 works well for N up to 1024 on Hopper.
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    stride_out_c = out_flat.stride(0)
    stride_out_m = out_flat.stride(1)
    stride_out_n = out_flat.stride(2)

    stride_a_c = left_flat.stride(0)
    stride_a_m = left_flat.stride(1)
    stride_a_k = left_flat.stride(2)

    stride_b_c = right_flat.stride(0)
    # right is accessed as (K,N) = (last axis, middle axis)
    stride_b_k = right_flat.stride(2)   # last axis (K)
    stride_b_n = right_flat.stride(1)   # middle axis (N)

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # blocks along M (i)
        (N + BLOCK_N - 1) // BLOCK_N,   # blocks along N (j)
        C_total,                        # one program per hidden channel
    )

    # Launch the batched GEMM
    tri_mul_kernel[grid](
        out_flat,
        left_flat,
        right_flat,
        N, N, N,
        stride_out_m,
        stride_out_n,
        stride_out_c,
        stride_a_m,
        stride_a_k,
        stride_a_c,
        stride_b_k,
        stride_b_n,
        stride_b_c,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=2,
    )

    # ------------------------------------------------------------------
    # 9) Reshape back to (B, N, N, H)
    out = out_flat.reshape(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 10) Post‑processing – layer‑norm, output gate, final projection
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    # Multiply by the output gate (FP16 → FP32 promotion happens automatically)
    out = out * out_gate

    # Final linear projection back to the original dimension (use FP16 for speed)
    to_out_w = weights["to_out.weight"].to(torch.float16)
    out = F.linear(out.to(torch.float16), to_out_w).to(torch.float32)

    return out
