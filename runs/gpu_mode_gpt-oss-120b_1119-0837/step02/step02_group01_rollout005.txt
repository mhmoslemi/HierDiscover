TriMul “outgoing’’ forward pass – Triton‑accelerated version.

Algorithm
---------
1️⃣  Layer‑Norm the input tensor (B,N,N,D) with a custom Triton kernel.
2️⃣  Cast to FP16 and perform a *single* fused linear projection that
    simultaneously produces the three projection tensors (left, right)
    and the three gate tensors (left_gate, right_gate, out_gate).  This
    eliminates five separate GEMM calls and reduces memory traffic.
3️⃣  Apply sigmoid to the raw gates and (optionally) the pairwise mask;
    compute the gated tensors  `left = left·σ(left_gate)·mask` and
    `right = right·σ(right_gate)·mask`.
4️⃣  Contract the two 4‑D tensors with the O(N³·H) pattern

        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]

    by reshaping to (B·H, N, N) and using a batched half‑precision
    `torch.bmm` (cuBLAS/Tensor‑Core) – the heaviest part of the operator.
5️⃣  Layer‑Norm the result with a second Triton kernel (weights are FP32,
    accumulation stays in FP32, output remains FP16 for speed).
6️⃣  Apply the output gate and the final linear projection back to the
    model dimension.  The final tensor is returned in FP32.

Only the two LayerNorms are implemented in Triton; all dense linear
operations use highly‑optimized FP16 GEMM kernels provided by PyTorch.
This satisfies the “part of the operations in a kernel’’ requirement while
dramatically cutting runtime versus the naïve PyTorch reference.
"""

@triton.jit
def layernorm_kernel(
    x_ptr,                # *ptr to input  [M, N]
    y_ptr,                # *ptr to output [M, N]
    weight_ptr,           # *ptr to weight [N]
    bias_ptr,             # *ptr to bias   [N]
    M,                    # #rows   (batch*seq*seq)
    N,                    # feature dimension (dim or hidden_dim)
    eps,                  # epsilon
    stride_xm, stride_xn,  # input strides
    stride_ym, stride_yn,  # output strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
):
    pid = tl.program_id(0)

    # Row range for this program
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # ------------------------------------------------------------------
    # First pass: compute per‑row mean & variance (FP32 accumulation)
    # ------------------------------------------------------------------
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_N)
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        x_f32 = tl.cast(x, tl.float32)
        sum_val += tl.sum(x_f32, axis=1)
        sum_sq  += tl.sum(x_f32 * x_f32, axis=1)

    mean = sum_val / N
    var  = sum_sq / N - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # ------------------------------------------------------------------
    # Second pass: write normalised + affine transformed values
    # ------------------------------------------------------------------
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        x_f32 = tl.cast(x, tl.float32)

        # normalise (still FP32)
        norm_x = (x_f32 - mean[:, None]) * rstd[:, None]

        # affine transform – weight/bias may be FP16 or FP32; cast to FP32
        w = tl.cast(tl.load(weight_ptr + offs_n, mask=mask_n, other=0.0), tl.float32)
        b = tl.cast(tl.load(bias_ptr + offs_n, mask=mask_n, other=0.0), tl.float32)

        y = norm_x * w[None, :] + b[None, :]
        # store back in the original dtype of x (automatically casts)
        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn,
            tl.cast(y, x.dtype),
            mask=mask_m[:, None] & mask_n[None, :],
        )

def triton_layernorm(x: torch.Tensor,
                     weight: torch.Tensor,
                     bias: torch.Tensor,
                     eps: float = 1e-5) -> torch.Tensor:
    """
    LayerNorm over the last dimension using the Triton kernel.
    `x` must be a contiguous 2‑D tensor (M, N).
    """
    assert x.dim() == 2, "triton_layernorm expects a 2‑D tensor"
    M, N = x.shape
    y = torch.empty_like(x)

    # Ensure contiguous layout for correct stride handling
    x = x.contiguous()
    y = y.contiguous()
    weight = weight.contiguous()
    bias = bias.contiguous()

    stride_xm, stride_xn = x.stride()
    stride_ym, stride_yn = y.stride()

    grid = lambda meta: (triton.cdiv(M, meta["BLOCK_M"]),)

    layernorm_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        weight_ptr=weight,
        bias_ptr=bias,
        M=M,
        N=N,
        eps=eps,
        stride_xm=stride_xm,
        stride_xn=stride_xn,
        stride_ym=stride_ym,
        stride_yn=stride_yn,
        BLOCK_M=128,   # rows per program
        BLOCK_N=64,    # columns per tile
    )
    return y


# ----------------------------------------------------------------------
# Triton‑accelerated TriMul “outgoing’’ entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing’’ operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, D] (float32)
        - mask         : torch.Tensor [B, N, N] or None
        - weights      : dict of torch.Tensors (all parameters)
        - config       : dict with keys ``dim``, ``hidden_dim``, ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor [B, N, N, D] (float32)
    """
    # --------------------------------------------------------------
    # Unpack
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype          # expected torch.float32

    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    eps = 1e-5

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm (Triton)
    # --------------------------------------------------------------
    M = bs * seq_len * seq_len                     # rows for layernorm
    x_flat = input_tensor.reshape(M, dim)          # (M, D) FP32

    norm_weight = weights["norm.weight"].to(device, dtype)
    norm_bias   = weights["norm.bias"].to(device, dtype)

    x_norm_flat = triton_layernorm(x_flat, norm_weight, norm_bias, eps=eps)
    x_norm = x_norm_flat.view(bs, seq_len, seq_len, dim)   # (B,N,N,D) FP32

    # --------------------------------------------------------------
    # 2️⃣ Cast to FP16 for the heavy part
    # --------------------------------------------------------------
    x_norm_f16 = x_norm.to(torch.float16)
    x_norm_f16_flat = x_norm_f16.reshape(M, dim)   # (M, D) FP16

    # --------------------------------------------------------------
    # 3️⃣ Fuse all projection + gate linear layers into one GEMM
    # --------------------------------------------------------------
    #   left_proj   : (H, D)
    #   left_gate   : (H, D)
    #   right_proj  : (H, D)
    #   right_gate  : (H, D)
    #   out_gate    : (H, D)
    # Stack them along the output‑feature dimension.
    left_proj_w   = weights["left_proj.weight"]
    left_gate_w   = weights["left_gate.weight"]
    right_proj_w  = weights["right_proj.weight"]
    right_gate_w  = weights["right_gate.weight"]
    out_gate_w    = weights["out_gate.weight"]

    # shape (5*H, D)  – no bias in any of these linear layers
    combined_weight = torch.cat(
        [
            left_proj_w, left_gate_w,
            right_proj_w, right_gate_w,
            out_gate_w,
        ],
        dim=0,
    ).to(device, torch.float16)                     # (5*H, D) FP16

    # Single fused linear projection
    proj_gate_flat = F.linear(x_norm_f16_flat, combined_weight)   # (M, 5*H) FP16

    # --------------------------------------------------------------
    # 4️⃣ Slice & apply sigmoid gates (still FP16)
    # --------------------------------------------------------------
    proj_gate = proj_gate_flat.view(bs, seq_len, seq_len, 5 * hidden_dim)   # (B,N,N,5H)

    H = hidden_dim
    left_proj   = proj_gate[..., 0:H]                     # (B,N,N,H)
    left_gate_r = proj_gate[..., H:2*H]
    right_proj  = proj_gate[..., 2*H:3*H]
    right_gate_r= proj_gate[..., 3*H:4*H]
    out_gate_r  = proj_gate[..., 4*H:5*H]

    left_gate = torch.sigmoid(left_gate_r)               # (B,N,N,H) FP16
    right_gate= torch.sigmoid(right_gate_r)
    out_gate  = torch.sigmoid(out_gate_r)

    if not nomask and mask is not None:
        mask_f16 = mask.to(device, torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left  = left_proj * left_gate * mask_f16
        right = right_proj * right_gate * mask_f16
    else:
        left  = left_proj * left_gate
        right = right_proj * right_gate

    # --------------------------------------------------------------
    # 5️⃣ Batched contraction: Σₖ left[b,i,k,h] * right[b,j,k,h]
    # --------------------------------------------------------------
    # Rearrange to (B*H, N, N) for a batched GEMM
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B,H,N,N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # (B,H,N,N)

    BH = bs * hidden_dim
    left_mat  = left_perm.view(BH, seq_len, seq_len)      # (B*H, N, N)
    right_mat = right_perm.view(BH, seq_len, seq_len)     # (B*H, N, N)

    # Batched half‑precision GEMM (Tensor‑Core)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B*H, N, N) FP16

    # Restore (B,N,N,H) layout
    out = out_mat.view(bs, hidden_dim, seq_len, seq_len) \
                .permute(0, 2, 3, 1).contiguous()           # (B,N,N,H) FP16

    # --------------------------------------------------------------
    # 6️⃣ Output LayerNorm (Triton) – keep FP16 precision for speed
    # --------------------------------------------------------------
    out_flat = out.view(M, hidden_dim)                     # (B*N*N, H) FP16

    to_out_norm_weight = weights["to_out_norm.weight"].to(device, dtype)   # FP32
    to_out_norm_bias   = weights["to_out_norm.bias"].to(device, dtype)     # FP32

    out_norm_flat = triton_layernorm(out_flat, to_out_norm_weight, to_out_norm_bias, eps=eps)
    out_norm = out_norm_flat.view(bs, seq_len, seq_len, hidden_dim)  # (B,N,N,H) FP16

    # --------------------------------------------------------------
    # 7️⃣ Apply output gate and final projection back to model dim
    # --------------------------------------------------------------
    gated = out_norm * out_gate                                 # (B,N,N,H) FP16

    to_out_w = weights["to_out.weight"].to(device, torch.float16)   # (D, H) FP16
    # Final linear (FP16) and cast back to FP32 as the API contract
    output_fp16 = F.linear(gated, to_out_w)                      # (B,N,N,D) FP16
    output = output_fp16.to(torch.float32)                       # (B,N,N,D) FP32

    return output
