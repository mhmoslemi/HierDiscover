"""
TriMul (outgoing) – Triton‑accelerated core contraction.

The heavy O(N³) step
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
is a batched matrix multiplication (left @ rightᵀ) for every hidden
channel h.  This implementation
* keeps all cheap operations (layer‑norm, gating, final projection) in
  half‑precision PyTorch for minimal memory traffic,
* fuses the N³ contraction into a single Triton kernel that:
    - works in fp16 (half) for the operands,
    - accumulates in fp32,
    - tiles M‑ and N‑dimensions with 64×64 blocks and K‑tiles of 64,
    - uses 4 warps and 4 shared‑memory stages (double buffering) to
      hide latency,
* flattens the batch × hidden dimension into one GEMM batch,
* returns the final result in fp32 as required by the reference API.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# -----------------------------------------------------------------------------#
# Triton kernel: batched GEMM C = A @ Bᵀ
#   A : (batch, M, K)   – fp16
#   B : (batch, K, N)   – fp16 (already transposed)
#   C : (batch, M, N)   – fp16 (accumulated in fp32)
# -----------------------------------------------------------------------------#
@triton.jit
def _tri_mul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,            # pointers (fp16)
    M, N, K,                        # matrix shapes (M=N=seq_len, K=seq_len)
    stride_am, stride_ak,           # A row/col stride inside a batch
    stride_bk, stride_bn,           # B row/col stride (B is K×N)
    stride_cm, stride_cn,           # C row/col stride
    stride_a_batch, stride_b_batch, stride_c_batch,  # batch strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched fp16 GEMM with fp32 accumulation.

    One program per (M‑tile, N‑tile) × batch.
    """
    pid = tl.program_id(axis=0)      # tile index over (M,N)
    batch_id = tl.program_id(axis=1) # flattened (batch*hidden) dimension

    # --------------------------------------------------------------- #
    # tile coordinates
    # --------------------------------------------------------------- #
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    block_m = pid // num_blocks_n
    block_n = pid % num_blocks_n

    offs_m = block_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = block_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # --------------------------------------------------------------- #
    # batch‑specific pointers
    # --------------------------------------------------------------- #
    a_batch = a_ptr + batch_id * stride_a_batch
    b_batch = b_ptr + batch_id * stride_b_batch
    c_batch = c_ptr + batch_id * stride_c_batch

    # --------------------------------------------------------------- #
    # accumulator (fp32)
    # --------------------------------------------------------------- #
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------- #
    # main K‑loop
    # --------------------------------------------------------------- #
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_off = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_off < K

        # A tile: (BLOCK_M, BLOCK_K) – row‑major
        a_ptrs = a_batch + (offs_m[:, None] * stride_am) + (k_off[None, :] * stride_ak)
        a = tl.load(a_ptrs,
                    mask=(offs_m[:, None] < M) & mask_k[None, :],
                    other=0.0)

        # B tile (already transposed): (BLOCK_K, BLOCK_N)
        b_ptrs = b_batch + (k_off[:, None] * stride_bk) + (offs_n[None, :] * stride_bn)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & (offs_n[None, :] < N),
                    other=0.0)

        # rank‑K update (fp16×fp16 → fp32)
        acc += tl.dot(a, b)

    # --------------------------------------------------------------- #
    # write back (store as fp16)
    # --------------------------------------------------------------- #
    c_ptrs = c_batch + (offs_m[:, None] * stride_cm) + (offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=(offs_m[:, None] < M) & (offs_n[None, :] < N))


# -----------------------------------------------------------------------------#
# Public entry point
# -----------------------------------------------------------------------------#
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul block (no gradients).

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor, shape [B, N, N, D] (float32)
            - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
            - weights_dict : dict of parameter tensors (float32)
            - config_dict  : {"dim": D, "hidden_dim": H, "nomask": bool, ...}

    Returns:
        torch.Tensor of shape [B, N, N, D] (float32)
    """
    # ----------------------------------------------------------------- #
    # unpack arguments
    # ----------------------------------------------------------------- #
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    eps = 1e-5

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ----------------------------------------------------------------- #
    # 1) Layer‑norm on the feature dimension (float32 for numeric stability)
    # ----------------------------------------------------------------- #
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )   # [B, N, N, dim]  (float32)

    # ----------------------------------------------------------------- #
    # 2) Cast to fp16 and pull half‑precision weights once
    # ----------------------------------------------------------------- #
    x_fp16 = x.half()

    left_proj_w   = weights["left_proj.weight"].half()
    right_proj_w  = weights["right_proj.weight"].half()
    left_gate_w   = weights["left_gate.weight"].half()
    right_gate_w  = weights["right_gate.weight"].half()
    out_gate_w    = weights["out_gate.weight"].half()
    to_out_norm_w = weights["to_out_norm.weight"].half()
    to_out_norm_b = weights["to_out_norm.bias"].half()
    to_out_w      = weights["to_out.weight"].half()

    # ----------------------------------------------------------------- #
    # 3) Linear projections (no bias) – stay in fp16
    # ----------------------------------------------------------------- #
    left  = F.linear(x_fp16, left_proj_w)    # [B, N, N, H]
    right = F.linear(x_fp16, right_proj_w)   # [B, N, N, H]

    # ----------------------------------------------------------------- #
    # 4) Optional mask (broadcast over hidden dim)
    # ----------------------------------------------------------------- #
    if not nomask:
        # mask is expected to be either bool or float (0/1)
        mask_f = mask.to(x_fp16.dtype).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ----------------------------------------------------------------- #
    # 5) Gating (sigmoid) – fp16
    # ----------------------------------------------------------------- #
    left_gate  = torch.sigmoid(F.linear(x_fp16, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_fp16, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x_fp16, out_gate_w))

    left  = left * left_gate
    right = right * right_gate

    # ----------------------------------------------------------------- #
    # 6) Core contraction: batched GEMM (left @ rightᵀ)
    #    layout  (B, H, N, N) → (batch, M, K) and (batch, K, N)
    # ----------------------------------------------------------------- #
    # permute to bring hidden dim in front and make the matrices row‑major
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    # transpose right so that we can compute A @ Bᵀ
    right_T = right_perm.permute(0, 1, 3, 2).contiguous()  # (B, H, N, N)

    B_, H_, M, K = left_perm.shape   # M == K == N
    _, _, _, N_   = right_T.shape
    assert M == K == N_ == N, "Unexpected shape after permute"

    total_batches = B_ * H_               # flatten batch × hidden

    # reshape to classic batched GEMM layout
    a = left_perm.view(total_batches, M, K)          # (batch, M, K)  fp16
    b = right_T.view(total_batches, K, N_)           # (batch, K, N)  fp16
    c = torch.empty((total_batches, M, N_), dtype=torch.float16, device=device)

    # --------------------------------------------------------------- #
    # 7) Launch the Triton kernel
    # --------------------------------------------------------------- #
    BLOCK_M, BLOCK_N, BLOCK_K = 64, 64, 64   # tile sizes tuned for H100 fp16 TensorCore
    grid = (
        ((M + BLOCK_M - 1) // BLOCK_M) * ((N_ + BLOCK_N - 1) // BLOCK_N),  # tiles over (M,N)
        total_batches,                                                         # one program per GEMM batch
    )

    # strides for the flattened batch tensors (row‑major)
    stride_a_batch = M * K
    stride_am = K
    stride_ak = 1

    stride_b_batch = K * N_
    stride_bk = N_
    stride_bn = 1

    stride_c_batch = M * N_
    stride_cm = N_
    stride_cn = 1

    _tri_mul_gemm_kernel[grid](
        a, b, c,
        M, N_, K,
        stride_am, stride_ak,
        stride_bk, stride_bn,
        stride_cm, stride_cn,
        stride_a_batch, stride_b_batch, stride_c_batch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,        # 4 warps map nicely onto a 64×64 tile (16×16 TensorCores per warp)
        num_stages=4,       # double‑buffering depth for latency hiding
    )

    # --------------------------------------------------------------- #
    # 8) Reshape back to [B, N, N, H] (still fp16)
    # --------------------------------------------------------------- #
    out = c.view(B_, hidden_dim, M, N_).permute(0, 2, 3, 1).contiguous()  # [B, N, N, H]

    # --------------------------------------------------------------- #
    # 9) Hidden‑dim layer‑norm, out‑gate and final linear (all fp16)
    # --------------------------------------------------------------- #
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=eps,
    )
    out = out * out_gate                     # broadcast over hidden dim
    out = F.linear(out, to_out_w)            # back to original dim D (fp16)

    # --------------------------------------------------------------- #
    # 10) Cast to fp32 for the API contract
    # --------------------------------------------------------------- #
    return out.float()
