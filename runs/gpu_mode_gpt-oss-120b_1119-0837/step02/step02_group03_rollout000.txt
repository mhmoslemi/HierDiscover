1.  Input LayerNorm  (row‑wise) – implemented in a tiny Triton kernel.
2.  Cast to FP16 for the heavy compute.
3.  Fuse all per‑position linear projections (left‑proj, left‑gate,
    right‑proj, right‑gate, out‑gate) into a *single* matmul:
       [B·L·L, dim] @ [(5·H)·dim].T  →  [B·L·L, 5·H]
    Split the result, apply sigmoid to gate tensors and (optional) mask.
4.  Assemble tensors of shape (B, H, L, L) and run a *batched* matrix‑multiply
       out[b,h] = left[b,h] @ right[b,h].T
    using `torch.bmm`, which calls the highly‑optimized cuBLAS StridedBatchedGEMM
    (fp16 compute, fp32 accumulation on H100).
5.  Reshape back to (B, L, L, H) and apply a second LayerNorm (again Triton).
6.  Apply the output gate, then a final FP32 linear projection back to `dim`.

Only the two LayerNorms are written in Triton; the remaining work
leverages Tensor‑core matmuls (FP16) and cuBLAS batched GEMM for maximal
throughput.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton LayerNorm (row‑wise) – operates on a 2‑D tensor [M, N].
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,                # *ptr to input  [M, N]
    y_ptr,                # *ptr to output [M, N]
    weight_ptr,           # *ptr to weight [N]
    bias_ptr,             # *ptr to bias   [N]
    M,                    # number of rows
    N,                    # feature dimension
    eps,                  # epsilon
    stride_xm, stride_xn,  # strides for input
    stride_ym, stride_yn,  # strides for output
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
):
    pid = tl.program_id(0)

    # ------------------------------------------------------------------
    # Row offsets for this program
    # ------------------------------------------------------------------
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # ------------------------------------------------------------------
    # First pass: compute mean & variance
    # ------------------------------------------------------------------
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_N)
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        sum_val += tl.sum(x, axis=1)
        sum_sq  += tl.sum(x * x, axis=1)

    mean = sum_val / N
    var  = sum_sq / N - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # ------------------------------------------------------------------
    # Second pass: write normalised + affine transformed values
    # ------------------------------------------------------------------
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[0, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        # Normalise
        norm_x = (x - mean[:, None]) * rstd[:, None]

        # Load affine parameters
        w = tl.load(weight_ptr + offs_n, mask=mask_n, other=0.0)
        b = tl.load(bias_ptr   + offs_n, mask=mask_n, other=0.0)

        y = norm_x * w[None, :] + b[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn,
            y,
            mask=mask_m[:, None] & mask_n[None, :],
        )


def triton_layernorm(x: torch.Tensor, weight: torch.Tensor, bias: torch.Tensor, eps: float = 1e-5):
    """
    LayerNorm over the last dimension of ``x`` using the Triton kernel.
    ``x`` must be 2‑D (M, N). Returned tensor has the same dtype as ``x``.
    """
    assert x.dim() == 2, "triton_layernorm expects a 2‑D tensor"
    M, N = x.shape
    y = torch.empty_like(x)

    # ensure contiguous for stride correctness
    x = x.contiguous()
    y = y.contiguous()
    weight = weight.contiguous()
    bias = bias.contiguous()

    stride_xm, stride_xn = x.stride()
    stride_ym, stride_yn = y.stride()

    grid = lambda meta: (triton.cdiv(M, meta["BLOCK_M"]),)

    layernorm_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        weight_ptr=weight,
        bias_ptr=bias,
        M=M,
        N=N,
        eps=eps,
        stride_xm=stride_xm,
        stride_xn=stride_xn,
        stride_ym=stride_ym,
        stride_yn=stride_yn,
        BLOCK_M=128,
        BLOCK_N=64,
    )
    return y


# ----------------------------------------------------------------------
# Triton‑driven forward pass
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor   [B, L, L, D]   (float32)
        - mask         : torch.Tensor   [B, L, L] or None
        - weights      : dict of torch tensors (model parameters)
        - config       : dict with keys ``dim``, ``hidden_dim`` and optional ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, L, L, D] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    bs, L, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (Triton) – over the channel dimension ``dim``
    # ------------------------------------------------------------------
    M = bs * L * L
    x_flat = input_tensor.view(M, dim)
    norm_w = weights["norm.weight"].to(device, dtype).contiguous()
    norm_b = weights["norm.bias"].to(device, dtype).contiguous()
    x_norm_flat = triton_layernorm(x_flat, norm_w, norm_b, eps=eps)
    x_norm = x_norm_flat.view(bs, L, L, dim)

    # ------------------------------------------------------------------
    # 2️⃣ Cast to FP16 for the heavy compute
    # ------------------------------------------------------------------
    x_norm_h = x_norm.to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Fuse all per‑position linear projections (5 * hidden_dim) into one matmul
    # ------------------------------------------------------------------
    # Gather and concatenate weights (all are (hidden_dim, dim) in FP16)
    w_left   = weights["left_proj.weight"]
    w_left_g = weights["left_gate.weight"]
    w_right  = weights["right_proj.weight"]
    w_right_g = weights["right_gate.weight"]
    w_out_g  = weights["out_gate.weight"]

    W_all = torch.cat(
        [w_left, w_left_g, w_right, w_right_g, w_out_g], dim=0
    ).to(device, torch.float16).contiguous()          # (5*H, dim)

    # Linear projection of the whole [B·L·L, dim] matrix
    proj_all = F.linear(x_norm_h.view(-1, dim), W_all)  # (B·L·L, 5*H)
    proj_all = proj_all.view(bs, L, L, 5 * hidden_dim)

    # Split results
    left          = proj_all[..., :hidden_dim]                           # (B, L, L, H)
    left_gate     = torch.sigmoid(proj_all[..., hidden_dim:2 * hidden_dim])
    right         = proj_all[..., 2 * hidden_dim:3 * hidden_dim]
    right_gate    = torch.sigmoid(proj_all[..., 3 * hidden_dim:4 * hidden_dim])
    out_gate      = torch.sigmoid(proj_all[..., 4 * hidden_dim:])         # (B, L, L, H)

    # Apply gates
    left  = left * left_gate
    right = right * right_gate

    # Optional mask (broadcast on the hidden dimension)
    if not nomask and mask is not None:
        mask_h = mask.to(device, torch.float16).unsqueeze(-1)   # (B, L, L, 1)
        left  = left * mask_h
        right = right * mask_h

    # ------------------------------------------------------------------
    # 4️⃣ Batched matrix multiplication: out[b,h] = left[b,h] @ right[b,h].T
    #     using torch.bmm (cuBLAS StridedBatched GEMM, FP16 compute)
    # ------------------------------------------------------------------
    # Rearrange to (B, H, L, L) and flatten batch dimension
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, L, L)
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    batch = bs * hidden_dim
    left_flat  = left_perm.view(batch, L, L)            # (B*H, L, L)
    right_flat = right_perm.view(batch, L, L)           # (B*H, L, L)

    # BMM with transpose on the second operand
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))   # (B*H, L, L)

    # Reshape back to (B, L, L, H)
    out = out_flat.view(bs, hidden_dim, L, L).permute(0, 2, 3, 1).contiguous()  # (B, L, L, H)

    # ------------------------------------------------------------------
    # 5️⃣ Output LayerNorm (Triton) over the hidden dimension ``H``
    # ------------------------------------------------------------------
    out_M = bs * L * L
    out_flat = out.view(out_M, hidden_dim)
    to_out_norm_w = weights["to_out_norm.weight"].to(device, dtype).contiguous()
    to_out_norm_b = weights["to_out_norm.bias"].to(device, dtype).contiguous()
    out_norm_flat = triton_layernorm(out_flat, to_out_norm_w, to_out_norm_b, eps=eps)
    out_norm = out_norm_flat.view(bs, L, L, hidden_dim)

    # ------------------------------------------------------------------
    # 6️⃣ Apply the output gate and final projection back to ``dim``
    # ------------------------------------------------------------------
    # out_gate currently FP16 – promote to model dtype for the multiplication
    out_gate_f32 = out_gate.to(dtype)               # (B, L, L, H)
    out = out_norm * out_gate_f32

    # Final linear projection (hidden_dim -> dim) – stays in FP32
    to_out_w = weights["to_out.weight"].to(device, dtype).contiguous()  # (dim, hidden_dim)
    out = F.linear(out, to_out_w)                     # (B, L, L, dim)

    return out
```
