"""
Optimized Triton+PyTorch implementation of the *outgoing* TriMul operator
(AlphaFold‑3).  The heavy N³ contraction is performed with a single
batched `torch.bmm` (cuBLAS Tensor‑core GEMM) while the two LayerNorms are
fused into a custom Triton kernel that computes mean/variance per row and
applies the affine transformation in one pass.  All compute‑intensive
operations run in FP16 and the final result is cast back to FP32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton row‑wise LayerNorm (operates on a 2‑D tensor [M, N]).
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,                # *ptr to input  [M, N]
    y_ptr,                # *ptr to output [M, N]
    w_ptr,                # *ptr to weight [N]
    b_ptr,                # *ptr to bias   [N]
    M,                    # total rows
    N,                    # feature dimension
    eps,                  # epsilon
    stride_xm, stride_xn,  # strides for input
    stride_ym, stride_yn,  # strides for output
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
):
    pid = tl.program_id(0)

    # --------------------------------------------------------------
    # Row offsets for this program
    # --------------------------------------------------------------
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # --------------------------------------------------------------
    # First pass: compute mean & variance for each row
    # --------------------------------------------------------------
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_N)
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        sum_val += tl.sum(x, axis=1)
        sum_sq  += tl.sum(x * x, axis=1)

    mean = sum_val / N
    var  = sum_sq / N - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # --------------------------------------------------------------
    # Second pass: write normalized & affine‑transformed values
    # --------------------------------------------------------------
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        norm_x = (x - mean[:, None]) * rstd[:, None]

        w = tl.load(w_ptr + offs_n, mask=mask_n, other=0.0)
        b = tl.load(b_ptr + offs_n, mask=mask_n, other=0.0)

        y = norm_x * w[None, :] + b[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn,
            y,
            mask=mask_m[:, None] & mask_n[None, :],
        )


def triton_layernorm(x: torch.Tensor, weight: torch.Tensor, bias: torch.Tensor,
                    eps: float = 1e-5) -> torch.Tensor:
    """
    Row‑wise LayerNorm using the Triton kernel above.
    Arguments:
        x      – (M, N) 2‑D tensor (any dtype, but weight/bias must match dtype)
        weight – (N,) 1‑D tensor
        bias   – (N,) 1‑D tensor
    Returns:
        Tensor of shape (M, N) with the same dtype as *x*.
    """
    assert x.dim() == 2, "triton_layernorm expects a 2‑D tensor"
    M, N = x.shape
    y = torch.empty_like(x)

    # Ensure contiguous tensors (required for correct stride handling)
    x = x.contiguous()
    y = y.contiguous()
    weight = weight.contiguous()
    bias = bias.contiguous()

    stride_xm, stride_xn = x.stride()
    stride_ym, stride_yn = y.stride()

    grid = (triton.cdiv(M, 128),)  # one program per BLOCK_M rows

    layernorm_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        w_ptr=weight,
        b_ptr=bias,
        M=M,
        N=N,
        eps=eps,
        stride_xm=stride_xm,
        stride_xn=stride_xn,
        stride_ym=stride_ym,
        stride_yn=stride_yn,
        BLOCK_M=128,
        BLOCK_N=64,
    )
    return y


# ----------------------------------------------------------------------
# Entry point required by the test harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor   [B, L, L, D] (float32)
        - mask         : torch.Tensor   [B, L, L] (float32 or bool) or ``None``
        - weights      : dict of torch tensors (model parameters)
        - config       : dict with keys ``dim``, ``hidden_dim`` and optional ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, L, L, D] (float32)
    """
    # --------------------------------------------------------------
    # Unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    bs, L, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    eps = 1e-5

    # --------------------------------------------------------------
    # 1️⃣  Input LayerNorm (FP16) – Triton
    # --------------------------------------------------------------
    # Cast input to FP16 for the whole pipeline
    x_fp16 = input_tensor.to(torch.float16)
    norm_w = weights["norm.weight"].to(device, torch.float16).contiguous()
    norm_b = weights["norm.bias"].to(device, torch.float16).contiguous()
    x_norm = triton_layernorm(x_fp16, norm_w, norm_b, eps=eps)  # [B, L, L, D] FP16

    # --------------------------------------------------------------
    # 2️⃣  Linear projections (bias‑free, FP16)
    # --------------------------------------------------------------
    left_w  = weights["left_proj.weight"].to(device, torch.float16).contiguous()
    right_w = weights["right_proj.weight"].to(device, torch.float16).contiguous()
    left  = F.linear(x_norm, left_w)   # [B, L, L, H]
    right = F.linear(x_norm, right_w)  # [B, L, L, H]

    # --------------------------------------------------------------
    # 3️⃣  Optional mask (broadcast over hidden dim)
    # --------------------------------------------------------------
    if not nomask and mask is not None:
        # mask may be bool or float – cast to FP16 and add a trailing dim
        mask_h = mask.to(device, torch.float16).unsqueeze(-1)  # [B, L, L, 1]
        left = left * mask_h
        right = right * mask_h

    # --------------------------------------------------------------
    # 4️⃣  Gating (sigmoid) – FP16
    # --------------------------------------------------------------
    left_gate_w  = weights["left_gate.weight"].to(device, torch.float16).contiguous()
    right_gate_w = weights["right_gate.weight"].to(device, torch.float16).contiguous()
    out_gate_w   = weights["out_gate.weight"].to(device, torch.float16).contiguous()

    left_gate  = torch.sigmoid(F.linear(x_norm, left_gate_w))   # [B, L, L, H]
    right_gate = torch.sigmoid(F.linear(x_norm, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x_norm, out_gate_w))

    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5️⃣  Batched matrix multiplication (Tensor‑core GEMM via torch.bmm)
    # --------------------------------------------------------------
    # Rearrange to (batch, M, K) where batch = B * H
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, L, L]
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    batch = bs * hidden_dim

    left_flat  = left_perm.view(batch, L, L)             # (batch, L, K)
    right_flat = right_perm.view(batch, L, L)            # (batch, L, K)

    # Right needs to be transposed on the reduction axis (k)
    right_t = right_flat.transpose(1, 2).contiguous()    # (batch, K, N)

    # Batched GEMM: (batch, L, K) @ (batch, K, L) -> (batch, L, L)
    out_flat = torch.bmm(left_flat, right_t)            # FP16 result

    # Reshape back to (B, L, L, H)
    out_fp16 = out_flat.view(bs, hidden_dim, L, L).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 6️⃣  Output LayerNorm (FP16) – Triton
    # --------------------------------------------------------------
    out_M = bs * L * L
    out_flat2 = out_fp16.view(out_M, hidden_dim)
    to_out_norm_w = weights["to_out_norm.weight"].to(device, torch.float16).contiguous()
    to_out_norm_b = weights["to_out_norm.bias"].to(device, torch.float16).contiguous()
    out_norm_flat = triton_layernorm(out_flat2, to_out_norm_w, to_out_norm_b, eps=eps)
    out_norm = out_norm_flat.view(bs, L, L, hidden_dim)

    # --------------------------------------------------------------
    # 7️⃣  Apply output gate and final projection (FP16 → FP16)
    # --------------------------------------------------------------
    out = out_norm * out_gate  # element‑wise gate, still FP16
    to_out_w = weights["to_out.weight"].to(device, torch.float16).contiguous()
    out = F.linear(out, to_out_w)  # [B, L, L, D] FP16

    # --------------------------------------------------------------
    # 8️⃣  Cast back to FP32 for the final output
    # --------------------------------------------------------------
    return out.to(torch.float32)
