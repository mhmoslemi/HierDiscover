"""
TriMul “outgoing” forward pass – Triton‑accelerated.

Algorithm
---------
1.  Apply LayerNorm on the last dimension of the input tensor (B,N,N,D) with a
    custom 2‑pass Triton kernel (FP32 -> FP32).
2.  Cast the normalized tensor to FP16 for the compute‑bound stages.
3.  Fuse the two linear projections (left/right) into a single GEMM:
        X @ [W_left ; W_right]ᵀ   →   (B,N,N,2*H)
    and split the result.
4.  Fuse the three gate linear layers (left_gate, right_gate, out_gate) into a
    single GEMM, apply a sigmoid and split.
5.  Apply the optional mask (broadcast over the hidden dimension) and the
    element‑wise gates.
6.  Compute the core O(N³·H) contraction:
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    using `torch.einsum`, which internally reduces to a very efficient batched
    half‑precision GEMM without any extra data layout transformations.
7.  Layer‑normalize the intermediate result with a second Triton kernel
    (FP32).
8.  Apply the output gate (sigmoid) and project back to the model dimension
    with a final FP32 linear layer.

Only the LayerNorms are written in Triton; all dense linear operations are
performed with cuBLAS / Tensor‑Core GEMMs (FP16), and the N³ contraction is
expressed as a single `einsum` that avoids costly permutes and copies.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 2‑pass LayerNorm kernel (operates on the last dimension)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,                # *ptr to input  [M, N]
    y_ptr,                # *ptr to output [M, N]
    weight_ptr,           # *ptr to weight [N]
    bias_ptr,             # *ptr to bias   [N]
    M,                    # #rows   (batch*seq*seq)
    N,                    # feature dimension (dim or hidden_dim)
    eps,                  # epsilon
    stride_xm, stride_xn,  # input strides
    stride_ym, stride_yn,  # output strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
):
    pid = tl.program_id(0)

    # ------------------------------------------------------------------
    # Row range for this program
    # ------------------------------------------------------------------
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # ------------------------------------------------------------------
    # First pass: compute per‑row mean & variance (float32 accumulator)
    # ------------------------------------------------------------------
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_N)
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        sum_val += tl.sum(x, axis=1)
        sum_sq  += tl.sum(x * x, axis=1)

    mean = sum_val / N
    var  = sum_sq / N - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # ------------------------------------------------------------------
    # Second pass: write normalised + affine transformed values
    # ------------------------------------------------------------------
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        # normalise
        norm_x = (x - mean[:, None]) * rstd[:, None]

        # affine
        w = tl.load(weight_ptr + offs_n, mask=mask_n, other=0.0)
        b = tl.load(bias_ptr   + offs_n, mask=mask_n, other=0.0)

        y = norm_x * w[None, :] + b[None, :]

        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn,
            y,
            mask=mask_m[:, None] & mask_n[None, :],
        )


def triton_layernorm(x: torch.Tensor,
                     weight: torch.Tensor,
                     bias: torch.Tensor,
                     eps: float = 1e-5) -> torch.Tensor:
    """
    Triton implementation of LayerNorm over the last dimension.
    Expects a 2‑D tensor with shape (M, N) where N is the feature dimension.
    """
    assert x.dim() == 2, "triton_layernorm expects a 2‑D tensor"
    M, N = x.shape
    y = torch.empty_like(x)

    # Ensure contiguous tensors for correct stride handling
    x = x.contiguous()
    y = y.contiguous()
    weight = weight.contiguous()
    bias = bias.contiguous()

    stride_xm, stride_xn = x.stride()
    stride_ym, stride_yn = y.stride()

    grid = lambda meta: (triton.cdiv(M, meta["BLOCK_M"]),)

    layernorm_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        weight_ptr=weight,
        bias_ptr=bias,
        M=M,
        N=N,
        eps=eps,
        stride_xm=stride_xm,
        stride_xn=stride_xn,
        stride_ym=stride_ym,
        stride_yn=stride_yn,
        BLOCK_M=128,   # rows per program, good default for H100
        BLOCK_N=64,    # columns per tile, fits registers nicely
    )
    return y


# ----------------------------------------------------------------------
# TriMul “outgoing” entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, D] (float32)
        - mask         : torch.Tensor [B, N, N] or None
        - weights      : dict of torch.Tensors (all parameters)
        - config       : dict with keys ``dim``, ``hidden_dim``, ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor [B, N, N, D] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_fp32 = input_tensor.dtype       # expected torch.float32
    dtype_fp16 = torch.float16

    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (Triton) – collapse B,N,N into rows
    # ------------------------------------------------------------------
    M = bs * seq_len * seq_len
    x_flat = input_tensor.reshape(M, dim)               # (B*N*N, D)
    norm_w = weights["norm.weight"].to(device, dtype_fp32)
    norm_b = weights["norm.bias"].to(device, dtype_fp32)
    x_norm_flat = triton_layernorm(x_flat, norm_w, norm_b, eps=eps)
    x_norm = x_norm_flat.view(bs, seq_len, seq_len, dim)  # (B,N,N,D)

    # ------------------------------------------------------------------
    # Cast to FP16 for all heavy linear work
    # ------------------------------------------------------------------
    x_norm_f16 = x_norm.to(dtype_fp16)

    # ------------------------------------------------------------------
    # 2️⃣ Fuse left/right projections into a single GEMM
    # ------------------------------------------------------------------
    #   [W_left ; W_right]   → shape (2*H, D)
    left_proj_w = weights["left_proj.weight"].to(device, dtype_fp16)   # (H, D)
    right_proj_w = weights["right_proj.weight"].to(device, dtype_fp16) # (H, D)
    proj_w = torch.cat([left_proj_w, right_proj_w], dim=0)  # (2*H, D)

    proj = F.linear(x_norm_f16, proj_w)                    # (B,N,N,2*H)
    left_proj = proj[..., :hidden_dim]                     # (B,N,N,H)
    right_proj = proj[..., hidden_dim:]                    # (B,N,N,H)

    # ------------------------------------------------------------------
    # 3️⃣ Fuse the three gate linear layers (left_gate, right_gate, out_gate)
    # ------------------------------------------------------------------
    left_gate_w  = weights["left_gate.weight"].to(device, dtype_fp16)   # (H, D)
    right_gate_w = weights["right_gate.weight"].to(device, dtype_fp16)  # (H, D)
    out_gate_w   = weights["out_gate.weight"].to(device, dtype_fp16)    # (H, D)
    gate_w = torch.cat([left_gate_w, right_gate_w, out_gate_w], dim=0)  # (3*H, D)

    gates = F.linear(x_norm_f16, gate_w)                     # (B,N,N,3*H)
    left_gate_raw  = gates[..., :hidden_dim]                 # (B,N,N,H)
    right_gate_raw = gates[..., hidden_dim:2*hidden_dim]    # (B,N,N,H)
    out_gate_raw   = gates[..., 2*hidden_dim:]              # (B,N,N,H)

    left_gate  = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)
    out_gate   = torch.sigmoid(out_gate_raw)

    # ------------------------------------------------------------------
    # 4️⃣ Optional mask (broadcast over hidden_dim)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_f16 = mask.to(device, dtype_fp16).unsqueeze(-1)   # (B,N,N,1)
        left_proj = left_proj * mask_f16
        right_proj = right_proj * mask_f16

    # ------------------------------------------------------------------
    # 5️⃣ Apply element‑wise gates
    # ------------------------------------------------------------------
    left  = left_proj * left_gate   # (B,N,N,H)
    right = right_proj * right_gate # (B,N,N,H)

    # ------------------------------------------------------------------
    # 6️⃣ Core O(N³·H) contraction using a single einsum (half‑precision)
    # ------------------------------------------------------------------
    # out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    out = torch.einsum('b i k h, b j k h -> b i j h', left, right)  # (B,N,N,H)

    # ------------------------------------------------------------------
    # 7️⃣ Output LayerNorm (Triton) – operate in FP32
    # ------------------------------------------------------------------
    out_fp32 = out.to(torch.float32)
    out_flat = out_fp32.reshape(M, hidden_dim)               # (B*N*N, H)

    to_out_norm_w = weights["to_out_norm.weight"].to(device, torch.float32)
    to_out_norm_b = weights["to_out_norm.bias"].to(device, torch.float32)
    out_norm_flat = triton_layernorm(out_flat, to_out_norm_w, to_out_norm_b, eps=eps)
    out_norm = out_norm_flat.view(bs, seq_len, seq_len, hidden_dim)  # (B,N,N,H)

    # ------------------------------------------------------------------
    # 8️⃣ Gating + final projection back to model dimension
    # ------------------------------------------------------------------
    out_gate_fp32 = out_gate.to(torch.float32)               # (B,N,N,H)
    gated = out_norm * out_gate_fp32                         # (B,N,N,H)

    to_out_w = weights["to_out.weight"].to(device, torch.float32)  # (D, H)
    output = F.linear(gated, to_out_w)                      # (B,N,N,D) – FP32

    return output
```
