"""
Optimized Triton implementation of the TriMul “outgoing” operator.

Algorithm
---------
1. **Input LayerNorm** – a 2‑D Triton kernel normalises the
   (B·N·N, D) view of the input tensor.
2. **Mixed‑precision fusing** – all five linear transforms
   (left_proj, right_proj, left_gate, right_gate, out_gate) are
   realised by a single half‑precision matmul with a concatenated
   weight matrix.  This reduces memory traffic from 5 matmuls to 1.
3. **Masking & gating** – the optional mask is applied to the two
   projected tensors, then element‑wise sigmoid gates are applied.
4. **Core O(N³) contraction** – the gated tensors are reshaped to
   (B·H, N, N) and a batched half‑precision GEMM (torch.bmm) computes
   `out[b,i,j,d] = Σ_k left[b,i,k,d]·right[b,j,k,d]`.
5. **Output LayerNorm** – another Triton kernel normalises over the
   hidden dimension.
6. **Final gating & projection** – the output gate is applied,
   followed by a half‑precision linear back‑projection to the model
   dimension.  The result is returned as float32.

Only the LayerNorms are implemented with Triton kernels; all
large‑scale arithmetic uses half‑precision Tensor‑Core GEMMs,
which is optimal on H100 GPUs.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 2‑pass LayerNorm kernel (operates on the last dimension)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,                # *ptr to input  [M, N]
    y_ptr,                # *ptr to output [M, N]
    weight_ptr,           # *ptr to weight [N]
    bias_ptr,             # *ptr to bias   [N]
    M,                    # #rows   (batch*seq*seq)
    N,                    # feature dimension (dim or hidden_dim)
    eps,                  # epsilon
    stride_xm, stride_xn,  # input strides
    stride_ym, stride_yn,  # output strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
):
    pid = tl.program_id(0)

    # ------------------------------------------------------------------
    # Row range for this program
    # ------------------------------------------------------------------
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # ------------------------------------------------------------------
    # First pass: compute per‑row mean & variance
    # ------------------------------------------------------------------
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_N)
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        sum_val += tl.sum(x, axis=1)
        sum_sq  += tl.sum(x * x, axis=1)

    mean = sum_val / N
    var  = sum_sq / N - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # ------------------------------------------------------------------
    # Second pass: write normalized + affine transformed values
    # ------------------------------------------------------------------
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        # normalise
        norm_x = (x - mean[:, None]) * rstd[:, None]

        # affine (weights/bias may be FP16 or FP32)
        w = tl.load(weight_ptr + offs_n, mask=mask_n, other=0.0)
        b = tl.load(bias_ptr   + offs_n, mask=mask_n, other=0.0)

        y = norm_x * w[None, :] + b[None, :]

        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn,
            y,
            mask=mask_m[:, None] & mask_n[None, :],
        )


def triton_layernorm(x: torch.Tensor,
                     weight: torch.Tensor,
                     bias: torch.Tensor,
                     eps: float = 1e-5) -> torch.Tensor:
    """LayerNorm over the last dimension using Triton.
    `x` must be a contiguous 2‑D tensor (M, N)."""
    assert x.dim() == 2, "triton_layernorm expects a 2‑D tensor"
    M, N = x.shape
    y = torch.empty_like(x)

    # Ensure contiguous layout for correct stride handling
    x = x.contiguous()
    y = y.contiguous()
    weight = weight.contiguous()
    bias = bias.contiguous()

    stride_xm, stride_xn = x.stride()
    stride_ym, stride_yn = y.stride()

    grid = lambda meta: (triton.cdiv(M, meta["BLOCK_M"]),)

    layernorm_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        weight_ptr=weight,
        bias_ptr=bias,
        M=M,
        N=N,
        eps=eps,
        stride_xm=stride_xm,
        stride_xn=stride_xn,
        stride_ym=stride_ym,
        stride_yn=stride_yn,
        BLOCK_M=128,   # rows per program; good default for H100
        BLOCK_N=64,    # columns per tile; fits registers nicely
    )
    return y


def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, D] (float32)
        - mask         : torch.Tensor [B, N, N] or None
        - weights      : dict of torch.Tensors (model parameters)
        - config       : dict with keys ``dim``, ``hidden_dim``, ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor [B, N, N, D] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype  # expected float32

    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (Triton) over the model dimension D
    # ------------------------------------------------------------------
    M = bs * seq_len * seq_len
    x_flat = input_tensor.reshape(M, dim)                     # (M, D)
    norm_weight = weights["norm.weight"].to(device, dtype)
    norm_bias   = weights["norm.bias"].to(device, dtype)

    x_norm_flat = triton_layernorm(x_flat, norm_weight, norm_bias, eps=eps)
    x_norm = x_norm_flat.view(bs, seq_len, seq_len, dim)      # (B,N,N,D) FP32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to half precision for the heavy part
    # ------------------------------------------------------------------
    x_norm_f16 = x_norm.to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Fuse five linear maps (proj & gates) into a single matmul
    # ------------------------------------------------------------------
    # Gather weight tensors and cast to FP16
    left_proj_w   = weights["left_proj.weight"].to(device, torch.float16)   # (H, D)
    right_proj_w  = weights["right_proj.weight"].to(device, torch.float16)  # (H, D)
    left_gate_w   = weights["left_gate.weight"].to(device, torch.float16)   # (H, D)
    right_gate_w  = weights["right_gate.weight"].to(device, torch.float16)  # (H, D)
    out_gate_w    = weights["out_gate.weight"].to(device, torch.float16)    # (H, D)

    # Concatenate for a single matmul: shape (5*H, D)
    combined_weight = torch.cat(
        [left_proj_w, right_proj_w, left_gate_w, right_gate_w, out_gate_w],
        dim=0,
    )  # (5*H, D)

    # Perform the fused linear transformation (bias=False for all)
    # Input: (M, D) → Output: (M, 5*H)
    fused = F.linear(x_norm_f16.reshape(M, dim), combined_weight)  # (M, 5*H)

    # ------------------------------------------------------------------
    # 4️⃣ Slice the fused result into the individual tensors
    # ------------------------------------------------------------------
    h = hidden_dim
    left_proj   = fused[:, 0*h:1*h]                       # (M, H)
    right_proj  = fused[:, 1*h:2*h]                       # (M, H)
    left_gate   = fused[:, 2*h:3*h]                       # (M, H)
    right_gate  = fused[:, 3*h:4*h]                       # (M, H)
    out_gate    = fused[:, 4*h:5*h]                       # (M, H)

    # Sigmoids for gates (still FP16)
    left_gate   = torch.sigmoid(left_gate)
    right_gate  = torch.sigmoid(right_gate)
    out_gate    = torch.sigmoid(out_gate)

    # ------------------------------------------------------------------
    # 5️⃣ Optional mask (broadcast over hidden_dim)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # mask shape (B,N,N) → (M,1) broadcast
        mask_f16 = mask.to(device, torch.float16).reshape(M, 1)
        left_proj = left_proj * mask_f16
        right_proj = right_proj * mask_f16

    # ------------------------------------------------------------------
    # 6️⃣ Apply element‑wise gating
    # ------------------------------------------------------------------
    left  = left_proj * left_gate   # (M, H)
    right = right_proj * right_gate # (M, H)

    # ------------------------------------------------------------------
    # 7️⃣ Core N³ contraction via batched half‑precision GEMM
    # ------------------------------------------------------------------
    # Reshape to (B, N, N, H)
    left = left.view(bs, seq_len, seq_len, hidden_dim)
    right = right.view(bs, seq_len, seq_len, hidden_dim)

    # Permute to (B, H, N, N) for contiguous batches
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    BH = bs * hidden_dim
    left_mat = left_perm.view(BH, seq_len, seq_len)          # (B*H, N, N)
    right_mat = right_perm.view(BH, seq_len, seq_len)        # (B*H, N, N)

    # Batched matmul: (i,k)·(j,k)ᵀ
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*H, N, N)

    # Restore to (B, N, N, H)
    out = out_mat.view(bs, hidden_dim, seq_len, seq_len) \
                .permute(0, 2, 3, 1).contiguous()          # (B,N,N,H) FP16

    # ------------------------------------------------------------------
    # 8️⃣ Output LayerNorm (still half‑precision)
    # ------------------------------------------------------------------
    out_flat = out.view(M, hidden_dim)   # (B*N*N, H)

    to_out_norm_w = weights["to_out_norm.weight"].to(device, torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device, torch.float16)

    out_norm_flat = triton_layernorm(out_flat,
                                     to_out_norm_w,
                                     to_out_norm_b,
                                     eps=eps)          # (M, H) half
    out_norm = out_norm_flat.view(bs, seq_len, seq_len, hidden_dim)  # (B,N,N,H)

    # ------------------------------------------------------------------
    # 9️⃣ Apply output gate
    # ------------------------------------------------------------------
    out_gate = out_gate.view(bs, seq_len, seq_len, hidden_dim)  # (B,N,N,H)
    gated = out_norm * out_gate  # element‑wise, still FP16

    # ------------------------------------------------------------------
    # 🔟 Final projection back to model dimension D
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(device, torch.float16)  # (D, H)

    output_fp16 = F.linear(gated, to_out_w)  # (B,N,N,D) half
    output = output_fp16.to(torch.float32)   # cast back to FP32 for API compatibility

    return output
