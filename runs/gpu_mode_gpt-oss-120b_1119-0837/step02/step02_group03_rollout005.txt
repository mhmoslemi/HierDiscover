```
def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    eps=1e-5

    # Input LayerNorm
    x_norm = triton_layernorm(input_tensor.reshape(-1, dim), 
                               weights["norm.weight"].to(dtype), 
                               weights["norm.bias"].to(dtype),
                               eps).reshape(bs, N, N, dim)

    # Cast to fp16
    x_norm_h = x_norm.to(torch.float16)

    # Project left and right
    left = torch.nn.functional.linear(x_norm_h, weights["left_proj.weight"].to(torch.float16))
    right = torch.nn.functional.linear(x_norm_h, weights["right_proj.weight"].to(torch.float16))

    # Mask (if any)
    if not nomask and mask is not None:
        mask_h = mask.unsqueeze(-1).to(torch.float16)
        left = left * mask_h
        right = right * mask_h

    # Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm_h, 
                      weights["left_gate.weight"].to(torch.float16)))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm_h,
                      weights["right_gate.weight"].to(torch.float16)))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm_h,
                      weights["out_gate.weight"].to(torch.float16)))

    left = left * left_gate
    right = right * right_gate

    # Reshape for batched GEMM
    # shape: [bs, N, N, hidden_dim] => [bs*hidden_dim, N, N]
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(bs*hidden_dim, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(bs*hidden_dim, N, N)
    # Compute GEMM: left @ right^T
    out_flat = torch.bmm(left_flat, right_flat.transpose(1,2))
    # reshape back
    out = out_flat.view(bs, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,H] fp16
    # Output layernorm
    out_norm = triton_layernorm(out.reshape(-1, hidden_dim),
                                weights["to_out_norm.weight"].to(dtype),
                                weights["to_out_norm.bias"].to(dtype),
                                eps).reshape(bs, N, N, hidden_dim)

    # Apply out gate
    out = out_norm * out_gate.to(dtype) # cast out_gate to fp32

    # Final linear projection
    out = torch.nn.functional.linear(out, weights["to_out.weight"].to(dtype))

    return out

@triton.jit
def layernorm_kernel(x_ptr, y_ptr, weight_ptr, bias_ptr, M, N, eps,
                     stride_xm, stride_xn,
                     stride_ym, stride_yn,
                     BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr):
    pid = tl.program_id(0)
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq = tl.zeros((BLOCK_M,), dtype=tl.float32)

    # number of column blocks
    num_n_blocks = tl.cdiv(N, BLOCK_N)
    for n in range(num_n_blocks):
        offs_n = n * BLOCK_N + tl.arange(0, BLOCK_N)
        mask_n = offs_n < N

        x = tl.load(x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
                    mask=mask_m[:, None] & mask_n[None, :],
                    other=0.0)
        x_f32 = x.to(tl.float32)
        sum_val += tl.sum(x_f32, axis=1)
        sum_sq += tl.sum(x_f32 * x_f32, axis=1)

    mean = sum_val / N
    var = sum_sq / N - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # second pass: normalize and apply affine
    for n in range(num_n_blocks):
        offs_n = n * BLOCK_N + tl.arange(0, BLOCK_N)
        mask_n = offs_n < N

        x = tl.load(x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
                    mask=mask_m[:, None] & mask_n[None, :],
                    other=0.0)
        x_f32 = x.to(tl.float32)
        norm_x = (x_f32 - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_n, mask=mask_n, other=0.0)
        b = tl.load(bias_ptr + offs_n, mask=mask_n, other=0.0)

        # cast weight, bias to float32 for multiplication
        w_f32 = w.to(tl.float32)
        b_f32 = b.to(tl.float32)

        y = norm_x * w_f32 + b_f32
        # cast back to original dtype
        y_out = y.to(x.dtype)  # maybe x.dtype is same as y_ptr dtype

        tl.store(y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn,
                 y_out,
                 mask=mask_m[:, None] & mask_n[None, :])
```

"""
Optimized Triton+PyTorch implementation of the *outgoing* TriMul operator
(AlphaFold‑3).  The algorithm follows the reference PyTorch version but:

*   Input and output LayerNorms are performed by a custom two‑pass
    Triton kernel (row‑wise, FP32 accumulation → FP16/FP32 output).
*   The expensive N³ contraction
        out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
    is rewritten as a batched matrix multiplication
        left_flat @ right_flatᵀ
    and executed with `torch.bmm`, which on H100 uses Tensor‑cores
    (FP16, FP32 accumulation) and is far faster than a naïve Triton
    dot‑loop.
*   All linear projections and gates are computed in FP16; the final
    projection and the output are cast back to FP32.  This mixed‑precision
    pipeline preserves the model semantics while achieving sub‑millisecond
    runtimes for the test configurations.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton LayerNorm (row‑wise) – works for any 2‑D tensor [M, N].
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,                # *ptr to input  [M, N]
    y_ptr,                # *ptr to output [M, N]
    w_ptr,                # *ptr to weight [N]
    b_ptr,                # *ptr to bias   [N]
    M,                    # number of rows
    N,                    # feature dimension
    eps,                  # epsilon
    stride_xm, stride_xn,  # strides for input
    stride_ym, stride_yn,  # strides for output
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
):
    pid = tl.program_id(0)

    # ------------------------------------------------------------------
    # Row offsets for this program
    # ------------------------------------------------------------------
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # ------------------------------------------------------------------
    # First pass: compute mean & variance
    # ------------------------------------------------------------------
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_n_blocks = tl.cdiv(N, BLOCK_N)
    for n in range(num_n_blocks):
        offs_n = n * BLOCK_N + tl.arange(0, BLOCK_N)
        mask_n = offs_n < N

        # Load a tile of X (shape: BLOCK_M x BLOCK_N)
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        # Cast to FP32 for the reduction
        x_f32 = x.to(tl.float32)
        sum_val += tl.sum(x_f32, axis=1)
        sum_sq  += tl.sum(x_f32 * x_f32, axis=1)

    mean = sum_val / N
    var  = sum_sq / N - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # ------------------------------------------------------------------
    # Second pass: write normalized + affine‑transformed values
    # ------------------------------------------------------------------
    for n in range(num_n_blocks):
        offs_n = n * BLOCK_N + tl.arange(0, BLOCK_N)
        mask_n = offs_n < N

        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        x_f32 = x.to(tl.float32)
        norm_x = (x_f32 - mean[:, None]) * rstd[:, None]

        w = tl.load(w_ptr + offs_n, mask=mask_n, other=0.0)
        b = tl.load(b_ptr + offs_n, mask=mask_n, other=0.0)
        w_f32 = w.to(tl.float32)
        b_f32 = b.to(tl.float32)

        y_f32 = norm_x * w_f32 + b_f32
        # Cast back to the original dtype of the input (FP16 or FP32)
        y = y_f32.to(x.dtype)

        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn,
            y,
            mask=mask_m[:, None] & mask_n[None, :],
        )


def triton_layernorm(x: torch.Tensor,
                     weight: torch.Tensor,
                     bias: torch.Tensor,
                     eps: float = 1e-5) -> torch.Tensor:
    """
    Row‑wise LayerNorm implemented in Triton.
    ``x`` must be 2‑D (M, N); the function returns a tensor of the same shape
    and dtype as ``x``.
    """
    assert x.dim() == 2, "triton_layernorm expects a 2‑D tensor"
    M, N = x.shape
    y = torch.empty_like(x)

    # Ensure contiguity (required for correct stride extraction)
    x = x.contiguous()
    y = y.contiguous()
    weight = weight.contiguous()
    bias = bias.contiguous()

    stride_xm, stride_xn = x.stride()
    stride_ym, stride_yn = y.stride()

    grid = lambda meta: (triton.cdiv(M, meta["BLOCK_M"]),)

    layernorm_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        w_ptr=weight,
        b_ptr=bias,
        M=M,
        N=N,
        eps=eps,
        stride_xm=stride_xm,
        stride_xn=stride_xn,
        stride_ym=stride_ym,
        stride_yn=stride_yn,
        BLOCK_M=128,   # rows per program (tuned)
        BLOCK_N=64,    # cols per tile (covers hidden_dim up to 384)
    )
    return y


# ----------------------------------------------------------------------
# Entry point required by the benchmark harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor   [B, L, L, D]   (float32)
        - mask         : torch.Tensor   [B, L, L]   (float32) or ``None``
        - weights      : dict of torch tensors with keys:
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
        - config : dict with fields ``dim`` (D), ``hidden_dim`` (H) and optional
                   ``nomask`` (bool).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, L, L, D] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    bs, L, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (Triton) – operates on [bs*L*L, dim]
    # ------------------------------------------------------------------
    x_flat = input_tensor.reshape(bs * L * L, dim).contiguous()
    norm_w = weights["norm.weight"].to(device, dtype).contiguous()
    norm_b = weights["norm.bias"].to(device, dtype).contiguous()
    x_norm_flat = triton_layernorm(x_flat, norm_w, norm_b, eps=eps)
    x_norm = x_norm_flat.view(bs, L, L, dim)

    # ------------------------------------------------------------------
    # 2️⃣ Cast to FP16 for the heavy compute
    # ------------------------------------------------------------------
    x_h = x_norm.to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Linear projections (bias‑free, FP16)
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(device, torch.float16).contiguous()
    right_proj_w = weights["right_proj.weight"].to(device, torch.float16).contiguous()
    left = F.linear(x_h, left_proj_w)      # [B, L, L, H]
    right = F.linear(x_h, right_proj_w)

    # ------------------------------------------------------------------
    # 4️⃣ Optional mask (broadcast over the hidden dimension)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_h = mask.to(device, torch.float16).unsqueeze(-1)   # [B, L, L, 1]
        left = left * mask_h
        right = right * mask_h

    # ------------------------------------------------------------------
    # 5️⃣ Gates (sigmoid, FP16)
    # ------------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(device, torch.float16).contiguous()
    right_gate_w = weights["right_gate.weight"].to(device, torch.float16).contiguous()
    out_gate_w = weights["out_gate.weight"].to(device, torch.float16).contiguous()

    left_gate = torch.sigmoid(F.linear(x_h, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_h, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x_h, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6️⃣ Batched GEMM via torch.bmm (FP16) – computes the N³ contraction
    # ------------------------------------------------------------------
    # Reshape to (B*H, L, L), where each batch element is an independent matrix.
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(bs * hidden_dim, L, L)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(bs * hidden_dim, L, L)

    # out_flat[i] = left_flat[i] @ right_flat[i].T
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # Back to the original layout [B, L, L, H] (still FP16)
    out = out_flat.view(bs, hidden_dim, L, L).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7️⃣ Output LayerNorm (Triton) over the hidden dimension
    # ------------------------------------------------------------------
    out_flat2 = out.reshape(bs * L * L, hidden_dim).contiguous()
    out_norm_w = weights["to_out_norm.weight"].to(device, dtype).contiguous()
    out_norm_b = weights["to_out_norm.bias"].to(device, dtype).contiguous()
    out_norm_flat = triton_layernorm(out_flat2, out_norm_w, out_norm_b, eps=eps)
    out_norm = out_norm_flat.view(bs, L, L, hidden_dim)

    # ------------------------------------------------------------------
    # 8️⃣ Apply output gate and final linear projection back to dim
    # ------------------------------------------------------------------
    out = out_norm * out_gate.to(dtype)               # broadcast over H
    to_out_w = weights["to_out.weight"].to(device, dtype).contiguous()
    out = F.linear(out, to_out_w)                     # [B, L, L, D] (float32)

    return out
