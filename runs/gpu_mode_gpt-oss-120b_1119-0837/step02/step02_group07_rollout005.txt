"""
TriMul (outgoing) – fast forward pass.

The core of the operation is a batched GEMM over the “hidden” dimension:
    out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h] .
For each (batch * hidden) slice this is a matrix multiplication
    (N × N) · (N × N)ᵀ  (over the shared axis k).
We implement this GEMM with a custom Triton kernel using fp16 inputs,
fp32 accumulation and Tensor‑Core friendly tiling.
All other layers (LayerNorms, linear projections, sigmoids and
the final projection) stay in PyTorch, but we fuse the four linear
projections (proj + gate) into two matmul calls to halve the number of
passes over the large input tensor.

The kernel tiles the output with a 2‑D tile (BLOCK_M × BLOCK_N) and a
1‑D reduction tile (BLOCK_K).  Tile sizes are chosen adaptively to keep
register usage low while maximizing occupancy on an H100.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    out_ptr,            # float32*   output (C, M, N)
    a_ptr,              # float16*   left  (C, M, K)
    b_ptr,              # float16*   right (C, K, N) – stored as (C, K, N) but accessed as transposed
    M, N, K,            # matrix dimensions (all equal to seq_len)
    stride_out_c, stride_out_m, stride_out_n,   # output strides
    stride_a_c, stride_a_m, stride_a_k,        # left strides
    stride_b_c, stride_b_k, stride_b_n,        # right strides (K‑major)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched GEMM for a single hidden channel:
        out_c[M,N] += left_c[M,K] @ right_c[K,N]
    """
    pid_c = tl.program_id(2)          # channel index (batch * hidden)
    pid_m = tl.program_id(0)          # tile row
    pid_n = tl.program_id(1)          # tile col

    # compute offsets inside the full matrix
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -------------------------------------------------------------
    # Loop over the reduction dimension K
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # pointers to the current tiles
        a_tile = a_ptr + pid_c * stride_a_c \
                 + offs_m[:, None] * stride_a_m \
                 + offs_k[None, :] * stride_a_k          # (M, K)

        b_tile = b_ptr + pid_c * stride_b_c \
                 + offs_k[:, None] * stride_b_k \
                 + offs_n[None, :] * stride_b_n          # (K, N)

        # load tiles, zero‑pad out‑of‑bounds entries
        a = tl.load(a_tile,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)                               # fp16
        b = tl.load(b_tile,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)                               # fp16

        # GEMM accumulate (fp16 * fp16 -> fp32)
        acc += tl.dot(a, b)

    # -------------------------------------------------------------
    # write the result tile
    out_tile = out_ptr + pid_c * stride_out_c \
               + offs_m[:, None] * stride_out_m \
               + offs_n[None, :] * stride_out_n
    tl.store(out_tile, acc,
             mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module (AlphaFold3 style).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor[B, N, N, C] (float32)
            - mask         : torch.Tensor[B, N, N] (float32/bool) or None
            - weights      : dict of model parameters
            - config       : dict with keys
                * "dim"        – input channel dimension C
                * "hidden_dim"– hidden dimension H
                * "nomask"     – bool, True → ignore mask

    Returns:
        torch.Tensor[B, N, N, C] (float32) – module output
    """
    # -------------------------------------------------------------
    # unpack
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    device = input_tensor.device

    # -------------------------------------------------------------
    # 1) Input LayerNorm (float32)
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, C]  (FP32)

    # -------------------------------------------------------------
    # 2) Cast to fp16 for all linear ops
    x_h = x.to(torch.float16)

    # -------------------------------------------------------------
    # 3) Fuse projection + gate for the left side:
    #    Concatenate the proj and gate weights → (2*H, C)
    left_concat_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16)  # shape (2*H, C)

    left_concat = F.linear(x_h, left_concat_w)  # [B, N, N, 2*H] fp16
    # split
    left_proj, left_gate = torch.split(left_concat, hidden_dim, dim=-1)

    # 4) Fuse projection + gate for the right side:
    right_concat_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16)  # shape (2*H, C)

    right_concat = F.linear(x_h, right_concat_w)  # [B, N, N, 2*H] fp16
    right_proj, right_gate = torch.split(right_concat, hidden_dim, dim=-1)

    # -------------------------------------------------------------
    # 5) Apply sigmoid to the gate tensors
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)

    # -------------------------------------------------------------
    # 6) Combine projection, gate and (optional) mask in one step.
    #    This avoids an extra write‑back of the masked/gated tensors.
    left = left_proj * left_gate
    right = right_proj * right_gate

    if not nomask and mask is not None:
        mask_h = mask.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_h
        right = right * mask_h

    # -------------------------------------------------------------
    # 7) Prepare tensors for the Triton kernel.
    #    Move hidden dimension before the two sequence dims:
    #    (B, N, N, H) → (B, H, N, N) → flatten B*H as leading dim.
    B, N, _, _ = left.shape
    left = left.permute(0, 3, 1, 2).contiguous()    # [B, H, N, N]
    right = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    C_total = B * hidden_dim                         # = batch * hidden
    left_flat = left.reshape(C_total, N, N)          # (C, M, K)  K=N
    right_flat = right.reshape(C_total, N, N)        # (C, K, N)

    # output buffer (FP32)
    out_flat = torch.empty_like(left_flat, dtype=torch.float32, device=device)

    # -------------------------------------------------------------
    # 8) Choose tile sizes adaptively.
    if N <= 256:
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 32
        num_warps = 4
    elif N <= 512:
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 64
        num_warps = 8
    else:
        # for the largest sequences we use a larger M/N tile to increase work per block
        BLOCK_M = BLOCK_N = 256
        BLOCK_K = 64
        num_warps = 8

    # strides for the kernel (convert to int64 for Triton)
    stride_out_c, stride_out_m, stride_out_n = (
        out_flat.stride(0),
        out_flat.stride(1),
        out_flat.stride(2),
    )
    stride_a_c, stride_a_m, stride_a_k = (
        left_flat.stride(0),
        left_flat.stride(1),
        left_flat.stride(2),
    )
    stride_b_c, stride_b_k, stride_b_n = (
        right_flat.stride(0),
        right_flat.stride(2),   # K‑major stride
        right_flat.stride(1),   # N‑major stride
    )

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # grid dim 0 (M)
        (N + BLOCK_N - 1) // BLOCK_N,   # grid dim 1 (N)
        C_total,                        # grid dim 2 (C)
    )

    tri_mul_kernel[grid](
        out_flat,
        left_flat,
        right_flat,
        N, N, N,
        stride_out_c, stride_out_m, stride_out_n,
        stride_a_c, stride_a_m, stride_a_k,
        stride_b_c, stride_b_k, stride_b_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=num_warps,
    )

    # -------------------------------------------------------------
    # 9) Reshape back to (B, N, N, H)
    out = out_flat.reshape(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # -------------------------------------------------------------
    # 10) Post‑processing: LayerNorm over hidden dimension → out‑gate → final projection
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    # out_gate: compute once (fp16) and cast back to fp32 for the final scaling
    out_gate = torch.sigmoid(F.linear(x_h, weights["out_gate.weight"].to(torch.float16)))
    out = out * out_gate.to(torch.float32)

    # final linear projection back to the original dimension
    out = F.linear(out, weights["to_out.weight"])

    return out
