    x_ptr, proj_w_ptr, gate_w_ptr, out_ptr,
    M, dim, hidden,
    stride_xm, stride_xn,
    stride_pw_h, stride_pw_d,
    stride_gw_h, stride_gw_d,

    BLOCK_H: tl.constexpr,
    BLOCK_K: tl.constexpr,
):

   pid_h = tl.program_id(1)
   ... (row block, hidden block)

@triton.jit
def linear_gated_kernel(
    x_ptr, proj_w_ptr, gate_w_ptr, out_ptr,
    M, dim, hidden,
    stride_xm, stride_xn,
    stride_pw_h, stride_pw_d,
    stride_gw_h, stride_gw_d,
    stride_om, stride_on,
    BLOCK_M: tl.constexpr,
    BLOCK_H: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)
    pid_h = tl.program_id(1)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_h = pid_h * BLOCK_H + tl.arange(0, BLOCK_H)

    mask_m = offs_m < M
    mask_h = offs_h < hidden

    acc_proj = tl.zeros((BLOCK_M, BLOCK_H), dtype=tl.float32)
    acc_gate = tl.zeros((BLOCK_M, BLOCK_H), dtype=tl.float32)

    # Loop over dim blocks
    for k in range(0, tl.cdiv(dim, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < dim

        # load x tile
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_k[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )  # shape (BLOCK_M, BLOCK_K)

        # load projection weight tile (transposed)
        proj_w = tl.load(
            proj_w_ptr + offs_k[:, None] * stride_pw_d + offs_h[None, :] * stride_pw_h,
            mask=mask_k[:, None] & mask_h[None, :],
            other=0.0,
        )  # shape (BLOCK_K, BLOCK_H)

        # load gate weight tile (transposed)
        gate_w = tl.load(
            gate_w_ptr + offs_k[:, None] * stride_gw_d + offs_h[None, :] * stride_gw_h,
            mask=mask_k[:, None] & mask_h[None, :],
            other=0.0,
        )  # shape (BLOCK_K, BLOCK_H)

        # dot product
        acc_proj += tl.dot(x, proj_w)   # (BLOCK_M, BLOCK_H)
        acc_gate += tl.dot(x, gate_w)

    # Apply sigmoid to gate
    gate_sig = 1.0 / (1.0 + tl.exp(-acc_gate))

    # Multiply
    out = (acc_proj * gate_sig).to(tl.float16)

    # store
    tl.store(
        out_ptr + offs_m[:, None] * stride_om + offs_h[None, :] * stride_on,
        out,
        mask=mask_m[:, None] & mask_h[None, :]
    )

"""
custom_kernel
==============

Fast “outgoing” TriMul (AlphaFold‑3) implementation.

Key ideas
---------

*   Input and output LayerNorms are performed by a two‑pass Triton kernel
    (row‑wise mean/variance → affine transform).  This replaces the
    PyTorch implementation and guarantees we execute at least one custom
    kernel.

*   The four linear layers that produce ``left``/``right`` *and* their
    gates are fused into two custom Triton kernels
    ``linear_gated_kernel``.  A single read of the normalized input
    ``x`` produces a gated tensor in half‑precision, dramatically
    reducing memory traffic (the naïve version reads ``x`` five times).

*   The expensive tensor contraction  

        out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]

    is exactly a batched matrix multiplication  
    ``(B·H, L, L)  @  (B·H, L, L)ᵀ`` and is delegated to PyTorch’s
    ``torch.bmm`` which maps to the highly‑optimised cuBLAS‑Lt kernel on
    H100.  This is considerably faster than the hand‑written Triton GEMM
    used previously.

*   All heavy arithmetic is carried out in float16 (FP16) so the
    Tensor‑cores are fully utilised; accumulators are FP32 for stability.
    The final output, LayerNorms and the output gate are kept in FP32
    to match the reference implementation.

*   Optional masking is handled by a cheap element‑wise broadcast
    multiplication after the fused linear+gate step.

The function returns a float32 tensor of shape ``[B, L, L, D]`` and
conforms to the signature required by the test harness.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton row‑wise LayerNorm (2‑D tensor [M, N])
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr, y_ptr, weight_ptr, bias_ptr,
    M, N, eps,
    stride_xm, stride_xn,
    stride_ym, stride_yn,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
):
    pid = tl.program_id(0)

    # ------------------------------------------------------------------
    # Row offsets for this program
    # ------------------------------------------------------------------
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # ------------------------------------------------------------------
    # First pass – compute mean and variance
    # ------------------------------------------------------------------
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_N)
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        sum_val += tl.sum(x,  axis=1)
        sum_sq  += tl.sum(x * x, axis=1)

    mean = sum_val / N
    var  = sum_sq / N - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # ------------------------------------------------------------------
    # Second pass – normalise + affine transform
    # ------------------------------------------------------------------
    for k in range(num_k):
        offs_n = tl.arange(0, BLOCK_N) + k * BLOCK_N
        mask_n = offs_n < N
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_n[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        norm_x = (x - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_n, mask=mask_n, other=0.0)
        b = tl.load(bias_ptr   + offs_n, mask=mask_n, other=0.0)

        y = norm_x * w[None, :] + b[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn,
            y,
            mask=mask_m[:, None] & mask_n[None, :],
        )


def triton_layernorm(x: torch.Tensor, w: torch.Tensor, b: torch.Tensor, eps: float = 1e-5):
    """Apply LayerNorm on the last dimension of a 2‑D tensor using Triton."""
    assert x.dim() == 2, "triton_layernorm expects a 2‑D tensor"
    M, N = x.shape
    y = torch.empty_like(x)

    # ensure contiguity for stride correctness
    x = x.contiguous()
    y = y.contiguous()
    w = w.contiguous()
    b = b.contiguous()

    stride_xm, stride_xn = x.stride()
    stride_ym, stride_yn = y.stride()

    grid = lambda meta: (triton.cdiv(M, meta["BLOCK_M"]),)

    layernorm_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        weight_ptr=w,
        bias_ptr=b,
        M=M,
        N=N,
        eps=eps,
        stride_xm=stride_xm,
        stride_xn=stride_xn,
        stride_ym=stride_ym,
        stride_yn=stride_yn,
        BLOCK_M=128,   # tuned for our shapes
        BLOCK_N=64,
    )
    return y


# ----------------------------------------------------------------------
# Fused Linear + Sigmoid‑Gate kernel
#   out = (X @ W_proj.T) * sigmoid(X @ W_gate.T)
#   X : [M, D]            (float16)
#   W_proj, W_gate : [H, D] (float16)
#   out : [M, H]          (float16)
# ----------------------------------------------------------------------
@triton.jit
def linear_gated_kernel(
    x_ptr,               # *ptr to input matrix   [M, D]
    proj_w_ptr,          # *ptr to proj weight   [H, D]
    gate_w_ptr,          # *ptr to gate weight   [H, D]
    out_ptr,             # *ptr to output matrix [M, H]
    M, D, H,
    stride_xm, stride_xn,
    stride_pw_h, stride_pw_d,
    stride_gw_h, stride_gw_d,
    stride_om, stride_on,
    BLOCK_M: tl.constexpr,
    BLOCK_H: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)
    pid_h = tl.program_id(1)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_h = pid_h * BLOCK_H + tl.arange(0, BLOCK_H)

    mask_m = offs_m < M
    mask_h = offs_h < H

    acc_proj = tl.zeros((BLOCK_M, BLOCK_H), dtype=tl.float32)
    acc_gate = tl.zeros((BLOCK_M, BLOCK_H), dtype=tl.float32)

    # iterate over the reduction dimension D in blocks
    for k in range(0, tl.cdiv(D, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < D

        # load a tile of X   (BLOCK_M × BLOCK_K)
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_k[None, :] * stride_xn,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # load a tile of proj weight transposed (BLOCK_K × BLOCK_H)
        proj_w = tl.load(
            proj_w_ptr
            + offs_k[:, None] * stride_pw_d
            + offs_h[None, :] * stride_pw_h,
            mask=mask_k[:, None] & mask_h[None, :],
            other=0.0,
        )
        # load a tile of gate weight transposed (BLOCK_K × BLOCK_H)
        gate_w = tl.load(
            gate_w_ptr
            + offs_k[:, None] * stride_gw_d
            + offs_h[None, :] * stride_gw_h,
            mask=mask_k[:, None] & mask_h[None, :],
            other=0.0,
        )

        acc_proj += tl.dot(x, proj_w)   # (BLOCK_M, BLOCK_H)
        acc_gate += tl.dot(x, gate_w)

    # sigmoid on the gate (still in FP32)
    gate_sig = 1.0 / (1.0 + tl.exp(-acc_gate))

    # final gated output, cast back to FP16 for the rest of the pipeline
    out = (acc_proj * gate_sig).to(tl.float16)

    tl.store(
        out_ptr + offs_m[:, None] * stride_om + offs_h[None, :] * stride_on,
        out,
        mask=mask_m[:, None] & mask_h[None, :],
    )


# ----------------------------------------------------------------------
# Triton‑backed LayerNorm wrapper
# ----------------------------------------------------------------------
def triton_ln(x: torch.Tensor, weight: torch.Tensor, bias: torch.Tensor, eps: float = 1e-5):
    """LayerNorm over the last dimension of a 2‑D tensor using Triton."""
    return triton_layernorm(x, weight, bias, eps)

    # ------------------------------------------------------------------
    out_flat = out.reshape(bs * L * L, hidden)                  # [M, H] (float16)
    out_flat_fp32 = out_flat.to(torch.float32)                  # convert before LN
    to_out_norm_w = weights["to_out_norm.weight"].to(device, dtype).contiguous()
    to_out_norm_b = weights["to_out_norm.bias"  ].to(device, dtype).contiguous()
    out_norm_flat = triton_ln(out_flat_fp32, to_out_norm_w, to_out_norm_b, eps=eps)
    out_norm = out_norm_flat.view(bs, L, L, hidden)            # [B, L, L, H] (float32)

    # ------------------------------------------------------------------
    # 9️⃣  Output gate (computed in float32 from the *original* normed input)
    # ------------------------------------------------------------------
    out_gate_w = weights["out_gate.weight"].to(device, dtype).contiguous()
    # reuse the earlier flattened normalized input (float32)
    out_gate = torch.sigmoid(F.linear(x_norm_flat, out_gate_w))   # [M, H] FP32
    out_gate = out_gate.view(bs, L, L, hidden)                    # [B, L, L, H]

    # ------------------------------------------------------------------
    # 10️⃣ Apply output gate and final linear projection back to D
    # ------------------------------------------------------------------
    gated = out_norm * out_gate                                   # [B, L, L, H] FP32
    to_out_w = weights["to_out.weight"].to(device, dtype).contiguous()  # [D, H]
    out_final = F.linear(gated.view(-1, hidden), to_out_w)        # [M, D] FP32
    out_final = out_final.view(bs, L, L, dim)                     # [B, L, L, D]

    return out_final
```
