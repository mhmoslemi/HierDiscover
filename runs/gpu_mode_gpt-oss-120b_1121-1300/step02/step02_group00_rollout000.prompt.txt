import torch
from torch import nn, einsum
import math

# Reference code in PyTorch
class TriMul(nn.Module):
    def __init__(
        self,
        dim: int,
        hidden_dim: int,
    ):
        super().__init__()

        self.norm = nn.LayerNorm(dim)

        self.left_proj = nn.Linear(dim, hidden_dim, bias=False)
        self.right_proj = nn.Linear(dim, hidden_dim, bias=False)

        self.left_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.right_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.out_gate = nn.Linear(dim, hidden_dim, bias=False)

        self.to_out_norm = nn.LayerNorm(hidden_dim)
        self.to_out = nn.Linear(hidden_dim, dim, bias=False)

    def forward(self, x: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
        """
        x: [bs, seq_len, seq_len, dim]
        mask: [bs, seq_len, seq_len]

        Returns:
            output: [bs, seq_len, seq_len, dim]
        """
        batch_size, seq_len, _, dim = x.shape

        x = self.norm(x)

        left = self.left_proj(x)
        right = self.right_proj(x)

        mask = mask.unsqueeze(-1)
        left = left * mask
        right = right * mask

        left_gate = self.left_gate(x).sigmoid()
        right_gate = self.right_gate(x).sigmoid()
        out_gate = self.out_gate(x).sigmoid()

        left = left * left_gate
        right = right * right_gate

        out = einsum('... i k d, ... j k d -> ... i j d', left, right)
        # This einsum is the same as the following:
        # out = torch.zeros(batch_size, seq_len, seq_len, dim, device=x.device)
        
        # # Compute using nested loops
        # for b in range(batch_size):
        #     for i in range(seq_len):
        #         for j in range(seq_len):
        #             # Compute each output element
        #             for k in range(seq_len):
        #                 out[b, i, j] += left[b, i, k, :] * right[b, j, k, :]

        out = self.to_out_norm(out)
        out = out * out_gate
        return self.to_out(out)

"""
TriMul (outgoing) – high‑performance Triton implementation
-----------------------------------------------------------------
Algorithm
---------
1. Layer‑norm the input tensor over the feature dimension `dim`.
2. Compute five linear projections from the normalized tensor:
      left_proj, right_proj, left_gate, right_gate, out_gate
   (the three gates are passed through a sigmoid).
3. Optionally mask `left_proj` and `right_proj`.
4. Apply the left/right gates → `left = left_proj * left_gate`,
   `right = right_proj * right_gate`.
5. Heavy multiplicative update:
      out_h = left_h @ right_hᵀ   for each hidden channel `h`.
   The tensors are reshaped to (batch*hidden, N, N) and a
   custom Triton kernel performs a batched GEMM in FP16
   (Tensor‑Core) accumulating in FP32.
6. Layer‑norm over the hidden dimension, multiply by `out_gate`,
   and finally project back to the original feature dimension.
The function returns a torch.Tensor of shape [B, N, N, dim] (float32).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched matmul C = A @ B
#   A : (batch, M, K)   – row‑major
#   B : (batch, K, N)   – row‑major (already transposed on the PyTorch side)
#   C : (batch, M, N)
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # Grid: (num_blocks_M, num_blocks_N, batch_heads)
    pid_batch = tl.program_id(2)          # which batch‑head
    pid_m = tl.program_id(0)              # block row
    pid_n = tl.program_id(1)              # block column

    # Tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Base pointers (including the batch‑head offset)
    a_ptrs = a_ptr + pid_batch * stride_a_batch \
                    + offs_m[:, None] * stride_a_m \
                    + offs_k[None, :] * stride_a_k
    b_ptrs = b_ptr + pid_batch * stride_b_batch \
                    + offs_k[:, None] * stride_b_k \
                    + offs_n[None, :] * stride_b_n

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in tiles of size BLOCK_K
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs,
                    mask=(offs_m[:, None] < M) & (offs_k[None, :] < K),
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=(offs_k[:, None] < K) & (offs_n[None, :] < N),
                    other=0.0)
        # fp16·fp16 -> fp32 accumulation using Tensor‑Cores
        acc += tl.dot(a, b)

        # Advance pointers to the next K‑tile
        a_ptrs += BLOCK_K * stride_a_k
        b_ptrs += BLOCK_K * stride_b_k

    # Write result (store as fp16, later cast back to fp32)
    c = acc.to(tl.float16)
    c_ptrs = c_ptr + pid_batch * stride_c_batch \
                     + offs_m[:, None] * stride_c_m \
                     + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs,
             c,
             mask=(offs_m[:, None] < M) & (offs_n[None, :] < N))


# ----------------------------------------------------------------------
# Entry point used by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, dim]
        - mask_tensor  : torch.Tensor of shape [B, N, N] (or None)
        - weights_dict : mapping from string names to torch.Tensor
        - config_dict  : contains "dim", "hidden_dim", "nomask", ...

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    device = input_tensor.device
    dtype = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1) Input LayerNorm over `dim`
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 2) Linear projections (cast to fp16 for the heavy GEMM)
    # ------------------------------------------------------------------
    # keep everything in fp16 from here on to maximise Tensor‑Core utilisation
    x_fp16 = x.to(torch.float16)

    left = F.linear(x_fp16, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x_fp16, weights["right_proj.weight"].to(torch.float16))

    # ------------------------------------------------------------------
    # 3) Optional mask (broadcasted on the hidden dimension)
    # ------------------------------------------------------------------
    if not config.get("nomask", False) and mask is not None:
        mask_ = mask.to(x_fp16.dtype).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4) Gates (sigmoid) – also computed in fp16
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(
        F.linear(x_fp16, weights["left_gate.weight"].to(torch.float16))
    )
    right_gate = torch.sigmoid(
        F.linear(x_fp16, weights["right_gate.weight"].to(torch.float16))
    )
    out_gate = torch.sigmoid(
        F.linear(x_fp16, weights["out_gate.weight"].to(torch.float16))
    )

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Heavy multiplicative update  (batched GEMM)
    # ------------------------------------------------------------------
    # reshape to (batch * hidden, N, N) and transpose the right operand
    left_perm = left.permute(0, 3, 1, 2).contiguous()      # [B, H, N, N]
    right_perm = right.permute(0, 3, 2, 1).contiguous()    # [B, H, N, N] (k ↔ j)

    B, H, N, _ = left_perm.shape
    batch_heads = B * H

    A = left_perm.view(batch_heads, N, N)          # (BH, M=N, K=N)
    Bmat = right_perm.view(batch_heads, N, N)      # (BH, K=N, N)

    # Cast to FP16 for the kernel (the kernel itself works in FP16)
    A_fp16 = A
    B_fp16 = Bmat
    C_fp16 = torch.empty_like(A_fp16)  # (BH, N, N)

    # Tuned tile sizes for H100 Tensor‑Cores
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # blocks along M dimension
        (N + BLOCK_N - 1) // BLOCK_N,   # blocks along N dimension
        batch_heads                     # one block per (batch, head)
    )

    batched_matmul_kernel[grid](
        A_fp16, B_fp16, C_fp16,
        A_fp16.stride(0), A_fp16.stride(1), A_fp16.stride(2),
        B_fp16.stride(0), B_fp16.stride(1), B_fp16.stride(2),
        C_fp16.stride(0), C_fp16.stride(1), C_fp16.stride(2),
        M=N, N=N, K=N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,        # H100 has 64 warps per SM, 8 gives high occupancy
        num_stages=4,       # double‑buffering for better latency hiding
    )
    torch.cuda.synchronize()

    # Reshape back to [B, N, N, H] and cast to FP32
    out = (
        C_fp16.to(torch.float32)
               .view(B, H, N, N)          # (B, H, N, N)
               .permute(0, 2, 3, 1)        # (B, N, N, H)
    )

    # ------------------------------------------------------------------
    # 6) Output LayerNorm + output gate
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate

    # ------------------------------------------------------------------
    # 7) Final projection back to the original `dim`
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])

    return out
