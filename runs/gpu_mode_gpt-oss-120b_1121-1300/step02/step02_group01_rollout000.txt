    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    eps = 1e-5

    # 1. Input LN
    x = F.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=eps)  # fp32
    # cast to bf16 for heavy projection
    x_bf16 = x.to(torch.bfloat16)

    # 2. Stacked projection (left and right)
    proj_weight = torch.cat([weights["left_proj.weight"], weights["right_proj.weight"]], dim=0).to(torch.bfloat16)  # (2*hidden_dim, dim)
    proj = F.linear(x_bf16, proj_weight)   # shape [B,N,N,2*hidden_dim] bf16
    left, right = torch.split(proj, hidden_dim, dim=-1)  # each bf16

    # 3. Mask (optional)
    if mask is not None and not nomask:
        mask_bf16 = mask.unsqueeze(-1).to(torch.bfloat16)
        left = left * mask_bf16
        right = right * mask_bf16

    # 4. Stacked gating
    gate_weight = torch.cat([weights["left_gate.weight"], weights["right_gate.weight"], weights["out_gate.weight"]], dim=0)  # (3*hidden_dim, dim)
    gate = torch.sigmoid(F.linear(x, gate_weight))  # fp32
    left_gate, right_gate, out_gate = torch.split(gate, hidden_dim, dim=-1)  # each fp32

    # cast gates to bf16 for multiplication
    left_gate_bf16 = left_gate.to(torch.bfloat16)
    right_gate_bf16 = right_gate.to(torch.bfloat16)

    # 5. Apply per-channel gates
    left = left * left_gate_bf16
    right = right * right_gate_bf16

    # 6. Core multiplication via Triton
    B, N, _, _ = left.shape
    # permute to (B, hidden_dim, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    # reshape to (B*hidden_dim, N, N)
    left_mat = left_perm.view(B * hidden_dim, N, N)
    right_mat = right_perm.view(B * hidden_dim, N, N)

    out_mat = torch.empty_like(left_mat)  # bf16

    # launch kernel
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

            (N + BLOCK_N - 1) // BLOCK_N)

            left_mat,
            right_mat,
            out_mat,
            B * hidden_dim,

            right_mat.stride(0), right_mat.stride(1), right_mat.stride(2),
            out_mat.stride(0), out_mat.stride(1), out_mat.stride(2),
            BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        )
    except triton.compiler.exceptions.KernelLaunchException:
        # fallback to torch.bmm
        out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # reshape back
    out = out_mat.view(B, hidden_dim, N, N).to(torch.float32).permute(0, 2, 3, 1)  # [B,N,N,hidden_dim] fp32

                       (hidden_dim,),

                       bias=weights["to_out_norm.bias"],
                       eps=eps)

"""
TriMul (outgoing) – fused Triton implementation.

The core of the TriMul layer is a batch of matrix‑multiplications
   out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]
where `left` and `right` have shape [B, N, N, H] (H = hidden_dim).
We launch a Triton kernel that treats the (B·H) planes as independent
matmul problems and computes them in BF16 with an FP32 accumulator.
All cheap linear / LayerNorm steps are performed with PyTorch,
while the heavy N³ update is fully Triton‑accelerated.

The implementation also:
* stacks the two projection matrices (left/right) into a single
  GEMM, and similarly stacks the three gating matrices,
  reducing kernel launch overhead.
* works with optional masking, mixed precision (BF16 for GEMMs,
  FP32 for LayerNorm/gating), and any allowed configuration.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _trmul_batch_matmul_kernel(
    left_ptr,                     # BF16 left matrix   (B*H, N, N)
    right_ptr,                    # BF16 right matrix  (B*H, N, N)
    out_ptr,                      # BF16 output matrix (B*H, N, N)
    batch_h,                      # total number of (B*H) planes
    N,                            # sequence length
    stride_l_batch, stride_l_m, stride_l_k,
    stride_r_batch, stride_r_k, stride_r_n,
    stride_o_batch, stride_o_m, stride_o_n,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Each program instance computes a BLOCK_M×BLOCK_N tile of one
    (B·H)‑plane of the result matrix.
    """
    b = tl.program_id(0)               # which (batch×head) plane
    pid_m = tl.program_id(1)           # tile row
    pid_n = tl.program_id(2)           # tile col

    # Offsets within the N×N matrix
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # Accumulator in FP32 for numerical stability
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in blocks
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile: shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_ptr + \
            b * stride_l_batch + \
            offs_m[:, None] * stride_l_m + \
            offs_k[None, :] * stride_l_k
        left = tl.load(left_ptrs,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)

        # right tile (transposed access): shape (BLOCK_K, BLOCK_N)
        # we need right[k, n] -> row = k, col = n
        right_ptrs = right_ptr + \
            b * stride_r_batch + \
            offs_k[:, None] * stride_r_k + \
            offs_n[None, :] * stride_r_n
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_n[None, :],
                        other=0.0).to(tl.float32)

        acc += tl.dot(left, right)

    # Write the result back as BF16
    out_ptrs = out_ptr + \
        b * stride_o_batch + \
        offs_m[:, None] * stride_o_m + \
        offs_n[None, :] * stride_o_n
    tl.store(out_ptrs,
             acc.to(tl.bfloat16),
             mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul “outgoing” operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor   [B, N, N, C]
            - mask: torch.Tensor (or None) [B, N, N]
            - weights: dict of torch.Tensors containing all linear / layer‑norm params
            - config: dict with keys `dim`, `hidden_dim`, `nomask` (bool)

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # 1. Input LayerNorm (FP32)
    # ---------------------------------------------------------

        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]  FP32

    # ---------------------------------------------------------
    x_bf16 = x.to(torch.bfloat16)

    # ---------------------------------------------------------

        right = right * mask_bf16

    # ---------------------------------------------------------

        [weights["left_gate.weight"], weights["right_gate.weight"], weights["out_gate.weight"]],
        dim=0,

    #    out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
    # ---------------------------------------------------------
    B, N, _, _ = left.shape

    right_perm = right.permute(0, 3, 1, 2).contiguous()

    right_mat = right_perm.view(B * hidden_dim, N, N) # BF16

    out_mat = torch.empty_like(left_mat)

    # launch Triton kernel
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    grid = (
        B * hidden_dim,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

        _trmul_batch_matmul_kernel[grid](
            left_mat,
            right_mat,
            out_mat,
            B * hidden_dim,
            N,
            left_mat.stride(0), left_mat.stride(1), left_mat.stride(2),
            right_mat.stride(0), right_mat.stride(1), right_mat.stride(2),
            out_mat.stride(0), out_mat.stride(1), out_mat.stride(2),
            BLOCK_M=BLOCK_M,
            BLOCK_N=BLOCK_N,
            BLOCK_K=BLOCK_K,
        )
    except Exception:  # fallback to torch if Triton compilation fails
        out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # reshape to original layout and cast back to FP32
    out = (
        out_mat.view(B, hidden_dim, N, N)
        .to(torch.float32)
        .permute(0, 2, 3, 1)               # [B, N, N, hidden_dim] FP32
    )

    # ---------------------------------------------------------
    # 8. Post‑norm, gating and final projection
    # ---------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate                       # element‑wise (FP32)
    out = F.linear(out, weights["to_out.weight"])  # → [B, N, N, dim] FP32

    return out
```
