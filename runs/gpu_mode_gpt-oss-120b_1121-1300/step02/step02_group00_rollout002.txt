"""
TriMul (outgoing) – Triton‑accelerated cubic update
==================================================
Algorithm
---------
1. Layer‑norm the 4‑D input over its feature dimension (dim).
2. Fuse the five 𝑥→𝑦 linear maps (left_proj, right_proj,
   left_gate, right_gate, out_gate) into a single matmul:
        X_flat (B·N·N, dim)  @  W_stack (5·H, dim)ᵀ → Y (B·N·N, 5·H)
   and then split Y into the five tensors.
   This reduces the number of cuBLAS launches from 5 → 1.
3. Optional pair‑wise mask is applied to *left* and *right*.
4. Gating: sigmoid on the three gate tensors and element‑wise
   multiplication with the projected tensors.
5. Heavy multiplicative update:
        out_h = left_h @ right_hᵀ   for each hidden head h.
   Implemented as a *batched* GEMM in Triton (FP16→FP32 accumulation).
   The kernel is tuned for H100 Tensor‑Cores (BLOCK_M/N = 128,
   BLOCK_K = 64, 8 warps, 4 stages).  No per‑element masking is needed
   because the mask was already applied in step 3.
6. Layer‑norm over the hidden dimension, multiply by *out_gate*.
7. Final linear projection back to the original feature dimension.

All heavy arithmetic is performed in FP16 for Tensor‑Core utilisation,
while the final result is returned in FP32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiplication C = A @ Bᵀ
#   A : (batch_head, M, K)   – row‑major
#   B : (batch_head, K, N)   – row‑major (already transposed)
#   C : (batch_head, M, N)
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # ---------------------------------------------------------------
    # program identifiers
    # ---------------------------------------------------------------
    pid_batch = tl.program_id(2)          # which (batch * head)
    pid_m = tl.program_id(0)              # block row
    pid_n = tl.program_id(1)              # block column

    # ---------------------------------------------------------------
    # tile offsets
    # ---------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # predicates for out‑of‑bounds handling
    mask_m = offs_m < M
    mask_n = offs_n < N

    # ---------------------------------------------------------------
    # base pointers for the current tile (including the batch‑head offset)
    # ---------------------------------------------------------------
    a_ptrs = a_ptr + pid_batch * stride_a_batch \
                   + offs_m[:, None] * stride_a_m \
                   + offs_k[None, :] * stride_a_k
    b_ptrs = b_ptr + pid_batch * stride_b_batch \
                   + offs_k[:, None] * stride_b_k \
                   + offs_n[None, :] * stride_b_n

    # accumulator in FP32 (Tensor‑Core accumulation)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---------------------------------------------------------------
    # loop over K dimension in tiles of size BLOCK_K
    # ---------------------------------------------------------------
    iters = tl.cdiv(K, BLOCK_K)
    for _ in range(0, iters):
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (offs_k[None, :] < K),
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=(offs_k[:, None] < K) & mask_n[None, :],
                    other=0.0)
        acc += tl.dot(a, b)                     # fp16·fp16 → fp32

        # advance pointers to next K‑tile
        a_ptrs += BLOCK_K * stride_a_k
        b_ptrs += BLOCK_K * stride_b_k

    # ---------------------------------------------------------------
    # store result (convert back to FP16, later cast to FP32 in PyTorch)
    # ---------------------------------------------------------------
    c = acc.to(tl.float16)
    c_ptrs = c_ptr + pid_batch * stride_c_batch \
                     + offs_m[:, None] * stride_c_m \
                     + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs,
             c,
             mask=mask_m[:, None] & mask_n[None, :])

# Entrypoint used by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the *outgoing* TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, dim]
        - mask_tensor  : torch.Tensor of shape [B, N, N] (or None)
        - weights_dict : mapping from string names to torch.Tensor
        - config_dict  : contains ``dim``, ``hidden_dim`` and ``nomask``

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim         = config["dim"]
    hidden_dim  = config["hidden_dim"]
    nomask      = config.get("nomask", False)
    eps = 1e-5

    device = input_tensor.device
    dtype  = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1) Input LayerNorm over ``dim``
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 2) Fuse the five linear projections (left, right, three gates)
    # ------------------------------------------------------------------
    #   stacked_weight → (5*hidden_dim, dim)
    stacked_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                     # fp16 for cuBLAS Tensor‑Core

    # flatten to (B·N·N, dim) and apply the fused linear layer
    B, N, _, _ = input_tensor.shape
    x_fp16 = x.to(torch.float16)
    x_flat = x_fp16.view(-1, dim)           # (B·N·N, dim)
    fused = F.linear(x_flat, stacked_weight)   # (B·N·N, 5*H)

    # reshape and split
    fused = fused.view(B, N, N, 5 * hidden_dim)
    left      = fused[..., 0 * hidden_dim : 1 * hidden_dim]
    right     = fused[..., 1 * hidden_dim : 2 * hidden_dim]
    left_gate = fused[..., 2 * hidden_dim : 3 * hidden_dim]
    right_gate= fused[..., 3 * hidden_dim : 4 * hidden_dim]
    out_gate  = fused[..., 4 * hidden_dim : 5 * hidden_dim]

    # ------------------------------------------------------------------
    # 3) Optional pair‑wise mask (broadcasted on the hidden dim)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_ = mask.to(x_fp16.dtype).unsqueeze(-1)   # [B, N, N, 1]
        left  = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4) Sigmoid gates (still in fp16) and element‑wise gating
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate   = torch.sigmoid(out_gate)

    left  = left  * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Heavy multiplicative update (batched GEMM)
    # ------------------------------------------------------------------
    #   left : [B, N, N, H]   →   (B, H, N, N) → (BH, N, N)
    #   right: [B, N, N, H]   →   transpose on the *i* axis → (B, H, N, N) where rows = K
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N) = (B, H, K, N)

    B_, H_, N_, _ = left_perm.shape
    batch_heads = B_ * H_

    A = left_perm.view(batch_heads, N_, N_)   # (BH, M=N, K=N)
    Bmat = right_perm.view(batch_heads, N_, N_)  # (BH, K=N, N)

    # Allocate output buffer (FP16, will be cast to FP32 later)
    C_fp16 = torch.empty_like(A, dtype=torch.float16)

    # ---------------------------------------------------------------
    # Triton launch parameters (tuned for H100 Tensor‑Cores)
    # ---------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        (N_ + BLOCK_M - 1) // BLOCK_M,   # blocks along M
        (N_ + BLOCK_N - 1) // BLOCK_N,   # blocks along N
        batch_heads,                     # one grid slice per (batch, head)
    )

    batched_matmul_kernel[grid](
        A, Bmat, C_fp16,
        A.stride(0), A.stride(1), A.stride(2),
        Bmat.stride(0), Bmat.stride(1), Bmat.stride(2),
        C_fp16.stride(0), C_fp16.stride(1), C_fp16.stride(2),
        M=N_, N=N_, K=N_,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=4,
    )
    torch.cuda.synchronize()

    # ------------------------------------------------------------------
    # 6) Reshape back to [B, N, N, H] and convert to FP32
    # ------------------------------------------------------------------
    out = (
        C_fp16
        .view(B_, H_, N_, N_)      # (B, H, N, N)
        .permute(0, 2, 3, 1)       # (B, N, N, H)
        .to(torch.float32)
    )

    # ------------------------------------------------------------------
    # 7) Output LayerNorm + output gate
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate

    # ------------------------------------------------------------------
    # 8) Final linear projection back to the original ``dim``
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])

    return out
```
