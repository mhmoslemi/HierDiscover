def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    eps = 1e-5
    # 1. LayerNorm
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=weights["norm.weight"],
                     bias=weights["norm.bias"],
                     eps=eps)
    # cast to fp16 for compute
    x_fp16 = x.to(torch.float16)
    # 2. Fused left projection+gate
    left_combined_weight = torch.cat([weights["left_proj.weight"], weights["left_gate.weight"]], dim=0).to(torch.float16)
    left_out = F.linear(x_fp16, left_combined_weight)
    left_proj = left_out[..., :hidden_dim]
    left_gate = torch.sigmoid(left_out[..., hidden_dim:])
    # 3. Fused right projection+gate
    right_combined_weight = torch.cat([weights["right_proj.weight"], weights["right_gate.weight"]], dim=0).to(torch.float16)
    right_out = F.linear(x_fp16, right_combined_weight)
    right_proj = right_out[..., :hidden_dim]
    right_gate = torch.sigmoid(right_out[..., hidden_dim:])
    # 4. Optional mask
    if not config.get("nomask", False) and mask is not None:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)  # [B,N,N,1]
        left_proj = left_proj * mask_fp16
        right_proj = right_proj * mask_fp16
    # 5. Apply gates
    left = left_proj * left_gate
    right = right_proj * right_gate
    # 6. Compute out_gate
    out_gate_weight = weights["out_gate.weight"].to(torch.float16)
    out_gate = torch.sigmoid(F.linear(x_fp16, out_gate_weight))
    # 7. Batched GEMM
    # Permute for matmul
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, H, N, N) as (K,N)
    B, H, N, _ = left_perm.shape
    batch_heads = B * H
    A = left_perm.view(batch_heads, N, N)
    Bmat = right_perm.view(batch_heads, N, N)
    # output tensor
    C_fp16 = torch.empty_like(A)
    # kernel launch
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64
    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        batch_heads,
    )
    batched_matmul_kernel[grid](
        A, Bmat, C_fp16,
        A.stride(0), A.stride(1), A.stride(2),
        Bmat.stride(0), Bmat.stride(1), Bmat.stride(2),
        C_fp16.stride(0), C_fp16.stride(1), C_fp16.stride(2),
        M=N, N=N, K=N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=4,
    )
    torch.cuda.synchronize()
    # Reshape back
    out = C_fp16.to(torch.float32).view(B, H, N, N).permute(0, 2, 3, 1)  # (B,N,N,H)
    # 8. Output LayerNorm
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=weights["to_out_norm.weight"],
                       bias=weights["to_out_norm.bias"],
                       eps=eps)
    # 9. Apply out_gate and final projection
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])
    return out

"""
Outgoing TriMul (AlphaFold3) – Triton implementation
---------------------------------------------------
We compute the forward pass of the TriMul module:

1. Layer‑Norm over the feature dimension.
2. Two *fused* linear layers that simultaneously produce a projection and a
   gating vector (proj+gate).  This halves the number of torch‑linear calls.
3. Optional masking of the projected tensors.
4. Element‑wise gating (sigmoid) and multiplication.
5. Heavy multiplicative update:
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   which is a batched matrix‑multiply (BH, N, N) × (BH, N, N)ᵀ.
   We implement this with a custom Triton kernel that works in FP16,
   accumulates in FP32 and uses 128×128×64 tiling (8 warps, 4 stages).
6. Layer‑Norm over the hidden dimension, multiplication by the output gate.
7. Final linear projection back to the original feature dimension.

All heavy arithmetic is done in FP16 to exploit H100 Tensor‑Cores, while
the final result is returned as FP32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Batched GEMM kernel: C = A @ Bᵀ  (A: (BH,M,K), B: (BH,N,K) → Bᵀ: (BH,K,N))
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # grid axes: (block_m, block_n, batch_head)
    pid_batch = tl.program_id(2)   # which (batch * head)
    pid_m = tl.program_id(0)       # block row index
    pid_n = tl.program_id(1)       # block column index

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # base pointers for the current tile (including batch‑head offset)
    a_ptrs = a_ptr + pid_batch * stride_a_batch \
                    + offs_m[:, None] * stride_a_m \
                    + offs_k[None, :] * stride_a_k
    b_ptrs = b_ptr + pid_batch * stride_b_batch \
                    + offs_k[:, None] * stride_b_k \
                    + offs_n[None, :] * stride_b_n

    # accumulator in FP32 (Tensor‑Core fp16·fp16→fp32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension in tiles of size BLOCK_K
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs,
                    mask=(offs_m[:, None] < M) & (offs_k[None, :] < K),
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=(offs_k[:, None] < K) & (offs_n[None, :] < N),
                    other=0.0)
        acc += tl.dot(a, b)            # fp16 dot → fp32 accumulation
        a_ptrs += BLOCK_K * stride_a_k
        b_ptrs += BLOCK_K * stride_b_k

    # write C (store as FP16, later cast back to FP32)
    c = acc.to(tl.float16)
    c_ptrs = c_ptr + pid_batch * stride_c_batch \
                     + offs_m[:, None] * stride_c_m \
                     + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs,
             c,
             mask=(offs_m[:, None] < M) & (offs_n[None, :] < N))


# ----------------------------------------------------------------------
# Entry point for the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, dim]
        - mask_tensor  : torch.Tensor of shape [B, N, N] (or None)
        - weights_dict : mapping from string names to torch.Tensor
        - config_dict  : contains "dim", "hidden_dim", "nomask", ...

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1) Input LayerNorm over the feature dimension `dim`
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )
    # cast to FP16 for the heavy part
    x_fp16 = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2) Fused projection + gate for the *left* side
    #    (concat proj and gate weights → one linear)
    # ------------------------------------------------------------------
    left_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16)                     # (2*H, dim)
    left_out = F.linear(x_fp16, left_w)    # (B,N,N,2*H)
    left_proj = left_out[..., :hidden_dim]
    left_gate = torch.sigmoid(left_out[..., hidden_dim:])

    # ------------------------------------------------------------------
    # 3) Fused projection + gate for the *right* side
    # ------------------------------------------------------------------
    right_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16)                    # (2*H, dim)
    right_out = F.linear(x_fp16, right_w)  # (B,N,N,2*H)
    right_proj = right_out[..., :hidden_dim]
    right_gate = torch.sigmoid(right_out[..., hidden_dim:])

    # ------------------------------------------------------------------
    # 4) Optional mask (broadcast on hidden dimension)
    # ------------------------------------------------------------------
    if not config.get("nomask", False) and mask is not None:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left_proj = left_proj * mask_fp16
        right_proj = right_proj * mask_fp16

    # ------------------------------------------------------------------
    # 5) Apply gates
    # ------------------------------------------------------------------
    left = left_proj * left_gate
    right = right_proj * right_gate

    # free intermediate tensors early
    del left_proj, left_gate, right_proj, right_gate, left_out, right_out, left_w, right_w

    # ------------------------------------------------------------------
    # 6) Output gate (single linear + sigmoid)
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(
        F.linear(x_fp16, weights["out_gate.weight"].to(torch.float16))
    )

    # ------------------------------------------------------------------
    # 7) Heavy multiplicative update (batched GEMM)
    #     out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    # reshape to (batch*head, N, N) and transpose the right operand
    # left  : (B, N, N, H) → (B, H, N, N)   (i,k)
    # right : (B, N, N, H) → (B, H, N, N)   (k,j)   <-- note swapped axes
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, H, N, N) where dim2 is K

    B, H, N, _ = left_perm.shape
    batch_heads = B * H

    A = left_perm.view(batch_heads, N, N)   # (BH, M=N, K=N)
    Bmat = right_perm.view(batch_heads, N, N)  # (BH, K=N, N)

    # output buffer in FP16
    C_fp16 = torch.empty_like(A)

    # ------------------------------------------------------------------
    # launch parameters (tuned for H100 Tensor‑Cores)
    # ------------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,  # blocks along M
        (N + BLOCK_N - 1) // BLOCK_N,  # blocks along N
        batch_heads,                    # one launch per (batch,head)
    )

    batched_matmul_kernel[grid](
        A, Bmat, C_fp16,
        A.stride(0), A.stride(1), A.stride(2),
        Bmat.stride(0), Bmat.stride(1), Bmat.stride(2),
        C_fp16.stride(0), C_fp16.stride(1), C_fp16.stride(2),
        M=N, N=N, K=N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,          # 8 warps → 256 threads per block
        num_stages=4,         # double‑buffering
    )
    torch.cuda.synchronize()

    # reshape back to (B, N, N, H) and convert to FP32
    out = (
        C_fp16.to(torch.float32)
               .view(B, H, N, N)   # (B, H, N, N)
               .permute(0, 2, 3, 1)  # (B, N, N, H)
    )

    # ------------------------------------------------------------------
    # 8) Output LayerNorm + output gate
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate

    # ------------------------------------------------------------------
    # 9) Final projection back to the original feature dimension
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])

    return out
