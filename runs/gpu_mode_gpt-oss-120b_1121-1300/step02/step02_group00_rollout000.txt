"""
TriMul (outgoing) – Triton‑accelerated forward pass
--------------------------------------------------

The module takes a 4‑D tensor *x* of shape [B, N, N, D] and computes

    out = Linear( LayerNorm(  Σ_k  ( (x·W_L)·σ(x·W_GL) )  @
                             ( (x·W_R)·σ(x·W_GR) )ᵀ ) · σ(x·W_GO) )
    out = LayerNorm(out) · σ(x·W_GO)  →  Linear →  final tensor [B,N,N,D]

where
* W_L, W_R, W_GL, W_GR, W_GO are linear weights,
* σ is the sigmoid, and
* the heavy Σ_k multiplication is a batched matrix‑multiply
  (one GEMM per head) that dominates the runtime.

Only the matrix‑multiply (step 5) is off‑loaded to a custom Triton
kernel; everything else stays in PyTorch for simplicity.

The kernel works on FP16 operands and accumulates in FP32 (Tensor‑Core
friendly).  Input / output tensors are cast back to FP32 before being
returned.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: C = A @ B   (batched GEMM)
#   A : (batch, M, K) – row‑major
#   B : (batch, K, N) – row‑major (already transposed on the PyTorch side)
#   C : (batch, M, N) – row‑major, stored as fp16 (accumulated in fp32)
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # --------------------------------------------------------------
    # 1️⃣ program ids
    # --------------------------------------------------------------
    pid_batch = tl.program_id(2)          # which (batch*head)
    pid_m = tl.program_id(0)              # block row
    pid_n = tl.program_id(1)              # block column

    # --------------------------------------------------------------
    # 2️⃣ tile offsets
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # --------------------------------------------------------------
    # 3️⃣ base pointers for this tile (including the batch‑head offset)
    # --------------------------------------------------------------
    a_ptrs = a_ptr + pid_batch * stride_a_batch \
                  + offs_m[:, None] * stride_a_m \
                  + offs_k[None, :] * stride_a_k
    b_ptrs = b_ptr + pid_batch * stride_b_batch \
                  + offs_k[:, None] * stride_b_k \
                  + offs_n[None, :] * stride_b_n

    # --------------------------------------------------------------
    # 4️⃣ accumulator (fp32)
    # --------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # 5️⃣ loop over K‑tiles
    # --------------------------------------------------------------
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(
            a_ptrs,
            mask=(offs_m[:, None] < M) & (offs_k[None, :] < K),
            other=0.0,
        )
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & (offs_n[None, :] < N),
            other=0.0,
        )
        acc += tl.dot(a, b)           # fp16·fp16 → fp32

        # advance to next K‑tile
        a_ptrs += BLOCK_K * stride_a_k
        b_ptrs += BLOCK_K * stride_b_k

    # --------------------------------------------------------------
    # 6️⃣ write result (store as fp16; later cast back to fp32)
    # --------------------------------------------------------------
    c = acc.to(tl.float16)
    c_ptrs = c_ptr + pid_batch * stride_c_batch \
                     + offs_m[:, None] * stride_c_m \
                     + offs_n[None, :] * stride_c_n
    tl.store(
        c_ptrs,
        c,
        mask=(offs_m[:, None] < M) & (offs_n[None, :] < N),
    )


# ----------------------------------------------------------------------
# Entrypoint used by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the *outgoing* TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor  [B, N, N, D]   (float32)
        - mask_tensor  : torch.Tensor  [B, N, N]    (bool/float) or None
        - weights_dict : mapping name → torch.Tensor (already on device)
        - config_dict  : {"dim": D, "hidden_dim": H, "nomask": bool, ...}

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, D] (float32)
    """
    # --------------------------------------------------------------
    # 0️⃣ unpack arguments
    # --------------------------------------------------------------
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    device = x.device

    # --------------------------------------------------------------
    # 1️⃣ Layer‑norm over the feature dimension `dim`
    # --------------------------------------------------------------
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # --------------------------------------------------------------
    # 2️⃣ Project to hidden space (FP16 for the heavy part)
    # --------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    left = F.linear(x_fp16, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x_fp16, weights["right_proj.weight"].to(torch.float16))

    # --------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast on hidden dimension)
    # --------------------------------------------------------------
    if not config.get("nomask", False) and mask is not None:
        mask_fp16 = mask.to(x_fp16.dtype).unsqueeze(-1)        # [B,N,N,1]
        left = left * mask_fp16
        right = right * mask_fp16

    # --------------------------------------------------------------
    # 4️⃣ Gates (sigmoid) – still in FP16
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(
        F.linear(x_fp16, weights["left_gate.weight"].to(torch.float16))
    )
    right_gate = torch.sigmoid(
        F.linear(x_fp16, weights["right_gate.weight"].to(torch.float16))
    )
    out_gate = torch.sigmoid(
        F.linear(x_fp16, weights["out_gate.weight"].to(torch.float16))
    )

    # Apply gate activations
    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5️⃣ Heavy multiplicative update (batched GEMM)
    # --------------------------------------------------------------
    # Bring head dimension to the second axis and make tensors contiguous.
    # After permute: (B, H, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # left[b,h,i,k]
    right_perm = right.permute(0, 3, 2, 1).contiguous() # right[b,h,k,j]  (already transposed)

    B, H, N, _ = left_perm.shape
    batch_heads = B * H

    # Collapse batch*head → single leading dimension for the kernel
    A = left_perm.view(batch_heads, N, N)   # (BH, M=N, K=N)
    Bmat = right_perm.view(batch_heads, N, N)   # (BH, K=N, N)

    # Allocate output buffer (FP16, will be cast back)
    C = torch.empty_like(A, dtype=torch.float16)

    # --------------------------------------------------------------
    # 5️⃣1️⃣ Choose tile sizes (tuned for H100 Tensor‑Cores)
    # --------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # blocks along M
        (N + BLOCK_N - 1) // BLOCK_N,   # blocks along N
        batch_heads                     # one block per (batch, head)
    )

    batched_matmul_kernel[grid](
        A, Bmat, C,
        A.stride(0), A.stride(1), A.stride(2),
        Bmat.stride(0), Bmat.stride(1), Bmat.stride(2),
        C.stride(0), C.stride(1), C.stride(2),
        M=N, N=N, K=N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=4,
    )
    torch.cuda.synchronize()

    # --------------------------------------------------------------
    # 5️⃣2️⃣ Reshape back to [B, N, N, H] and go to FP32
    # --------------------------------------------------------------
    out = (
        C.float()                                   # fp16 → fp32
          .view(B, H, N, N)                         # (B, H, N, N)
          .permute(0, 2, 3, 1)                      # (B, N, N, H)
    )

    # --------------------------------------------------------------
    # 6️⃣ Output LayerNorm + out‑gate
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate

    # --------------------------------------------------------------
    # 7️⃣ Final projection back to the original `dim`
    # --------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])

    return out
```
