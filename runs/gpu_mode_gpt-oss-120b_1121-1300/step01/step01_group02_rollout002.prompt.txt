import torch
from torch import nn, einsum
import math

# Reference code in PyTorch
class TriMul(nn.Module):
    def __init__(
        self,
        dim: int,
        hidden_dim: int,
    ):
        super().__init__()

        self.norm = nn.LayerNorm(dim)

        self.left_proj = nn.Linear(dim, hidden_dim, bias=False)
        self.right_proj = nn.Linear(dim, hidden_dim, bias=False)

        self.left_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.right_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.out_gate = nn.Linear(dim, hidden_dim, bias=False)

        self.to_out_norm = nn.LayerNorm(hidden_dim)
        self.to_out = nn.Linear(hidden_dim, dim, bias=False)

    def forward(self, x: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
        """
        x: [bs, seq_len, seq_len, dim]
        mask: [bs, seq_len, seq_len]

        Returns:
            output: [bs, seq_len, seq_len, dim]
        """
        batch_size, seq_len, _, dim = x.shape

        x = self.norm(x)

        left = self.left_proj(x)
        right = self.right_proj(x)

        mask = mask.unsqueeze(-1)
        left = left * mask
        right = right * mask

        left_gate = self.left_gate(x).sigmoid()
        right_gate = self.right_gate(x).sigmoid()
        out_gate = self.out_gate(x).sigmoid()

        left = left * left_gate
        right = right * right_gate

        out = einsum('... i k d, ... j k d -> ... i j d', left, right)
        # This einsum is the same as the following:
        # out = torch.zeros(batch_size, seq_len, seq_len, dim, device=x.device)
        
        # # Compute using nested loops
        # for b in range(batch_size):
        #     for i in range(seq_len):
        #         for j in range(seq_len):
        #             # Compute each output element
        #             for k in range(seq_len):
        #                 out[b, i, j] += left[b, i, k, :] * right[b, j, k, :]

        out = self.to_out_norm(out)
        out = out * out_gate
        return self.to_out(out)

"""
TriMul outgoing implementation.

The forward pass follows the reference PyTorch code and fuses the
costly N³ multiplicative update with a custom Triton kernel.

Steps
-----
1. Layer‑norm over the last channel dimension.
2. Linear projections (`left_proj`, `right_proj`) and optional mask.
3. Sigmoid gates (`left_gate`, `right_gate`, `out_gate`).
4. Core update:
   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is a batched matrix multiplication (A @ Bᵀ) performed **per
   hidden channel** with a tiled GEMM kernel written in Triton.
   The kernel works in FP16 (to save bandwidth) and accumulates in FP32.
5. Hidden‑dim layer‑norm, element‑wise `out_gate`, and a final linear
   projection back to the original channel dimension.

All tensors stay on the GPU; only the heavy N³ part is executed in the
custom Triton kernel, giving good utilisation on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Batched GEMM kernel for the TriMul multiplicative update.
    For each (batch, hidden) pair we compute:
        out[i, j] = Σ_k left[i, k] * right[j, k]   (i, j, k ∈ [0, N))
    which is exactly a matrix multiplication of left (N×N) with
    rightᵀ (N×N).  The kernel tiles the operation with
    BLOCK_M×BLOCK_N output tiles and accumulates over K in BLOCK_K‑wide
    slices, using FP16 loads and FP32 accumulation.
    """
    pid_m = tl.program_id(0)   # tile row
    pid_n = tl.program_id(1)   # tile column
    pid_z = tl.program_id(2)   # batch * hidden index

    # Decompose pid_z into batch and hidden indices
    hidden_idx = pid_z % H
    batch_idx  = pid_z // H

    # Offsets for the current M×N tile
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # Base pointers for the selected batch / hidden channel
    a_ptr = left_ptr + batch_idx * stride_left_b + hidden_idx * stride_left_h
    b_ptr = right_ptr + batch_idx * stride_right_b + hidden_idx * stride_right_h
    c_ptr = out_ptr + batch_idx * stride_out_b + hidden_idx * stride_out_h

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in BLOCK_K steps
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        k_offs = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offs < N

        # Load A tile (shape BLOCK_M × BLOCK_K)
        a_offs = a_ptr + offs_i[:, None] * stride_left_i + k_offs[None, :] * stride_left_k
        a = tl.load(a_offs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # Load Bᵀ tile (shape BLOCK_K × BLOCK_N)
        # Bᵀ[k, j] = right[batch, j, k, hidden]
        b_offs = b_ptr + k_offs[:, None] * stride_right_k + offs_j[None, :] * stride_right_j
        b = tl.load(b_offs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # Accumulate into FP32
        acc += tl.dot(a, b).to(tl.float32)

    # Store the tile back (cast to FP16 because the output buffer is FP16)
    c_offs = c_ptr + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j
    tl.store(c_offs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor  [B, N, N, dim]
        - mask         : torch.Tensor  [B, N, N]  (may be None)
        - weights      : dict of torch.Tensors containing model parameters
        - config       : dict with keys "dim", "hidden_dim" (and optional flags)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32).
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # Load model parameters
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias) – shape (B, N, N, hidden_dim)
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # left_proj: (hidden_dim, dim)
    right = F.linear(x, right_proj_weight) # right_proj: (hidden_dim, dim)

    # ------------------------------------------------------------------
    # 3. Apply mask if provided
    # ------------------------------------------------------------------
    if mask is None or mask.numel() == 0:
        mask = torch.ones((bs, N, N), dtype=x.dtype, device=x.device)
    mask = mask.to(x.dtype).unsqueeze(-1)  # (B, N, N, 1)
    left = left * mask
    right = right * mask

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid of linear layers) and gated multiplication
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core multiplicative update (Σ_k left[i,k,d] * right[j,k,d])
    #    Implemented with a Triton kernel.
    # ------------------------------------------------------------------
    # Cast to fp16 to reduce bandwidth; the kernel accumulates in fp32.
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # Output buffer (fp16)
    out_fp16 = torch.empty_like(left_fp16)

    # Triton block sizes – tuned for H100 (good shared‑memory utilisation)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Strides in element units (not bytes)
    stride_left_b = left_fp16.stride(0)
    stride_left_i = left_fp16.stride(1)
    stride_left_k = left_fp16.stride(2)
    stride_left_h = left_fp16.stride(3)

    stride_right_b = right_fp16.stride(0)
    stride_right_j = right_fp16.stride(1)   # “row” after transpose
    stride_right_k = right_fp16.stride(2)
    stride_right_h = right_fp16.stride(3)

    stride_out_b = out_fp16.stride(0)
    stride_out_i = out_fp16.stride(1)
    stride_out_j = out_fp16.stride(2)
    stride_out_h = out_fp16.stride(3)

    # Grid: (tiles over i, tiles over j, batch * hidden)
    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        bs * hidden_dim,
    )

    # Launch the Triton kernel
    trimul_outgoing_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        bs,
        N,
        hidden_dim,
        stride_left_b,
        stride_left_i,
        stride_left_k,
        stride_left_h,
        stride_right_b,
        stride_right_j,
        stride_right_k,
        stride_right_h,
        stride_out_b,
        stride_out_i,
        stride_out_j,
        stride_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Convert back to FP32 for the remaining layers
    out = out_fp16.to(torch.float32)   # (B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 6. Hidden‑dim LayerNorm, out‑gate, and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )
    out = out * out_gate
    out = F.linear(out, to_out_weight)   # (B, N, N, dim)

    return out
