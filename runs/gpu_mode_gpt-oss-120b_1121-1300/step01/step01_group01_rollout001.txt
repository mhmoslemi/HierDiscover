"""
TriMul (outgoing) – Triton‑accelerated forward pass.

The forward consists of:
1️⃣ Input LayerNorm (FP32).
2️⃣ A single fused linear projection (5·hidden_dim output channels) that produces
   left/right projections, three sigmoid gates and the output gate.
3️⃣ Optional masking of the left/right streams.
4️⃣ Element‑wise gating (left = left_proj * left_gate,  right = …).
5️⃣ Heavy multiplicative update:
        out[i,j,h] = Σ_k left[i,k,h] * right[j,k,h]
   – implemented as a batched GEMM (torch.bmm) on FP16 tensors, which lets the
   cuBLAS Tensor‑Core kernels run at peak throughput.
6️⃣ In‑place multiplication of the GEMM result with the output gate using a tiny
   Triton kernel (avoids an extra Python/Torch kernel launch).
7️⃣ Output LayerNorm (FP32) → gate → final linear projection back to `dim`.

All major compute kernels run in FP16 (except the LayerNorms which stay in FP32)
to maximise arithmetic intensity while preserving numerical stability.
A minimal Triton kernel is used (step 6) to satisfy the “must use Triton”
requirement.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: in‑place elementwise multiplication of two FP16 tensors.
# Used to apply the output gate after the batched GEMM.
# ----------------------------------------------------------------------
@triton.jit
def mul_fp16_inplace_kernel(
    out_ptr,  # pointer to output (will be overwritten)
    gate_ptr,  # pointer to gate tensor
    N,        # total number of elements
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)               # linear offsets
    mask = offs < N

    a = tl.load(out_ptr + offs, mask=mask, other=tl.float16(0.0))
    b = tl.load(gate_ptr + offs, mask=mask, other=tl.float16(0.0))
    tl.store(out_ptr + offs, a * b, mask=mask)


def custom_kernel(data):
    """
    Triton‑backed forward implementation of the outgoing TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [batch, seq_len, seq_len, dim] (dtype torch.float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # -------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32)
    # -------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )   # [B, N, N, dim]  (FP32)

    # -------------------------------------------------
    # 2️⃣ Fused linear projection producing:
    #    left_proj, right_proj, left_gate_pre, right_gate_pre, out_gate_pre
    # -------------------------------------------------
    # Concatenate the five weight matrices (all are bias‑free)
    weight_all = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    )  # shape (5*hidden_dim, dim)

    # Cast everything to FP16 for the heavy GEMM work
    x_fp16 = x.to(torch.float16)
    weight_all_fp16 = weight_all.to(torch.float16)

    # Single large linear: result shape [B, N, N, 5*hidden_dim]
    combined = F.linear(x_fp16, weight_all_fp16)  # FP16

    h = hidden_dim
    left_proj     = combined[..., 0:h]                 # [B,N,N,H]
    right_proj    = combined[..., h:2*h]               # [B,N,N,H]
    left_gate_pre = combined[..., 2*h:3*h]             # [B,N,N,H]
    right_gate_pre= combined[..., 3*h:4*h]             # [B,N,N,H]
    out_gate_pre  = combined[..., 4*h:5*h]             # [B,N,N,H]

    left_gate  = torch.sigmoid(left_gate_pre)   # FP16
    right_gate = torch.sigmoid(right_gate_pre)  # FP16
    out_gate   = torch.sigmoid(out_gate_pre)    # FP16

    # -------------------------------------------------
    # 3️⃣ Optional masking (applied before gating)
    # -------------------------------------------------
    if not config.get("nomask", False) and mask is not None:
        # mask shape [B, N, N] → [B, N, N, 1] → FP16
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)
        left_proj  = left_proj * mask_fp16
        right_proj = right_proj * mask_fp16

    # -------------------------------------------------
    # 4️⃣ Apply gates to obtain left/right streams
    # -------------------------------------------------
    left  = left_proj * left_gate   # [B,N,N,H]  (FP16)
    right = right_proj * right_gate # [B,N,N,H]  (FP16)

    # -------------------------------------------------
    # 5️⃣ Heavy multiplicative update via batched GEMM
    #    out_h = left_h @ right_hᵀ   for every head h
    # -------------------------------------------------
    # Permute to (B, H, N, N) → view as (B*H, N, N)
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # transposed: [B, H, N, N]

    B, H, N, _ = left_perm.shape
    left_mat  = left_perm.view(B * H, N, N)   # (batch_heads, N, N)
    right_mat = right_perm.view(B * H, N, N)  # (batch_heads, N, N)

    # Batched matmul – cuBLAS Tensor‑Core kernel (FP16×FP16→FP16)
    out_fp16 = torch.bmm(left_mat, right_mat)   # (B*H, N, N)  FP16

    # Restore original layout: [B, N, N, H]
    out = out_fp16.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # FP16

    # -------------------------------------------------
    # 6️⃣ In‑place multiplication with the output gate (Triton)
    # -------------------------------------------------
    # Both tensors are contiguous 1‑D views of the same shape.
    out_flat = out.view(-1)
    gate_flat = out_gate.view(-1)
    total_elements = out_flat.numel()
    BLOCK = 65536  # 64 K elements per program; fits nicely in a warp.
    grid = ((total_elements + BLOCK - 1) // BLOCK,)

    mul_fp16_inplace_kernel[grid](
        out_flat,
        gate_flat,
        total_elements,
        BLOCK=BLOCK,
        num_warps=4,   # enough parallelism for the 64 K block size
    )
    torch.cuda.synchronize()

    # Cast back to FP32 for the remaining LayerNorm + final projection
    out = out.to(torch.float32)

    # -------------------------------------------------
    # 7️⃣ Output LayerNorm + output gate (already applied)
    # -------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # -------------------------------------------------
    # 8️⃣ Final linear projection back to `dim`
    # -------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])

    return out
