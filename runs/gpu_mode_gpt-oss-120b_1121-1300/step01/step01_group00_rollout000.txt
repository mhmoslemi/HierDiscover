```
@triton.jit
def trimul_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, D, N,
    stride_lb, stride_ld, stride_li, stride_lk,
    stride_rb, stride_rd, stride_rj, stride_rk,
    stride_ob, stride_od, stride_oi, stride_oj,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid0 = tl.program_id(0)  # i tile
    pid1 = tl.program_id(1)  # j tile
    pid2 = tl.program_id(2)  # b*d tile

    b = pid2 // D
    d = pid2 % D

    i = pid0 * BLOCK_M
    j = pid1 * BLOCK_N

    # Pointers
    left_tile_ptr = left_ptr + b*stride_lb + d*stride_ld + i*stride_li
    right_tile_ptr = right_ptr + b*stride_rb + d*stride_rd + j*stride_rj

    # Accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K

        left_ptrs = left_tile_ptr + tl.arange(0, BLOCK_M)[:, None] * stride_li + (k_offset + tl.arange(0, BLOCK_K))[None, :] * stride_lk
        right_ptrs = right_tile_ptr + (k_offset + tl.arange(0, BLOCK_K)[:, None]) * stride_rk + tl.arange(0, BLOCK_N)[None, :] * stride_rj

        # load and mask
        mask_l = (i + tl.arange(0, BLOCK_M)[:, None] < N) & (k_offset + tl.arange(0, BLOCK_K)[None, :] < N)
        mask_r = (k_offset + tl.arange(0, BLOCK_K)[:, None] < N) & (j + tl.arange(0, BLOCK_N)[None, :] < N)

        left = tl.load(left_ptrs, mask=mask_l, other=0.0)
        right = tl.load(right_ptrs, mask=mask_r, other=0.0)
        acc += tl.dot(left, right)   # default FP16->FP32 accumulation

    # write out
    out_ptrs = out_ptr + b*stride_ob + d*stride_od + i*stride_oi + j*stride_oj
    mask_o = (i + tl.arange(0, BLOCK_M)[:, None] < N) & (j + tl.arange(0, BLOCK_N)[None, :] < N)
    tl.store(out_ptrs, acc, mask=mask_o)
```

"""
TriMul (outgoing) – Triton‑accelerated forward pass

Algorithm
---------
1. Layer‑norm the 4‑D input tensor  [B, N, N, C].
2. Linear projections (no bias) → left/right  (shape [B, N, N, H]).
3. Optional binary mask multiplication (skip when config["nomask"] is True).
4. Gating: sigmoid‑gated linear layers on the *pre‑norm* tensor `x`.
   left  = left  * left_gate
   right = right * right_gate
5. Core multiplicative update:
       out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]   (left @ rightᵀ)
   Implemented as a fused batched GEMM in Triton.
   * left/right are cast to FP16 and the GEMM uses Tensor‑cores
     (`tl.dot` on FP16 → FP32 accumulation).
   * Grid: (tiles_M, tiles_N, B*H) – each program computes an
     (M×N) tile of the output for a specific (batch, hidden) pair.
6. Post‑norm, gating with `out_gate` and final linear projection back to `C`.

The heavy N³ work (step 5) is the only part written in Triton; all
other operations stay in PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _trimul_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, H, N,
    stride_lb, stride_lh, stride_li, stride_lk,
    stride_rb, stride_rh, stride_rj, stride_rk,
    stride_ob, stride_oh, stride_oi, stride_oj,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched GEMM for outgoing TriMul:
        out[b, h, i, j] = Σₖ left[b, h, i, k] * right[b, h, j, k]
    left/right are stored as [B, H, N, N] (row‑major, inner‑most dim = k).
    """
    pid_i = tl.program_id(0)          # tile along i (seq_len)
    pid_j = tl.program_id(1)          # tile along j
    pid_hb = tl.program_id(2)         # combined (batch, hidden) index

    # ----- decode batch and hidden indices ---------------------------------
    b = pid_hb // H
    h = pid_hb % H

    # ----- offsets for the current tile ------------------------------------
    i_base = pid_i * BLOCK_M
    j_base = pid_j * BLOCK_N

    # Base pointers for this (b,h) slice
    left_base  = left_ptr  + b * stride_lb + h * stride_lh
    right_base = right_ptr + b * stride_rb + h * stride_rh
    out_base   = out_ptr   + b * stride_ob + h * stride_oh

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ----- loop over reduction dimension k ---------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_off = k * BLOCK_K

        # left tile : [M, K]  (i, k)
        i_idx = i_base + tl.arange(0, BLOCK_M)[:, None]          # (M,1)
        k_idx_l = k_off + tl.arange(0, BLOCK_K)[None, :]         # (1,K)
        left_ptrs = left_base + i_idx * stride_li + k_idx_l * stride_lk
        mask_l = (i_idx < N) & (k_idx_l < N)
        left_tile = tl.load(left_ptrs, mask=mask_l, other=tl.float16(0))

        # right tile : [K, N]  (k, j) – we read transposed matrix
        k_idx_r = k_off + tl.arange(0, BLOCK_K)[:, None]         # (K,1)
        j_idx = j_base + tl.arange(0, BLOCK_N)[None, :]          # (1,N)
        right_ptrs = right_base + k_idx_r * stride_rk + j_idx * stride_rj
        mask_r = (k_idx_r < N) & (j_idx < N)
        right_tile = tl.load(right_ptrs, mask=mask_r, other=tl.float16(0))

        # FP16 dot → FP32 accumulation
        acc += tl.dot(left_tile, right_tile)   # (M,N)

    # ----- write the tile ---------------------------------------------------
    i_out = i_base + tl.arange(0, BLOCK_M)[:, None]
    j_out = j_base + tl.arange(0, BLOCK_N)[None, :]
    out_ptrs = out_base + i_out * stride_oi + j_out * stride_oj
    mask_o = (i_out < N) & (j_out < N)
    tl.store(out_ptrs, acc, mask=mask_o)


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul “outgoing” operator.

    Args:
        data (tuple): (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool/float) or None
            - weights: dict of all linear / layer‑norm parameters
            - config: dict with keys ``dim`` (C), ``hidden_dim`` (H) and
                      ``nomask`` (bool)

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32) – the TriMul forward output.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]                # C
    hidden_dim = config["hidden_dim"]  # H
    eps = 1e-5

    # ------------------------------------------------------------------ #
    # 1. Layer‑norm on the input (per channel)
    # ------------------------------------------------------------------ #
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )   # [B, N, N, C]  float32

    # ------------------------------------------------------------------ #
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------ #
    left  = F.linear(x, weights["left_proj.weight"])   # [B,N,N,H]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------ #
    # 3. Optional binary mask
    # ------------------------------------------------------------------ #
    if mask is not None and not config.get("nomask", False):
        mask_f = mask.unsqueeze(-1).to(left.dtype)   # broadcast over H
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------ #
    # 4. Gating (sigmoid)
    # ------------------------------------------------------------------ #
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # [B,N,N,H]
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))   # [B,N,N,H]

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------ #
    # 5. Core multiplicative update – Triton kernel (FP16, Tensor‑cores)
    # ------------------------------------------------------------------ #
    # Cast to FP16 to reduce memory traffic and enable Tensor‑cores
    left  = left.to(torch.float16)
    right = right.to(torch.float16)

    # Rearrange to [B, H, N, N] (contiguous)
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    B, H, N, _ = left_perm.shape

    # Output buffer (FP32 accumulation)
    out_perm = torch.empty((B, H, N, N), dtype=torch.float32, device=device)

    # Strides in element units (not bytes)
    stride_lb, stride_lh, stride_li, stride_lk = left_perm.stride()
    stride_rb, stride_rh, stride_rj, stride_rk = right_perm.stride()
    stride_ob, stride_oh, stride_oi, stride_oj = out_perm.stride()

    # Tile sizes – tuned for H100 (FP16 Tensor‑cores)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    tiles_m = (N + BLOCK_M - 1) // BLOCK_M
    tiles_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (tiles_m, tiles_n, B * H)   # 3‑D launch grid

    _trimul_matmul_kernel[grid](
        left_perm,
        right_perm,
        out_perm,
        B,
        H,
        N,
        stride_lb,
        stride_lh,
        stride_li,
        stride_lk,
        stride_rb,
        stride_rh,
        stride_rj,
        stride_rk,
        stride_ob,
        stride_oh,
        stride_oi,
        stride_oj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # ------------------------------------------------------------------ #
    # 6. Restore original layout [B, N, N, H]
    # ------------------------------------------------------------------ #
    out = out_perm.permute(0, 2, 3, 1)                # [B, N, N, H]  float32

    # ------------------------------------------------------------------ #
    # 7. Post‑norm, gating and final projection back to `dim`
    # ------------------------------------------------------------------ #
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate                                   # element‑wise
    out = F.linear(out, weights["to_out.weight"])          # → [B, N, N, C]

    return out
