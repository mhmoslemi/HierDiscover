"""
TriMul (outgoing) – highly‑fused forward pass.

Algorithm
---------
1. Layer‑norm over the input channels (dim)   →  x
2. Linear projection without bias:
      left  = x @ left_proj_weight    (B,N,N,hidden)
      right = x @ right_proj_weight   (B,N,N,hidden)
3. Optional mask (B,N,N) applied after the projections.
4. Three sigmoid gates (left_gate, right_gate, out_gate) computed with
   bias‑free linear layers and element‑wise multiplication.
5. Core multiplicative update (the N³ part):
      out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   This is a batched GEMM (left @ rightᵀ) for each hidden channel.
   Implemented with a custom Triton kernel that tiles the i‑j output
   plane (BLOCK_M × BLOCK_N) and accumulates over K in BLOCK_K‑wide
   slices.  All loads use FP16, the accumulation uses FP32, and the
   result is stored as FP16.  The kernel processes each (batch, hidden)
   pair independently (program‑id‑2).
6. Hidden‑dim layer‑norm, element‑wise out‑gate, final linear projection
   back to the original channel dimension.

The heavy N³ computation is off‑loaded to Triton; the rest of the
pipeline uses Torch primitives (highly‑optimised cuBLAS/LayerNorm).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel for the batched GEMM (left @ rightᵀ) per hidden channel
# ----------------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,   # pointers
    B, N, H,                        # problem size
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute out[b, i, j, h] = Σₖ left[b,i,k,h] * right[b,j,k,h].

    The kernel is launched with a 3‑D grid:
      * axis‑0 : tiles over the i dimension (rows)
      * axis‑1 : tiles over the j dimension (cols)
      * axis‑2 : (batch * hidden) pairs
    """
    pid_i = tl.program_id(0)            # tile row
    pid_j = tl.program_id(1)            # tile col
    pid_z = tl.program_id(2)            # batch*hidden index

    hidden_idx = pid_z % H
    batch_idx  = pid_z // H

    # Offsets of the current M×N tile
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # Base pointers for the selected batch & hidden channel
    a_ptr = left_ptr + batch_idx * stride_l_b + hidden_idx * stride_l_h
    b_ptr = right_ptr + batch_idx * stride_r_b + hidden_idx * stride_r_h
    c_ptr = out_ptr + batch_idx * stride_o_b + hidden_idx * stride_o_h

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        k_offs = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offs < N

        # Load A tile : shape (BLOCK_M, BLOCK_K)
        a_offs = a_ptr + offs_i[:, None] * stride_l_i + k_offs[None, :] * stride_l_k
        a = tl.load(
            a_offs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )  # FP16

        # Load Bᵀ tile : shape (BLOCK_K, BLOCK_N)
        # Bᵀ[k, j] = right[batch, j, k, hidden]
        b_offs = b_ptr + k_offs[:, None] * stride_r_k + offs_j[None, :] * stride_r_j
        b = tl.load(
            b_offs,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )  # FP16

        # FP16×FP16 → FP32 accumulation
        acc += tl.dot(a, b).to(tl.float32)

    # Write‑back (cast to FP16)
    c_offs = c_ptr + offs_i[:, None] * stride_o_i + offs_j[None, :] * stride_o_j
    tl.store(
        c_offs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


# ----------------------------------------------------------------------
# Entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul block.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor   [B, N, N, dim]  (float32)
        - mask         : torch.Tensor   [B, N, N]      (optional)
        - weights      : dict of torch.Tensor (model parameters)
        - config       : dict with keys ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor
        Output of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # 1️⃣ unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # --------------------------------------------------------------
    # 2️⃣ model parameters
    # --------------------------------------------------------------
    norm_weight          = weights["norm.weight"]
    norm_bias            = weights["norm.bias"]
    left_proj_weight     = weights["left_proj.weight"]   # (hidden_dim, dim)
    right_proj_weight    = weights["right_proj.weight"]
    left_gate_weight     = weights["left_gate.weight"]
    right_gate_weight    = weights["right_gate.weight"]
    out_gate_weight      = weights["out_gate.weight"]
    to_out_norm_weight   = weights["to_out_norm.weight"]
    to_out_norm_bias     = weights["to_out_norm.bias"]
    to_out_weight        = weights["to_out.weight"]      # (dim, hidden_dim)

    # --------------------------------------------------------------
    # 3️⃣ Input layer‑norm (over the last dim)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
    )   # (B, N, N, dim)

    # --------------------------------------------------------------
    # 4️⃣ Linear projections (bias‑free) → shape (B, N, N, hidden_dim)
    # --------------------------------------------------------------
    left  = F.linear(x, left_proj_weight)   # (B,N,N,hidden)
    right = F.linear(x, right_proj_weight)  # (B,N,N,hidden)

    # --------------------------------------------------------------
    # 5️⃣ Optional mask (mask is broadcast along hidden_dim)
    # --------------------------------------------------------------
    if mask is not None and mask.numel() > 0:
        # expand to (B,N,N,1) and cast to same dtype as left/right
        mask_exp = mask.to(left.dtype).unsqueeze(-1)
        left  = left * mask_exp
        right = right * mask_exp

    # --------------------------------------------------------------
    # 6️⃣ Gates (sigmoid of linear layers) – same shape as left/right
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_weight))

    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 7️⃣ Core multiplicative update – Triton GEMM
    # --------------------------------------------------------------
    # Cast to FP16 for bandwidth‑friendly loads
    left_fp16  = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # Allocate output buffer (FP16)
    out_fp16 = torch.empty_like(left_fp16)

    # Tile sizes – tuned for H100 tensor‑core utilisation
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    # Compute strides in element units (not bytes)
    stride_l_b = left_fp16.stride(0)
    stride_l_i = left_fp16.stride(1)
    stride_l_k = left_fp16.stride(2)
    stride_l_h = left_fp16.stride(3)

    stride_r_b = right_fp16.stride(0)
    stride_r_j = right_fp16.stride(1)   # “row” after transpose
    stride_r_k = right_fp16.stride(2)
    stride_r_h = right_fp16.stride(3)

    stride_o_b = out_fp16.stride(0)
    stride_o_i = out_fp16.stride(1)
    stride_o_j = out_fp16.stride(2)
    stride_o_h = out_fp16.stride(3)

    # Grid dimensions: (tiles_i, tiles_j, batch*hidden)
    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        bs * hidden_dim,
    )

    # Launch Triton kernel
    trimul_outgoing_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        bs,
        N,
        hidden_dim,
        stride_l_b,
        stride_l_i,
        stride_l_k,
        stride_l_h,
        stride_r_b,
        stride_r_j,
        stride_r_k,
        stride_r_h,
        stride_o_b,
        stride_o_i,
        stride_o_j,
        stride_o_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Back‑to FP32 for the subsequent layers
    out = out_fp16.to(torch.float32)   # (B,N,N,hidden)

    # --------------------------------------------------------------
    # 8️⃣ Hidden‑dim layer‑norm, out‑gate, final projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )
    out = out * out_gate
    out = F.linear(out, to_out_weight)   # → (B,N,N,dim)

    return out
