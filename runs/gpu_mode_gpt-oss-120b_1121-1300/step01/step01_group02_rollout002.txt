"""TriMul outgoing kernel.

Implements the forward pass of the TriMul module from AlphaFold3.
The core O(N³) multiplicative update
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is realized as a batched GEMM (A @ Bᵀ) per hidden channel using a customized Triton kernel.
All other layers (LayerNorm, linear projections, sigmoid gates) are kept in PyTorch.
The kernel works in FP16 for inputs/outputs and accumulates in FP32, handling arbitrary
batch size (≤2), sequence length (≤1024) and hidden dimension (≤128) while correctly
supporting optional masking.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Batched GEMM for the TriMul multiplicative update.
    
    For each (batch, hidden) slice computes:
        out[i, j] = Σ_k left[i, k] * right[j, k]
    which is exactly A @ Bᵀ where A = left, B = right.
    
    Parameters
    ----------
    left_ptr, right_ptr, out_ptr : pointers to fp16 tensors
        Layout: (B, N, N, H) – contiguous in memory.
    B, N, H : int
        Batch, sequence length, hidden dimension.
    stride_* : int
        Strides (in elements) for each dimension.
    BLOCK_M, BLOCK_N, BLOCK_K : compile‑time block sizes.
    """
    pid_m = tl.program_id(0)   # tile row (i)
    pid_n = tl.program_id(1)   # tile col (j)
    pid_z = tl.program_id(2)   # batch * hidden index

    # Decode batch and hidden channel
    hidden_idx = pid_z % H
    batch_idx  = pid_z // H

    # Compute offsets for the tile
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Base pointers for this (batch, hidden) slice
    a_base = left_ptr + batch_idx * stride_left_b + hidden_idx * stride_left_h
    b_base = right_ptr + batch_idx * stride_right_b + hidden_idx * stride_right_h
    c_base = out_ptr + batch_idx * stride_out_b + hidden_idx * stride_out_h

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        k_offs = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offs < N

        # Load A tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = a_base + offs_i[:, None] * stride_left_i + k_offs[None, :] * stride_left_k
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)   # fp16

        # Load Bᵀ tile: shape (BLOCK_K, BLOCK_N)
        # Bᵀ[k, j] = right[batch, j, k, hidden]
        b_ptrs = b_base + k_offs[:, None] * stride_right_k + offs_j[None, :] * stride_right_j
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)   # fp16

        # FP16 dot, accumulate in FP32
        acc += tl.dot(a, b)    # result promoted to FP32 automatically

    # Write back the tile
    c_ptrs = c_base + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])

    """
    Forward pass for the outgoing TriMul module.

    Parameters
    ----------
    data : tuple

"""
TriMul outgoing kernel.

Implements the forward pass of the TriMul module (AlphaFold3 “outgoing’’
variant).  The expensive N³ multiplicative update

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

is reduced to a batched matrix multiplication (A @ Bᵀ) per hidden channel
and executed with a custom Triton kernel.  All other layers
(LayerNorm, linear projections, sigmoid gates) are performed in PyTorch.
The kernel works in FP16 for inputs/outputs, accumulates in FP32, and
handles optional masking, arbitrary batch size (≤2), sequence length
(≤1024) and hidden dimension (≤384).  It is tuned for H100 GPUs.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Batched GEMM for the TriMul multiplicative update.

    For each (batch, hidden) slice computes:
        out[i, j] = Σ_k left[i, k] * right[j, k]
    i.e. a matrix multiplication between left (N×N) and rightᵀ (N×N).

    Arguments
    ---------
    left_ptr, right_ptr, out_ptr : pointers to fp16 tensors
        Layout = (B, N, N, H) – contiguous.
    B, N, H : int
        Batch size, sequence length, hidden dimension.
    stride_* : int
        Element‑wise strides for each dimension.
    BLOCK_M, BLOCK_N, BLOCK_K : compile‑time tile sizes.
    """
    pid_m = tl.program_id(0)   # tile row (i)
    pid_n = tl.program_id(1)   # tile col (j)
    pid_z = tl.program_id(2)   # batch * hidden index

    # decode batch and hidden channel from pid_z
    hidden_idx = pid_z % H
    batch_idx  = pid_z // H

    # tile offsets
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # base pointers for this (batch, hidden) slice
    a_base = left_ptr + batch_idx * stride_left_b + hidden_idx * stride_left_h
    b_base = right_ptr + batch_idx * stride_right_b + hidden_idx * stride_right_h
    c_base = out_ptr + batch_idx * stride_out_b + hidden_idx * stride_out_h

    # accumulator (FP32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        k_offs = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offs < N

        # load A tile  (M × K)
        a_ptrs = a_base + offs_i[:, None] * stride_left_i + k_offs[None, :] * stride_left_k
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)   # fp16

        # load Bᵀ tile (K × N)
        # Bᵀ[k, j] = right[batch, j, k, hidden]
        b_ptrs = b_base + k_offs[:, None] * stride_right_k + offs_j[None, :] * stride_right_j
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0)   # fp16

        # dot product, accumulate in FP32
        acc += tl.dot(a, b)          # result automatically promoted to FP32

    # store tile (cast back to fp16)
    c_ptrs = c_base + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j
    tl.store(c_ptrs,
             acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])
