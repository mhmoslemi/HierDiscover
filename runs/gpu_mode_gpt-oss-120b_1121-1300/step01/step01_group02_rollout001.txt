"""
TriMul outgoing forward pass (AlphaFold3 style)

The heavy N³ “multiplicative update”

    out[b, i, j, d] = Σₖ left[b, i, k, d] * right[b, j, k, d]

is a batched matrix‑multiply (L @ Rᵀ) performed independently for each
hidden channel *d*.  We fuse this operation into a single Triton kernel.
All other parts (LayerNorms, linear projections, gating, final projection)
are kept in PyTorch for simplicity.

The kernel works in FP16 to reduce memory traffic and accumulates in FP32.
The output of the kernel is therefore FP16 and is cast back to FP32 before
the remaining layers.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def trimul_outgoing_kernel(
    left_ptr,               # FP16 left   [B,N,N,H]
    right_ptr,              # FP16 right  [B,N,N,H]
    out_ptr,                # FP16 out    [B,N,N,H]
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Batched GEMM for TriMul:
        out[i, j] = Σₖ left[i, k] * right[j, k]    (per batch & hidden channel)
    """
    pid_m = tl.program_id(0)          # tile row   (i)
    pid_n = tl.program_id(1)          # tile col   (j)
    pid_z = tl.program_id(2)          # batch * hidden channel

    # decode batch and hidden index
    hidden_idx = pid_z % H
    batch_idx  = pid_z // H

    # i‑ and j‑coordinates of the current tile
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # base pointers for this (batch, hidden) pair
    a_base = left_ptr  + batch_idx * stride_left_b  + hidden_idx * stride_left_h
    b_base = right_ptr + batch_idx * stride_right_b + hidden_idx * stride_right_h
    c_base = out_ptr   + batch_idx * stride_out_b   + hidden_idx * stride_out_h

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension in BLOCK_K slices
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        k_offs = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offs < N

        # left tile   (M × K)
        a_ptrs = a_base + offs_i[:, None] * stride_left_i + k_offs[None, :] * stride_left_k
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float16)

        # rightᵀ tile (K × N)   – note the transpose via strides
        b_ptrs = b_base + k_offs[:, None] * stride_right_k + offs_j[None, :] * stride_right_j
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0).to(tl.float16)

        # accumulate in FP32
        acc += tl.dot(a, b).to(tl.float32)

    # write result (cast back to FP16)
    c_ptrs = c_base + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, dim]
        - mask         : torch.Tensor of shape [B, N, N] or ``None``
        - weights      : dict of torch.Tensors containing model parameters
        - config       : dict with keys ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # model parameters
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]

    left_proj_weight  = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]

    left_gate_weight  = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight   = weights["out_gate.weight"]

    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    to_out_weight      = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1. Input LayerNorm over the last dimension
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )   # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias) → (B, N, N, hidden_dim)
    # ------------------------------------------------------------------
    left  = F.linear(x, left_proj_weight)   # (B, N, N, H)
    right = F.linear(x, right_proj_weight)  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast on hidden dimension)
    # ------------------------------------------------------------------
    if mask is None:
        mask = torch.ones((bs, N, N), dtype=x.dtype, device=x.device)
    mask = mask.to(x.dtype).unsqueeze(-1)   # (B, N, N, 1)
    left = left * mask
    right = right * mask

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid) – computed from the normalized tensor ``x``
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_weight))   # (B,N,N,H)
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))  # (B,N,N,H)
    out_gate   = torch.sigmoid(F.linear(x, out_gate_weight))    # (B,N,N,H)

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core TriMul update (batched GEMM) – Triton kernel
    # ------------------------------------------------------------------
    # Cast to fp16 for the kernel (memory‑bandwidth friendly)
    left_fp16  = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()
    out_fp16   = torch.empty_like(left_fp16)   # (B,N,N,H)

    # Triton block size – works well on H100 (tune if needed)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # strides in element units (not bytes)
    stride_left_b = left_fp16.stride(0)
    stride_left_i = left_fp16.stride(1)
    stride_left_k = left_fp16.stride(2)
    stride_left_h = left_fp16.stride(3)

    stride_right_b = right_fp16.stride(0)
    stride_right_j = right_fp16.stride(1)   # "row" after transpose
    stride_right_k = right_fp16.stride(2)
    stride_right_h = right_fp16.stride(3)

    stride_out_b = out_fp16.stride(0)
    stride_out_i = out_fp16.stride(1)
    stride_out_j = out_fp16.stride(2)
    stride_out_h = out_fp16.stride(3)

    # launch grid: (tiles over i, tiles over j, batch*hidden)
    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        bs * hidden_dim,
    )

    trimul_outgoing_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        bs,
        N,
        hidden_dim,
        stride_left_b,
        stride_left_i,
        stride_left_k,
        stride_left_h,
        stride_right_b,
        stride_right_j,
        stride_right_k,
        stride_right_h,
        stride_out_b,
        stride_out_i,
        stride_out_j,
        stride_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # back to FP32 for the rest of the pipeline
    out = out_fp16.to(torch.float32)   # (B,N,N,H)

    # ------------------------------------------------------------------
    # 6. Hidden‑dim LayerNorm, out‑gate, final linear projection to dim
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )
    out = out * out_gate
    out = F.linear(out, to_out_weight)   # (B, N, N, dim)

    return out
