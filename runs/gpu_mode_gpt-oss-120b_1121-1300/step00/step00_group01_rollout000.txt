def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_bh = tl.program_id(2)
    b = pid_bh // H
    h = pid_bh % H

    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)

    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(N, BLOCK_K)):

"""
TriMul (outgoing) forward pass implemented with a fused Triton kernel.

Algorithm
---------
1. Layer‑norm the input tensor x ∈ ℝ^{B×N×N×C}.
2. Linear projections → left, right ∈ ℝ^{B×N×N×H}.
3. Optional mask (broadcast on the channel dim) is applied to left/right.
4. Compute three sigmoidal gates (left_gate, right_gate, out_gate) from x.
5. Apply left/right gates: left←left·left_gate, right←right·right_gate.
6. Compute the “triangle multiplicative” contraction
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   for every batch b and hidden head h.  This is a batched
   matrix‑multiply (N×N)·(N×N)ᵀ for each head.  The heavy N³ work
   is performed inside a Triton kernel that tiles the i/j dimensions
   (BLOCK_M × BLOCK_N) and streams over k (BLOCK_K).  The kernel
   loads left and the transpose of right (so the dot uses
   tl.dot) and accumulates in FP32, finally storing the result in
   FP16 to reduce memory traffic.
7. Convert the intermediate result to FP32 and apply a second
   Layer‑norm (to_out_norm) over the hidden dimension.
8. Multiply by out_gate (broadcast) and a final linear layer to
   recover the original channel dimension C.
9. Return the tensor of shape [B, N, N, C] (float32).

The implementation mixes PyTorch (layer‑norms, linear layers, gating)
with a custom Triton kernel for the O(N³) contraction, satisfying the
requirement that “part of the operations” be done on‑device.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel for the core contraction:
#   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr,               # const * half   (B,N,N,H) – left projection after gating
    right_ptr,              # const * half   (B,N,N,H) – right projection after gating
    out_ptr,                # * half         (B,N,N,H) – destination
    B, N, H,                # int32 scalars
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr,  # tile size for i dimension
    BLOCK_N: tl.constexpr,  # tile size for j dimension
    BLOCK_K: tl.constexpr,  # tile size for k dimension
):
    # ------------------------------------------------------------------
    # grid dimensions:
    #   axis 0 -> i‑tile,
    #   axis 1 -> j‑tile,
    #   axis 2 -> (batch * head) linear index
    # ------------------------------------------------------------------
    pid_bh = tl.program_id(2)
    b = pid_bh // H                     # batch index
    h = pid_bh % H                      # head index

    pid_i = tl.program_id(0)            # i‑tile index
    pid_j = tl.program_id(1)            # j‑tile index

    # i/j coordinates for this tile
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32 for precision
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over the reduction axis k in BLOCK_K sized chunks
    # ------------------------------------------------------------------
    for k_iter in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k_iter * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # -------- load left[b, i, k, h]  (shape BLOCK_M × BLOCK_K) ----------
        # left layout: (B, i, k, H)
        left_ptrs = (
            left_ptr
            + b * stride_l_b
            + h * stride_l_h
            + offs_i[:, None] * stride_l_i
            + offs_k[None, :] * stride_l_k
        )
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0,
                       dtype=tl.float16)

        # -------- load rightᵀ[b, k, j, h]  (shape BLOCK_K × BLOCK_N) ----------
        # right layout: (B, j, k, H).  We want its transpose in k‑j order.
        right_ptrs = (
            right_ptr
            + b * stride_r_b
            + h * stride_r_h
            + offs_k[:, None] * stride_r_k   # k   → row
            + offs_j[None, :] * stride_r_j   # j   → column
        )
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0,
                        dtype=tl.float16)

        # Promote to FP32, perform dot, accumulate
        acc += tl.dot(left.to(tl.float32), right.to(tl.float32))

    # ------------------------------------------------------------------
    # Store the result (cast back to FP16 to match the output tensor dtype)
    # ------------------------------------------------------------------
    out_ptrs = (
        out_ptr
        + b * stride_o_b
        + h * stride_o_h
        + offs_i[:, None] * stride_o_i
        + offs_j[None, :] * stride_o_j
    )
    tl.store(out_ptrs, acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Public entry‑point that follows the required signature
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask_tensor, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32)
            - mask_tensor:  torch.Tensor of shape [B, N, N] (float32 or bool)
            - weights:      dict with all model parameters (float32 tensors)
            - config:       dict containing at least:
                * "dim"          – input channel dimension C
                * "hidden_dim"   – hidden dimension H
                * "nomask"       – bool, True if no masking should be applied

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    B, N, N2, C = input_tensor.shape
    assert N == N2, "Input must be square on the two middle axes."
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------------------
    # 1) First LayerNorm over channel dimension C
    # --------------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = F.layer_norm(input_tensor,
                     normalized_shape=(C,),
                     weight=norm_weight,
                     bias=norm_bias,
                     eps=1e-5)

    # --------------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # --------------------------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"]   # (H, C)
    right_proj_w = weights["right_proj.weight"]  # (H, C)

    left  = F.linear(x, left_proj_w)   # (B, N, N, H)
    right = F.linear(x, right_proj_w)  # (B, N, N, H)

    # --------------------------------------------------------------------------
    # 3) Optional mask (broadcast on the hidden dimension)
    # --------------------------------------------------------------------------
    if not nomask:
        # mask may be bool or float; make it float for multiplication
        mask = mask_tensor.to(dtype).unsqueeze(-1)          # (B, N, N, 1)
        left  = left * mask
        right = right * mask

    # --------------------------------------------------------------------------
    # 4) Compute three sigmoid gates from the *original* normalized x
    # --------------------------------------------------------------------------
    left_gate_w  = weights["left_gate.weight"]   # (H, C)
    right_gate_w = weights["right_gate.weight"]  # (H, C)
    out_gate_w   = weights["out_gate.weight"]    # (H, C)

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))   # (B,N,N,H)
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    # --------------------------------------------------------------------------
    # 5) Apply the gates to the projected tensors
    # --------------------------------------------------------------------------
    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------------------
    # 6) Core TriMul contraction via Triton
    # --------------------------------------------------------------------------
    # Cast to FP16 before the kernel to reduce memory traffic.
    left_fp16  = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # Allocate output tensor for the hidden representation
    out_hidden_fp16 = torch.empty((B, N, N, hidden_dim),
                                 dtype=torch.float16,
                                 device=device)

    # Strides required by the kernel (in elements, not bytes)
    s_l_b, s_l_i, s_l_k, s_l_h = left_fp16.stride()
    s_r_b, s_r_j, s_r_k, s_r_h = right_fp16.stride()
    s_o_b, s_o_i, s_o_j, s_o_h = out_hidden_fp16.stride()

    # Tunable block sizes (empirically good on H100)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,               # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,               # j‑tiles
        B * hidden_dim                              # (batch, head) dimension
    )

    # Launch the Triton kernel
    tri_mul_kernel[grid](
        left_fp16,
        right_fp16,
        out_hidden_fp16,
        B, N, hidden_dim,
        s_l_b, s_l_i, s_l_k, s_l_h,
        s_r_b, s_r_j, s_r_k, s_r_h,
        s_o_b, s_o_i, s_o_j, s_o_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,        # hardware‑friendly default
    )

    # Convert back to FP32 for the remaining FP32‑only ops
    out_hidden = out_hidden_fp16.to(torch.float32)

    # --------------------------------------------------------------------------
    # 7) Second LayerNorm over the hidden dimension H
    # --------------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out = F.layer_norm(out_hidden,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_w,
                       bias=to_out_norm_b,
                       eps=1e-5)

    # --------------------------------------------------------------------------
    # 8) Multiply by the output gate and final linear projection back to C
    # --------------------------------------------------------------------------
    out = out * out_gate                     # (B,N,N,H) element‑wise
    to_out_w = weights["to_out.weight"]      # (C, H)
    out = F.linear(out, to_out_w)            # (B,N,N,C)

    return out
```
