    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_bnh, stride_left_nkh, stride_left_h,  # Actually stride for left: dims order [B,N,N,H]

    stride_out_bnh, stride_out_njh, stride_out_h,

):

   pid = tl.program_id(axis=0)
   # Compute combined batch index = b * H + h
   total_hw = B * H

def custom_kernel(data):
    input_tensor, mask, weights, config = data

    dim = config["dim"]; hidden_dim = config["hidden_dim"]; nomask = config.get("nomask", True) # might be given in config.

    # Set dtype to float16 for intermediate linear transforms to reduce memory bandwidth.
    # Compute layernorm
    eps = 1e-5

    # container for weight and bias; all weight shapes as provided.

    # Perform layernorm manually:
    x = input_tensor
    # compute mean and var across dim axis=-1 (dim)
    x_shape = x.shape
    # compute mean/var using torch.mean and var along last dim, keepdim
    mean = x.mean(-1, keepdim=True)
    var = x.var(-1, unbiased=False, keepdim=True)
    x_norm = (x - mean) / torch.sqrt(var + eps)
    if norm_weight is not None:
        x_norm = x_norm * norm_weight + norm_bias   # broadcast weight shape [dim]; bias shape [dim]
    
    # Linear transforms - we'll cast to float16
    # define linear function:
    def linear(x, weight):
        # weight shape [out, in]; x shape [B,N,N,in]
        # compute x @ weight.T, using torch.einsum or torch.matmul? 
        # x shape (B,N,N,in)
        # weight shape (out,in)
        # perform torch.tensordot: out = torch.einsum('...i,oi->...o', x, weight)
        # but we can use F.linear: torch.nn.functional.linear(x, weight) where weight shape (out,in) automatically.
        return torch.nn.functional.linear(x, weight)  # bias not used
    
    left = linear(x_norm, left_proj_weight)   # shape [B,N,N,hidden_dim]
    right = linear(x_norm, right_proj_weight)

    left_gate = torch.sigmoid(linear(x_norm, left_gate_weight))
    right_gate = torch.sigmoid(linear(x_norm, right_gate_weight))
    out_gate = torch.sigmoid(linear(x_norm, out_gate_weight))

    # apply mask if provided and not nomask
    if not config.get("nomask", False):
        mask = mask.to(x.dtype).unsqueeze(-1)   # [B,N,N,1]
        left = left * mask
        right = right * mask

    # apply gates
    left = left * left_gate
    right = right * right_gate

    # cast to half for kernel
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    B, N, _, H = left_fp16.shape

    # Prepare output buffer
    out_h = torch.empty((B, N, N, H), dtype=torch.float32, device=left_fp16.device)

    # Compute strides in elements (not bytes). In Triton, stride args are number of elements (not bytes) by default typical; but they require in terms of number of elements (like pointer arithmetic). Triton expects stride in number of elements (not bytes). Provide strides as int.

    # Compute strides for left/right/out using .stride()
    # left_strides = left_fp16.stride() yields tuple (B_stride, N1_stride, N2_stride, H_stride)
    # For dims (B,N,N,H): stride_b, stride_i, stride_k, stride_h
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left_fp16.stride()
    stride_right_b, stride_right_i, stride_right_k, stride_right_h = right_fp16.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out_h.stride()

    # Launch kernel
    # Determine block sizes
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32
    # Compute grid
    total_hw = B * H
    grid = (total_hw, (N + BLOCK_M - 1) // BLOCK_M, (N + BLOCK_N - 1) // BLOCK_N)
    # Call kernel
    trimul_einsum_kernel[grid](
        left_fp16, right_fp16, out_h,
        B, N, H,
        stride_left_b, stride_left_i, stride_left_k, stride_left_h,
        stride_right_b, stride_right_i, stride_right_k, stride_right_h,
        stride_out_b, stride_out_i, stride_out_j, stride_out_h,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )

    # After kernel, apply to_out_norm and out_gate and linear to output dim.

    # to_out_norm: LayerNorm on hidden dim H
    # use weight and bias for to_out_norm (size hidden_dim)
    # apply to_out_norm:
    out_norm = (out_h - out_h.mean(-1, keepdim=True)) / torch.sqrt(out_h.var(-1, unbiased=False, keepdim=True) + eps)
    out_norm = out_norm * to_out_norm_weight + to_out_norm_bias   # broadcast

    # Multiply by out_gate
    out_norm = out_norm * out_gate

    # Linear to original dim: to_out is weight [dim, hidden_dim]
    out = torch.nn.functional.linear(out_norm, to_out_weight)  # shape [B,N,N,dim] as bias=0

    return out

"""
TriMul “outgoing” kernel (AlphaFold3 style)

The operator works on a 4‑D tensor `x` of shape [B, N, N, C] (C = dim).
The forward pass consists of

1. Layer‑norm on the last dimension.
2. Linear projections for left/right values and three sigmoidal gates.
3. Optional masking of left/right.
4. Gating of left/right values.
5. A massive N×N×N contraction:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   which is a batched matrix‑multiply L_d @ R_dᵀ for each hidden dim d.
6. Layer‑norm on the hidden dimension, gating, and a final linear map
   back to the original dimension.

The heavy N³ contraction (step 5) is implemented with a custom Triton
kernel that treats each pair (batch, hidden) as an independent GEMM.
The kernel loops over the K‑dimension in tiles, loads FP16 tiles,
accumulates in FP32, and writes the result back in FP32.
All other operations are performed in PyTorch (float32) for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# -------------------------------------------------------------------------
# Triton kernel for the einsum part:
#   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
# -------------------------------------------------------------------------
@triton.jit
def trimul_einsum_kernel(
    # pointers
    left_ptr, right_ptr, out_ptr,
    # problem size
    B, N, H,
    # strides (in #elements, not bytes)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    # compile‑time block sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # -------------------------------
    # program IDs
    # -------------------------------
    pid_batch = tl.program_id(0)   # combined batch * hidden dimension
    pid_i = tl.program_id(1)       # block row   (i dimension)
    pid_j = tl.program_id(2)       # block col   (j dimension)

    # decode combined batch index into (batch, hidden)
    b = pid_batch // H
    h = pid_batch % H

    # ---------------------------------
    # block offsets
    # ---------------------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K tiles
    for k_block in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k_block * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # -------------------------
        # load left tile (M x K)
        # -------------------------
        left_offset = (
            left_ptr
            + b * stride_left_b
            + h * stride_left_h
            + offs_i[:, None] * stride_left_i
            + offs_k[None, :] * stride_left_k
        )
        left = tl.load(
            left_offset,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )

        # -------------------------
        # load right tile (N x K)
        # -------------------------
        right_offset = (
            right_ptr
            + b * stride_right_b
            + h * stride_right_h
            + offs_j[:, None] * stride_right_i
            + offs_k[None, :] * stride_right_k
        )
        right = tl.load(
            right_offset,
            mask=mask_j[:, None] & mask_k[None, :],
            other=0.0,
        )

        # -------------------------
        # dot product over K
        # -------------------------
        acc += tl.dot(left, right)   # (M,N) result accumulated in FP32

    # -------------------------
    # store result
    # -------------------------
    out_offset = (
        out_ptr
        + b * stride_out_b
        + h * stride_out_h
        + offs_i[:, None] * stride_out_i
        + offs_j[None, :] * stride_out_j
    )
    tl.store(
        out_offset,
        acc,
        mask=mask_i[:, None] & mask_j[None, :],
    )


# -------------------------------------------------------------------------
# Main entry point expected by the evaluator
# -------------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” module.
    Arguments:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)
    Returns:
        Tensor of shape [B, N, N, dim] (float32) on the same device.
    """
    # Unpack inputs
    input_tensor, mask_tensor, weights, config = data
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    dim = config["dim"]        # same as input last dim; kept for clarity
    nomask = config.get("nomask", True)

    # -----------------------------------------------------------------
    # 1. LayerNorm on input (dim)
    # -----------------------------------------------------------------
    eps = 1e-5
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    # torch.nn.functional.layer_norm expects normalized_shape = (dim,)
    x_norm = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )   # [B,N,N,dim]  (float32)

    # -----------------------------------------------------------------
    # 2. Linear projections (no bias) and gates (with sigmoid)
    # -----------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"]   # (hidden_dim, dim)
    right_proj_w  = weights["right_proj.weight"]
    left_gate_w   = weights["left_gate.weight"]
    right_gate_w  = weights["right_gate.weight"]
    out_gate_w    = weights["out_gate.weight"]
    to_out_norm_w = weights["to_out_norm.weight"]  # (hidden_dim,)
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w      = weights["to_out.weight"]       # (dim, hidden_dim)

    # linear without bias
    left = F.linear(x_norm, left_proj_w)          # [B,N,N,hidden_dim]
    right = F.linear(x_norm, right_proj_w)

    # sigmoid gates
    left_gate  = torch.sigmoid(F.linear(x_norm, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_norm, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x_norm, out_gate_w))

    # -----------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden dim)
    # -----------------------------------------------------------------
    if not nomask:
        # mask shape [B,N,N] -> [B,N,N,1] for broadcasting
        mask = mask_tensor.to(x_norm.dtype).unsqueeze(-1)
        left = left * mask
        right = right * mask

    # -----------------------------------------------------------------
    # 4. Apply element‑wise gates
    # -----------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5. Compute the N³ contraction via Triton
    # -----------------------------------------------------------------
    # Cast to FP16 for better bandwidth, keep result FP32
    left_fp16 = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()

    # Output buffer (FP32)
    out_hidden = torch.empty((B, N, N, hidden_dim), dtype=torch.float32,
                             device=left_fp16.device)

    # Strides in #elements (not bytes) for Triton
    s_left_b, s_left_i, s_left_k, s_left_h = left_fp16.stride()
    s_right_b, s_right_i, s_right_k, s_right_h = right_fp16.stride()
    s_out_b, s_out_i, s_out_j, s_out_h = out_hidden.stride()

    # Tile sizes – chosen to fit H100 tensor‑core resources
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32

    # Grid: (combined batch*hidden, i‑tiles, j‑tiles)
    grid = (
        B * hidden_dim,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    trimul_einsum_kernel[grid](
        left_fp16,
        right_fp16,
        out_hidden,
        B,
        N,
        hidden_dim,
        s_left_b,
        s_left_i,
        s_left_k,
        s_left_h,
        s_right_b,
        s_right_i,
        s_right_k,
        s_right_h,
        s_out_b,
        s_out_i,
        s_out_j,
        s_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # -----------------------------------------------------------------
    # 6. Post‑processing: hidden‑dim LayerNorm, gating, final linear
    # -----------------------------------------------------------------
    # Hidden‑dim LayerNorm (on last dimension = hidden_dim)
    out_ln = F.layer_norm(
        out_hidden,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=eps,
    )   # [B,N,N,hidden_dim]

    # Apply output gate
    out_gated = out_ln * out_gate

    # Final linear back to original dim (no bias)
    output = F.linear(out_gated, to_out_w)   # [B,N,N,dim] (float32)

    return output
