Pseudo-code:

```
def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]; hidden_dim = config["hidden_dim"]

    # 1. LayerNorm (weight and bias)
    # Compute mean and var across dim (last dim) for each [bs, seq_len, seq_len, dim] slice.
    # Use torch.nn.functional.layer_norm
    norm_weight = weights["norm.weight"]; norm_bias = weights["norm.bias"]
    eps = 1e-5
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # 2. Linear projections
    left_proj = torch.nn.functional.linear(x, weights["left_proj.weight"]) # shape [B,N,N,H]
    right_proj = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # Mask handling: if mask is not None (maybe shape [B,N,N])
    mask = mask_tensor
    if mask is not None:
        mask = mask[..., None]  # broadcast to hidden dim
        left_proj = left_proj * mask
        right_proj = right_proj * mask

    # 3. Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    left = left_proj * left_gate
    right = right_proj * right_gate

    # 4. TriMul via Triton kernel
    out_hidden = triton_tri_mul_outgoing(left, right) # shape [B,N,N,H]

    # 5. to_out_norm
    out = torch.nn.functional.layer_norm(out_hidden, (hidden_dim,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"], eps=eps)

    # 6. Gating
    out = out * out_gate

    # 7. Linear to dim
    out = torch.nn.functional.linear(out, weights["to_out.weight"]) # shape [B,N,N,dim]

    return out
```

import triton.language as tl

def tri_mul_outgoing_triton(left, right, BLOCK_M=128, BLOCK_N=128, BLOCK_K=32):
    # left,right: torch tensors shape [B,N,N,H] dtype float16
    B, N, _, H = left.shape
    out = torch.empty((B, N, N, H), dtype=torch.float32, device=left.device)

@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_batch, stride_l_i, stride_l_k, stride_l_h,
    stride_r_batch, stride_r_i, stride_r_k, stride_r_h,
    stride_o_batch, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0)
    total_bh = B * H
    num_pid_m = (N + BLOCK_M - 1) // BLOCK_M
    num_pid_n = (N + BLOCK_N - 1) // BLOCK_N
    bh = pid // (num_pid_m * num_pid_n)
    pid_in_bh = pid % (num_pid_m * num_pid_n)
    pid_m = pid_in_bh // num_pid_n
    pid_n = pid_in_bh % num_pid_n

    # decode batch and hidden dim
    b = bh // H
    h = bh % H

    # offsets
    i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # k tiles loop
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # compute boundaries for i and j
    i_valid = i < N
    j_valid = j < N

    # pointer to the start of the batch and hidden dimension
    # left base = b*stride_l_batch + h*stride_l_h
    left_base = b * stride_l_batch + h * stride_l_h
    right_base = b * stride_r_batch + h * stride_r_h
    out_base = b * stride_o_batch + h * stride_o_h

    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k_tile * BLOCK_K
        k = k_offset + tl.arange(0, BLOCK_K)  # shape (BLOCK_K,)
        k_valid = k < N

        # pointers for left block: shape (BLOCK_M, BLOCK_K)
        a_ptrs = left_ptr + left_base + i[:, None] * stride_l_i + k[None, :] * stride_l_k
        a_mask = i_valid[:, None] & k_valid[None, :]
        a = tl.load(a_ptrs, mask=a_mask, other=0.0).to(tl.float32)

        # pointers for right block (transposed): shape (BLOCK_K, BLOCK_N)
        # we need right[j,k], but we load with first axis k, second axis j.
        b_ptrs = right_ptr + right_base + k[:, None] * stride_r_k + j[None, :] * stride_r_i
        b_mask = k_valid[:, None] & j_valid[None, :]
        b = tl.load(b_ptrs, mask=b_mask, other=0.0).to(tl.float32)

        acc += tl.dot(a, b)

    # store result
    out_ptrs = out_ptr + out_base + i[:, None] * stride_o_i + j[None, :] * stride_o_j
    out_mask = i_valid[:, None] & j_valid[None, :]
    tl.store(out_ptrs, acc, mask=out_mask)

"""
TriMul (outgoing) forward pass implemented with Triton.

Algorithm
---------
1. Layer‑norm the input tensor over the channel dimension (dim).
2. Linear projections (left_proj, right_proj) from dim → hidden_dim.
3. Apply an optional mask (broadcasted over hidden_dim).
4. Compute three gates (left_gate, right_gate, out_gate) as sigmoid(linear(dim→hidden_dim)).
5. Fuse the gates with the projected tensors:
        left  = left  * left_gate
        right = right * right_gate
6. Core operation: for every batch b and hidden channel h,
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix‑multiply A @ Bᵀ where A and B have shape (N,N)
   and the batch dimension is B·hidden_dim.  The heavy computation is
   performed in a fused Triton kernel using fp16 arithmetic with
   fp32 accumulation.
7. Layer‑norm over hidden_dim, multiply by out_gate and finally a linear
   projection back to dim.

The Triton kernel tiles the (i,j) output plane with blocks of size
(128×128) and iterates over k in 32‑element tiles.  Masks handle the
edge cases where N is not a multiple of the tile size.  The kernel
operates on fp16 inputs (left/right) to exploit Tensor‑cores on an
H100 and writes the result in fp32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _tri_mul_outgoing_kernel(
    left_ptr,                     # fp16 *
    right_ptr,                    # fp16 *
    out_ptr,                      # fp32 *
    B, N, H,                      # int64
    stride_l_batch, stride_l_i, stride_l_k, stride_l_h,   # int64
    stride_r_batch, stride_r_i, stride_r_k, stride_r_h,   # int64
    stride_o_batch, stride_o_i, stride_o_j, stride_o_h,   # int64
    BLOCK_M: tl.constexpr,        # tile size in i dimension
    BLOCK_N: tl.constexpr,        # tile size in j dimension
    BLOCK_K: tl.constexpr,        # tile size in k dimension
):
    pid = tl.program_id(0)

    # ----------- split program id into (batch·hidden), i‑tile, j‑tile ----------
    total_bh = B * H                                    # number of (b,h) combos
    num_pid_m = (N + BLOCK_M - 1) // BLOCK_M           # tiles along i
    num_pid_n = (N + BLOCK_N - 1) // BLOCK_N           # tiles along j

    bh = pid // (num_pid_m * num_pid_n)               # which (b,h) we are processing
    pid_in_bh = pid % (num_pid_m * num_pid_n)
    pid_m = pid_in_bh // num_pid_n                     # tile index in i
    pid_n = pid_in_bh % num_pid_n                      # tile index in j

    # Decode batch and hidden‑dim indices
    b = bh // H
    h = bh % H

    # ----------- compute thread‑level offsets ----------
    i_range = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)   # (BLOCK_M,)
    j_range = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)   # (BLOCK_N,)

    i_valid = i_range < N
    j_valid = j_range < N

    # Base pointers for the current (b,h)
    left_base  = b * stride_l_batch + h * stride_l_h
    right_base = b * stride_r_batch + h * stride_r_h
    out_base   = b * stride_o_batch + h * stride_o_h

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension in tiles
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k_tile * BLOCK_K
        k_range = k_offset + tl.arange(0, BLOCK_K)       # (BLOCK_K,)
        k_valid = k_range < N

        # ----- load A block : left[b,i,k,h] (size BLOCK_M×BLOCK_K) -----
        a_ptrs = (
            left_ptr
            + left_base
            + i_range[:, None] * stride_l_i
            + k_range[None, :] * stride_l_k
        )
        a_mask = i_valid[:, None] & k_valid[None, :]
        a = tl.load(a_ptrs, mask=a_mask, other=0.0).to(tl.float32)

        # ----- load B block : right[b,j,k,h] transposed to (k,j) -----
        # we need B_T[k,j] = right[b,j,k,h]
        b_ptrs = (
            right_ptr
            + right_base
            + k_range[:, None] * stride_r_k
            + j_range[None, :] * stride_r_i
        )
        b_mask = k_valid[:, None] & j_valid[None, :]
        b = tl.load(b_ptrs, mask=b_mask, other=0.0).to(tl.float32)

        # ----- accumulate dot product -----
        acc += tl.dot(a, b)

    # ----------- write out tile ----------
    out_ptrs = (
        out_ptr
        + out_base
        + i_range[:, None] * stride_o_i
        + j_range[None, :] * stride_o_j
    )
    out_mask = i_valid[:, None] & j_valid[None, :]
    tl.store(out_ptrs, acc, mask=out_mask)


def _tri_mul_outgoing_triton(left_fp16: torch.Tensor,
                             right_fp16: torch.Tensor,
                             block_m: int = 128,
                             block_n: int = 128,
                             block_k: int = 32) -> torch.Tensor:
    """
    Triton implementation of the core TriMul einsum:
        out[...,i,j,h] = Σ_k left[...,i,k,h] * right[...,j,k,h]

    Arguments
    ---------
    left_fp16, right_fp16 : torch.Tensor[float16] of shape [B,N,N,H]
    block_*               : tiling sizes (must satisfy BLOCK_M·BLOCK_N ≤ 1024*1024 for H100)

    Returns
    -------
    out : torch.Tensor[float32] of shape [B,N,N,H]
    """
    assert left_fp16.dtype == torch.float16 and right_fp16.dtype == torch.float16
    B, N, _, H = left_fp16.shape
    out = torch.empty((B, N, N, H), dtype=torch.float32, device=left_fp16.device)

    # strides (in elements, not bytes)
    l0, l1, l2, l3 = left_fp16.stride()
    r0, r1, r2, r3 = right_fp16.stride()
    o0, o1, o2, o3 = out.stride()

    # grid = total (b,h) tiles × i‑tiles × j‑tiles
    num_m = (N + block_m - 1) // block_m
    num_n = (N + block_n - 1) // block_n
    total_programs = B * H * num_m * num_n
    grid = (total_programs,)

    _tri_mul_outgoing_kernel[grid](
        left_fp16,
        right_fp16,
        out,
        B, N, H,
        l0, l1, l2, l3,
        r0, r1, r2, r3,
        o0, o1, o2, o3,
        BLOCK_M=block_m,
        BLOCK_N=block_n,
        BLOCK_K=block_k,
    )
    return out


def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor of shape [B, N, N] (bool or float)
        - weights_dict : mapping of layer‑norm and linear weights (all float32)
        - config_dict  : contains "dim", "hidden_dim", "nomask" ...

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # unpack
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------ #
    # 1. Input layer‑norm over the last dimension (dim)
    # ------------------------------------------------------------------ #
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B,N,N,dim]

    # ------------------------------------------------------------------ #
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------ #
    left = F.linear(x, weights["left_proj.weight"])   # → [B,N,N,hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------ #
    # 3. Optional mask (broadcast over hidden_dim)
    # ------------------------------------------------------------------ #
    if not config.get("nomask", False):
        # mask may be bool; cast to same dtype as tensors
        mask = mask_tensor.to(x.dtype).unsqueeze(-1)   # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------ #
    # 4. Gates (sigmoid of linear layers)
    # ------------------------------------------------------------------ #
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # apply gates
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------ #
    # 5. Core TriMul – fused batched GEMM via Triton
    # ------------------------------------------------------------------ #
    # Cast to fp16 to exploit Tensor Cores; the kernel will accumulate in fp32.
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # Triton kernel (handles arbitrary N up to 1024 efficiently)
    out_hidden = _tri_mul_outgoing_triton(left_fp16, right_fp16,
                                          block_m=128, block_n=128, block_k=32)  # [B,N,N,hidden_dim]

    # ------------------------------------------------------------------ #
    # 6. Output layer‑norm, out‑gate and final projection back to dim
    # ------------------------------------------------------------------ #
    out = F.layer_norm(
        out_hidden,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate                     # gating in hidden space
    out = F.linear(out, weights["to_out.weight"])  # [B,N,N,dim]

    return out
