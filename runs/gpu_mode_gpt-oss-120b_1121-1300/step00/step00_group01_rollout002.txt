import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def trimul_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, D, N,
    stride_lb, stride_ld, stride_li, stride_lk,
    stride_rb, stride_rd, stride_rj, stride_rk,
    stride_ob, stride_od, stride_oi, stride_oj,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # compute program id
    pid = tl.program_id(0)
    # number of tiles
    total_tiles_m = tl.cdiv(N, BLOCK_M)
    total_tiles_n = tl.cdiv(N, BLOCK_N)
    tiles_per_pair = total_tiles_m * total_tiles_n  # per (b,d) pair

    pair_idx = pid // tiles_per_pair      # which (b,d)
    tile_idx = pid % tiles_per_pair

    # decode batch and hidden dimension
    b = pair_idx // D
    d = pair_idx % D

    # tile coarse coordinates
    tile_i = tile_idx // total_tiles_n
    tile_j = tile_idx % total_tiles_n

    # offsets in the i/j dimensions
    i_offset = tile_i * BLOCK_M
    j_offset = tile_j * BLOCK_N

    # pointers to the beginning of the (b,d) slice
    left_base = left_ptr + b * stride_lb + d * stride_ld
    right_base = right_ptr + b * stride_rb + d * stride_rd
    out_base = out_ptr + b * stride_ob + d * stride_od

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K

        # Offsets for left tile (i,k)
        i_idx = i_offset + tl.arange(0, BLOCK_M)[:, None]
        k_idx_left = k_offset + tl.arange(0, BLOCK_K)[None, :]
        left_ptrs = left_base + i_idx * stride_li + k_idx_left * stride_lk
        # mask for left load
        mask_left = (i_idx < N) & (k_idx_left < N)

        # Offsets for right tile (k,j) after transposition
        k_idx_right = k_offset + tl.arange(0, BLOCK_K)[:, None]
        j_idx = j_offset + tl.arange(0, BLOCK_N)[None, :]
        right_ptrs = right_base + k_idx_right * stride_rk + j_idx * stride_rj
        mask_right = (k_idx_right < N) & (j_idx < N)

        left_tile = tl.load(left_ptrs, mask=mask_left, other=0.0).to(tl.float32)
        right_tile = tl.load(right_ptrs, mask=mask_right, other=0.0).to(tl.float32)

        acc += tl.dot(left_tile, right_tile)

    # write back to out
    i_idx_out = i_offset + tl.arange(0, BLOCK_M)[:, None]
    j_idx_out = j_offset + tl.arange(0, BLOCK_N)[None, :]
    out_ptrs = out_base + i_idx_out * stride_oi + j_idx_out * stride_oj
    mask_out = (i_idx_out < N) & (j_idx_out < N)
    tl.store(out_ptrs, acc, mask=mask_out)

def custom_kernel(data):
    """
    Triton-accelerated forward pass of the Triangle Multiplicative Update (TriMul) *outgoing* variant.
    The kernel fuses the final N×N matrix multiplication for every hidden dimension channel.
    Steps:
      1. Layer‑norm the input.
      2. Linear projections for left/right and their gates (PyTorch).
      3. Apply mask (if provided) and gating.
      4. Compute out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d] using a custom Triton GEMM.
      5. Post‑norm, gating, and final linear projection back to the original dimension.
    """
    import torch
    import torch.nn.functional as F

    # unpack
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # -------- LayerNorm ----------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # -------- Linear projections ----------
    left = F.linear(x, weights["left_proj.weight"])       # [B,N,N,hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # -------- Optional mask ----------
    if mask is not None and mask.numel() > 0:
        # mask shape [B,N,N] -> unsqueeze to broadcast
        mask_f = mask.unsqueeze(-1).type_as(left)
        left = left * mask_f
        right = right * mask_f

    # -------- Gates ----------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # -------- Rearrange for batched matmul ----------
    # shape -> [B, hidden_dim, N, N]
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    B, D, N, _ = left_perm.shape

    # Allocate output buffer
    out_perm = torch.empty_like(left_perm)

    # Strides (in elements)
    stride_lb, stride_ld, stride_li, stride_lk = left_perm.stride()
    stride_rb, stride_rd, stride_rj, stride_rk = right_perm.stride()
    stride_ob, stride_od, stride_oi, stride_oj = out_perm.stride()

    # Choose block sizes (tuned for H100, fp32)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    total_tiles_m = (N + BLOCK_M - 1) // BLOCK_M
    total_tiles_n = (N + BLOCK_N - 1) // BLOCK_N
    total_programs = B * D * total_tiles_m * total_tiles_n

    grid = (total_programs,)

    trimul_matmul_kernel[grid](
        left_perm,
        right_perm,
        out_perm,
        B,
        D,
        N,
        stride_lb,
        stride_ld,
        stride_li,
        stride_lk,
        stride_rb,
        stride_rd,
        stride_rj,
        stride_rk,
        stride_ob,
        stride_od,
        stride_oi,
        stride_oj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Back to original layout [B,N,N,hidden_dim]
    out = out_perm.permute(0, 2, 3, 1)

    # -------- Post‑norm, gating and final projection ----------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])
    return out

# -*- coding: utf-8 -*-
"""
TriMul (outgoing) – Triton‑accelerated forward pass

The implementation follows the reference PyTorch module:

1. Layer‑norm the 4‑D input [B, N, N, C].
2. Linear projections for “left” and “right” (weight only, no bias).
3. Optional mask multiplication.
4. Gating (sigmoid‑gated linear layers) and element‑wise multiplication.
5. Core operation:
       out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
   This is a batched matrix multiplication for each hidden‑dim channel.
   A custom Triton kernel fuses the N×N GEMM for every (batch, hidden) pair.
6. Post‑norm, gating and final linear projection back to the original channel size.

Only the costly N‑by‑N multiplication is performed in Triton; the remaining
operations stay in PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def trimul_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, D, N,
    stride_lb, stride_ld, stride_li, stride_lk,
    stride_rb, stride_rd, stride_rj, stride_rk,
    stride_ob, stride_od, stride_oi, stride_oj,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute, for every (batch, hidden) pair,
        out = left @ right.T   (left, right : N×N matrices)
    where left and right are stored as [B, D, N, N] tensors.
    """
    pid = tl.program_id(0)

    # tiling layout ---------------------------------------------------------
    total_tiles_m = tl.cdiv(N, BLOCK_M)   # tiles along i
    total_tiles_n = tl.cdiv(N, BLOCK_N)   # tiles along j
    tiles_per_pair = total_tiles_m * total_tiles_n  # per (b,d)

    pair_idx = pid // tiles_per_pair          # flattened (b,d) index
    tile_idx = pid % tiles_per_pair           # which i‑j tile inside that pair

    b = pair_idx // D
    d = pair_idx % D

    tile_i = tile_idx // total_tiles_n
    tile_j = tile_idx % total_tiles_n

    i_offset = tile_i * BLOCK_M
    j_offset = tile_j * BLOCK_N

    # base pointers for the selected (b,d) slice
    left_base  = left_ptr  + b * stride_lb + d * stride_ld
    right_base = right_ptr + b * stride_rb + d * stride_rd
    out_base   = out_ptr   + b * stride_ob + d * stride_od

    # accumulator -----------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the reduction dimension k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K

        # ---- left tile (i,k) ------------------------------------------------
        i_idx = i_offset + tl.arange(0, BLOCK_M)[:, None]           # (M,1)
        k_idx_l = k_offset + tl.arange(0, BLOCK_K)[None, :]         # (1,K)
        left_ptrs = left_base + i_idx * stride_li + k_idx_l * stride_lk
        mask_l = (i_idx < N) & (k_idx_l < N)

        # ---- right tile (k,j) after transposition --------------------------
        k_idx_r = k_offset + tl.arange(0, BLOCK_K)[:, None]         # (K,1)
        j_idx = j_offset + tl.arange(0, BLOCK_N)[None, :]           # (1,N)
        right_ptrs = right_base + k_idx_r * stride_rk + j_idx * stride_rj
        mask_r = (k_idx_r < N) & (j_idx < N)

        left_tile  = tl.load(left_ptrs,  mask=mask_l, other=0.0).to(tl.float32)
        right_tile = tl.load(right_ptrs, mask=mask_r, other=0.0).to(tl.float32)

        # accumulate product
        acc += tl.dot(left_tile, right_tile)   # (M,N)

    # ---- write output -------------------------------------------------------
    i_idx_out = i_offset + tl.arange(0, BLOCK_M)[:, None]
    j_idx_out = j_offset + tl.arange(0, BLOCK_N)[None, :]
    out_ptrs = out_base + i_idx_out * stride_oi + j_idx_out * stride_oj
    mask_o = (i_idx_out < N) & (j_idx_out < N)
    tl.store(out_ptrs, acc, mask=mask_o)


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul “outgoing” operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor   [B, N, N, C]
            - mask: torch.Tensor (or None) [B, N, N]
            - weights: dict with all linear / layer‑norm parameters
            - config: dict with keys `dim`, `hidden_dim`, `nomask` (bool)

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # --------------------------------------------------------------------- #
    # 1. Input layer‑norm
    # --------------------------------------------------------------------- #
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # --------------------------------------------------------------------- #
    # 2. Linear projections (no bias)
    # --------------------------------------------------------------------- #
    left  = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------------- #
    # 3. Optional mask (skip when `nomask` is True)
    # --------------------------------------------------------------------- #
    if mask is not None and not config.get("nomask", False):
        mask_f = mask.unsqueeze(-1).type_as(left)   # broadcast over hidden_dim
        left  = left * mask_f
        right = right * mask_f

    # --------------------------------------------------------------------- #
    # 4. Gating
    # --------------------------------------------------------------------- #
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------------- #
    # 5. Core N×N multiplicative update – Triton kernel
    # --------------------------------------------------------------------- #
    # Rearrange to [B, hidden_dim, N, N] for batched GEMM
    left_perm  = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    B, D, N, _ = left_perm.shape

    # output buffer in the same layout
    out_perm = torch.empty_like(left_perm)

    # strides (elements, not bytes)
    stride_lb, stride_ld, stride_li, stride_lk = left_perm.stride()
    stride_rb, stride_rd, stride_rj, stride_rk = right_perm.stride()
    stride_ob, stride_od, stride_oi, stride_oj = out_perm.stride()

    # block sizes – experimentally good on H100 (fp32)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    tiles_m = (N + BLOCK_M - 1) // BLOCK_M
    tiles_n = (N + BLOCK_N - 1) // BLOCK_N
    total_progs = B * D * tiles_m * tiles_n
    grid = (total_progs,)

    trimul_matmul_kernel[grid](
        left_perm,
        right_perm,
        out_perm,
        B,
        D,
        N,
        stride_lb,
        stride_ld,
        stride_li,
        stride_lk,
        stride_rb,
        stride_rd,
        stride_rj,
        stride_rk,
        stride_ob,
        stride_od,
        stride_oi,
        stride_oj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Back to original layout [B, N, N, hidden_dim]
    out = out_perm.permute(0, 2, 3, 1)

    # --------------------------------------------------------------------- #
    # 6. Post‑norm, gating and final projection back to `dim`
    # --------------------------------------------------------------------- #
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])   # → [B, N, N, dim]

    return out
