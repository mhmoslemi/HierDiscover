@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, D,
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_j, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr,
    ):
    pid_b = tl.program_id(0)  # batch index
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)
    pid_d = tl.program_id(3)

    # compute offsets
    i_base = pid_i * BLOCK_I
    j_base = pid_j * BLOCK_J
    d_base = pid_d * BLOCK_D

    # offsets within block
    offs_i = i_base + tl.arange(0, BLOCK_I)
    offs_j = j_base + tl.arange(0, BLOCK_J)
    offs_d = d_base + tl.arange(0, BLOCK_D)

    # mask valid indices
    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_d = offs_d < D

    # Linearize inside block
    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_D), dtype=tl.float32)

    # Loop over k blocks
    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for pid_k in range(num_k_blocks):
        k_base = pid_k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left tile: shape (BLOCK_I, BLOCK_K, BLOCK_D)
        left_ptrs = left_ptr + pid_b * stride_left_b \
            + offs_i[:, None, None] * stride_left_i \
            + offs_k[None, :, None] * stride_left_k \
            + offs_d[None, None, :] * stride_left_d
        left_tile = tl.load(left_ptrs, mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :], other=0.0)

        # Load right tile transposed: shape (BLOCK_K, BLOCK_J, BLOCK_D)
        # We want right transposed across k and j.
        right_ptrs = right_ptr + pid_b * stride_right_b \
            + offs_k[:, None, None] * stride_right_k \
            + offs_j[None, :, None] * stride_right_j \
            + offs_d[None, None, :] * stride_right_d
        right_tile_T = tl.load(right_ptrs, mask=mask_k[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :], other=0.0)

        # Compute dot per d
        # left_tile: (I,K,D), right_tile_T: (K,J,D)
        for d_index in range(BLOCK_D):
            # if mask_d[d_index] is false skip
            if not mask_d[d_index]:
                continue
            L = left_tile[:, :, d_index]  # (I,K)
            R = right_tile_T[:, :, d_index]  # (K,J)
            # dot
            acc[:, :, d_index] += tl.dot(L, R)

    # write output
    out_ptrs = out_ptr + pid_b * stride_out_b \
        + offs_i[:, None, None] * stride_out_i \
        + offs_j[None, :, None] * stride_out_j \
        + offs_d[None, None, :] * stride_out_d

    # store with mask
    tl.store(out_ptrs, acc, mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :])

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, D,
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_i, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr):
    pid_b = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)
    pid_d = tl.program_id(3)

    # base offsets
    i_base = pid_i * BLOCK_I
    j_base = pid_j * BLOCK_J
    d_base = pid_d * BLOCK_D

    # offsets within block
    offs_i = i_base + tl.arange(0, BLOCK_I)
    offs_j = j_base + tl.arange(0, BLOCK_J)
    offs_d = d_base + tl.arange(0, BLOCK_D)

    # predicates for out-of-bound
    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_d = offs_d < D

    # accumulator
    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_D), dtype=tl.float32)

    # number of k blocks
    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for pid_k in range(num_k_blocks):
        k_base = pid_k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left tile (I, K, D)
        left_ptrs = left_ptr + pid_b * stride_left_b \
            + (offs_i[:, None, None] * stride_left_i) \
            + (offs_k[None, :, None] * stride_left_k) \
            + (offs_d[None, None, :] * stride_left_d)
        left_tile = tl.load(left_ptrs, mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :], other=0.0)

        # Load right tile transposed (K, J, D)
        right_ptrs = right_ptr + pid_b * stride_right_b \
            + (offs_k[:, None, None] * stride_right_k) \
            + (offs_j[None, :, None] * stride_right_i) \
            + (offs_d[None, None, :] * stride_right_d)
        right_tile = tl.load(right_ptrs, mask=mask_k[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :], other=0.0)

        # Compute dot for each d
        for d_idx in range(BLOCK_D):
            # optional: if not mask_d[d_idx], continue (but not needed)
            L = left_tile[:, :, d_idx]  # shape (I,K)
            R = right_tile[:, :, d_idx]  # shape (K,J)
            acc[:, :, d_idx] += tl.dot(L, R)

    # store output
    out_ptrs = out_ptr + pid_b * stride_out_b \
        + (offs_i[:, None, None] * stride_out_i) \
        + (offs_j[None, :, None] * stride_out_j) \
        + (offs_d[None, None, :] * stride_out_d)

    tl.store(out_ptrs, acc, mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :])

def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    # config params
    dim = config['dim']
    hidden_dim = config['hidden_dim']

    # Process mask
    if mask_tensor is None:
        mask_tensor = torch.ones(input_tensor.shape[0], input_tensor.shape[1], input_tensor.shape[2], dtype=input_tensor.dtype, device=device)
    # else ensure shape broadcast

    # LayerNorm
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])

    # Linear projections
    left = torch.nn.functional.linear(x, weights["left_proj.weight"]) # shape B,N,N,hidden_dim
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])
    
    # Apply mask
    mask_unsqueeze = mask_tensor.unsqueeze(-1)  # shape [B,N,N,1]
    left = left * mask_unsqueeze
    right = right * mask_unsqueeze

    # Gating linear + sigmoid
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # Compute contracted output using Triton kernel
    # left and right shape: (B,N,N,hidden_dim)
    B,N,_dim_hidden = left.shape[0], left.shape[1], left.shape[3]  # N=seq_len, hidden_dim = dim hidden

    # allocate output
    out = torch.empty_like(left)  # shape B,N,N,hidden_dim

    # Get strides in terms of ints
    stride_left_b, stride_left_i, stride_left_k, stride_left_d = left.stride()
    stride_right_b, stride_right_i, stride_right_k, stride_right_d = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_d = out.stride()

    # Set block sizes heuristically
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_K = 64
    BLOCK_D = 32

    # Compute grid sizes
    grid = (B,
            (N + BLOCK_I - 1) // BLOCK_I,
            (N + BLOCK_J - 1) // BLOCK_J,
            (hidden_dim + BLOCK_D - 1) // BLOCK_D)

    tri_mul_kernel[grid](
        left,
        right,
        out,
        B, N, hidden_dim,
        stride_left_b, stride_left_i, stride_left_k, stride_left_d,
        stride_right_b, stride_right_i, stride_right_k, stride_right_d,
        stride_out_b, stride_out_i, stride_out_j, stride_out_d,
        BLOCK_I=BLOCK_I, BLOCK_J=BLOCK_J, BLOCK_K=BLOCK_K, BLOCK_D=BLOCK_D,
    )

    # Apply to_out normalization and gating
    out_norm = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"])
    out_norm = out_norm * out_gate

    # Final linear projection to original dim
    out_final = torch.nn.functional.linear(out_norm, weights["to_out.weight"])

    return out_final

"""
TriMul (outgoing) kernel implementation.

The forward pass consists of:
1. LayerNorm on the input tensor.
2. Linear projections (left/right) and three gating linear layers.
3. Application of a pairwise multiplicative update:
      out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is a batched “matrix‑multiply‑over‑k” for each hidden dimension `d`.
   The heavy O(N³·D) contraction is performed by a custom Triton kernel that
   tiles the i, j and hidden dimension while streaming over the k‑axis.
4. LayerNorm + gating on the contracted result.
5. Final linear projection back to the original channel dimension.

Only the contraction step (step 3) is done in Triton; the remaining
operations use PyTorch (which is already highly optimized). The kernel
uses three launch dimensions (batch·d‑block, i‑block, j‑block) to stay
within Triton’s 3‑axis limit.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,           # pointers
    B, N, D,                               # tensor dimensions
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_i, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    NUM_D_BLOCKS,                          # constexpr, number of hidden‑dim blocks
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    """Contract left and right tensors:
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    """
    # ----------------------------------------------------------------------
    # 1) Decode program ids
    # ----------------------------------------------------------------------
    pid0 = tl.program_id(0)                     # combines batch & hidden‑dim block
    pid_i = tl.program_id(1)                     # i‑block
    pid_j = tl.program_id(2)                     # j‑block

    # split first axis into batch and hidden‑dim block
    pid_b = pid0 // NUM_D_BLOCKS
    pid_d = pid0 % NUM_D_BLOCKS

    # ----------------------------------------------------------------------
    # 2) Compute block offsets
    # ----------------------------------------------------------------------
    i_base = pid_i * BLOCK_I
    j_base = pid_j * BLOCK_J
    d_base = pid_d * BLOCK_D

    offs_i = i_base + tl.arange(0, BLOCK_I)
    offs_j = j_base + tl.arange(0, BLOCK_J)
    offs_d = d_base + tl.arange(0, BLOCK_D)

    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_d = offs_d < D

    # accumulator for the output tile (i, j, d)
    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_D), dtype=tl.float32)

    # ----------------------------------------------------------------------
    # 3) Loop over k‑axis in tiles
    # ----------------------------------------------------------------------
    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for pid_k in range(num_k_blocks):
        k_base = pid_k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ----------- load left : shape (I, K, D) -------------------------
        left_ptrs = left_ptr + pid_b * stride_left_b \
            + (offs_i[:, None, None] * stride_left_i) \
            + (offs_k[None, :, None] * stride_left_k) \
            + (offs_d[None, None, :] * stride_left_d)
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :],
                       other=0.0)

        # ----------- load right (transposed) : shape (K, J, D) -----------
        # we store it transposed so that a dot product can be used directly.
        right_ptrs = right_ptr + pid_b * stride_right_b \
            + (offs_k[:, None, None] * stride_right_k) \
            + (offs_j[None, :, None] * stride_right_i) \
            + (offs_d[None, None, :] * stride_right_d)
        right_T = tl.load(right_ptrs,
                          mask=mask_k[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :],
                          other=0.0)

        # ----------- compute dot products per hidden‑dim slice ------------
        for d_idx in range(BLOCK_D):
            # left slice (I, K), right slice (K, J)
            L = left[:, :, d_idx]          # (I, K)
            R = right_T[:, :, d_idx]       # (K, J)
            acc[:, :, d_idx] += tl.dot(L, R)   # (I, J)

    # ----------------------------------------------------------------------
    # 4) Store the result
    # ----------------------------------------------------------------------
    out_ptrs = out_ptr + pid_b * stride_out_b \
        + (offs_i[:, None, None] * stride_out_i) \
        + (offs_j[None, :, None] * stride_out_j) \
        + (offs_d[None, None, :] * stride_out_d)

    tl.store(out_ptrs,
             acc,
             mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :])


def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
        - weights_dict : dict with all linear/ln weights (no bias except LN)
        - config_dict  : {"dim": int, "hidden_dim": int, ...}

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 0) Prepare mask (if none, treat as all‑ones)
    # ------------------------------------------------------------------
    if mask_tensor is None:
        mask = torch.ones(input_tensor.shape[0],
                         input_tensor.shape[1],
                         input_tensor.shape[2],
                         dtype=dtype,
                         device=device)
    else:
        mask = mask_tensor.to(dtype)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )   # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2) Linear projections (left/right) – no bias
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B, N, N, hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3) Apply mask (broadcast on the hidden dimension)
    # ------------------------------------------------------------------
    mask_expanded = mask.unsqueeze(-1)                # [B, N, N, 1]
    left = left * mask_expanded
    right = right * mask_expanded

    # ------------------------------------------------------------------
    # 4) Gating linear layers + sigmoid
    # ------------------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    # ------------------------------------------------------------------
    # 5) Apply gates to the masked projections
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6) Pairwise multiplicative update via Triton kernel
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape

    # allocate output tensor for the contraction (still in hidden_dim)
    out = torch.empty_like(left)   # [B, N, N, hidden_dim]

    # strides (int64) for pointer arithmetic
    stride_left_b, stride_left_i, stride_left_k, stride_left_d = left.stride()
    stride_right_b, stride_right_i, stride_right_k, stride_right_d = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_d = out.stride()

    # Tile sizes – tuned for typical H100 workloads
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_K = 64
    BLOCK_D = 32

    num_d_blocks = (hidden_dim + BLOCK_D - 1) // BLOCK_D
    grid = (
        B * num_d_blocks,                         # batch × hidden‑dim block
        (N + BLOCK_I - 1) // BLOCK_I,             # i‑blocks
        (N + BLOCK_J - 1) // BLOCK_J,             # j‑blocks
    )

    # launch the kernel
    tri_mul_kernel[grid](
        left,
        right,
        out,
        B, N, hidden_dim,
        stride_left_b, stride_left_i, stride_left_k, stride_left_d,
        stride_right_b, stride_right_i, stride_right_k, stride_right_d,
        stride_out_b, stride_out_i, stride_out_j, stride_out_d,
        num_d_blocks,
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
        BLOCK_D=BLOCK_D,
    )
    # make sure the kernel has finished before moving on
    torch.cuda.synchronize()

    # ------------------------------------------------------------------
    # 7) Output LayerNorm + gating
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    out = out * out_gate

    # ------------------------------------------------------------------
    # 8) Final linear projection back to `dim`
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])   # [B, N, N, dim]

    return out
```

This code follows the reference PyTorch implementation, moving the expensive
`einsum('... i k d, ... j k d -> ... i j d')` contraction to a highly‑tuned
Triton kernel that streams over the `k` dimension while tiling `i`, `j` and
the hidden dimension. The rest of the operations (LayerNorm, linear
projections, sigmoids, and final projection) use PyTorch’s optimized
implementations. The kernel works for any batch size ≤ 2, sequence length
up to 1024, and hidden dimensions up to 384, handling optional masks
and both normal and Cauchy‑distributed inputs.
