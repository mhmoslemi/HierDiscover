    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    batch, seq_len, _, _ = input_tensor.shape

    # Load weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    right = torch.nn.functional.linear(x, right_proj_weight)

    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

import torch
import triton
import triton.language as tl

@triton.jit
def tri_mul_kernel(l_ptr, r_ptr, out_ptr,
                   M, N, K,
                   stride_l_batch, stride_l_m, stride_l_k,
                   stride_r_batch, stride_r_m, stride_r_k,
                   stride_o_batch, stride_o_m, stride_o_n,
                   BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute block offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load L tile
        l_ptrs = l_ptr + pid_batch * stride_l_batch + \
                 offs_m[:, None] * stride_l_m + \
                 offs_k[None, :] * stride_l_k
        a = tl.load(l_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        a = a.to(tl.float32)

        # we need element R[n, k] for transposed (k, n)
        r_ptrs = r_ptr + pid_batch * stride_r_batch + \

import torch
import triton
import triton.language as tl

def tri_mul_gemm(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM with transposed RHS: out = a @ b^T.
    a, b shape: (batch, N, K) where K = N.
    Returns tensor (batch, N, N) in float32.
    """
    assert a.is_cuda and b.is_cuda
    B, M, K = a.shape
    N = K
    # block sizes tuned for H100
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64
    grid = (B,
            (M + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    out = torch.empty((B, M, N), dtype=torch.float32, device=a.device)

    tri_mul_kernel[grid](
        a,
        b,
        out,
        M, N, K,
        a.stride(0), a.stride(1), a.stride(2),
        b.stride(0), b.stride(1), b.stride(2),
        out.stride(0), out.stride(1), out.stride(2),
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=8,
    )
    return out

@triton.jit
def tri_mul_kernel(
    l_ptr, r_ptr, out_ptr,
    M, N, K,
    stride_l_batch, stride_l_m, stride_l_k,
    stride_r_batch, stride_r_m, stride_r_k,
    stride_o_batch, stride_o_m, stride_o_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    """Batched GEMM kernel computing out = L @ R^T."""
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # offsets within the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over k dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # load tile of L (shape BLOCK_M x BLOCK_K)
        l_ptrs = l_ptr + pid_batch * stride_l_batch \
                 + offs_m[:, None] * stride_l_m \
                 + offs_k[None, :] * stride_l_k
        a = tl.load(l_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        a = a.to(tl.float32)

        # load tile of R transposed.
        # Original R layout: (batch, M, K). We need R[n, k] -> transposed(k, n).
        r_ptrs = r_ptr + pid_batch * stride_r_batch \
                 + offs_n[None, :] * stride_r_m \
                 + offs_k[:, None] * stride_r_k
        b = tl.load(r_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)
        b = b.to(tl.float32)

        acc += tl.dot(a, b)

    # store the result tile
    out_ptrs = out_ptr + pid_batch * stride_o_batch \
               + offs_m[:, None] * stride_o_m \
               + offs_n[None, :] * stride_o_n
    tl.store(out_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul operation
    (AlphaFold3 style).  The heavy N³ reduction is performed by a dedicated
    Triton GEMM kernel (left @ rightᵀ) while the surrounding linear
    projections, layer‑norms and sigmoids are handled with PyTorch.
    """
    input_tensor, mask, weights, config = data
    # config contains dim and hidden_dim
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    B, N, _, _ = input_tensor.shape

    # ---------- 1. LayerNorm on the input ----------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor, (dim,), weight=norm_weight, bias=norm_bias
    )

    # ---------- 2. Linear projections (no bias) ----------
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]

    left = torch.nn.functional.linear(x, left_proj_weight)   # [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_weight)

    # ---------- 3. Optional mask ----------
    if mask is not None:
        left = left * mask.unsqueeze(-1)
        right = right * mask.unsqueeze(-1)

    # ---------- 4. Gating ----------
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight)).to(torch.float16)
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight)).to(torch.float16)
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    # Cast the projected tensors to fp16 for the heavy reduction
    left = left.to(torch.float16) * left_gate
    right = right.to(torch.float16) * right_gate

    # ---------- 5. Prepare tensors for the batched GEMM ----------
    #   shape needed for kernel: (B*hidden_dim, N, N)
    left = left.permute(0, 3, 1, 2).contiguous()   # B, hidden_dim, N, N
    right = right.permute(0, 3, 1, 2).contiguous()
    left_mat = left.view(-1, N, N)   # (B*hidden_dim, N, N)
    right_mat = right.view(-1, N, N)

    # ---------- 6. Core N³ reduction via Triton ----------
    # out_mat shape: (B*hidden_dim, N, N)  (float32)
    out_mat = tri_mul_gemm(left_mat, right_mat)

    # ---------- 7. Reshape back to [B,N,N,hidden_dim] ----------
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ---------- 8. Output LayerNorm, gating and final linear ----------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    out = torch.nn.functional.layer_norm(
        out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_weight)   # [B,N,N,dim]

    return out

"""
Triton‑accelerated forward pass of the “outgoing” TriMul operator (AlphaFold‑3).

The heavy N³ reduction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is implemented as a batched GEMM with a transposed RHS:
    out_batch = left_batch @ right_batchᵀ
where each batch corresponds to a (hidden_dim) channel of a given
example.  The surrounding LayerNorms, linear projections and sigmoids
are kept in PyTorch – only the N³ core is executed on the GPU via a
custom Triton kernel.
"""
import torch
import triton
import triton.language as tl


# --------------------------------------------------------------
# Triton kernel: batched GEMM computing out = L @ Rᵀ
# --------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    l_ptr, r_ptr, out_ptr,               # pointers
    M, N, K,                             # matrix sizes (M=N=K=N_seq)
    stride_l_batch, stride_l_m, stride_l_k,
    stride_r_batch, stride_r_m, stride_r_k,
    stride_o_batch, stride_o_m, stride_o_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    """
    out[batch, i, j] = Σ_k L[batch, i, k] * R[batch, j, k]
    (R is accessed transposed, i.e. Rᵀ[k, j] = R[j, k]).
    """
    pid_batch = tl.program_id(0)       # batch × hidden channel
    pid_m = tl.program_id(1)           # tile row index
    pid_n = tl.program_id(2)           # tile column index

    # ----------------------------------------------------------
    # Tile offsets
    # ----------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ----------------------------------------------------------
    # Loop over K dimension
    # ----------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # ------- L tile (M × K) --------------------------------
        l_ptrs = l_ptr + pid_batch * stride_l_batch \
                 + offs_m[:, None] * stride_l_m \
                 + offs_k[None, :] * stride_l_k
        a = tl.load(l_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=tl.float16(0))
        a = a.to(tl.float32)                     # promote for accumulation

        # ------- R tile (N × K) transposed -> (K × N) ---------
        # Original layout: (batch, N, K). We need element R[j, k].
        # Load as Rᵀ[k, j] = R[j, k].
        r_ptrs = r_ptr + pid_batch * stride_r_batch \
                 + offs_n[None, :] * stride_r_m \
                 + offs_k[:, None] * stride_r_k
        b = tl.load(r_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=tl.float16(0))
        b = b.to(tl.float32)

        # ------- Accumulate -------------------------------------------------
        acc += tl.dot(a, b)           # (BLOCK_M, BLOCK_K)×(BLOCK_K, BLOCK_N)

    # ----------------------------------------------------------
    # Store output tile
    # ----------------------------------------------------------
    out_ptrs = out_ptr + pid_batch * stride_o_batch \
               + offs_m[:, None] * stride_o_m \
               + offs_n[None, :] * stride_o_n
    tl.store(out_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


def tri_mul_gemm(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM with transposed RHS.
    Input shapes: (B*H, N, N)  (fp16)
    Returns:       (B*H, N, N)  (fp32)
    """
    assert left.is_cuda and right.is_cuda
    B, N, K = left.shape
    assert K == N, "TriMul requires square inner dimension"
    # Block sizes tuned for H100 (large shared memory, 8‑warp occupancy)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (B,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    out = torch.empty((B, N, N), dtype=torch.float32, device=left.device)

    tri_mul_kernel[grid](
        left,
        right,
        out,
        N, N, K,
        left.stride(0), left.stride(1), left.stride(2),
        right.stride(0), right.stride(1), right.stride(2),
        out.stride(0), out.stride(1), out.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,          # 8 warps ≈ 256 threads per block
    )
    return out


# --------------------------------------------------------------
# Public entry point
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul module.

    Args:
        data: (input, mask, weights, config)
            - input : torch.Tensor[bs, N, N, dim] (float32)
            - mask  : torch.Tensor[bs, N, N]  (float32 or bool) – may be all‑ones
            - weights: dict of torch.Tensor with the exact keys used in the reference
            - config : dict containing "dim" and "hidden_dim"

    Returns:
        torch.Tensor[bs, N, N, dim]  (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    bs, N, _, _ = input_tensor.shape

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = torch.nn.functional.linear(x, left_proj_weight)   # [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # 3. Mask (if supplied)
    # ------------------------------------------------------------------
    if mask is not None:
        left = left * mask.unsqueeze(-1)
        right = right * mask.unsqueeze(-1)

    # ------------------------------------------------------------------
    # 4. Gating (sigmoids)
    # ------------------------------------------------------------------
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = torch.sigmoid(
        torch.nn.functional.linear(x, left_gate_weight)
    ).to(torch.float16)
    right_gate = torch.sigmoid(
        torch.nn.functional.linear(x, right_gate_weight)
    ).to(torch.float16)
    out_gate = torch.sigmoid(
        torch.nn.functional.linear(x, out_gate_weight)
    )   # stays FP32 – later multiplication is FP32‑compatible

    # Cast projections to fp16 before heavy reduction
    left = left.to(torch.float16) * left_gate
    right = right.to(torch.float16) * right_gate

    # ------------------------------------------------------------------
    # 5. Prepare tensors for the batched GEMM
    #    Shape needed for the kernel: (B*hidden_dim, N, N)
    # ------------------------------------------------------------------
    left = left.permute(0, 3, 1, 2).contiguous()   # B, hidden_dim, N, N
    right = right.permute(0, 3, 1, 2).contiguous()
    left_mat = left.view(-1, N, N)                 # (B*hidden_dim, N, N)
    right_mat = right.view(-1, N, N)

    # ------------------------------------------------------------------
    # 6. N³ reduction via Triton
    # ------------------------------------------------------------------
    out_mat = tri_mul_gemm(left_mat, right_mat)   # (B*hidden_dim, N, N) FP32

    # ------------------------------------------------------------------
    # 7. Restore original layout
    # ------------------------------------------------------------------
    out = out_mat.view(bs, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8. Output LayerNorm, gating and final linear projection
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )
    out = out * out_gate                         # gating (FP32)
    out = torch.nn.functional.linear(out, to_out_weight)  # back to [B,N,N,dim]

    return out
