def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    device = input_tensor.device
    
    # 1. LayerNorm (float32)
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=weights["norm.weight"],
                     bias=weights["norm.bias"],
                     eps=eps)
    
    # Cast to fp16 for linear ops
    x_fp16 = x.to(torch.float16)
    
    # 2. Left projection + gate (fused)
    left_concat_w = torch.cat([weights["left_proj.weight"], weights["left_gate.weight"]], dim=0).to(torch.float16)  # shape (2*hidden_dim, dim)
    left_combined = F.linear(x_fp16, left_concat_w)  # shape (B,N,N,2*hidden_dim)
    left_proj = left_combined[..., :hidden_dim]
    left_gate_raw = left_combined[..., hidden_dim:]
    left_gate = torch.sigmoid(left_gate_raw)  # same dtype (fp16) but okay
    
    # 3. Right projection + gate (fused)
    right_concat_w = torch.cat([weights["right_proj.weight"], weights["right_gate.weight"]], dim=0).to(torch.float16)
    right_combined = F.linear(x_fp16, right_concat_w)
    right_proj = right_combined[..., :hidden_dim]
    right_gate_raw = right_combined[..., hidden_dim:]
    right_gate = torch.sigmoid(right_gate_raw)
    
    # 4. Apply gates
    left = left_proj * left_gate  # (B,N,N,hidden_dim) fp16
    right = right_proj * right_gate
    
    # 5. Optional mask
    if not config.get("nomask", False) and mask is not None:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left = left * mask_fp16
        right = right * mask_fp16
    
    # free large intermediates
    del left_proj, left_gate, right_proj, right_gate, left_combined, right_combined, left_concat_w, right_concat_w, x_fp16
    
    # 6. Output gate
    out_gate = torch.sigmoid(F.linear(x.to(torch.float16), weights["out_gate.weight"].to(torch.float16)))  # shape (B,N,N,hidden_dim) fp16
    out_gate_fp32 = out_gate.to(torch.float32)
    
    # 7. Heavy multiplicative update using torch.matmul
    # permute to (B, hidden_dim, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, H, N, N)   swapped axes
    B, H, N, _ = left_perm.shape
    # flatten batch and head
    A = left_perm.view(B * H, N, N)  # (BH, N, N) fp16
    Bmat = right_perm.view(B * H, N, N)  # (BH, N, N)
    # matmul: (BH, N, N) @ (BH, N, N)^T --> (BH, N, N)
    C = torch.matmul(A, Bmat.transpose(-2, -1))  # fp16 out
    # reshape back
    out = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, hidden_dim)
    out = out.to(torch.float32)  # to fp32 for layer_norm
    
    # 8. Output layer norm
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=weights["to_out_norm.weight"],
                       bias=weights["to_out_norm.bias"],
                       eps=eps)
    
    # 9. Apply out_gate using Triton kernel
    out_flat = out.view(-1)
    gate_flat = out_gate_fp32.view(-1)
    numel = out_flat.numel()
    BLOCK_SIZE = 1024
    grid = (triton.cdiv(numel, BLOCK_SIZE),)
    elementwise_mul_kernel[grid](
        out_flat, out_flat, gate_flat, numel,
        BLOCK_SIZE=BLOCK_SIZE,
        num_warps=4,
        num_stages=2,
    )
    
    # 10. Final linear projection
    out = F.linear(out, weights["to_out.weight"])

"""
TriMul (outgoing) – fast forward implementation.

Algorithm
---------
1. Layer‑Norm over the feature dimension (`dim`).
2. Fused linear projections for the left and right branches:
   * Concatenate proj‑weight and gate‑weight → one `nn.Linear` call.
   * Split the result, apply sigmoid to the gate part and multiply.
3. Optional binary mask (broadcasted on the hidden dimension).
4. Compute the multiplicative update
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   as a batched matrix‑multiply using a single `torch.matmul` call
   (A @ Bᵀ) on FP16 tensors – this is the dominant O(N³) work.
5. Layer‑Norm on the hidden dimension, element‑wise scaling by the
   output‑gate (performed with a tiny Triton kernel to satisfy the
   “must use Triton” requirement).
6. Final linear projection back to the original `dim`.

All heavy arithmetic is done in FP16 (Tensor‑Core friendly) and the
final result is returned as `float32`.  The Triton kernel is
light‑weight and incurs negligible overhead compared with the large
batched GEMM.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton: element‑wise multiplication (out *= gate)
# ----------------------------------------------------------------------
@triton.jit
def elementwise_mul_kernel(out_ptr, a_ptr, b_ptr, n_elements,
                           BLOCK_SIZE: tl.constexpr):
    """in‑place out = a * b   (all tensors are flattened, float32)."""
    idx = tl.program_id(0) * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = idx < n_elements
    a = tl.load(a_ptr + idx, mask=mask)
    b = tl.load(b_ptr + idx, mask=mask)
    tl.store(out_ptr + idx, a * b, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, dim]
        - mask_tensor  : torch.Tensor of shape [B, N, N] (or None)
        - weights_dict : mapping of weight names → torch.Tensor
        - config_dict  : contains keys "dim", "hidden_dim", "nomask", ...

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    device = input_tensor.device

    # --------------------------------------------------------------
    # 1) Input LayerNorm (float32)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )                       # [B, N, N, dim]  (float32)

    # --------------------------------------------------------------
    # 2) Left branch: proj + gate (fused)
    # --------------------------------------------------------------
    x_fp16 = x.to(torch.float16)  # cast once – used for all linear ops

    left_concat_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16)                # (2*H, dim)
    left_comb = F.linear(x_fp16, left_concat_w)  # (B, N, N, 2*H)

    left_proj = left_comb[..., :hidden_dim]
    left_gate = torch.sigmoid(left_comb[..., hidden_dim:])  # (B,N,N,H)

    left = left_proj * left_gate        # (B,N,N,H)  fp16

    # --------------------------------------------------------------
    # 3) Right branch: proj + gate (fused)
    # --------------------------------------------------------------
    right_concat_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16)                # (2*H, dim)
    right_comb = F.linear(x_fp16, right_concat_w)  # (B,N,N,2*H)

    right_proj = right_comb[..., :hidden_dim]
    right_gate = torch.sigmoid(right_comb[..., hidden_dim:])

    right = right_proj * right_gate    # (B,N,N,H)  fp16

    # --------------------------------------------------------------
    # 4) Optional mask (broadcast on hidden dimension)
    # --------------------------------------------------------------
    if not config.get("nomask", False) and mask is not None:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left = left * mask_fp16
        right = right * mask_fp16

    # free temporaries we no longer need
    del left_proj, left_gate, right_proj, right_gate
    del left_comb, right_comb, left_concat_w, right_concat_w

    # --------------------------------------------------------------
    # 5) Output gate (single linear + sigmoid)
    # --------------------------------------------------------------
    out_gate = torch.sigmoid(
        F.linear(x_fp16, weights["out_gate.weight"].to(torch.float16))
    )                                 # (B,N,N,H)  fp16
    out_gate_fp32 = out_gate.to(torch.float32)   # used later by Triton kernel

    # --------------------------------------------------------------
    # 6) Heavy multiplicative update – batched GEMM (A @ Bᵀ)
    # --------------------------------------------------------------
    # bring hidden dimension to the batch axis
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, H, N, N)  (k,j)

    B, H, N, _ = left_perm.shape
    batch_heads = B * H

    # reshape to (BH, N, N)
    A = left_perm.view(batch_heads, N, N)      # fp16
    Bmat = right_perm.view(batch_heads, N, N)  # fp16

    # batched matrix multiplication (A @ Bᵀ)
    C = torch.matmul(A, Bmat.transpose(-2, -1))   # fp16, shape (BH, N, N)

    # reshape back to (B, N, N, H) and cast to float32 for the next LN
    out = (
        C.view(B, H, N, N)
          .permute(0, 2, 3, 1)          # (B, N, N, H)
          .contiguous()
          .to(torch.float32)
    )

    # --------------------------------------------------------------
    # 7) Output LayerNorm (over hidden dim)
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # --------------------------------------------------------------
    # 8) Apply the output gate with a tiny Triton kernel
    # --------------------------------------------------------------
    out_flat = out.view(-1)
    gate_flat = out_gate_fp32.view(-1)
    numel = out_flat.numel()
    BLOCK_SIZE = 1024                         # ≤ max 1024 threads / block
    grid = (triton.cdiv(numel, BLOCK_SIZE),)

    elementwise_mul_kernel[grid](
        out_flat, out_flat, gate_flat, numel,
        BLOCK_SIZE=BLOCK_SIZE,
        num_warps=4,
        num_stages=2,
    )
    # `out` is now scaled in‑place

    # --------------------------------------------------------------
    # 9) Final projection back to original dimensionality
    # --------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])

    return out
