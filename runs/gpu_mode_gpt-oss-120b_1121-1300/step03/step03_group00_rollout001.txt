def custom_kernel(data):
    x, mask, weights, config = data
    dim = config['dim']; hidden_dim = config['hidden_dim']; eps = 1e-5; device = x.device

    # Input x is float32; cast to float16 for heavy ops
    x_fp16 = x.to(torch.float16)

    # Apply layernorm on dim
    x_ln = F.layer_norm(x_fp16, (dim,), weight=weights['norm.weight'].to(torch.float16), bias=weights['norm.bias'].to(torch.float16), eps=eps)

    # Linear projections left and right: weight shapes (hidden_dim, dim). Use torch.nn.functional.linear.
    left = F.linear(x_ln, weights['left_proj.weight'].to(torch.float16))
    right = F.linear(x_ln, weights['right_proj.weight'].to(torch.float16))

    # mask: optional
    if not config.get('nomask', False) and mask is not None:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)
        left = left * mask_fp16
        right = right * mask_fp16

    # gating:
    left_gate = torch.sigmoid(F.linear(x_ln, weights['left_gate.weight'].to(torch.float16)))
    right_gate = torch.sigmoid(F.linear(x_ln, weights['right_gate.weight'].to(torch.float16)))
    out_gate = torch.sigmoid(F.linear(x_ln, weights['out_gate.weight'].to(torch.float16)))

    left = left * left_gate
    right = right * right_gate

    # Prepare for heavy multiplication
    # left shape (B,N,N,H) ; right shape (B,N,N,H)
    # transpose right for correct orientation
    right_t = right.transpose(1,2) # shape (B,N,N,H) with swapped strides

    # Output buffer
    out = torch.empty_like(left)  # same shape, fp16

    B, N, _, H = left.shape

    # Determine grid sizes:
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64    # K = N

    # Compute strides
    # left:
    stride_a_batch = left.stride(0)
    stride_a_i = left.stride(1)
    stride_a_k = left.stride(2)
    # right_t:
    stride_b_batch = right_t.stride(0)
    stride_b_k = right_t.stride(1)   # after transpose, dim1 is K (original j)
    stride_b_n = right_t.stride(2)   # after transpose, dim2 is N (original i)
    # out:
    stride_c_batch = out.stride(0)
    stride_c_i = out.stride(1)
    stride_c_j = out.stride(2)
    # Validate correctness.

    # Launch grid
    grid = ( (N + BLOCK_M - 1)// BLOCK_M,
             (N + BLOCK_N - 1)// BLOCK_N,
             B * H)

    # Call kernel

"""
TriMul (outgoing) – Triton‑accelerated forward pass
---------------------------------------------------
Implements the “outgoing” TriMul operation from AlphaFold3.
Only the N³ heavy matrix‑multiply is off‑loaded to Triton; all
other linear / LayerNorm / gating steps stay in PyTorch.

The kernel operates on the original layout [B, N, N, H] (no
contiguous copy).  `right` is only a view with dimensions 1 and 2
transposed, so the GEMM obtains the required (i,k)·(k,j) product
by using strided loads.  This removes the costly `permute(...).contiguous()`
that the reference Triton implementation performed.

The kernel works with FP16 operands (Tensor‑Core friendly) and
accumulates in FP32.  The final tensor is cast back to FP32 before
the last LayerNorm and linear projection.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for 4‑D tensors [B, N, N, H]
#   A : left   – shape (B, N, N, H)   (row = i,   col = k)
#   B : rightᵀ – shape (B, N, N, H)   (row = k,   col = j)   (obtained via transpose)
#   C : output – shape (B, N, N, H)   (row = i,   col = j)
# Strides are passed explicitly; no materialised copies are required.
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    stride_a_batch, stride_a_i, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_j,
    stride_c_batch, stride_c_i, stride_c_j,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute C = A @ B  (batched over the leading dimension)
    A : (batch, M, K)  – row‑major with arbitrary strides
    B : (batch, K, N)  – row‑major with arbitrary strides
    C : (batch, M, N)  – stored with the same layout as A (last dim fastest)
    All tensors are fp16; accumulation is fp32.
    """
    # --------------------------------------------------------------
    # 1️⃣ program ids
    # --------------------------------------------------------------
    pid_batch = tl.program_id(2)          # (batch * head) index
    pid_m = tl.program_id(0)              # block row   (i)
    pid_n = tl.program_id(1)              # block col   (j)

    # --------------------------------------------------------------
    # 2️⃣ tile offsets
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # --------------------------------------------------------------
    # 3️⃣ base pointers for this tile (including the batch‑head offset)
    # --------------------------------------------------------------
    a_ptrs = a_ptr + pid_batch * stride_a_batch \
                  + offs_m[:, None] * stride_a_i \
                  + offs_k[None, :] * stride_a_k

    b_ptrs = b_ptr + pid_batch * stride_b_batch \
                  + offs_k[:, None] * stride_b_k \
                  + offs_n[None, :] * stride_b_j

    # --------------------------------------------------------------
    # 4️⃣ accumulator (fp32)
    # --------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # 5️⃣ loop over K‑tiles
    # --------------------------------------------------------------
    num_tiles_k = tl.cdiv(K, BLOCK_K)
    for _ in range(num_tiles_k):
        a = tl.load(
            a_ptrs,
            mask=(offs_m[:, None] < M) & (offs_k[None, :] < K),
            other=0.0,
        )
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & (offs_n[None, :] < N),
            other=0.0,
        )
        acc += tl.dot(a, b)          # fp16·fp16 → fp32
        a_ptrs += BLOCK_K * stride_a_k
        b_ptrs += BLOCK_K * stride_b_k

    # --------------------------------------------------------------
    # 6️⃣ write result (store as fp16)
    # --------------------------------------------------------------
    c = acc.to(tl.float16)
    c_ptrs = c_ptr + pid_batch * stride_c_batch \
                     + offs_m[:, None] * stride_c_i \
                     + offs_n[None, :] * stride_c_j
    tl.store(
        c_ptrs,
        c,
        mask=(offs_m[:, None] < M) & (offs_n[None, :] < N),
    )


# ----------------------------------------------------------------------
# Entry point used by the benchmark harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the *outgoing* TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor  [B, N, N, D] (float32)
        - mask_tensor  : torch.Tensor  [B, N, N]   (bool/float) or None
        - weights_dict : name → torch.Tensor (already on the correct device)
        - config_dict  : {"dim": D, "hidden_dim": H, "nomask": bool, ...}

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, D] (float32)
    """
    # --------------------------------------------------------------
    # 0️⃣ unpack arguments
    # --------------------------------------------------------------
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # --------------------------------------------------------------
    # 1️⃣ Layer‑norm over the channel dimension `dim` (FP16 for speed)
    # --------------------------------------------------------------
    x = x.to(torch.float16)
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"].to(torch.float16),
        bias=weights["norm.bias"].to(torch.float16),
        eps=eps,
    )

    # --------------------------------------------------------------
    # 2️⃣ Linear projections (FP16)
    # --------------------------------------------------------------
    left = F.linear(
        x, weights["left_proj.weight"].to(torch.float16)
    )  # [B, N, N, H]
    right = F.linear(
        x, weights["right_proj.weight"].to(torch.float16)
    )

    # --------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast on hidden dim)
    # --------------------------------------------------------------
    if (not config.get("nomask", False)) and mask is not None:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_fp16
        right = right * mask_fp16

    # --------------------------------------------------------------
    # 4️⃣ Gates (sigmoid) – still FP16
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(
        F.linear(x, weights["left_gate.weight"].to(torch.float16))
    )
    right_gate = torch.sigmoid(
        F.linear(x, weights["right_gate.weight"].to(torch.float16))
    )
    out_gate = torch.sigmoid(
        F.linear(x, weights["out_gate.weight"].to(torch.float16))
    )

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5️⃣ Heavy multiplicative update (batched GEMM)
    # --------------------------------------------------------------
    # rightᵀ is obtained by swapping dimensions 1 and 2 (no copy)
    right_t = right.transpose(1, 2)   # shape (B, N, N, H), strides swapped

    # Output buffer (FP16)
    out = torch.empty_like(left)

    B, N, _, H = left.shape
    batch_heads = B * H

    # ------------------------------------------------------------------
    # Tile sizes – tuned for H100 Tensor‑Cores (FP16·FP16 → FP32)
    # ------------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    # ------------------------------------------------------------------
    # Strides (all tensors are contiguous in memory)
    # ------------------------------------------------------------------
    # A = left
    stride_a_batch = left.stride(0)   # B * N * N * H
    stride_a_i = left.stride(1)       # N * H
    stride_a_k = left.stride(2)       # H

    # B = rightᵀ (after transpose)
    stride_b_batch = right_t.stride(0)
    stride_b_k = right_t.stride(1)    # = H   (row dimension after transpose)
    stride_b_j = right_t.stride(2)    # = N*H (column dimension after transpose)

    # C = out (same layout as A)
    stride_c_batch = out.stride(0)
    stride_c_i = out.stride(1)
    stride_c_j = out.stride(2)

    # ------------------------------------------------------------------
    # Kernel launch grid
    # ------------------------------------------------------------------
    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # blocks along M (i)
        (N + BLOCK_N - 1) // BLOCK_N,   # blocks along N (j)
        batch_heads,                    # one block per (batch, head)
    )

    tri_mul_batched_matmul_kernel[grid](
        left, right_t, out,
        stride_a_batch,
        stride_a_i,
        stride_a_k,
        stride_b_batch,
        stride_b_k,
        stride_b_j,
        stride_c_batch,
        stride_c_i,
        stride_c_j,
        M=N,
        N=N,
        K=N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=4,
    )
    torch.cuda.synchronize()

    # --------------------------------------------------------------
    # 6️⃣ Cast back to FP32, apply output LayerNorm and out‑gate
    # --------------------------------------------------------------
    out = out.float()                     # (B, N, N, H) fp32
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate.float()

    # --------------------------------------------------------------
    # 7️⃣ Final linear projection back to original `dim`
    # --------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])

    return out
