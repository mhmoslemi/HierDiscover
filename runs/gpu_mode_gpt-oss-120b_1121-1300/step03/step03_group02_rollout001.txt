@triton.jit
def gate_and_mask_kernel(
    left_ptr, right_ptr, out_gate_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    total_elems: tl.constexpr,
    H: tl.constexpr,
    mask_present: tl.constexpr,
    BLOCK_SIZE: tl.constexpr = 1024
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems
    # load data
    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
    out_gate = tl.load(out_gate_ptr + offs, mask=mask, other=0.0)
    # sigmoid
    l_gate_sig = 1.0 / (1.0 + tl.exp(-left_gate))
    r_gate_sig = 1.0 / (1.0 + tl.exp(-right_gate))
    out_gate_sig = 1.0 / (1.0 + tl.exp(-out_gate))
    left = left * l_gate_sig
    right = right * r_gate_sig
    # mask if needed
    if mask_present:
        # compute mask index = offsets // H
        mask_idx = offs // H
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask_idx < (total_elems // H), other=1.0)
        left = left * mask_val
        right = right * mask_val
    # store back
    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)
    tl.store(out_gate_ptr + offs, out_gate_sig, mask=mask)
```

"""
TriMul (outgoing) – Triton‑accelerated cubic update
==================================================
The implementation follows the reference PyTorch module but aggressively fuses
operations with two custom Triton kernels:

1. **gate_and_mask_kernel**
   * Computes the three sigmoids (left_gate, right_gate, out_gate) and
     applies the element‑wise gating.
   * If a pairwise mask is provided it is broadcast over the hidden dimension
     and multiplied into *left* and *right*.
   * All tensors are kept in fp16, the sigmoid uses a fast 1/(1+exp(-x))
     formulation.

2. **batched_matmul_kernel**
   * Performs the heavy O(N³) multiplicative update:
         out[b, i, j, h] = Σₖ left[b, i, k, h] * right[b, j, k, h]
   * The kernel works on the flattened batch‑head dimension
     (B·H matrices of size N×N) and accumulates in FP32 (Tensor‑Core
     friendly).  Block sizes are chosen dynamically (64/128) so that
     there is no out‑of‑bounds work for the tested sequence lengths.

The rest of the pipeline (LayerNorms, the fused five‑linear projection,
final LayerNorm, output gate and the final linear projection) is done with
torch.nn.functional for simplicity; the dominant O(N³) work is fully
off‑loaded to the kernels above.

The kernel works for any batch size `B ∈ {1,2}`, sequence length
`N ∈ {128,256,512,1024}`, feature dimension `dim` (e.g. 128/384) and
hidden dimension `hidden_dim` (e.g. 128).  The output is a float32 tensor
with shape `[B, N, N, dim]`.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1) ELEMENT‑WISE GATING + OPTIONAL MASK
# ----------------------------------------------------------------------
@triton.jit
def gate_and_mask_kernel(
    left_ptr,               # fp16*   [B*N*N*H]
    right_ptr,              # fp16*
    out_gate_ptr,           # fp16*
    left_gate_ptr,          # fp16*
    right_gate_ptr,         # fp16*
    mask_ptr,               # fp16*   [B*N*N]  (may be dummy)
    TOTAL_ELEMS: tl.constexpr,   # B*N*N*H
    H: tl.constexpr,                # hidden dimension
    MASK: tl.constexpr,             # 0 -> no mask, 1 -> mask present
    BLOCK_SIZE: tl.constexpr = 1024,
):
    """
    In‑place:
        left  = left  * sigmoid(left_gate)  * mask
        right = right * sigmoid(right_gate) * mask
        out_gate = sigmoid(out_gate)
    All tensors are stored in flat layout.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    valid = offs < TOTAL_ELEMS

    # ------------------------------------------------------------------
    # load everything (masked loads return 0.0 for out‑of‑bounds)
    # ------------------------------------------------------------------
    left      = tl.load(left_ptr      + offs, mask=valid, other=0.0)
    right     = tl.load(right_ptr     + offs, mask=valid, other=0.0)
    left_g    = tl.load(left_gate_ptr + offs, mask=valid, other=0.0)
    right_g   = tl.load(right_gate_ptr+ offs, mask=valid, other=0.0)
    out_gate  = tl.load(out_gate_ptr + offs, mask=valid, other=0.0)

    # ------------------------------------------------------------------
    # sigmoid (1/(1+exp(-x))) – works in fp16
    # ------------------------------------------------------------------
    left_sig  = 1.0 / (1.0 + tl.exp(-left_g))
    right_sig = 1.0 / (1.0 + tl.exp(-right_g))
    out_sig   = 1.0 / (1.0 + tl.exp(-out_gate))

    left = left * left_sig
    right = right * right_sig

    # ------------------------------------------------------------------
    # optional mask (broadcasted over the hidden axis)
    # ------------------------------------------------------------------
    if MASK:
        # each hidden element repeats the same mask entry
        # mask_index = floor(offs / H)
        mask_idx = offs // H
        mask_val = tl.load(mask_ptr + mask_idx,
                           mask=mask_idx < (TOTAL_ELEMS // H),
                           other=1.0)          # 1.0 == no masking
        left = left * mask_val
        right = right * mask_val

    # ------------------------------------------------------------------
    # store results back (in‑place)
    # ------------------------------------------------------------------
    tl.store(left_ptr      + offs, left,      mask=valid)
    tl.store(right_ptr     + offs, right,     mask=valid)
    tl.store(out_gate_ptr  + offs, out_sig,   mask=valid)


# ----------------------------------------------------------------------
# 2) BATCHED GEMM:  out = left @ rightᵀ  (one GEMM per (batch, head))
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_batch = tl.program_id(2)          # batch * head dimension
    pid_m = tl.program_id(0)              # block row
    pid_n = tl.program_id(1)              # block column

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for the current tile (including batch‑head offset)
    a_ptrs = a_ptr + pid_batch * stride_a_batch \
                   + offs_m[:, None] * stride_a_m \
                   + offs_k[None, :] * stride_a_k
    b_ptrs = b_ptr + pid_batch * stride_b_batch \
                   + offs_k[:, None] * stride_b_k \
                   + offs_n[None, :] * stride_b_n

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    iters = tl.cdiv(K, BLOCK_K)
    for _ in range(iters):
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (offs_k[None, :] < K),
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=(offs_k[:, None] < K) & mask_n[None, :],
                    other=0.0)
        acc += tl.dot(a, b)                     # fp16·fp16 → fp32

        a_ptrs += BLOCK_K * stride_a_k
        b_ptrs += BLOCK_K * stride_b_k

    # store FP16 result (will be cast to FP32 later)
    c = acc.to(tl.float16)
    c_ptrs = c_ptr + pid_batch * stride_c_batch \
                     + offs_m[:, None] * stride_c_m \
                     + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs,
             c,
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# ENTRypoint used by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the *outgoing* TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, dim]
        - mask_tensor  : torch.Tensor of shape [B, N, N] (or None)
        - weights_dict : mapping from string names to torch.Tensor
        - config_dict  : contains ``dim``, ``hidden_dim`` and ``nomask``

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim         = config["dim"]
    hidden_dim  = config["hidden_dim"]
    nomask      = config.get("nomask", False)
    eps = 1e-5

    device = input_tensor.device
    dtype  = input_tensor.dtype

    # --------------------------------------------------------------
    # 1) Layer‑norm over the feature dimension (dim)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # --------------------------------------------------------------
    # 2) Fuse the five linear maps into one matmul
    # --------------------------------------------------------------
    stacked_weight = torch.cat(
        [
            weights["left_proj.weight"],    # (H, dim)
            weights["right_proj.weight"],   # (H, dim)
            weights["left_gate.weight"],    # (H, dim)
            weights["right_gate.weight"],   # (H, dim)
            weights["out_gate.weight"],     # (H, dim)
        ],
        dim=0,
    ).to(torch.float16)                      # (5*H, dim)

    B, N, _, _ = input_tensor.shape
    x_fp16 = x.to(torch.float16)
    x_flat = x_fp16.view(-1, dim)            # (B·N·N, dim)

    fused = F.linear(x_flat, stacked_weight) # (B·N·N, 5*H)  fp16

    fused = fused.view(B, N, N, 5 * hidden_dim)

    left, right, left_gate, right_gate, out_gate = torch.split(
        fused, hidden_dim, dim=3
    )  # each -> [B, N, N, H]

    # --------------------------------------------------------------
    # 3) Gating + optional mask (all in fp16)
    # --------------------------------------------------------------
    total_elements = left.numel()
    mask_present = 0 if (nomask or mask is None) else 1

    # cast mask once if it exists (broadcasted later)
    if mask_present:
        mask_fp16 = mask.to(torch.float16)   # [B, N, N]
    else:
        # dummy tensor – Triton will never read it because MASK==0
        mask_fp16 = torch.empty(0, dtype=torch.float16, device=device)

    # launch gating kernel
    BLOCK_SIZE = 1024
    grid_g = (total_elements + BLOCK_SIZE - 1) // BLOCK_SIZE

    gate_and_mask_kernel[grid_g](
        left,
        right,
        out_gate,
        left_gate,
        right_gate,
        mask_fp16,
        total_elements,
        hidden_dim,
        mask_present,
        BLOCK_SIZE=BLOCK_SIZE,
        num_warps=4,
    )
    torch.cuda.synchronize()

    # --------------------------------------------------------------
    # 4) Heavy multiplicative update (batched GEMM)
    # --------------------------------------------------------------
    #   left  : [B, N, N, H] → (B, H, N, N)
    #   right : [B, N, N, H] → (B, H, N, N) with K/N swapped
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, K=N, N)

    B_, H_, N_, _ = left_perm.shape
    batch_heads = B_ * H_

    A = left_perm.view(batch_heads, N_, N_)    # (BH, M=N, K=N)
    Bmat = right_perm.view(batch_heads, N_, N_)  # (BH, K=N, N)

    # output buffer for the GEMM (still fp16)
    C_fp16 = torch.empty_like(A, dtype=torch.float16)

    # Choose block sizes that tile the matrix without remainder when possible
    if N_ % 128 == 0:
        BLOCK_M = BLOCK_N = 128
    else:
        BLOCK_M = BLOCK_N = 64
    BLOCK_K = 64 if N_ >= 64 else N_

    grid = (
        (N_ + BLOCK_M - 1) // BLOCK_M,   # grid dim M
        (N_ + BLOCK_N - 1) // BLOCK_N,   # grid dim N
        batch_heads,                     # grid dim batch*head
    )

    batched_matmul_kernel[grid](
        A, Bmat, C_fp16,
        A.stride(0), A.stride(1), A.stride(2),   # a: batch, M, K
        Bmat.stride(0), Bmat.stride(1), Bmat.stride(2),  # b: batch, K, N
        C_fp16.stride(0), C_fp16.stride(1), C_fp16.stride(2),  # c: batch, M, N
        M=N_, N=N_, K=N_,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=4,
    )
    torch.cuda.synchronize()

    # --------------------------------------------------------------
    # 5) Reshape back to [B, N, N, H] (float32) and apply output norm
    # --------------------------------------------------------------
    out = (
        C_fp16
        .view(B_, H_, N_, N_)      # (B, H, N, N)
        .permute(0, 2, 3, 1)       # (B, N, N, H)
        .to(torch.float32)         # to fp32 for layer‑norm
    )

    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    # out_gate already sigmoid‑ed (fp16) → cast to fp32 for multiplication
    out = out * out_gate.to(torch.float32)

    # --------------------------------------------------------------
    # 6) Final linear projection back to original feature dimension
    # --------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])

    return out
