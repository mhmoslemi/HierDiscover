        apply_gate_mask_kernel[grid](
            left_proj,
            left_gate,
            mask_fp16,

            total,

        )

    else:

        right = right_proj * right_gate

    # free intermediate tensors we no longer need
    del left_proj, left_gate, right_proj, right_gate, left_out, right_out, left_w, right_w

    # 5) compute out_gate (shared across hidden dim)
    out_gate = torch.sigmoid(F.linear(x_fp16, weights["out_gate.weight"].to(torch.float16)))  # (B,N,N,H)

    # 6) heavy multiplicative update via batched GEMM (torch.bmm)
    # reshape to (batch*head, N, N) and do A @ B^T
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N)

    B, H, N, _ = left_perm.shape
    batch_heads = B * H
    A = left_perm.view(batch_heads, N, N)   # (BH, M=N, K=N)
    Bmat = right_perm.view(batch_heads, N, N)  # (BH, K=N, N)

    # torch.bmm expects (batch, M, K) @ (batch, K, N)
    C = torch.bmm(A, Bmat.transpose(1, 2))  # (BH, N, N)

    # reshape back
    out = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

    # 7) output layernorm and gating
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=weights["to_out_norm.weight"],
                       bias=weights["to_out_norm.bias"],
                       eps=eps)
    out = out * out_gate

    # 8) final linear back to original dim (float32)
    out = F.linear(out, weights["to_out.weight"])

    return out

"""
TriMul (outgoing) – fast forward implementation.

Algorithm
---------
1. Layer‑Norm over the last dimension.
2. Fused linear projections for the left and right sides:
   * concat(proj, gate) weight → one matmul, then split.
3. Optional mask: a tiny Triton kernel fuses
   (proj * gate * mask) in a single pass, avoiding two separate
   element‑wise kernels.
4. Compute the heavy multiplicative update
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   as a batched matrix‑multiply using torch.bmm (highly‑optimised
   cuBLAS/Tensor‑Core GEMM).  Shapes are rearranged to (B·H, N, N) and
   (B·H, N, N)ᵀ before the call.
5. Layer‑Norm over the hidden dimension, element‑wise multiplication by
   the output gate.
6. Final linear projection back to the original feature dimension.

Only the mask‑gate fusion kernel is written in Triton; all other heavy
arithmetic uses PyTorch + cuBLAS, which on an H100 easily meets the
sub‑millisecond target.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _apply_gate_mask_kernel(
    proj_ptr, gate_ptr, mask_ptr, out_ptr,
    total_elements, hidden_dim, mask_elements,
    BLOCK: tl.constexpr,
):
    """
    out = proj * gate * mask
    *proj*, *gate* have shape (B, N, N, H)   → flattened length = total_elements
    *mask*  has shape (B, N, N)            → flattened length = mask_elements
    The mask is broadcast across the hidden dimension, i.e. the same mask
    value is used for all H entries of a given (b,i,j) location.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)                     # linear offset
    valid = offs < total_elements

    # load projection and gating values
    proj = tl.load(proj_ptr + offs, mask=valid, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=valid, other=0.0)

    # compute index into the 3‑D mask (repeat every hidden_dim entries)
    mask_offs = offs // hidden_dim
    mask_val = tl.load(mask_ptr + mask_offs,
                       mask=mask_offs < mask_elements,
                       other=1.0)                     # 1.0 → no masking when absent

    out = proj * gate * mask_val
    tl.store(out_ptr + offs, out, mask=valid)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor of shape [B, N, N] (bool/float) or None
        - weights_dict : mapping from string names to torch.Tensor (float32)
        - config_dict  : contains "dim", "hidden_dim", "nomask", ...

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1) Input LayerNorm over the feature dimension
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )
    x_fp16 = x.to(torch.float16)                     # heavy part in fp16

    # ------------------------------------------------------------------
    # 2) Fused projection + gate for the *left* side
    # ------------------------------------------------------------------
    left_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16)                               # (2*H, dim)
    left_all = F.linear(x_fp16, left_w)               # (B,N,N,2*H)
    left_proj = left_all[..., :hidden_dim]            # (B,N,N,H)
    left_gate = torch.sigmoid(left_all[..., hidden_dim:])  # (B,N,N,H)

    # ------------------------------------------------------------------
    # 3) Fused projection + gate for the *right* side
    # ------------------------------------------------------------------
    right_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16)                               # (2*H, dim)
    right_all = F.linear(x_fp16, right_w)             # (B,N,N,2*H)
    right_proj = right_all[..., :hidden_dim]
    right_gate = torch.sigmoid(right_all[..., hidden_dim:])

    # ------------------------------------------------------------------
    # 4) Optional mask & element‑wise gating (fused Triton kernel)
    # ------------------------------------------------------------------
    use_mask = mask is not None and not config.get("nomask", False)
    if use_mask:
        mask_fp16 = mask.to(torch.float16)           # (B,N,N)
        # flatten tensors so that the Triton kernel can use a 1‑D pointer
        total = left_proj.numel()                    # B*N*N*H
        mask_total = mask_fp16.numel()               # B*N*N

        BLOCK = 256
        grid = ((total + BLOCK - 1) // BLOCK,)

        # left = left_proj * left_gate * mask
        left_flat = torch.empty_like(left_proj).reshape(-1)
        _apply_gate_mask_kernel[grid](
            left_proj.reshape(-1),
            left_gate.reshape(-1),
            mask_fp16.reshape(-1),
            left_flat,
            total,
            hidden_dim,
            mask_total,
            BLOCK=BLOCK,
            num_warps=4,
        )
        left = left_flat.view_as(left_proj)

        # right = right_proj * right_gate * mask
        right_flat = torch.empty_like(right_proj).reshape(-1)
        _apply_gate_mask_kernel[grid](
            right_proj.reshape(-1),
            right_gate.reshape(-1),
            mask_fp16.reshape(-1),
            right_flat,
            total,
            hidden_dim,
            mask_total,
            BLOCK=BLOCK,
            num_warps=4,
        )
        right = right_flat.view_as(right_proj)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # free temporaries that are no longer needed
    del left_proj, left_gate, right_proj, right_gate, left_all, right_all, left_w, right_w

    # ------------------------------------------------------------------
    # 5) Output gate (single linear + sigmoid)
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(
        F.linear(x_fp16, weights["out_gate.weight"].to(torch.float16))
    )   # (B,N,N,H)

    # ------------------------------------------------------------------
    # 6) Heavy multiplicative update – batched GEMM via cuBLAS
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    #   left : (B,N,N,H) → (B,H,N,N) → (BH,N,N)
    #   right: (B,N,N,H) → (B,H,N,N) with swapped axes (k,j) → (BH,N,N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()    # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous()   # (B, H, N, N)

    B, H, N, _ = left_perm.shape
    batch_heads = B * H

    A = left_perm.view(batch_heads, N, N)                # (BH, M=N, K=N)
    Bmat = right_perm.view(batch_heads, N, N)            # (BH, K=N, N)

    # batched matrix multiply (FP16 inputs, FP32 accumulation inside cuBLAS)
    C = torch.bmm(A, Bmat.transpose(1, 2))               # (BH, N, N)

    # reshape back to (B, N, N, H)
    out = (
        C.view(B, H, N, N)
         .permute(0, 2, 3, 1)
         .contiguous()
    )   # (B, N, N, H)

    # ------------------------------------------------------------------
    # 7) Output LayerNorm + output gate
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate

    # ------------------------------------------------------------------
    # 8) Final projection back to the original feature dimension
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])   # (B, N, N, dim), float32

    return out
