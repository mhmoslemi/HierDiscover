"""
TriMul (outgoing) – fast Triton‑accelerated forward pass
=======================================================

The implementation follows the reference PyTorch module:

1. Layer‑norm on the input tensor (float32).
2. Two bias‑free linear projections (dim → hidden_dim) for the left and right
   branches and two linear gates (dim → hidden_dim).  All heavy ops are done
   in fp16 (Tensor‑Core friendly) and the gates are applied with a sigmoid.
   An optional binary mask (broadcast on the hidden dimension) is applied
   after the projections.
3. The core O(N³·hidden_dim) contraction is evaluated as a batch of
   matrix‑multiplications:

        out_d = left_d @ right_dᵀ   for d = 0 … hidden_dim‑1

   where left_d and right_d have shape (B, N, N).  This is realised with a
   custom Triton kernel that computes many independent GEMMs (batch*head =
   B·hidden_dim) in fp16 and accumulates in fp32.

4. Output‑layer‑norm, an output gate, and a final linear projection back to
   the original feature dimension (hidden_dim → dim).

All tensors are returned in float32.  The Triton kernel is the only part of
the computation that is not covered by PyTorch and provides the speed‑up
required for the large N cases.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute C = A @ B for a batch of matrices.
    A : (batch, M, K)   – row‑major
    B : (batch, K, N)   – row‑major
    C : (batch, M, N)   – row‑major, stored as fp16 (accumulated in fp32)
    """
    pid_batch = tl.program_id(2)              # which (batch*head)
    pid_m = tl.program_id(0)                  # block‑row index
    pid_n = tl.program_id(1)                  # block‑col index

    # block offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # base pointers for the current tile (including batch‑head offset)
    a_ptrs = a_ptr + pid_batch * stride_a_batch \
                  + offs_m[:, None] * stride_a_m \
                  + offs_k[None, :] * stride_a_k
    b_ptrs = b_ptr + pid_batch * stride_b_batch \
                  + offs_k[:, None] * stride_b_k \
                  + offs_n[None, :] * stride_b_n

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K‑tiles
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs,
                    mask=(offs_m[:, None] < M) & (offs_k[None, :] < K),
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=(offs_k[:, None] < K) & (offs_n[None, :] < N),
                    other=0.0)
        acc += tl.dot(a, b)        # fp16·fp16 → fp32

        a_ptrs += BLOCK_K * stride_a_k
        b_ptrs += BLOCK_K * stride_b_k

    # write out (store as fp16)
    c = acc.to(tl.float16)
    c_ptrs = c_ptr + pid_batch * stride_c_batch \
                     + offs_m[:, None] * stride_c_m \
                     + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs,
             c,
             mask=(offs_m[:, None] < M) & (offs_n[None, :] < N))


def custom_kernel(data):
    """
    Forward pass of the *outgoing* TriMul operator (no gradient).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor   [B, N, N, D]   (float32)
        - mask_tensor  : torch.Tensor   [B, N, N]      (bool / float) or None
        - weights_dict : name → torch.Tensor (already on the device)
        - config_dict  : {"dim": D, "hidden_dim": H, "nomask": bool, ...}

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, D] (float32)
    """
    # ------------------------------------------------------------------
    # 0️⃣ unpack arguments
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    device = x.device

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Cast to fp16 for the heavy part
    # ------------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Linear projections (bias‑free)
    # ------------------------------------------------------------------
    left = F.linear(x_fp16,
                    weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x_fp16,
                     weights["right_proj.weight"].to(torch.float16))

    # ------------------------------------------------------------------
    # 4️⃣ Optional mask (broadcast on hidden dimension)
    # ------------------------------------------------------------------
    if not config.get("nomask", False) and mask is not None:
        mask_fp16 = mask.to(x_fp16.dtype).unsqueeze(-1)      # [B,N,N,1]
        left = left * mask_fp16
        right = right * mask_fp16

    # ------------------------------------------------------------------
    # 5️⃣ Gates (sigmoid) – still in fp16
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(
        F.linear(x_fp16, weights["left_gate.weight"].to(torch.float16))
    )
    right_gate = torch.sigmoid(
        F.linear(x_fp16, weights["right_gate.weight"].to(torch.float16))
    )
    out_gate = torch.sigmoid(
        F.linear(x_fp16, weights["out_gate.weight"].to(torch.float16))
    )

    # Apply the gates
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6️⃣ Core contraction: batched GEMM (left @ rightᵀ) over the sequence dim
    # ------------------------------------------------------------------
    # Reshape to (B, hidden_dim, N, N) and make them contiguous so that the
    # hidden dimension can be merged into the batch dimension.
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)  -> A[i,k]
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, H, N, N)  -> B[k,j]

    B, H, N, _ = left_perm.shape
    batch_heads = B * H

    A = left_perm.view(batch_heads, N, N)       # (BH, M=N, K=N)
    Bmat = right_perm.view(batch_heads, N, N)   # (BH, K=N, N)

    # Output buffer (fp16)
    C = torch.empty_like(A, dtype=torch.float16)

    # ------------------------------------------------------------------
    # 6️⃣1️⃣ Triton GEMM launch parameters
    # ------------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 128                     # larger tile reduces the K‑loop count

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # blocks along M
        (N + BLOCK_N - 1) // BLOCK_N,   # blocks along N
        batch_heads                     # one batch‑head per GEMM
    )

    batched_matmul_kernel[grid](
        A, Bmat, C,
        A.stride(0), A.stride(1), A.stride(2),
        Bmat.stride(0), Bmat.stride(1), Bmat.stride(2),
        C.stride(0), C.stride(1), C.stride(2),
        M=N, N=N, K=N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=4,
    )
    torch.cuda.synchronize()

    # ------------------------------------------------------------------
    # 6️⃣2️⃣ Reshape back to [B, N, N, hidden_dim] (fp32)
    # ------------------------------------------------------------------
    out = (
        C.float()                                   # fp16 → fp32
          .view(B, H, N, N)                         # (B, H, N, N)
          .permute(0, 2, 3, 1)                      # (B, N, N, H)
    )

    # ------------------------------------------------------------------
    # 7️⃣ Output LayerNorm + out‑gate (fp32)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate

    # ------------------------------------------------------------------
    # 8️⃣ Final projection back to the original dimension (fp16 matmul → fp32)
    # ------------------------------------------------------------------
    out_fp16 = out.to(torch.float16)
    out_fp16 = F.linear(out_fp16,
                        weights["to_out.weight"].to(torch.float16))
    out = out_fp16.float()

    return out
