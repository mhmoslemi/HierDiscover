"""
TriMul (outgoing) – Triton‑accelerated implementation
====================================================

The forward pass consists of:
1. Layer‑norm over the feature dimension (dim) – torch (fp32).
2. One fused linear projection that produces the three projections
   (left, right) and the three gates (left_gate, right_gate, out_gate).
3. Element‑wise mask (if present) and sigmoid gating.
4. Batched matrix multiplication C = A @ Bᵀ for every head h:
       A = left  [B,N,N,H]  (M = i, K = k)
       B = right [B,N,N,H]  (M = j, K = k) → we load it as Bᵀ
   This step is performed by a custom Triton kernel that works
   directly on the original (non‑contiguous) tensor layout, thus
   avoiding the costly ``.permute(...).contiguous()`` copies that
   dominate memory traffic for large N.
5. Layer‑norm over the hidden dimension, multiplication by the
   output gate, and a final linear projection back to the original
   feature dimension.

All heavy arithmetic is carried out in FP16 (Tensor‑Core friendly);
accumulation is in FP32.  The final result is returned in FP32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ
#   A : (B, H, M=N, K=N)   – row‑major (M = i, K = k)
#   B : (B, H, K=N, N=N)   – row‑major (already transposed,
#                              i.e. Bᵀ stored as (k, j))
#   C : (B, H, M=N, N=N)
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    stride_a_batch, stride_a_head, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_head, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_head, stride_c_m, stride_c_n,
    M, N, K,
    H,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # ------------------------------------------------------------------
    # program IDs (grid: (num_M_blocks, num_N_blocks, B*H))
    # ------------------------------------------------------------------
    pid_m = tl.program_id(0)          # block row (i)
    pid_n = tl.program_id(1)          # block col (j)
    pid_bh = tl.program_id(2)         # combined batch*head index

    # ------------------------------------------------------------------
    # derive batch and head indices
    # ------------------------------------------------------------------
    batch_idx = pid_bh // H
    head_idx  = pid_bh % H

    # ------------------------------------------------------------------
    # tile offsets
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # ------------------------------------------------------------------
    # base pointers for this (batch,head) pair
    # ------------------------------------------------------------------
    a_base = a_ptr + batch_idx * stride_a_batch + head_idx * stride_a_head
    b_base = b_ptr + batch_idx * stride_b_batch + head_idx * stride_b_head
    c_base = c_ptr + batch_idx * stride_c_batch + head_idx * stride_c_head

    a_ptrs = a_base \
             + offs_m[:, None] * stride_a_m \
             + offs_k[None, :] * stride_a_k
    b_ptrs = b_base \
             + offs_k[:, None] * stride_b_k \
             + offs_n[None, :] * stride_b_n

    # accumulator in FP32 (Tensor‑Core accumulation)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # loop over K dimension
    # ------------------------------------------------------------------
    num_iters = tl.cdiv(K, BLOCK_K)
    for _ in range(0, num_iters):
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (offs_k[None, :] < K),
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=(offs_k[:, None] < K) & mask_n[None, :],
                    other=0.0)
        acc += tl.dot(a, b)   # fp16·fp16 → fp32

        a_ptrs += BLOCK_K * stride_a_k
        b_ptrs += BLOCK_K * stride_b_k

    # ------------------------------------------------------------------
    # write result (FP16)
    # ------------------------------------------------------------------
    c = acc.to(tl.float16)
    c_ptrs = c_base \
             + offs_m[:, None] * stride_c_m \
             + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# PyTorch entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the *outgoing* TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor of shape [B, N, N] or ``None``
        - weights_dict : mapping from string names to torch.Tensor
        - config_dict  : contains ``dim``, ``hidden_dim``, ``nomask``

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    eps = 1e-5

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1) Input LayerNorm over ``dim``
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 2) Fuse the five linear maps (left, right, three gates)
    # ------------------------------------------------------------------
    stacked_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)        # fp16 for Tensor‑Core

    B, N, _, _ = input_tensor.shape    # B, N, N, dim
    x_fp16 = x.to(torch.float16)
    x_flat = x_fp16.view(-1, dim)                # (B·N·N, dim)
    fused = F.linear(x_flat, stacked_weight)     # (B·N·N, 5*H)
    fused = fused.view(B, N, N, 5 * hidden_dim)

    left      = fused[..., 0 * hidden_dim : 1 * hidden_dim]
    right     = fused[..., 1 * hidden_dim : 2 * hidden_dim]
    left_gate = fused[..., 2 * hidden_dim : 3 * hidden_dim]
    right_gate= fused[..., 3 * hidden_dim : 4 * hidden_dim]
    out_gate  = fused[..., 4 * hidden_dim : 5 * hidden_dim]

    # ------------------------------------------------------------------
    # 3) Optional mask (broadcast on hidden dim)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_ = mask.to(torch.float16).unsqueeze(-1)   # [B, N, N, 1]
        left  = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4) Sigmoid gates and element‑wise gating (still fp16)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate   = torch.sigmoid(out_gate)

    left  = left  * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Heavy multiplicative update C = left @ rightᵀ
    #    Implemented with the Triton kernel defined above.
    #    No extra ``.contiguous()`` copies – we work directly on the
    #    original (B,N,N,H) layout.
    # ------------------------------------------------------------------
    # Output buffer: shape (B, H, N, N) in FP16
    out_fp16 = torch.empty(
        (B, hidden_dim, N, N), dtype=torch.float16, device=device
    )

    # Choose block sizes according to N (avoid tiny‑grid overhead)
    if N <= 256:
        BLOCK_M, BLOCK_N, BLOCK_K = 64, 64, 32
        num_warps = 4
    else:
        BLOCK_M, BLOCK_N, BLOCK_K = 128, 128, 64
        num_warps = 8

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # blocks over M (i)
        (N + BLOCK_N - 1) // BLOCK_N,   # blocks over N (j)
        B * hidden_dim,                 # one slice per (batch, head)
    )

    # Strides (in element units, not bytes)
    stride_a_batch, stride_a_head, stride_a_m, stride_a_k = (
        left.stride(0), left.stride(3), left.stride(1), left.stride(2)
    )
    stride_b_batch, stride_b_head, stride_b_k, stride_b_n = (
        right.stride(0), right.stride(3), right.stride(2), right.stride(1)
    )
    stride_c_batch, stride_c_head, stride_c_m, stride_c_n = (
        out_fp16.stride(0), out_fp16.stride(1), out_fp16.stride(2), out_fp16.stride(3)
    )

    batched_matmul_kernel[grid](
        left, right, out_fp16,
        stride_a_batch, stride_a_head, stride_a_m, stride_a_k,
        stride_b_batch, stride_b_head, stride_b_k, stride_b_n,
        stride_c_batch, stride_c_head, stride_c_m, stride_c_n,
        M=N, N=N, K=N,
        H=hidden_dim,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=num_warps,
        num_stages=4,
    )
    torch.cuda.synchronize()

    # ------------------------------------------------------------------
    # 6) Reshape back to [B, N, N, H] (convert to FP32)
    # ------------------------------------------------------------------
    out = out_fp16.permute(0, 2, 3, 1).to(torch.float32)   # (B, N, N, H)

    # ------------------------------------------------------------------
    # 7) Output LayerNorm + out‑gate
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate.to(out.dtype)    # broadcast over hidden dim

    # ------------------------------------------------------------------
    # 8) Final linear projection back to original ``dim``
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])

    return out
